# 权限与沙箱

上一章的 agent 能改文件了。这一章给它加最后一个、也是最危险的工具：**执行命令**。

危险在哪儿？编译、跑测试、看 git 状态，这些都要执行命令，agent 没有它几乎干不了活。
但同一个工具也能执行 `rm -rf`。

这一章的主线是一句话：

> **提示词不是安全边界。**

在系统提示里写「不要删除文件」，挡不住任何事情——模型可能误解、可能被输入里的指令带偏、
也可能只是判断失误。真正的边界必须是**代码里的机制**。

我们建三道防线，从内到外：

| 防线 | 拦住什么 | 拦不住什么 |
|---|---|---|
| 1. 不给 shell | 命令注入 | 命令本身就是危险的 |
| 2. 审批门 | 危险操作没经人同意就跑 | 人点了同意之后 |
| 3. 隔离 | 跑起来之后的破坏范围 | 隔离本身的漏洞 |

```go
// file: core.go
package main

import (
	"bufio"
	"bytes"
	"context"
	"crypto/rand"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"net/http"

func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)
}

type Block interface{ BlockType() string }

type TextBlock struct{ Text string }

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string
	Name string
	Args map[string]any
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string
	Content   string
	IsError   bool
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

type Message struct {
	ID      string
	Role    string
	Content []Block
}

func (m Message) Text() string {
	var b strings.Builder
	for _, block := range m.Content {
		if t, ok := block.(TextBlock); ok {
			b.WriteString(t.Text)
		}
	}
	return b.String()
}

func (m Message) ToolUses() []ToolUseBlock {
	var calls []ToolUseBlock
	for _, b := range m.Content {
		if c, ok := b.(ToolUseBlock); ok {
			calls = append(calls, c)
		}
	}
	return calls
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks}
}

type ToolSpec struct {
	Name        string
	Description string
	Parameters  map[string]any
}

type Delta struct {
	Kind string
	Text string
	Tool *ToolUseBlock
}

type Model interface {
	Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error)
}

type wireFunction struct {
	Name      string `json:"name"`
	Arguments string `json:"arguments"`
}

type wireToolCall struct {
	Index    int          `json:"index,omitempty"`
	ID       string       `json:"id,omitempty"`
	Type     string       `json:"type,omitempty"`
	Function wireFunction `json:"function"`
}

type wireMessage struct {
	Role       string         `json:"role"`
	Content    *string        `json:"content"`
	ToolCallID string         `json:"tool_call_id,omitempty"`
	ToolCalls  []wireToolCall `json:"tool_calls,omitempty"`
}

type wireToolDef struct {
	Type     string `json:"type"`
	Function struct {
		Name        string         `json:"name"`
		Description string         `json:"description"`
		Parameters  map[string]any `json:"parameters"`
	} `json:"function"`
}

type wireRequest struct {
	Model    string        `json:"model"`
	Messages []wireMessage `json:"messages"`
	Tools    []wireToolDef `json:"tools,omitempty"`
	Stream   bool          `json:"stream,omitempty"`
}

type wireChoice struct {
	Delta wireMessage `json:"delta"`
}

type wireResponse struct {
	Choices []wireChoice `json:"choices"`
}

func toWireMessages(messages []Message) []wireMessage {
	var out []wireMessage
	for _, m := range messages {
		if m.Role == "tool" {
			for _, b := range m.Content {
				if r, ok := b.(ToolResultBlock); ok {
					content := r.Content
					out = append(out, wireMessage{Role: "tool", Content: &content, ToolCallID: r.ToolUseID})
				}
			}
			continue
		}
		text, calls := m.Text(), m.ToolUses()
		wm := wireMessage{Role: m.Role}
		if text != "" {
			wm.Content = &text
		}
		for _, c := range calls {
			args, err := json.Marshal(c.Args)
			if err != nil {
				args = []byte("{}")
			}
			wm.ToolCalls = append(wm.ToolCalls, wireToolCall{ID: c.ID, Type: "function",
				Function: wireFunction{Name: c.Name, Arguments: string(args)}})
		}
		out = append(out, wm)
	}
	return out
}

func toWireTools(specs []ToolSpec) []wireToolDef {
	var out []wireToolDef
	for _, s := range specs {
		var d wireToolDef
		d.Type = "function"
		d.Function.Name = s.Name
		d.Function.Description = s.Description
		if s.Parameters == nil {
			d.Function.Parameters = map[string]any{"type": "object", "properties": map[string]any{}}
		} else {
			d.Function.Parameters = s.Parameters
		}
		out = append(out, d)
	}
	return out
}

func parseArguments(raw string) map[string]any {
	if raw == "" {
		return map[string]any{}
	}
	var args map[string]any
	if err := json.Unmarshal([]byte(raw), &args); err != nil {
		return map[string]any{"_invalid_json": raw}
	}
	return args
}

type toolCallSlot struct{ id, name, arguments string }

type toolCallAccumulator struct{ slots map[int]*toolCallSlot }

func newToolCallAccumulator() *toolCallAccumulator {
	return &toolCallAccumulator{slots: map[int]*toolCallSlot{}}
}

func (a *toolCallAccumulator) Feed(calls []wireToolCall) {
	for _, item := range calls {
		slot, ok := a.slots[item.Index]
		if !ok {
			slot = &toolCallSlot{}
			a.slots[item.Index] = slot
		}
		if item.ID != "" {
			slot.id = item.ID
		}
		if item.Function.Name != "" {
			slot.name = item.Function.Name
		}
		if item.Function.Arguments != "" {
			slot.arguments += item.Function.Arguments
		}
	}
}

func (a *toolCallAccumulator) Finish() []ToolUseBlock {
	indexes := make([]int, 0, len(a.slots))
	for i := range a.slots {
		indexes = append(indexes, i)
	}
	sort.Ints(indexes)
	var blocks []ToolUseBlock
	for _, i := range indexes {
		slot := a.slots[i]
		id := slot.id
		if id == "" {
			id = newID("call")
		}
		blocks = append(blocks, ToolUseBlock{ID: id, Name: slot.name, Args: parseArguments(slot.arguments)})
	}
	return blocks
}

func parseSSELine(line string) (wireResponse, bool) {
	line = strings.TrimSpace(line)
	if line == "" || strings.HasPrefix(line, ":") || !strings.HasPrefix(line, "data:") {
		return wireResponse{}, false
	}
	payload := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
	if payload == "[DONE]" {
		return wireResponse{}, false
	}
	var parsed wireResponse
	if err := json.Unmarshal([]byte(payload), &parsed); err != nil {
		return wireResponse{}, false
	}
	return parsed, true
}

type OpenAICompatModel struct {
	BaseURL string
	ModelID string
	Timeout time.Duration
}

func NewOpenAICompatModel(baseURL, modelID string) *OpenAICompatModel {
	return &OpenAICompatModel{BaseURL: baseURL, ModelID: modelID, Timeout: 180 * time.Second}
}

func (m *OpenAICompatModel) Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta, 32)
	errs := make(chan error, 1)
	go func() {
		defer close(deltas)
		defer close(errs)
		body := wireRequest{Model: m.ModelID, Messages: toWireMessages(messages),
			Tools: toWireTools(tools), Stream: true}
		raw, err := json.Marshal(body)
		if err != nil {
			errs <- fmt.Errorf("请求体序列化失败：%w", err)
			return
		}
		req, err := http.NewRequestWithContext(ctx, "POST", m.BaseURL+"/chat/completions", bytes.NewReader(raw))
		if err != nil {
			errs <- fmt.Errorf("构造请求失败：%w", err)
			return
		}
		req.Header.Set("Content-Type", "application/json")
		resp, err := (&http.Client{Timeout: m.Timeout}).Do(req)
		if err != nil {
			errs <- fmt.Errorf("请求失败：%w", err)
			return
		}
		defer resp.Body.Close()
		if resp.StatusCode != http.StatusOK {
			buf := make([]byte, 300)
			n, _ := resp.Body.Read(buf)
			errs <- fmt.Errorf("模型服务返回 %d：%s", resp.StatusCode, string(buf[:n]))
			return
		}
		acc := newToolCallAccumulator()
		scanner := bufio.NewScanner(resp.Body)
		scanner.Buffer(make([]byte, 0, 64*1024), 4*1024*1024)
		for scanner.Scan() {
			chunk, ok := parseSSELine(scanner.Text())
			if !ok || len(chunk.Choices) == 0 {
				continue
			}
			d := chunk.Choices[0].Delta
			if d.Content != nil && *d.Content != "" {
				deltas <- Delta{Kind: "text", Text: *d.Content}
			}
			if len(d.ToolCalls) > 0 {
				acc.Feed(d.ToolCalls)
			}
		}
		if err := scanner.Err(); err != nil {
			errs <- fmt.Errorf("读流中断：%w", err)
			return
		}
		for _, block := range acc.Finish() {
			b := block
			deltas <- Delta{Kind: "tool_use", Tool: &b}
		}
		errs <- nil
	}()
	return deltas, errs
}

type ToolResult struct {
	Content string
	IsError bool
}

type Tool interface {
	Spec() ToolSpec
	Run(ctx context.Context, args map[string]any) (string, error)
}

type FuncTool struct {
	spec ToolSpec
	fn   func(ctx context.Context, args map[string]any) (string, error)
}

func NewFuncTool(name, description string, parameters map[string]any,
	fn func(ctx context.Context, args map[string]any) (string, error)) *FuncTool {
	return &FuncTool{spec: ToolSpec{Name: name, Description: description, Parameters: parameters}, fn: fn}
}

func (t *FuncTool) Spec() ToolSpec { return t.spec }

func (t *FuncTool) Run(ctx context.Context, args map[string]any) (string, error) {
	return t.fn(ctx, args)
}

type ToolRegistry struct {
	tools map[string]Tool
	order []string
}

func NewToolRegistry(tools ...Tool) *ToolRegistry {
	reg := &ToolRegistry{tools: map[string]Tool{}}
	for _, t := range tools {
		reg.tools[t.Spec().Name] = t
		reg.order = append(reg.order, t.Spec().Name)
	}
	return reg
}

func (reg *ToolRegistry) Specs() []ToolSpec {
	var out []ToolSpec
	for _, name := range reg.order {
		out = append(out, reg.tools[name].Spec())
	}
	return out
}

func (reg *ToolRegistry) Execute(ctx context.Context, name string, args map[string]any) (result ToolResult) {
	tool, ok := reg.tools[name]
	if !ok {
		return ToolResult{Content: "没有叫 " + name + " 的工具", IsError: true}
	}
	defer func() {
		if r := recover(); r != nil {
			result = ToolResult{Content: fmt.Sprintf("工具崩溃：%v", r), IsError: true}
		}
	}()
	output, err := tool.Run(ctx, args)
	if err != nil {
		return ToolResult{Content: err.Error(), IsError: true}
	}
	return ToolResult{Content: output}
}

type Agent struct {
	Model        Model
	Registry     *ToolRegistry
	SystemPrompt string
	MaxRounds    int
	OnEvent      func(name string, args map[string]any, result ToolResult)
}

func (a *Agent) Run(ctx context.Context, userText string) (string, error) {
	history := []Message{
		NewMessage("system", TextBlock{Text: a.SystemPrompt}),
		NewMessage("user", TextBlock{Text: userText}),
	}
	for round := 0; round < a.MaxRounds; round++ {
		reply, err := a.oneRound(ctx, history)
		if err != nil {
			return "", err
		}
		history = append(history, reply)
		calls := reply.ToolUses()
		if len(calls) == 0 {
			return reply.Text(), nil
		}
		var results []Block
		for _, c := range calls {
			r := a.Registry.Execute(ctx, c.Name, c.Args)
			if a.OnEvent != nil {
				a.OnEvent(c.Name, c.Args, r)
			}
			results = append(results, ToolResultBlock{
				ToolUseID: c.ID, Content: r.Content, IsError: r.IsError})
		}
		history = append(history, NewMessage("tool", results...))
	}
	return "（轮数用完了）", nil
}

func (a *Agent) oneRound(ctx context.Context, messages []Message) (Message, error) {
	deltas, errs := a.Model.Stream(ctx, messages, a.Registry.Specs())
	var textParts []string
	var blocks []Block
	flushText := func() {
		if len(textParts) > 0 {
			joined := ""
			for _, p := range textParts {
				joined += p
			}
			blocks = append(blocks, TextBlock{Text: joined})
			textParts = nil
		}
	}
	for d := range deltas {
		switch d.Kind {
		case "text":
			textParts = append(textParts, d.Text)
		case "tool_use":
			if d.Tool != nil {
				flushText()
				blocks = append(blocks, *d.Tool)
			}
		}
	}
	flushText()
	if err := <-errs; err != nil {
		return Message{}, err
	}
	return NewMessage("assistant", blocks...), nil
}
```

## 第一道防线：不给 shell

这一点上 Go 比 Python 省心：`exec.Command("cat", path)` **本来就不经过 shell**。
要用 shell 得自己显式写出来 `exec.Command("sh", "-c", …)`——
危险的写法在 Go 里是一件要动手做的事，而不是默认行为。

```go
// file: main.go
package main

import (
	"fmt"
	"os/exec"
)

func main() {
	if err := SetupSandbox(); err != nil {
		fmt.Println("准备沙箱失败：", err)
		return
	}
	untrusted := "notes.txt; echo 我是被注入的命令"

	fmt.Println("── 交给 shell 执行（错误做法）──")
	out, _ := exec.Command("sh", "-c", "cat "+untrusted).CombinedOutput()
	fmt.Print(string(out))
	fmt.Println("→ 分号后面那条也执行了。模型只是想读一个文件，参数里的分号却变成了第二条命令。")

	fmt.Println("\n── 用参数数组执行（正确做法）──")
	cmd := exec.Command("cat", untrusted)
	cmd.Dir = sandboxRoot
	out, err := cmd.CombinedOutput()
	fmt.Printf("输出: %q\n", string(out))
	fmt.Println("错误:", err)
	fmt.Println("→ 整个字符串被当成一个文件名。没有 shell，就没有 shell 注入。")
}
```

（第一条命令没设 `cmd.Dir`，所以它在别处找 `notes.txt`，找不到——
但**分号后面那条照样执行了**，这才是重点。）

这不是「模型会写恶意命令」的问题。更常见的情况是：模型把一段**从网页或文件里读来的文本**
当参数传进来，而那段文本里正好有分号、反引号或者 `$(...)`。
不给 shell，这类问题**从根上不存在**。

代价是失去了管道和重定向。这个代价值得付：真要用管道，让 agent 分两步做，
或者提供一个专门的工具。

## 第二道防线：审批门

命令分三类。分类的依据不是「危不危险」，而是**「出了事能不能撤回」**。

```go
// file: policy.go
package main

import (
	"path/filepath"
	"strings"
)

type Decision string

const (
	DecisionAllow Decision = "allow" // 自动放行
	DecisionAsk   Decision = "ask"   // 要人点头
	DecisionDeny  Decision = "deny"  // 直接拒绝，不问
)

type Verdict struct {
	Decision Decision
	Reason   string
}

type commandKey struct {
	Program string
	Sub     string // 空串表示「不看子命令」
}

// autoAllow 只读、无副作用、跑错了也不心疼的命令。
//
// 两级匹配——只看程序名不够：git status 是只读的，git push 不是。
var autoAllow = map[commandKey]bool{
	{"ls", ""}: true, {"cat", ""}: true, {"head", ""}: true, {"tail", ""}: true,
	{"wc", ""}: true, {"grep", ""}: true, {"find", ""}: true, {"file", ""}: true,
	{"git", "status"}: true, {"git", "diff"}: true, {"git", "log"}: true, {"git", "show"}: true,
	{"go", "build"}: true, {"go", "vet"}: true, {"go", "test"}: true, {"go", "fmt"}: true,
	{"go", "version"}: true, {"go", "run"}: true,
}

// never 这些直接拒绝。
//
// 说清楚一件事：**这不是安全边界**。绕过一张名单的办法太多了
// （改名、软链接、用别的程序做同一件事）。它的作用是「明显不该由 agent 自己决定的事，
// 连问都不问」——省掉一次无意义的审批打扰。真正的边界是第三道防线。
var never = map[string]bool{
	"rm": true, "rmdir": true, "dd": true, "mkfs": true,
	"shutdown": true, "reboot": true, "halt": true, "poweroff": true,
	"sudo": true, "su": true, "chmod": true, "chown": true,
	"kill": true, "pkill": true, "killall": true,
	"curl": true, "wget": true, "ssh": true, "scp": true, "nc": true, "systemctl": true,
}

func Classify(argv []string) Verdict {
	if len(argv) == 0 {
		return Verdict{DecisionDeny, "命令是空的"}
	}
	program := filepath.Base(argv[0])
	sub := ""
	if len(argv) > 1 {
		sub = argv[1]
	}

	if never[program] {
		return Verdict{DecisionDeny, program + " 不在这个 agent 的能力范围内，请人工执行"}
	}
	if autoAllow[commandKey{program, ""}] {
		return Verdict{DecisionAllow, program + " 是只读命令"}
	}
	if autoAllow[commandKey{program, sub}] {
		return Verdict{DecisionAllow, program + " " + sub + " 是只读命令"}
	}
	return Verdict{DecisionAsk, program + " 可能有副作用，需要你确认"}
}

// ApprovalStore 记住用户在这次会话里同意过什么。
//
// 两种粒度：
//   - 这一条命令（含全部参数）——最保守。
//   - 这个程序的这个子命令（不含其余参数）——好用，范围仍然可控。
//
// 没有「永久同意一切」这一档：那等于把审批门关掉，
// 留着一个关不掉的门比留一个默认敞开的门诚实。
type ApprovalStore struct {
	exact  map[string]bool
	prefix map[commandKey]bool
}

func NewApprovalStore() *ApprovalStore {
	return &ApprovalStore{exact: map[string]bool{}, prefix: map[commandKey]bool{}}
}

func exactKey(argv []string) string { return strings.Join(argv, "\x00") }

func prefixKey(argv []string) commandKey {
	sub := ""
	if len(argv) > 1 {
		sub = argv[1]
	}
	return commandKey{filepath.Base(argv[0]), sub}
}

func (s *ApprovalStore) Remembered(argv []string) bool {
	return s.exact[exactKey(argv)] || s.prefix[prefixKey(argv)]
}

func (s *ApprovalStore) Remember(argv []string, scope string) {
	switch scope {
	case "exact":
		s.exact[exactKey(argv)] = true
	case "prefix":
		s.prefix[prefixKey(argv)] = true
	}
}
```

```go
// file: main.go
package main

import (
	"fmt"
	"strings"
)

func main() {
	for _, argv := range [][]string{
		{"ls", "-la"}, {"git", "status"}, {"git", "push"},
		{"go", "test", "./..."}, {"rm", "-rf", "/"}, {"make", "install"},
	} {
		v := Classify(argv)
		label := map[Decision]string{
			DecisionAllow: "自动放行", DecisionAsk: "要人确认", DecisionDeny: "直接拒绝"}[v.Decision]
		fmt.Printf("  %-24s %-8s （%s）\n", strings.Join(argv, " "), label, v.Reason)
	}
}
```

**审批疲劳会让审批形同虚设**：每跑一次 `make build` 都问一遍，
用户会开始无脑点同意。所以 `ApprovalStore` 要能记住选择。

## 第三道防线：隔离

前两道防线管的是**要不要跑**。第三道管的是**跑起来之后能碰到什么**。

```go
// file: runner.go
package main

import (
	"bytes"
	"context"
	"errors"
	"fmt"
	"os"
	"os/exec"
	"strings"
	"syscall"
	"time"
)

// envAllowlist 传给子进程的环境变量白名单。
//
// 这一条很容易被忽略，但它是最实在的一条：Go 里 cmd.Env 为 nil 时，
// 子进程**继承当前进程的全部环境变量**，里面通常有 API 密钥、令牌、数据库密码。
// 一条 `env` 命令就能把它们全打印出来，而输出会原样进对话历史、再发给模型服务。
var envAllowlist = []string{"PATH", "HOME", "LANG", "LC_ALL", "TZ", "TERM"}

const (
	maxOutputChars   = 4000
	defaultTimeoutS  = 30
	killGraceSeconds = 2
)

func buildEnv() []string {
	var env []string
	hasPath := false
	for _, key := range envAllowlist {
		if value, ok := os.LookupEnv(key); ok {
			env = append(env, key+"="+value)
			if key == "PATH" {
				hasPath = true
			}
		}
	}
	if !hasPath {
		env = append(env, "PATH=/usr/local/bin:/usr/bin:/bin")
	}
	// 明确告诉子进程「这里没有交互」，免得它弹出编辑器或分页器把执行卡死
	env = append(env, "GIT_PAGER=cat", "PAGER=cat", "NO_COLOR=1", "GOFLAGS=-mod=mod")
	return env
}

// clip 输出太长时保留头尾，中间省略。
//
// 保留头尾而不是只留开头：报错信息经常在末尾（栈回溯、测试汇总），
// 只截前面等于把最有用的部分扔了。
func clip(text string) string {
	r := []rune(text)
	if len(r) <= maxOutputChars {
		return text
	}
	headLen := maxOutputChars * 2 / 3
	tailLen := maxOutputChars / 3
	return fmt.Sprintf("%s\n…（中间省略 %d 字符）…\n%s",
		string(r[:headLen]), len(r)-headLen-tailLen, string(r[len(r)-tailLen:]))
}

type CommandOutcome struct {
	Argv     []string
	ExitCode int
	Stdout   string
	Stderr   string
	Duration time.Duration
	TimedOut bool
}

func (o CommandOutcome) Render() string {
	var parts []string
	parts = append(parts, "$ "+strings.Join(o.Argv, " "))
	if o.TimedOut {
		parts = append(parts, fmt.Sprintf("（超时，已终止，跑了 %.1f 秒）", o.Duration.Seconds()))
	} else {
		parts = append(parts, fmt.Sprintf("退出码 %d，用时 %.1f 秒", o.ExitCode, o.Duration.Seconds()))
	}
	if strings.TrimSpace(o.Stdout) != "" {
		parts = append(parts, "stdout:\n"+clip(o.Stdout))
	}
	if strings.TrimSpace(o.Stderr) != "" {
		parts = append(parts, "stderr:\n"+clip(o.Stderr))
	}
	if strings.TrimSpace(o.Stdout) == "" && strings.TrimSpace(o.Stderr) == "" {
		parts = append(parts, "（没有输出）")
	}
	return strings.Join(parts, "\n")
}

// RunIsolated 在受限环境里执行一条命令。
//
// 这里做到的隔离是**进程级**的：工作目录、环境变量、超时、输出上限。
// 做不到的是文件系统隔离和网络隔离——子进程照样能读整个磁盘、照样能联网。
// 真要那两样，得靠操作系统级的手段（容器、命名空间、seccomp 之类）。
// **知道自己的隔离到哪儿为止，比以为自己隔离了更重要。**
func RunIsolated(ctx context.Context, argv []string, dir string, timeout time.Duration) CommandOutcome {
	started := time.Now()
	runCtx, cancel := context.WithTimeout(ctx, timeout)
	defer cancel()

	cmd := exec.CommandContext(runCtx, argv[0], argv[1:]...)
	cmd.Dir = dir
	cmd.Env = buildEnv()
	// Stdin 留 nil：Go 会把它接到 /dev/null，等待输入的程序会立刻拿到文件结束。
	// 不这么做的话，一个等着读输入的程序会一直挂到超时。

	// 把子进程放进自己的进程组，取消时连它派生的孙进程一起杀。
	//
	// 只靠 CommandContext 是不够的：它只杀直接子进程。
	// `go test` 会派生编译器和测试二进制，杀掉 go 本身，那些还在跑。
	cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}
	cmd.Cancel = func() error {
		// 负号表示「整个进程组」
		return syscall.Kill(-cmd.Process.Pid, syscall.SIGKILL)
	}
	cmd.WaitDelay = killGraceSeconds * time.Second

	var stdout, stderr bytes.Buffer
	cmd.Stdout = &stdout
	cmd.Stderr = &stderr

	err := cmd.Run()
	duration := time.Since(started)

	outcome := CommandOutcome{Argv: argv, Stdout: stdout.String(),
		Stderr: stderr.String(), Duration: duration}

	switch {
	case errors.Is(runCtx.Err(), context.DeadlineExceeded):
		outcome.TimedOut = true
		outcome.ExitCode = -1
	case err == nil:
		outcome.ExitCode = 0
	default:
		var exitErr *exec.ExitError
		if errors.As(err, &exitErr) {
			outcome.ExitCode = exitErr.ExitCode()
		} else {
			outcome.ExitCode = -1
			if outcome.Stderr == "" {
				outcome.Stderr = err.Error()
			}
		}
	}
	return outcome
}
```

```go
package main

import (
	"context"
	"fmt"
	"path/filepath"

	"time"

// Approver 问人一句。返回 "no" / "once" / "always"。
//
// 真实程序里这里是一个界面弹窗；接口一样，行为由人决定。
type Approver func(argv []string, reason string) string

type auditEntry struct {
	Argv     []string
	Decision Decision
	Outcome  string
}

type CommandRunner struct {
	Dir      string
	Approve  Approver
	Store    *ApprovalStore
	Audit    []auditEntry
	Timeout  time.Duration
}

func NewCommandRunner(dir string, approve Approver) *CommandRunner {
	return &CommandRunner{Dir: dir, Approve: approve, Store: NewApprovalStore(),
		Timeout: defaultTimeoutS * time.Second}
}

func (r *CommandRunner) Run(ctx context.Context, argv []string, timeout time.Duration) (string, error) {
	if len(argv) == 0 {
		return "", fmt.Errorf("command 必须是一个非空的字符串数组，比如 [\"go\", \"test\", \"./...\"]")
	}
	verdict := Classify(argv)
	entry := auditEntry{Argv: argv, Decision: verdict.Decision}

	if verdict.Decision == DecisionDeny {
		entry.Outcome = "拒绝"
		r.Audit = append(r.Audit, entry)
		return "", fmt.Errorf("%s。这条命令没有执行。", verdict.Reason)
	}

	if verdict.Decision == DecisionAsk && !r.Store.Remembered(argv) {
		answer := r.Approve(argv, verdict.Reason)
		if answer == "no" {
			entry.Outcome = "用户拒绝"
			r.Audit = append(r.Audit, entry)
			return "", fmt.Errorf("用户拒绝了这条命令。换个办法，或者问问他想怎么做。")
		}
		if answer == "always" {
			r.Store.Remember(argv, "prefix")
		}
	}

	if timeout <= 0 {
		timeout = r.Timeout
	}
	outcome := RunIsolated(ctx, argv, r.Dir, timeout)
	entry.Outcome = fmt.Sprintf("退出码 %d", outcome.ExitCode)
	r.Audit = append(r.Audit, entry)
	return outcome.Render(), nil
}

func (r *CommandRunner) PrintAudit() {
	fmt.Printf("%-32s %-8s %s\n", "命令", "判定", "结果")
	fmt.Println(strings.Repeat("─", 56))
	for _, e := range r.Audit {
		line := strings.Join(e.Argv, " ")
		if len(line) > 32 {
			line = line[:32]
		}
		fmt.Printf("%-32s %-8s %s\n", line, e.Decision, e.Outcome)
	}
}

// MakeApprover 演示用的自动审批人：把「用户会点什么」写成一张表。
func MakeApprover(script map[string]string) Approver {
	return func(argv []string, reason string) string {
		answer := script[filepath.Base(argv[0])]
		if answer == "" {
			answer = "no"
		}
		label := map[string]string{"no": "拒绝", "once": "同意这一次",
			"always": "这个命令以后都同意"}[answer]
		fmt.Printf("    [审批] %s —— %s → 用户选择：%s\n",
			strings.Join(argv, " "), reason, label)
		return answer
	}
}
```

```go
// file: main.go
package main

import (
	"context"
	"fmt"
	"os"
	"strings"
	"time"
)

func main() {
	SetupSandbox()
	ctx := context.Background()
	os.Setenv("FAKE_API_KEY", "不该泄露的值")

	type check struct {
		label string
		ok    bool
	}
	var checks []check

	// 1. 参数里的分号不会变成第二条命令
	out := RunIsolated(ctx, []string{"echo", "a; echo b"}, sandboxRoot, 5*time.Second)
	checks = append(checks, check{"参数里的分号不被解释",
		strings.TrimSpace(out.Stdout) == "a; echo b"})

	// 2. 环境变量不外泄（printenv 找不到变量时退出码非 0）
	out = RunIsolated(ctx, []string{"printenv", "FAKE_API_KEY"}, sandboxRoot, 5*time.Second)
	checks = append(checks, check{"敏感环境变量没传下去",
		out.ExitCode != 0 && strings.TrimSpace(out.Stdout) == ""})

	// 3. 超时能终止
	out = RunIsolated(ctx, []string{"sleep", "10"}, sandboxRoot, time.Second)
	checks = append(checks, check{"超时能终止", out.TimedOut})

	// 4. 输出有上限
	out = RunIsolated(ctx, []string{"seq", "1", "50000"}, sandboxRoot, 5*time.Second)
	checks = append(checks, check{"输出被截断", len([]rune(out.Render())) < 10000})

	// 5. 拒绝名单生效
	strict := NewCommandRunner(sandboxRoot, MakeApprover(map[string]string{}))
	_, err := strict.Run(ctx, []string{"rm", "-rf", "."}, 0)
	checks = append(checks, check{"拒绝名单生效", err != nil})

	// 6. 没被同意的命令不会执行
	marker := sandboxRoot + "/should-not-exist.txt"
	strict.Run(ctx, []string{"touch", marker}, 0)
	_, statErr := os.Stat(marker)
	checks = append(checks, check{"没被同意的命令没有执行", statErr != nil})

	// 7. 空命令被挡回去
	_, err = strict.Run(ctx, nil, 0)
	checks = append(checks, check{"空命令被挡回去", err != nil})

	passed := 0
	for _, c := range checks {
		mark := "✗"
		if c.ok {
			mark = "✓"
			passed++
		}
		fmt.Printf("  %s %s\n", mark, c.label)
	}
	fmt.Printf("\n%d/%d 项通过\n", passed, len(checks))
}
```

## 说清楚防线到哪儿为止

这三道防线**不是**一个安全沙箱。诚实地列一下它们挡不住什么：

- **文件系统**：子进程照样能读你整个家目录。工作目录只是它的起始位置，不是牢笼。
- **网络**：进程级限制管不了网络。子进程能联网，就能把读到的东西发出去。
- **资源**：内存和 CPU 没有上限，一个死循环能把机器拖垮。

要挡住这三样，得上**操作系统级隔离**：容器、命名空间、seccomp，或者干脆一台隔离的机器。
那是另一个话题，但有一条现在就该记住：

> **在你没有操作系统级隔离的环境里，不要让 agent 自动执行未经审批的命令。**
> 审批门此时不是「防打扰」，它是唯一真正起作用的那道防线。

## 本章小结

- **提示词不是安全边界**。系统提示里写「不要删文件」挡不住任何事，
  真正的边界必须是代码里的机制。
- Go 在第一道防线上占便宜：`exec.Command` **本来就不经过 shell**，
  要用 shell 得自己写出 `sh -c`。危险写法是要动手做的事，而不是默认行为。
- 命令分三类的依据是**「出了事能不能撤回」**。分类要看（程序，子命令）两级。
- **拒绝名单不是安全边界**，它只是省掉无意义的审批打扰。
- **审批疲劳会让审批形同虚设**。要能记住选择，但**不要提供「永久同意一切」**。
- `cmd.Env` 为 nil 时子进程**继承全部环境变量**。必须显式给一份白名单，
  否则 API 密钥会随一条 `env` 的输出进对话历史。
- `exec.CommandContext` 只杀**直接子进程**。`go test` 派生的编译器和测试二进制不会死。
  要连孙进程一起杀，得 `Setpgid` 建进程组，再用 `cmd.Cancel` 杀掉整组（PID 取负号）。
- `cmd.Stdin` 留 nil，Go 会把它接到 `/dev/null`；等待输入的程序会立刻拿到文件结束，
  而不是挂到超时。
- 输出截断要**保留头尾**：报错信息经常在末尾。
- **知道自己的隔离到哪儿为止，比以为自己隔离了更重要**。
