# 有状态的 agent

到目前为止，我们的入口一直是一个函数：`Run(ctx, history)`。历史从外面传进来，跑完就完事。
这足够跑通一轮，但**没法做多轮对话**——第二次调用怎么知道第一次说了什么？

这一章把 agent 从「一个函数」变成「一个类型」，并且回答一个设计问题：
**状态到底放在谁身上。**

## 三样东西，三个归属

| 东西 | 例子 | 谁持有 | 为什么 |
|---|---|---|---|
| 配置 | 用哪个模型、几轮预算、系统提示 | **Agent** | 一个 agent 的性格是固定的 |
| 对话 | 消息列表、这段对话的标识 | **Session** | 同一个 agent 可以同时开几段对话 |
| 一次执行 | 这一轮用了几轮、为什么停 | **TurnResult** | 用完就丢 |

把对话放在 Agent 身上是最常见的设计错误。那样一个 agent 只能有一段对话，
换个话题就得重建整个类型，工具和配置全都白搭一遍。

```go
func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)

type Block interface{ BlockType() string }

type TextBlock struct{ Text string }

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string
	Name string
	Args map[string]any
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string
	Content   string
	IsError   bool
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

type Message struct {
	ID        string
	Role      string
	Content   []Block
	CreatedAt time.Time
}

func (m Message) Text() string {
	out := ""
	for _, b := range m.Content {
		if t, ok := b.(TextBlock); ok {
			out += t.Text
		}
	}
	return out
}

func (m Message) ToolUses() []ToolUseBlock {
	var calls []ToolUseBlock
	for _, b := range m.Content {
		if c, ok := b.(ToolUseBlock); ok {
			calls = append(calls, c)
		}
	}
	return calls
}

func (m Message) ToolResults() []ToolResultBlock {
	var out []ToolResultBlock
	for _, b := range m.Content {
		if r, ok := b.(ToolResultBlock); ok {
			out = append(out, r)
		}
	}
	return out
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks, CreatedAt: time.Now()}

func toWireMessages(messages []Message) []wireMessage {
	var out []wireMessage
	for _, m := range messages {
		if m.Role == "tool" {
			for _, b := range m.Content {
				if r, ok := b.(ToolResultBlock); ok {
					content := r.Content
					out = append(out, wireMessage{Role: "tool", Content: &content, ToolCallID: r.ToolUseID})
				}
			}
			continue
		}
		text, calls := m.Text(), m.ToolUses()
		wm := wireMessage{Role: m.Role}
		if text != "" {
			wm.Content = &text
		}
		for _, c := range calls {
			args, err := json.Marshal(c.Args)
			if err != nil {
				args = []byte("{}")
			}
			wm.ToolCalls = append(wm.ToolCalls, wireToolCall{ID: c.ID, Type: "function",
				Function: wireFunction{Name: c.Name, Arguments: string(args)}})
		}
		out = append(out, wm)
	}
	return out
}

func toWireTools(specs []ToolSpec) []wireToolDef {
	var out []wireToolDef
	for _, s := range specs {
		var d wireToolDef
		d.Type = "function"
		d.Function.Name = s.Name
		d.Function.Description = s.Description
		if s.Parameters == nil {
			d.Function.Parameters = map[string]any{"type": "object", "properties": map[string]any{}}
		} else {
			d.Function.Parameters = s.Parameters
		}
		out = append(out, d)
	}
	return out
}

func parseArguments(raw string) map[string]any {
	if raw == "" {
		return map[string]any{}
	}
	var args map[string]any
	if err := json.Unmarshal([]byte(raw), &args); err != nil {
		return map[string]any{"_invalid_json": raw}
	}
	return args
}

type toolCallSlot struct{ id, name, arguments string }

type toolCallAccumulator struct{ slots map[int]*toolCallSlot }

func newToolCallAccumulator() *toolCallAccumulator {
	return &toolCallAccumulator{slots: map[int]*toolCallSlot{}}
}

func (a *toolCallAccumulator) Feed(calls []wireToolCall) {
	for _, item := range calls {
		slot, ok := a.slots[item.Index]
		if !ok {
			slot = &toolCallSlot{}
			a.slots[item.Index] = slot
		}
		if item.ID != "" {
			slot.id = item.ID
		}
		if item.Function.Name != "" {
			slot.name = item.Function.Name
		}
		if item.Function.Arguments != "" {
			slot.arguments += item.Function.Arguments
		}
	}
}

func (a *toolCallAccumulator) Finish() []ToolUseBlock {
	indexes := make([]int, 0, len(a.slots))
	for i := range a.slots {
		indexes = append(indexes, i)
	}
	sort.Ints(indexes)
	var blocks []ToolUseBlock
	for _, i := range indexes {
		slot := a.slots[i]
		id := slot.id
		if id == "" {
			id = newID("call")
		}
		blocks = append(blocks, ToolUseBlock{ID: id, Name: slot.name, Args: parseArguments(slot.arguments)})
	}
	return blocks
}

func parseSSELine(line string) (wireResponse, bool) {
	line = strings.TrimSpace(line)
	if line == "" || strings.HasPrefix(line, ":") || !strings.HasPrefix(line, "data:") {
		return wireResponse{}, false
	}
	payload := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
	if payload == "[DONE]" {
		return wireResponse{}, false
	}
	var parsed wireResponse
	if err := json.Unmarshal([]byte(payload), &parsed); err != nil {
		return wireResponse{}, false
	}
	return parsed, true
}

type OpenAICompatModel struct {
	BaseURL string
	ModelID string
	APIKey  string
	Timeout time.Duration
}

func NewOpenAICompatModel(baseURL, modelID string) *OpenAICompatModel {
	return &OpenAICompatModel{BaseURL: baseURL, ModelID: modelID, Timeout: 180 * time.Second}
}

func (m *OpenAICompatModel) Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta, 32)
	errs := make(chan error, 1)
	go func() {
		defer close(deltas)
		defer close(errs)
		send := func(d Delta) bool {
			select {
			case deltas <- d:
				return true
			case <-ctx.Done():
				return false
			}
		}
		body := wireRequest{Model: m.ModelID, Messages: toWireMessages(messages),
			Tools: toWireTools(tools), Stream: true}
		raw, err := json.Marshal(body)
		if err != nil {
			errs <- fmt.Errorf("请求体序列化失败：%w", err)
			return
		}
		req, err := http.NewRequestWithContext(ctx, "POST", m.BaseURL+"/chat/completions", bytes.NewReader(raw))
		if err != nil {
			errs <- fmt.Errorf("构造请求失败：%w", err)
			return
		}
		req.Header.Set("Content-Type", "application/json")
		if m.APIKey != "" {
			req.Header.Set("Authorization", "Bearer "+m.APIKey)
		}
		resp, err := (&http.Client{Timeout: m.Timeout}).Do(req)
		if err != nil {
			errs <- fmt.Errorf("请求失败：%w", err)
			return
		}
		defer resp.Body.Close()
		if resp.StatusCode != http.StatusOK {
			buf := make([]byte, 300)
			n, _ := resp.Body.Read(buf)
			errs <- &StatusError{Code: resp.StatusCode, Body: string(buf[:n])}
			return
		}
		acc := newToolCallAccumulator()
		scanner := bufio.NewScanner(resp.Body)
		scanner.Buffer(make([]byte, 0, 64*1024), 4*1024*1024)
		for scanner.Scan() {
			chunk, ok := parseSSELine(scanner.Text())
			if !ok || len(chunk.Choices) == 0 {
				continue
			}
			d := chunk.Choices[0].Delta
			if d.Content != nil && *d.Content != "" {
				if !send(Delta{Kind: "text", Text: *d.Content}) {
					errs <- ctx.Err()
					return
				}
			}
			if len(d.ToolCalls) > 0 {
				acc.Feed(d.ToolCalls)
			}
		}
		if err := scanner.Err(); err != nil {
			if ctx.Err() != nil {
				errs <- ctx.Err()
			} else {
				errs <- fmt.Errorf("读流中断：%w", err)
			}
			return
		}
		for _, block := range acc.Finish() {
			b := block
			if !send(Delta{Kind: "tool_use", Tool: &b}) {
				errs <- ctx.Err()
				return
			}
		}
		errs <- nil
	}()
	return deltas, errs
}
```

```go
// file: tools.go
package main

import (
	"context"
	"fmt"
	"math"
	"sort"
	"strconv"
	"strings"
	"sync"
	"time"
)

type ToolResult struct {
	Content       string
	IsError       bool
	IsInvalidArgs bool
}

type Tool interface {
	Spec() ToolSpec
	Run(ctx context.Context, args map[string]any) (string, error)
}

type FuncTool struct {
	spec ToolSpec
	fn   func(ctx context.Context, args map[string]any) (string, error)
}

func NewFuncTool(name, description string, parameters map[string]any,
	fn func(ctx context.Context, args map[string]any) (string, error)) *FuncTool {
	return &FuncTool{spec: ToolSpec{Name: name, Description: description, Parameters: parameters}, fn: fn}
}

func (t *FuncTool) Spec() ToolSpec { return t.spec }

func (t *FuncTool) Run(ctx context.Context, args map[string]any) (string, error) {
	return t.fn(ctx, args)
}

func argString(args map[string]any, key string) string {
	s, _ := args[key].(string)
	return s
}

func jsonTypeOf(value any) string {
	switch v := value.(type) {
	case string:
		return "string"
	case bool:
		return "boolean"
	case float64:
		if v == math.Trunc(v) {
			return "integer"
		}
		return "number"
	case []any:
		return "array"
	case map[string]any:
		return "object"
	case nil:
		return "null"
	}
	return "unknown"
}

func requiredNames(schema map[string]any) []string {
	switch v := schema["required"].(type) {
	case []string:
		return v
	case []any:
		var out []string
		for _, item := range v {
			if s, ok := item.(string); ok {
				out = append(out, s)
			}
		}
		return out
	}
	return nil
}

func sortedKeys(m map[string]any) []string {
	keys := make([]string, 0, len(m))
	for k := range m {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	return keys
}

func ValidateArgs(args map[string]any, schema map[string]any) string {
	if raw, bad := args["_invalid_json"]; bad {
		return fmt.Sprintf("参数不是合法的 JSON：%v", raw)
	}
	props, _ := schema["properties"].(map[string]any)
	if props == nil {
		props = map[string]any{}
	}
	for _, name := range requiredNames(schema) {
		if _, ok := args[name]; !ok {
			return "缺少必填参数 " + name
		}
	}
	for _, key := range sortedKeys(args) {
		spec, known := props[key].(map[string]any)
		if !known {
			return fmt.Sprintf("不认识的参数 %s；这个工具接受 %s", key, strings.Join(sortedKeys(props), "、"))
		}
		if want, _ := spec["type"].(string); want != "" {
			got := jsonTypeOf(args[key])
			if !(got == want || (want == "number" && got == "integer")) {
				return fmt.Sprintf("参数 %s 应该是 %s，收到的是 %s", key, want, got)
			}
		}
	}
	return ""
}

type ToolRegistry struct {
	tools map[string]Tool
	order []string
}

func NewToolRegistry(tools ...Tool) *ToolRegistry {
	reg := &ToolRegistry{tools: map[string]Tool{}}
	for _, t := range tools {
		reg.Add(t)
	}
	return reg
}

func (reg *ToolRegistry) Add(tool Tool) {
	name := tool.Spec().Name
	if _, exists := reg.tools[name]; !exists {
		reg.order = append(reg.order, name)
	}
	reg.tools[name] = tool
}

func (reg *ToolRegistry) Specs() []ToolSpec {
	var out []ToolSpec
	for _, name := range reg.order {
		out = append(out, reg.tools[name].Spec())
	}
	return out
}

func (reg *ToolRegistry) Execute(ctx context.Context, name string, args map[string]any) (result ToolResult) {
	tool, ok := reg.tools[name]
	if !ok {
		return ToolResult{Content: fmt.Sprintf("没有叫 %s 的工具。可用的：%s",
			name, strings.Join(reg.order, "、")), IsError: true}
	}
	if problem := ValidateArgs(args, tool.Spec().Parameters); problem != "" {
		return ToolResult{Content: "参数不合格：" + problem, IsError: true, IsInvalidArgs: true}
	}
	defer func() {
		if r := recover(); r != nil {
			result = ToolResult{Content: fmt.Sprintf("工具 %s 执行时崩溃了：%v", name, r), IsError: true}
		}
	}()
	output, err := tool.Run(ctx, args)
	if err != nil {
		return ToolResult{Content: err.Error(), IsError: true}
	}
	return ToolResult{Content: output}
}

// 第 9 章那个算术求值器，原样带过来。
type calcParser struct {
	src []rune
	pos int
}

func (p *calcParser) peek() rune {
	for p.pos < len(p.src) && (p.src[p.pos] == ' ' || p.src[p.pos] == '\t') {
		p.pos++
	}
	if p.pos >= len(p.src) {
		return 0
	}
	return p.src[p.pos]
}

func (p *calcParser) parseExpr() (float64, error) {
	value, err := p.parseTerm()
	if err != nil {
		return 0, err
	}
	for {
		op := p.peek()
		if op != '+' && op != '-' {
			return value, nil
		}
		p.pos++
		rhs, err := p.parseTerm()
		if err != nil {
			return 0, err
		}
		if op == '+' {
			value += rhs
		} else {
			value -= rhs
		}
	}
}

func (p *calcParser) parseTerm() (float64, error) {
	value, err := p.parseAtom()
	if err != nil {
		return 0, err
	}
	for {
		op := p.peek()
		if op != '*' && op != '/' {
			return value, nil
		}
		p.pos++
		rhs, err := p.parseAtom()
		if err != nil {
			return 0, err
		}
		if op == '*' {
			value *= rhs
		} else {
			if rhs == 0 {
				return 0, fmt.Errorf("除数不能是 0")
			}
			value /= rhs
		}
	}
}

func (p *calcParser) parseAtom() (float64, error) {
	c := p.peek()
	switch {
	case c == '(':
		p.pos++
		value, err := p.parseExpr()
		if err != nil {
			return 0, err
		}
		if p.peek() != ')' {
			return 0, fmt.Errorf("括号没配对")
		}
		p.pos++
		return value, nil
	case c == '-':
		p.pos++
		value, err := p.parseAtom()
		return -value, err
	case c >= '0' && c <= '9':
		start := p.pos
		for p.pos < len(p.src) && ((p.src[p.pos] >= '0' && p.src[p.pos] <= '9') || p.src[p.pos] == '.') {
			p.pos++
		}
		return strconv.ParseFloat(string(p.src[start:p.pos]), 64)
	case c == 0:
		return 0, fmt.Errorf("表达式在这里就断了")
	default:
		return 0, fmt.Errorf("看不懂的字符 %q", string(c))
	}
}

func NewCalcTool() Tool {
	return NewFuncTool("calculate", "计算一个算术表达式，比如 (3+5)*2",
		map[string]any{"type": "object", "properties": map[string]any{
			"expression": map[string]any{"type": "string", "description": "只含数字和 + - * / ( )"}},
			"required": []string{"expression"}},
		func(ctx context.Context, args map[string]any) (string, error) {
			expr := argString(args, "expression")
			if len(expr) > 200 {
				return "", fmt.Errorf("表达式太长了")
			}
			p := &calcParser{src: []rune(expr)}
			value, err := p.parseExpr()
			if err != nil {
				return "", err
			}
			if p.peek() != 0 {
				return "", fmt.Errorf("表达式末尾有多余内容")
			}
			return strconv.FormatFloat(value, 'f', -1, 64), nil
		})
}

type Event struct {
	Kind string
	Data map[string]any
}

type EventStream struct {
	mu   sync.RWMutex
	subs map[int]func(Event)
	next int
}

func NewEventStream() *EventStream { return &EventStream{subs: map[int]func(Event){}} }

func (s *EventStream) Subscribe(fn func(Event)) func() {
	s.mu.Lock()
	defer s.mu.Unlock()
	id := s.next
	s.next++
	s.subs[id] = fn
	return func() {
		s.mu.Lock()
		defer s.mu.Unlock()
		delete(s.subs, id)
	}
}

func (s *EventStream) Emit(kind string, data map[string]any) {
	s.mu.RLock()
	handlers := make([]func(Event), 0, len(s.subs))
	for _, fn := range s.subs {
		handlers = append(handlers, fn)
	}
	s.mu.RUnlock()
	for _, fn := range handlers {
		fn(Event{Kind: kind, Data: data})
	}
}
```

## Session：一段对话

```go
// file: session.go
package main

import "time"

// Session 一段对话。
//
// 只装消息，不装模型也不装工具——那些是 Agent 的事。
// 这样同一个 agent 能同时开好几段对话，而一段对话也能换个 agent 接着聊。
type Session struct {
	ID        string
	Title     string
	Messages  []Message
	CreatedAt time.Time
	Metadata  map[string]string
}

func NewSession(title string) *Session {
	return &Session{ID: newID("ses"), Title: title, CreatedAt: time.Now(),
		Metadata: map[string]string{}}
}

// Append 往会话里追加消息。
//
// 接收者必须是**指针** *Session。写成值接收者 (s Session) 的话，
// append 改的是那份拷贝，函数一返回就丢了——会话看起来永远是空的。
// 这是 Go 里最经典的一类 bug，编译器不会拦你。
func (s *Session) Append(messages ...Message) {
	s.Messages = append(s.Messages, messages...)
}

func (s *Session) UserTurns() int {
	n := 0
	for _, m := range s.Messages {
		if m.Role == "user" {
			n++
		}
	}
	return n
}

func (s *Session) SummaryLine() string {
	if s.Title != "" {
		return s.Title
	}
	for _, m := range s.Messages {
		if m.Role == "user" {
			text := []rune(m.Text())
			if len(text) > 24 {
				return string(text[:24]) + "…"
			}
			return string(text)
		}
	}
	return "（空会话）"
}
```

## 系统提示：只有一条，永远在最前

系统提示不放进 `Session.Messages`，而是 Agent 的配置。原因有三条：

1. 它属于 agent 的性格，不属于这段对话。
2. 多数服务只认**开头那一条** system 消息，混在中间会被忽略。
3. 第 16 章要压缩上下文，系统提示是唯一**绝对不能压掉**的部分。
   把它单独放着，压缩逻辑就不用小心翼翼地绕开它。

```go
// file: agent.go
package main

import "context"

type AgentConfig struct {
	SystemPrompt string
	MaxRounds    int
}

func DefaultAgentConfig() AgentConfig {
	return AgentConfig{
		SystemPrompt: "你是一个helpful的助手。需要用工具时就调用，不要凭空编造。",
		MaxRounds:    8,
	}
}

// Agent 一个 agent：一套配置、一个模型、一组工具。
//
// 注意它**不持有会话**。会话是方法参数，每次调用传进来。
// 这样一个 Agent 值能服务很多段对话——服务端上这一点是刚需，
// 不然每来一个用户就要重建一遍工具表。
//
// 这里用配置结构体而不是一串可选参数：Go 没有默认参数，
// 另一种常见写法是函数式选项（NewAgent(model, reg, WithMaxRounds(4))），
// 参数多到十几个时才值得那么写，现在两三个用结构体更直白。
type Agent struct {
	Model    Model
	Registry *ToolRegistry
	Config   AgentConfig
	Events   *EventStream
}

func NewAgent(model Model, reg *ToolRegistry, config AgentConfig, events *EventStream) *Agent {
	if config.MaxRounds == 0 {
		config.MaxRounds = 8
	}
	if events == nil {
		events = NewEventStream()
	}
	return &Agent{Model: model, Registry: reg, Config: config, Events: events}
}

// prepare 拼出这一次要发给模型的消息。系统提示在最前，然后是会话历史。
func (a *Agent) prepare(session *Session) []Message {
	head := NewMessage("system", TextBlock{Text: a.Config.SystemPrompt})
	out := make([]Message, 0, len(session.Messages)+1)
	out = append(out, head)
	out = append(out, session.Messages...)
	return out
}

// Send 用户说一句话，agent 回一句。会话历史就地更新。
func (a *Agent) Send(ctx context.Context, session *Session, userText string) TurnResult {
	session.Append(NewMessage("user", TextBlock{Text: userText}))
	result := a.loop(ctx, session)
	// 这一轮产生的消息全部并进会话——包括中间那些工具调用与结果。
	// 不能只留最终答案：模型下一轮需要看到它自己调过什么、拿到了什么。
	session.Append(result.Messages...)
	return result
}

func (a *Agent) loop(ctx context.Context, session *Session) TurnResult {
	turnID := newID("turn")
	var newMessages []Message
	a.Events.Emit("turn_start", map[string]any{"turn_id": turnID, "session_id": session.ID})

	for round := 1; round <= a.Config.MaxRounds; round++ {
		base := append(a.prepare(session), newMessages...)
		reply, err := a.oneRound(ctx, base)
		if err != nil {
			return TurnResult{StopReason: StopError, RoundsUsed: round,
				Messages: newMessages, Err: err}
		}
		newMessages = append(newMessages, reply)

		calls := reply.ToolUses()
		if len(calls) == 0 {
			a.Events.Emit("turn_done", map[string]any{"reason": string(StopCompleted)})
			return TurnResult{Text: reply.Text(), StopReason: StopCompleted,
				RoundsUsed: round, Messages: newMessages}
		}

		var results []Block
		for _, call := range calls {
			r := a.Registry.Execute(ctx, call.Name, call.Args)
			a.Events.Emit("tool_end", map[string]any{
				"name": call.Name, "ok": !r.IsError, "output": r.Content})
			results = append(results, ToolResultBlock{
				ToolUseID: call.ID, Content: r.Content, IsError: r.IsError})
		}
		newMessages = append(newMessages, NewMessage("tool", results...))
	}
	return TurnResult{StopReason: StopBudget, RoundsUsed: a.Config.MaxRounds, Messages: newMessages}
}

func (a *Agent) oneRound(ctx context.Context, messages []Message) (Message, error) {
	deltas, errs := a.Model.Stream(ctx, messages, a.Registry.Specs())
	var textParts []string
	var blocks []Block
	flushText := func() {
		if len(textParts) > 0 {
			joined := ""
			for _, p := range textParts {
				joined += p
			}
			blocks = append(blocks, TextBlock{Text: joined})
			textParts = nil
		}
	}
	for d := range deltas {
		switch d.Kind {
		case "text":
			textParts = append(textParts, d.Text)
			a.Events.Emit("text_delta", map[string]any{"delta": d.Text})
		case "tool_use":
			if d.Tool != nil {
				flushText()
				blocks = append(blocks, *d.Tool)
			}
		}
	}
	flushText()
	if err := <-errs; err != nil {
		return Message{}, err
	}
	return NewMessage("assistant", blocks...), nil
}
```

`prepare` 那里有个 Go 的坑值得留意：`append(a.prepare(session), newMessages...)`
之所以安全，是因为 `prepare` 每次都新建一个切片。如果它返回的是
`append([]Message{head}, session.Messages...)` 之外的、和 `session.Messages`
共享底层数组的东西，这一句就可能把会话里的数据覆盖掉。
**切片共享底层数组是 Go 里另一类看不见的 bug。**

## 多轮对话：指代能被消解，说明历史真的带上了

```go
// file: main.go
package main

import (
	"context"
	"fmt"
)

func main() {
	ctx := context.Background()
	reg := NewToolRegistry(GetTimeTool{}, NewCalcTool())
	agent := NewAgent(
		NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B"),
		reg,
		AgentConfig{SystemPrompt: "你是一个简洁的助手。算数一律调用 calculate 工具，不要心算。",
			MaxRounds: 6},
		nil)

	session := NewSession("随手算几笔")
	for _, question := range []string{
		"帮我算 128 * 7",
		"再乘以 3 呢？", // 「再」——它得知道上一次的结果
		"刚才我一共问了你几个数学问题？",
	} {
		fmt.Printf("\n用户：%s\n", question)
		result := agent.Send(ctx, session, question)
		fmt.Printf("助手：%s\n", result.Text)
	}

	fmt.Printf("\n会话 %s 现在有 %d 条消息，%d 轮用户提问\n",
		session.ID, len(session.Messages), session.UserTurns())
}
```

第二个问题里的「再乘以 3」没有主语。它答对了，就说明**历史确实被带上了**——
这是有状态和无状态的分界线。

## 一个 agent，多段对话

```go
// file: main.go
package main

import (
	"context"
	"fmt"
)

func main() {
	ctx := context.Background()
	agent := NewAgent(
		NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B"),
		NewToolRegistry(GetTimeTool{}, NewCalcTool()),
		DefaultAgentConfig(), nil)

	work := NewSession("工作")
	life := NewSession("生活")

	agent.Send(ctx, work, "记住：我的项目代号叫「北极星」。")
	agent.Send(ctx, life, "记住：我养的猫叫「馒头」。")

	fmt.Println("在工作会话里问项目代号：")
	fmt.Println(" ", agent.Send(ctx, work, "我的项目代号是什么？").Text)

	fmt.Println("在生活会话里问项目代号（它不该知道）：")
	fmt.Println(" ", agent.Send(ctx, life, "我的项目代号是什么？").Text)

	for _, s := range []*Session{work, life} {
		fmt.Printf("  %s：%d 条消息\n", s.SummaryLine(), len(s.Messages))
	}
}
```

两段对话互不串味。**这正是把消息放在 Session 而不是 Agent 上的回报**——
如果历史挂在 agent 身上，上面这段代码要建两个 agent，工具表和配置都得重复一份。

## 会话里到底存了什么

```go
// file: main.go
package main

import (
	"context"
	"fmt"
	"strings"
)

func main() {
	agent := NewAgent(
		NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B"),
		NewToolRegistry(GetTimeTool{}), DefaultAgentConfig(), nil)

	demo := NewSession("看一眼内部结构")
	agent.Send(context.Background(), demo, "北京现在几点？")

	for i, m := range demo.Messages {
		var kinds []string
		if text := m.Text(); text != "" {
			kinds = append(kinds, fmt.Sprintf("文字(%d字)", len([]rune(text))))
		}
		for _, c := range m.ToolUses() {
			kinds = append(kinds, "调用 "+c.Name)
		}
		for _, r := range m.ToolResults() {
			mark := "✓"
			if r.IsError {
				mark = "✗"
			}
			kinds = append(kinds, "结果 "+mark)
		}
		fmt.Printf("%d. %-10s %s\n", i+1, m.Role, strings.Join(kinds, " + "))
	}
}
```

注意**中间过程也在会话里**：用户消息、助手的工具调用、工具结果、助手的最终回答。
界面上通常只显示第一条和最后一条，但发给模型的必须是全部——
少了工具调用与结果的对应关系，下一轮就会被服务端拒绝（第 11 章讲过）。

## 本章小结

- 三样东西三个归属：**配置归 Agent，对话归 Session，一次执行的结果用完就丢**。
- **Agent 不持有会话**。会话是方法参数，一个 Agent 值服务很多段对话。
- 修改会话的方法**必须用指针接收者**。值接收者上的 `append` 改的是拷贝，
  函数一返回就丢了——编译器不会拦你，会话看起来永远是空的。
- 拼消息时小心**切片共享底层数组**：拿到手的切片要确认是新建的，
  不然一次 `append` 可能覆盖掉别人的数据。
- **系统提示不进会话**，每次请求时拼在最前面。
- Go 没有默认参数，两三个可选项用**配置结构体**，多到十几个才值得写函数式选项。
- 验证「历史真的带上了」的最简单办法：问一个**带指代**的问题（「再乘以 3 呢？」）。
