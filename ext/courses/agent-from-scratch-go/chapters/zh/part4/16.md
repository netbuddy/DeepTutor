# 上下文压缩

对话越长，发给模型的消息越多。模型的上下文是有上限的——超了，服务端直接报错，
这一轮什么都拿不到。

这一章解决「满了怎么办」。它有一条**绝对不能违反的约束**，和三个可选的策略。

## 先说那条约束

压缩就是扔东西。但有些东西扔了会让这段历史**直接坏掉**：

| 不能动的 | 为什么 |
|---|---|
| 系统提示 | 扔了 agent 就失忆了，连自己是干嘛的都不知道 |
| 成对的工具调用与结果 | 第 11 章讲过，少一条结果，服务端拒绝整个请求 |
| 最近几轮 | 用户刚说的话被扔了，对话就接不上了 |

第二条最容易犯：按「最早的 N 条消息」截断，正好切在助手的工具调用和工具结果中间。

所以压缩的单位不是**消息**，是**轮**——从一条用户消息开始，到助手给出最终回答为止。
一轮里面的东西，要么全留，要么全扔。

## 本章起点

这一章不需要工具注册表，只需要能造出带工具块的消息、以及能调一次模型来写摘要。

```go
// file: core.go
package main

import (
	"bufio"
	"bytes"
	"context"
	"crypto/rand"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"net/http"
	"strings"
	"time"
)

func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)
}

type Block interface{ BlockType() string }

type TextBlock struct{ Text string }

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string
	Name string
	Args map[string]any
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string
	Content   string
	IsError   bool
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

// Meta 放我们自己的标记，不发给模型。这一章用它记「这条是摘要」。
type Meta struct {
	Summary bool
	Covers  int
}

type Message struct {
	ID        string
	Role      string
	Content   []Block
	CreatedAt time.Time
	Meta      Meta
}

func (m Message) Text() string {
	var b strings.Builder
	for _, block := range m.Content {
		if t, ok := block.(TextBlock); ok {
			b.WriteString(t.Text)
		}
	}
	return b.String()
}

func (m Message) ToolUses() []ToolUseBlock {
	var calls []ToolUseBlock
	for _, b := range m.Content {
		if c, ok := b.(ToolUseBlock); ok {
			calls = append(calls, c)
		}
	}
	return calls
}

func (m Message) ToolResults() []ToolResultBlock {
	var out []ToolResultBlock
	for _, b := range m.Content {
		if r, ok := b.(ToolResultBlock); ok {
			out = append(out, r)
		}
	}
	return out
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks, CreatedAt: time.Now()}
}

type Delta struct {
	Kind string
	Text string
}

type Model interface {
	Complete(ctx context.Context, messages []Message) (string, error)
}

type wireMessage struct {
	Role    string  `json:"role"`
	Content *string `json:"content"`
}

type wireRequest struct {
	Model    string        `json:"model"`
	Messages []wireMessage `json:"messages"`
	Stream   bool          `json:"stream"`
}

type wireChoice struct {
	Delta wireMessage `json:"delta"`
}

type wireResponse struct {
	Choices []wireChoice `json:"choices"`
}

func parseSSELine(line string) (wireResponse, bool) {
	line = strings.TrimSpace(line)
	if line == "" || strings.HasPrefix(line, ":") || !strings.HasPrefix(line, "data:") {
		return wireResponse{}, false
	}
	payload := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
	if payload == "[DONE]" {
		return wireResponse{}, false
	}
	var parsed wireResponse
	if err := json.Unmarshal([]byte(payload), &parsed); err != nil {
		return wireResponse{}, false
	}
	return parsed, true
}

// OpenAICompatModel 这一章只用它写摘要，所以给一个「一次要完整结果」的方法就够了。
type OpenAICompatModel struct {
	BaseURL string
	ModelID string
	Timeout time.Duration
}

func NewOpenAICompatModel(baseURL, modelID string) *OpenAICompatModel {
	return &OpenAICompatModel{BaseURL: baseURL, ModelID: modelID, Timeout: 180 * time.Second}
}

func (m *OpenAICompatModel) Complete(ctx context.Context, messages []Message) (string, error) {
	var wire []wireMessage
	for _, msg := range messages {
		text := msg.Text()
		wire = append(wire, wireMessage{Role: msg.Role, Content: &text})
	}
	raw, err := json.Marshal(wireRequest{Model: m.ModelID, Messages: wire, Stream: true})
	if err != nil {
		return "", err
	}
	req, err := http.NewRequestWithContext(ctx, "POST", m.BaseURL+"/chat/completions",
		bytes.NewReader(raw))
	if err != nil {
		return "", err
	}
	req.Header.Set("Content-Type", "application/json")
	resp, err := (&http.Client{Timeout: m.Timeout}).Do(req)
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		buf := make([]byte, 300)
		n, _ := resp.Body.Read(buf)
		return "", fmt.Errorf("模型服务返回 %d：%s", resp.StatusCode, string(buf[:n]))
	}
	var out strings.Builder
	scanner := bufio.NewScanner(resp.Body)
	scanner.Buffer(make([]byte, 0, 64*1024), 4*1024*1024)
	for scanner.Scan() {
		chunk, ok := parseSSELine(scanner.Text())
		if !ok || len(chunk.Choices) == 0 {
			continue
		}
		if d := chunk.Choices[0].Delta; d.Content != nil {
			out.WriteString(*d.Content)
		}
	}
	return strings.TrimSpace(out.String()), scanner.Err()
}
```

## 量：粗估就够

要知道「满没满」，得先能估算。精确算需要模型对应的分词器——一个几十兆的依赖，
而且换个模型就得换一个。**粗估的误差在 20% 以内，够用了**：真正的兜底是
服务端报错时的重试，不是估得准。

```go
// file: tokens.go
package main

import (
	"encoding/json"
	"unicode"
)

// EstimateTokens 粗估一段文本占多少 token。
//
// 经验规律：中日韩文字大约一个字一个 token；拉丁字母大约四个字符一个 token。
// 这个估法对中英混排的文本误差在 20% 以内，用来判断「要不要压缩」足够了。
//
// 注意要按 rune 数而不是 byte 数算：一个汉字在 UTF-8 里是 3 个字节，
// 按 len(string) 算会高估三倍。这是 Go 里处理中文最常见的错。
func EstimateTokens(text string) int {
	cjk, other := 0, 0
	for _, r := range text {
		if unicode.Is(unicode.Han, r) || unicode.Is(unicode.Hiragana, r) ||
			unicode.Is(unicode.Katakana, r) {
			cjk++
		} else {
			other++
		}
	}
	return cjk + other/4
}

// MessageTokens 一条消息的开销。除了正文，工具调用的参数和结果也要算进去。
func MessageTokens(m Message) int {
	total := EstimateTokens(m.Text())
	for _, c := range m.ToolUses() {
		total += EstimateTokens(c.Name)
		if raw, err := json.Marshal(c.Args); err == nil {
			total += EstimateTokens(string(raw))
		}
	}
	for _, r := range m.ToolResults() {
		total += EstimateTokens(r.Content)
	}
	return total + 4 // 每条消息的角色、分隔符等固定开销
}

func TotalTokens(messages []Message) int {
	total := 0
	for _, m := range messages {
		total += MessageTokens(m)
	}
	return total
}
```

```go
// file: main.go

	"fmt"
	"strings"
)

func main() {
	sample := []Message{

		fmt.Printf("%-10s %5d token\n", m.Role, MessageTokens(m))

	fmt.Printf("%-10s %5d token\n", "合计", TotalTokens(sample))
```

## 切成轮

```go
// file: turns.go
package main

import "fmt"

// Turn 一轮：一条用户消息，加上它引出的全部助手与工具消息。
//
// 压缩的最小单位。半轮是不能存在的东西。
type Turn struct{ Messages []Message }

func (t Turn) Tokens() int { return TotalTokens(t.Messages) }

func (t Turn) UserText() string {
	for _, m := range t.Messages {
		if m.Role == "user" {
			return m.Text()
		}
	}
	return ""
}

func (t Turn) IsSummary() bool {
	for _, m := range t.Messages {
		if m.Meta.Summary {
			return true
		}
	}
	return false
}

// SplitTurns 按用户消息切分。用户开口 = 新的一轮开始。
func SplitTurns(history []Message) []Turn {
	var turns []Turn
	var current []Message
	for _, m := range history {
		if m.Role == "user" && len(current) > 0 {
			turns = append(turns, Turn{Messages: current})
			current = nil
		}
		current = append(current, m)
	}
	if len(current) > 0 {
		turns = append(turns, Turn{Messages: current})
	}
	return turns
}

func Flatten(turns []Turn) []Message {
	var out []Message
	for _, t := range turns {
		out = append(out, t.Messages...)
	}
	return out
}

func CheckPairing(messages []Message) []string {
	calls, results := map[string]bool{}, map[string]bool{}
	for _, m := range messages {
		for _, c := range m.ToolUses() {
			calls[c.ID] = true
		}
		for _, r := range m.ToolResults() {
			results[r.ToolUseID] = true
		}
	}
	var problems []string
	for id := range calls {
		if !results[id] {
			problems = append(problems, fmt.Sprintf("调用 %s 没有结果", id))
		}
	}
	for id := range results {
		if !calls[id] {
			problems = append(problems, fmt.Sprintf("结果 %s 没有对应的调用", id))
		}
	}
	return problems
}
```

## 造一段长历史

```go
// file: fake.go
package main

import (
	"fmt"
	"strings"
)

var topics = []string{"数据库连接池", "缓存穿透", "分布式锁", "消息重复消费", "慢查询",
	"内存泄漏", "线程池打满", "证书过期", "磁盘写满", "时钟漂移"}

// FakeHistory 造一段像模像样的长对话，其中一部分带工具调用。
func FakeHistory(nTurns int) []Message {
	var history []Message
	for i := 0; i < nTurns; i++ {
		topic := topics[i%len(topics)]
		history = append(history, NewMessage("user",
			TextBlock{Text: fmt.Sprintf("第 %d 个问题：%s要怎么排查？", i+1, topic)}))

		if i%3 == 0 {
			call := ToolUseBlock{ID: newID("call"), Name: "search_docs",
				Args: map[string]any{"q": topic, "limit": 5}}
			history = append(history, NewMessage("assistant",
				TextBlock{Text: "我查一下资料。"}, call))
			history = append(history, NewMessage("tool", ToolResultBlock{
				ToolUseID: call.ID,
				Content: fmt.Sprintf("《%s排查手册》摘录：", topic) +
					strings.Repeat("这一段是资料正文。", 12)}))
		}
		history = append(history, NewMessage("assistant", TextBlock{
			Text: fmt.Sprintf("关于%s，先看监控指标，再看日志，最后复现。", topic) +
				strings.Repeat("展开说明若干。", 8)}))
	}
	return history
}
```

```go
// file: main.go
package main

import "fmt"

func main() {
	history := FakeHistory(20)
	turns := SplitTurns(history)
	withTools := 0
	for _, t := range turns {
		for _, m := range t.Messages {
			if len(m.ToolUses()) > 0 {
				withTools++
				break
			}
		}
	}
	fmt.Printf("造了 %d 轮、%d 条消息、约 %d token\n", len(turns), len(history), TotalTokens(history))
	fmt.Printf("其中带工具调用的轮：%d\n", withTools)
	if problems := CheckPairing(history); len(problems) == 0 {
		fmt.Println("成对检查：全部成对 ✓")
	} else {
		fmt.Println("成对检查：", problems)
	}
}
```

## 策略一：截断

最简单的办法——从最早的轮开始扔，扔到装得下为止。

```go
// file: compact.go
package main

import "fmt"

type CompactionResult struct {
	Messages     []Message
	Strategy     string
	TurnsBefore  int
	TurnsAfter   int
	TokensBefore int
	TokensAfter  int
}

func (r CompactionResult) Describe() string {
	saved := 0
	if r.TokensBefore > 0 {
		saved = 100 - r.TokensAfter*100/r.TokensBefore
	}
	return fmt.Sprintf("%s：%d 轮 → %d 轮，%d → %d token（省了 %d%%）",
		r.Strategy, r.TurnsBefore, r.TurnsAfter, r.TokensBefore, r.TokensAfter, saved)
}

// Truncate 从最早的轮开始扔。
//
// keepRecent 是底线：最近这几轮无论如何都留着。
// 没有这条底线，一段特别长的对话可能被压到只剩系统提示，用户会觉得 agent 突然失忆。
func Truncate(history []Message, budget, keepRecent int) CompactionResult {
	turns := SplitTurns(history)
	before := TotalTokens(history)

	kept := turns
	for len(kept) > keepRecent && TotalTokens(Flatten(kept)) > budget {
		kept = kept[1:] // 扔最早的一整轮，绝不切开
	}

	messages := Flatten(kept)
	return CompactionResult{Messages: messages, Strategy: "截断",
		TurnsBefore: len(turns), TurnsAfter: len(kept),
		TokensBefore: before, TokensAfter: TotalTokens(messages)}
}
```

```go
// file: main.go
package main

import "fmt"

func main() {
	history := FakeHistory(20)
	result := Truncate(history, 1500, 2)
	fmt.Println(result.Describe())
	if problems := CheckPairing(result.Messages); len(problems) == 0 {
		fmt.Println("成对检查：全部成对 ✓")
	} else {
		fmt.Println("成对检查：", problems)
	}
	fmt.Println("最早保留的一轮：", SplitTurns(result.Messages)[0].UserText())
}
```

截断的问题是**前面的信息彻底没了**。用户在第 3 轮说的「我的项目叫北极星」，
到第 20 轮就想不起来了。

## 策略二：摘要

让模型把要扔掉的那几轮总结成一段话，用这段话顶替它们。

```go
// file: summarize.go
package main

import (
	"context"
	"fmt"
	"strings"
)

const summaryPrompt = `把下面这段对话压缩成一份要点清单，供后续对话参考。

必须保留：用户提到的事实（名字、数字、偏好、约束）、已经做出的决定、还没解决的问题。
可以丢弃：寒暄、重复的解释、工具返回的原始长文。

用中文，不超过 200 字，直接给清单，不要开场白。

对话：
%s`

func renderTranscript(turns []Turn) string {
	var b strings.Builder
	for _, t := range turns {
		for _, m := range t.Messages {
			switch {
			case m.Role == "user":
				b.WriteString("用户：" + m.Text() + "\n")
			case m.Role == "assistant" && m.Text() != "":
				b.WriteString("助手：" + m.Text() + "\n")
			case m.Role == "tool":
				for _, r := range m.ToolResults() {
					b.WriteString("工具结果：" + truncate(r.Content, 120) + "\n")
				}
			}
		}
	}
	return b.String()
}

func truncate(s string, n int) string {
	r := []rune(s)
	if len(r) <= n {
		return s
	}
	return string(r[:n]) + "…"
}

// Summarize 把最早的若干轮换成一条摘要消息。
func Summarize(ctx context.Context, history []Message, budget int,
	model Model, keepRecent int) (CompactionResult, error) {
	turns := SplitTurns(history)
	before := TotalTokens(history)

	if before <= budget || len(turns) <= keepRecent {
		return CompactionResult{Messages: history, Strategy: "无需压缩",
			TurnsBefore: len(turns), TurnsAfter: len(turns),
			TokensBefore: before, TokensAfter: before}, nil
	}

	// 已经是摘要的轮不再进摘要——否则每次压缩都把摘要再摘要一遍，
	// 信息一轮比一轮糊，几次之后什么都不剩了。
	var head, carried []Turn
	for _, t := range turns[:len(turns)-keepRecent] {
		if t.IsSummary() {
			carried = append(carried, t)
		} else {
			head = append(head, t)
		}
	}
	tail := turns[len(turns)-keepRecent:]

	if len(head) == 0 {
		return Truncate(history, budget, keepRecent), nil
	}

	text, err := model.Complete(ctx, []Message{
		NewMessage("user", TextBlock{Text: fmt.Sprintf(summaryPrompt, renderTranscript(head))})})
	if err != nil {
		return CompactionResult{}, fmt.Errorf("写摘要失败：%w", err)
	}

	summary := NewMessage("user", TextBlock{Text: "【早前对话的要点，供你参考】\n" + text})
	summary.Meta = Meta{Summary: true, Covers: len(head)}

	messages := append(Flatten(carried), summary)
	messages = append(messages, Flatten(tail)...)
	return CompactionResult{Messages: messages, Strategy: "摘要",
		TurnsBefore: len(turns), TurnsAfter: len(SplitTurns(messages)),
		TokensBefore: before, TokensAfter: TotalTokens(messages)}, nil
}
```

**摘要消息用 user 角色发**，理由和第 12 章那些纠正提示一样：
用 assistant 角色模型会当成自己说过的话，用 system 角色很多服务只认开头那一条。
文字前面加个方括号标注，让模型知道这是转述而不是用户新说的话。

```go
// file: main.go
package main

import (
	"context"
	"fmt"
)

func main() {
	ctx := context.Background()
	model := NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
	history := FakeHistory(20)

	result, err := Summarize(ctx, history, 1500, model, 3)
	if err != nil {
		fmt.Println("出错：", err)
		return
	}
	fmt.Println(result.Describe())
	if problems := CheckPairing(result.Messages); len(problems) == 0 {
		fmt.Println("成对检查：全部成对 ✓")
	} else {
		fmt.Println("成对检查：", problems)
	}

	for _, m := range result.Messages {
		if m.Meta.Summary {
			fmt.Printf("\n摘要覆盖了 %d 轮，内容：\n%s\n", m.Meta.Covers, m.Text())
			break
		}
	}
}
```

## 别把摘要再摘要一遍

对话继续变长，压缩会被再次触发。这时候必须**跳过已有的摘要**：

```go
// file: main.go
package main

import (
	"context"
	"fmt"
)

func main() {
	ctx := context.Background()
	model := NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")

	first, err := Summarize(ctx, FakeHistory(20), 1500, model, 3)
	if err != nil {
		fmt.Println("第一次压缩失败：", err)
		return
	}
	fmt.Println("第一次压缩：", first.Describe())

	// 在压缩过的历史后面再接十轮
	extended := append(append([]Message{}, first.Messages...), FakeHistory(10)...)
	fmt.Printf("接着聊之后：%d 轮，%d token\n", len(SplitTurns(extended)), TotalTokens(extended))

	second, err := Summarize(ctx, extended, 1500, model, 3)
	if err != nil {
		fmt.Println("第二次压缩失败：", err)
		return
	}
	fmt.Println("第二次压缩：", second.Describe())

	count := 0
	fmt.Println("\n压缩两次之后，历史里的摘要：")
	for _, m := range second.Messages {
		if m.Meta.Summary {
			count++
			fmt.Printf("  覆盖 %d 轮：%s\n", m.Meta.Covers, truncate(m.Text(), 60))
		}
	}
	fmt.Printf("\n一共 %d 条摘要。注意第一条**原样保留**了，没有被二次压缩——\n", count)
	fmt.Println("反复摘要会让信息一轮比一轮糊。")
}
```

## 策略三：分层——按预算自己选

实际用的是个组合策略：**没超就不动，超了先摘要，摘要还装不下就截断**。

```go
// file: manager.go
package main

import "context"

// ContextManager 决定每次请求发多少历史。
//
// 压缩不是每轮都做的事——它要调一次模型，慢而且花钱。
// 只在超过阈值时做，而且阈值要留出余量给回复。
type ContextManager struct {
	Model           Model
	MaxContext      int
	ReserveForReply int
	KeepRecent      int
}

func NewContextManager(model Model, maxContext, reserve int) *ContextManager {
	return &ContextManager{Model: model, MaxContext: maxContext,
		ReserveForReply: reserve, KeepRecent: 3}
}

// Budget 留给历史的额度 = 上下文上限 - 给回复留的位置。
//
// 不留这个余量的后果：历史刚好塞满，模型一个字都吐不出来就报超限。
func (c *ContextManager) Budget() int { return c.MaxContext - c.ReserveForReply }

func (c *ContextManager) Prepare(ctx context.Context, systemPrompt string,
	history []Message) ([]Message, string, error) {
	head := NewMessage("system", TextBlock{Text: systemPrompt})
	room := c.Budget() - MessageTokens(head) // 系统提示不参与压缩，先扣掉

	if TotalTokens(history) <= room {
		return append([]Message{head}, history...), "无需压缩", nil
	}

	result, err := Summarize(ctx, history, room, c.Model, c.KeepRecent)
	if err != nil {
		return nil, "", err
	}
	if result.TokensAfter > room {
		// 摘要之后还是装不下——比如最近三轮本身就特别长。截断兜底。
		result = Truncate(result.Messages, room, 1)
		return append([]Message{head}, result.Messages...), "摘要 + 截断", nil
	}
	return append([]Message{head}, result.Messages...), result.Strategy, nil
}
```

## 验收：压缩之后还能接着聊

最实在的检验：在压缩过的历史里问一个只有早期才提到的事实。

```go
// file: main.go
package main

import (
	"context"
	"fmt"
)

func main() {
	ctx := context.Background()
	model := NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
	manager := NewContextManager(model, 3000, 800)

	facts := []Message{
		NewMessage("user", TextBlock{Text: "记住：我们的服务叫「北极星」，端口是 9187。"}),
		NewMessage("assistant", TextBlock{Text: "记住了。"}),
	}
	longHistory := append(append([]Message{}, facts...), FakeHistory(18)...)

	messages, strategy, err := manager.Prepare(ctx,
		"你是一个运维排障助手，回答控制在一句话。", longHistory)
	if err != nil {
		fmt.Println("压缩失败：", err)
		return
	}
	fmt.Printf("压缩策略：%s，发出去 %d token\n", strategy, TotalTokens(messages))

	question := "我们的服务叫什么？端口是多少？"
	messages = append(messages, NewMessage("user", TextBlock{Text: question}))
	answer, err := model.Complete(ctx, messages)
	if err != nil {
		fmt.Println("提问失败：", err)
		return
	}
	fmt.Printf("\n问：%s\n答：%s\n", question, answer)
	fmt.Println("\n→ 答得出「北极星 / 9187」，说明摘要把关键事实留住了；")
	fmt.Println("  答不出说明摘要提示词还得再明确一点。")
}
```

这个检验也是**调摘要提示词的方法**：把「必须保留：用户提到的事实」这句话删掉再跑一次，
你会看到它答不出来了。摘要提示词写什么，直接决定 agent 长期记得什么。

## 本章小结

- 精确算 token 要分词器，**粗估误差 20% 以内就够**——真正的兜底是服务端报错时的重试。
- 估算要**按 rune 数算，不能按 byte 数**：一个汉字在 UTF-8 里是 3 个字节，
  用 `len(string)` 会高估三倍。这是 Go 里处理中文最常见的错。
- 压缩的单位是**轮**，不是消息。切在工具调用和工具结果中间会让整段历史被服务端拒绝。
- 三样东西不能动：**系统提示、成对的调用与结果、最近几轮**。
- **截断**最简单，代价是早期信息彻底没了。
- **摘要**用模型把要扔的几轮压成一段话。摘要消息用 **user 角色**，
  文字里标注清楚这是转述。
- **已经是摘要的内容不要再摘要**。反复摘要会让信息一轮比一轮糊。
- 预算要**给回复留余量**。历史刚好塞满的后果是模型一个字都吐不出来就报超限。
- 验收方法：在压缩过的历史里问一个只有早期才提到的事实。
  这也是调摘要提示词的方法——**提示词写什么，直接决定 agent 长期记得什么**。
