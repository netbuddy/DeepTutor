# 会话持久化

上一章的会话活在内存里，进程一退全没了。这一章把它写到磁盘上。

看起来这是个「序列化一下就完了」的小事。其实有四个坑：

1. **怎么写**——每次全量覆写，还是追加？答案影响的是崩溃时你丢多少东西。
2. **怎么读回来**——`Content []Block` 是一个接口切片，而 **Go 没法把 JSON 解进接口**。
3. **写到一半崩了**——文件尾多半行，下次读会炸。
4. **格式以后要改**——存了半年的会话，加一个块类型就读不了了？

第二个坑是 Go 特有的，也是这一章最值得学的部分。

## 本章起点（一）：协议与模型

```go
// file: core.go
package main

import (
	"bufio"
	"bytes"
	"context"
	"crypto/rand"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"net/http"
	"sort"
	"strings"
	"time"
)

func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)
}

type Block interface{ BlockType() string }

type TextBlock struct {
	Text string `json:"text"`
}

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string         `json:"id"`
	Name string         `json:"name"`
	Args map[string]any `json:"args"`
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string `json:"tool_use_id"`
	Content   string `json:"content"`
	IsError   bool   `json:"is_error"`
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

type Message struct {
	ID        string
	Role      string
	Content   []Block
	CreatedAt time.Time
}

func (m Message) Text() string {
	out := ""
	for _, b := range m.Content {
		if t, ok := b.(TextBlock); ok {
			out += t.Text
		}
	}
	return out
}

func (m Message) ToolUses() []ToolUseBlock {
	var calls []ToolUseBlock
	for _, b := range m.Content {
		if c, ok := b.(ToolUseBlock); ok {
			calls = append(calls, c)
		}
	}
	return calls
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks, CreatedAt: time.Now()}
}

type ToolSpec struct {
	Name        string
	Description string
	Parameters  map[string]any
}

type Delta struct {
	Kind string
	Text string
	Tool *ToolUseBlock
}

type Model interface {
	Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error)
}

type wireFunction struct {
	Name      string `json:"name"`
	Arguments string `json:"arguments"`
}

type wireToolCall struct {
	Index    int          `json:"index,omitempty"`
	ID       string       `json:"id,omitempty"`
	Type     string       `json:"type,omitempty"`
	Function wireFunction `json:"function"`
}

type wireMessage struct {
	Role       string         `json:"role"`
	Content    *string        `json:"content"`
	ToolCallID string         `json:"tool_call_id,omitempty"`
	ToolCalls  []wireToolCall `json:"tool_calls,omitempty"`
}

type wireToolDef struct {
	Type     string `json:"type"`
	Function struct {
		Name        string         `json:"name"`
		Description string         `json:"description"`
		Parameters  map[string]any `json:"parameters"`
	} `json:"function"`
}

type wireRequest struct {
	Model    string        `json:"model"`
	Messages []wireMessage `json:"messages"`
	Tools    []wireToolDef `json:"tools,omitempty"`
	Stream   bool          `json:"stream,omitempty"`
}

type wireChoice struct {
	Delta wireMessage `json:"delta"`
}

type wireResponse struct {
	Choices []wireChoice `json:"choices"`
}

func toWireMessages(messages []Message) []wireMessage {
	var out []wireMessage
	for _, m := range messages {
		if m.Role == "tool" {
			for _, b := range m.Content {
				if r, ok := b.(ToolResultBlock); ok {
					content := r.Content
					out = append(out, wireMessage{Role: "tool", Content: &content, ToolCallID: r.ToolUseID})
				}
			}
			continue
		}
		text, calls := m.Text(), m.ToolUses()
		wm := wireMessage{Role: m.Role}
		if text != "" {
			wm.Content = &text
		}
		for _, c := range calls {
			args, err := json.Marshal(c.Args)
			if err != nil {
				args = []byte("{}")
			}
			wm.ToolCalls = append(wm.ToolCalls, wireToolCall{ID: c.ID, Type: "function",
				Function: wireFunction{Name: c.Name, Arguments: string(args)}})
		}
		out = append(out, wm)
	}
	return out
}

func toWireTools(specs []ToolSpec) []wireToolDef {
	var out []wireToolDef
	for _, s := range specs {
		var d wireToolDef
		d.Type = "function"
		d.Function.Name = s.Name
		d.Function.Description = s.Description
		if s.Parameters == nil {
			d.Function.Parameters = map[string]any{"type": "object", "properties": map[string]any{}}
		} else {
			d.Function.Parameters = s.Parameters
		}
		out = append(out, d)
	}
	return out
}

func parseArguments(raw string) map[string]any {
	if raw == "" {
		return map[string]any{}
	}
	var args map[string]any
	if err := json.Unmarshal([]byte(raw), &args); err != nil {
		return map[string]any{"_invalid_json": raw}
	}
	return args
}

type toolCallSlot struct{ id, name, arguments string }

type toolCallAccumulator struct{ slots map[int]*toolCallSlot }

func newToolCallAccumulator() *toolCallAccumulator {
	return &toolCallAccumulator{slots: map[int]*toolCallSlot{}}
}

func (a *toolCallAccumulator) Feed(calls []wireToolCall) {
	for _, item := range calls {
		slot, ok := a.slots[item.Index]
		if !ok {
			slot = &toolCallSlot{}
			a.slots[item.Index] = slot
		}
		if item.ID != "" {
			slot.id = item.ID
		}
		if item.Function.Name != "" {
			slot.name = item.Function.Name
		}
		if item.Function.Arguments != "" {
			slot.arguments += item.Function.Arguments
		}
	}
}

func (a *toolCallAccumulator) Finish() []ToolUseBlock {
	indexes := make([]int, 0, len(a.slots))
	for i := range a.slots {
		indexes = append(indexes, i)
	}
	sort.Ints(indexes)
	var blocks []ToolUseBlock
	for _, i := range indexes {
		slot := a.slots[i]
		id := slot.id
		if id == "" {
			id = newID("call")
		}
		blocks = append(blocks, ToolUseBlock{ID: id, Name: slot.name, Args: parseArguments(slot.arguments)})
	}
	return blocks
}

func parseSSELine(line string) (wireResponse, bool) {
	line = strings.TrimSpace(line)
	if line == "" || strings.HasPrefix(line, ":") || !strings.HasPrefix(line, "data:") {
		return wireResponse{}, false
	}
	payload := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
	if payload == "[DONE]" {
		return wireResponse{}, false
	}
	var parsed wireResponse
	if err := json.Unmarshal([]byte(payload), &parsed); err != nil {
		return wireResponse{}, false
	}
	return parsed, true
}

type OpenAICompatModel struct {
	BaseURL string
	ModelID string
	Timeout time.Duration
}

func NewOpenAICompatModel(baseURL, modelID string) *OpenAICompatModel {
	return &OpenAICompatModel{BaseURL: baseURL, ModelID: modelID, Timeout: 180 * time.Second}
}

func (m *OpenAICompatModel) Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta, 32)
	errs := make(chan error, 1)
	go func() {
		defer close(deltas)
		defer close(errs)
		body := wireRequest{Model: m.ModelID, Messages: toWireMessages(messages),
			Tools: toWireTools(tools), Stream: true}
		raw, err := json.Marshal(body)
		if err != nil {
			errs <- fmt.Errorf("请求体序列化失败：%w", err)
			return
		}
		req, err := http.NewRequestWithContext(ctx, "POST", m.BaseURL+"/chat/completions", bytes.NewReader(raw))
		if err != nil {
			errs <- fmt.Errorf("构造请求失败：%w", err)
			return
		}
		req.Header.Set("Content-Type", "application/json")
		resp, err := (&http.Client{Timeout: m.Timeout}).Do(req)
		if err != nil {
			errs <- fmt.Errorf("请求失败：%w", err)
			return
		}
		defer resp.Body.Close()
		if resp.StatusCode != http.StatusOK {
			buf := make([]byte, 300)
			n, _ := resp.Body.Read(buf)
			errs <- fmt.Errorf("模型服务返回 %d：%s", resp.StatusCode, string(buf[:n]))
			return
		}
		acc := newToolCallAccumulator()
		scanner := bufio.NewScanner(resp.Body)
		scanner.Buffer(make([]byte, 0, 64*1024), 4*1024*1024)
		for scanner.Scan() {
			chunk, ok := parseSSELine(scanner.Text())
			if !ok || len(chunk.Choices) == 0 {
				continue
			}
			d := chunk.Choices[0].Delta
			if d.Content != nil && *d.Content != "" {
				deltas <- Delta{Kind: "text", Text: *d.Content}
			}
			if len(d.ToolCalls) > 0 {
				acc.Feed(d.ToolCalls)
			}
		}
		if err := scanner.Err(); err != nil {
			errs <- fmt.Errorf("读流中断：%w", err)
			return
		}
		for _, block := range acc.Finish() {
			b := block
			deltas <- Delta{Kind: "tool_use", Tool: &b}
		}
		errs <- nil
	}()
	return deltas, errs
}
```

## 本章起点（二）：会话与 Agent

```go
// file: agent.go
package main

import (
	"context"
	"fmt"
	"time"
)

type ToolResult struct {
	Content string
	IsError bool
}

type Tool interface {
	Spec() ToolSpec
	Run(ctx context.Context, args map[string]any) (string, error)
}

type FuncTool struct {
	spec ToolSpec
	fn   func(ctx context.Context, args map[string]any) (string, error)
}

func NewFuncTool(name, description string, parameters map[string]any,
	fn func(ctx context.Context, args map[string]any) (string, error)) *FuncTool {
	return &FuncTool{spec: ToolSpec{Name: name, Description: description, Parameters: parameters}, fn: fn}
}

func (t *FuncTool) Spec() ToolSpec { return t.spec }

func (t *FuncTool) Run(ctx context.Context, args map[string]any) (string, error) {
	return t.fn(ctx, args)
}

type ToolRegistry struct {
	tools map[string]Tool
	order []string
}

func NewToolRegistry(tools ...Tool) *ToolRegistry {
	reg := &ToolRegistry{tools: map[string]Tool{}}
	for _, t := range tools {
		reg.tools[t.Spec().Name] = t
		reg.order = append(reg.order, t.Spec().Name)
	}
	return reg
}

func (reg *ToolRegistry) Specs() []ToolSpec {
	var out []ToolSpec
	for _, name := range reg.order {
		out = append(out, reg.tools[name].Spec())
	}
	return out
}

func (reg *ToolRegistry) Execute(ctx context.Context, name string, args map[string]any) (result ToolResult) {
	tool, ok := reg.tools[name]
	if !ok {
		return ToolResult{Content: "没有叫 " + name + " 的工具", IsError: true}
	}
	defer func() {
		if r := recover(); r != nil {
			result = ToolResult{Content: fmt.Sprintf("工具崩溃：%v", r), IsError: true}
		}
	}()
	output, err := tool.Run(ctx, args)
	if err != nil {
		return ToolResult{Content: err.Error(), IsError: true}
	}
	return ToolResult{Content: output}
}

type Session struct {
	ID        string
	Title     string
	Messages  []Message
	CreatedAt time.Time
	Metadata  map[string]string
}

func NewSession(title string) *Session {
	return &Session{ID: newID("ses"), Title: title, CreatedAt: time.Now(),
		Metadata: map[string]string{}}
}

func (s *Session) Append(messages ...Message) { s.Messages = append(s.Messages, messages...) }

type StopReason string

const (
	StopCompleted StopReason = "completed"
	StopBudget    StopReason = "budget"
	StopError     StopReason = "error"
)

type TurnResult struct {
	Text       string
	StopReason StopReason
	RoundsUsed int
	Messages   []Message
	Err        error
}

type AgentConfig struct {
	SystemPrompt string
	MaxRounds    int
}

type Agent struct {
	Model    Model
	Registry *ToolRegistry
	Config   AgentConfig
}

func NewAgent(model Model, reg *ToolRegistry, config AgentConfig) *Agent {
	if config.MaxRounds == 0 {
		config.MaxRounds = 6
	}
	if config.SystemPrompt == "" {
		config.SystemPrompt = "你是一个简洁的助手。需要用工具时就调用。"
	}
	return &Agent{Model: model, Registry: reg, Config: config}
}

func (a *Agent) prepare(session *Session) []Message {
	out := make([]Message, 0, len(session.Messages)+1)
	out = append(out, NewMessage("system", TextBlock{Text: a.Config.SystemPrompt}))
	out = append(out, session.Messages...)
	return out
}

func (a *Agent) Send(ctx context.Context, session *Session, userText string) TurnResult {
	session.Append(NewMessage("user", TextBlock{Text: userText}))
	result := a.loop(ctx, session)
	session.Append(result.Messages...)
	return result
}

func (a *Agent) loop(ctx context.Context, session *Session) TurnResult {
	var newMessages []Message
	for round := 1; round <= a.Config.MaxRounds; round++ {
		reply, err := a.oneRound(ctx, append(a.prepare(session), newMessages...))
		if err != nil {
			return TurnResult{StopReason: StopError, RoundsUsed: round, Messages: newMessages, Err: err}
		}
		newMessages = append(newMessages, reply)
		calls := reply.ToolUses()
		if len(calls) == 0 {
			return TurnResult{Text: reply.Text(), StopReason: StopCompleted,
				RoundsUsed: round, Messages: newMessages}
		}
		var results []Block
		for _, call := range calls {
			r := a.Registry.Execute(ctx, call.Name, call.Args)
			results = append(results, ToolResultBlock{
				ToolUseID: call.ID, Content: r.Content, IsError: r.IsError})
		}
		newMessages = append(newMessages, NewMessage("tool", results...))
	}
	return TurnResult{StopReason: StopBudget, RoundsUsed: a.Config.MaxRounds, Messages: newMessages}
}

func (a *Agent) oneRound(ctx context.Context, messages []Message) (Message, error) {
	deltas, errs := a.Model.Stream(ctx, messages, a.Registry.Specs())
	var textParts []string
	var blocks []Block
	flushText := func() {
		if len(textParts) > 0 {
			joined := ""
			for _, p := range textParts {
				joined += p
			}
			blocks = append(blocks, TextBlock{Text: joined})
			textParts = nil
		}
	}
	for d := range deltas {
		switch d.Kind {
		case "text":
			textParts = append(textParts, d.Text)
		case "tool_use":
			if d.Tool != nil {
				flushText()
				blocks = append(blocks, *d.Tool)
			}
		}
	}
	flushText()
	if err := <-errs; err != nil {
		return Message{}, err
	}
	return NewMessage("assistant", blocks...), nil
}

func NewTimeTool() Tool {
	return NewFuncTool("get_time", "查询某个城市的当前时间",
		map[string]any{"type": "object", "properties": map[string]any{
			"city": map[string]any{"type": "string"}}, "required": []string{"city"}},
		func(ctx context.Context, args map[string]any) (string, error) {
			city, _ := args["city"].(string)
			return fmt.Sprintf("%s 现在是 %s", city, time.Now().Format("15:04")), nil
		})
}
```

## 第一个坑：怎么写

两种写法：

| 写法 | 崩溃时丢什么 | 一次对话的写入量 |
|---|---|---|
| 每次把整个会话覆写一遍 | **整个会话**（写到一半停电，文件是半个 JSON） | 全部消息 |
| 只追加新增的那几条 | 最多最后一条 | 新增的几条 |

选追加。文件格式是 **JSONL**——每行一个独立的 JSON 对象。
第一行是会话本身的信息，后面每行一条消息。

这个选择还有个附带好处：**追加不需要先读**。一段有两千条消息的会话，
每加一句话都要先把两千条读进来再全写回去，是很荒唐的。

## 第二个坑：Go 没法把 JSON 解进接口

`Message.Content` 的类型是 `[]Block`，`Block` 是接口。
`json.Unmarshal` 拿到 `{"type":"text","text":"你好"}` 时，**它不知道该造哪个具体类型**——
接口只是一份方法清单，没有构造信息。这不是 Go 的缺陷，是静态类型的必然结果。

标准解法是**两趟解码**：第一趟只看判别字段，第二趟按判别结果解进具体类型。
中间那个「先不解、留着」的东西叫 `json.RawMessage`。

```go
// file: codec.go
package main

import (
	"encoding/json"
	"fmt"
	"time"
)

const SchemaVersion = 1

// UnknownBlockError 读到一个本版本不认识的块类型。
//
// 它和「格式坏了」是两回事：这条记录多半来自一个更新的版本。
// 两者的处置完全不同，所以要能区分开——这就是为什么它是一个独立的错误类型。
type UnknownBlockError struct{ Type string }

func (e *UnknownBlockError) Error() string {
	if e.Type == "" {
		return "块没有 type 字段"
	}
	return "不认识的块类型 " + e.Type
}

// blockToJSON 存一个块。
//
// 用匿名结构体现拼一个带 type 字段的形状：TextBlock 自己没有 type 字段，
// 而判别字段必须写进文件，否则读的时候无从判断。
func blockToJSON(block Block) ([]byte, error) {
	switch v := block.(type) {
	case TextBlock:
		return json.Marshal(struct {
			Type string `json:"type"`
			Text string `json:"text"`
		}{"text", v.Text})
	case ToolUseBlock:
		return json.Marshal(struct {
			Type string         `json:"type"`
			ID   string         `json:"id"`
			Name string         `json:"name"`
			Args map[string]any `json:"args"`
		}{"tool_use", v.ID, v.Name, v.Args})
	case ToolResultBlock:
		return json.Marshal(struct {
			Type      string `json:"type"`
			ToolUseID string `json:"tool_use_id"`
			Content   string `json:"content"`
			IsError   bool   `json:"is_error"`
		}{"tool_result", v.ToolUseID, v.Content, v.IsError})
	}
	return nil, fmt.Errorf("不知道怎么存这种块：%T", block)
}

// blockFromJSON 两趟解码。
func blockFromJSON(raw json.RawMessage) (Block, error) {
	// 第一趟：只看判别字段。这个匿名结构体只有一个字段，
	// 多出来的字段 json 包会自动忽略。
	var probe struct {
		Type string `json:"type"`
	}
	if err := json.Unmarshal(raw, &probe); err != nil {
		return nil, err
	}

	// 第二趟：按判别结果，把同一段字节解进具体类型。
	switch probe.Type {
	case "text":
		var b TextBlock
		return b, json.Unmarshal(raw, &b)
	case "tool_use":
		var b ToolUseBlock
		return b, json.Unmarshal(raw, &b)
	case "tool_result":
		var b ToolResultBlock
		return b, json.Unmarshal(raw, &b)
	}
	return nil, &UnknownBlockError{Type: probe.Type}
}

// storedMessage 一条消息在文件里的样子。
//
// Content 的类型是 []json.RawMessage 而不是 []Block——
// 这正是「留着，等会儿再解」的写法。
type storedMessage struct {
	Kind      string            `json:"kind"`
	ID        string            `json:"id"`
	Role      string            `json:"role"`
	CreatedAt time.Time         `json:"created_at"`
	Content   []json.RawMessage `json:"content"`
}

func messageToJSON(m Message) ([]byte, error) {
	stored := storedMessage{Kind: "message", ID: m.ID, Role: m.Role, CreatedAt: m.CreatedAt}
	for _, b := range m.Content {
		raw, err := blockToJSON(b)
		if err != nil {
			return nil, err
		}
		stored.Content = append(stored.Content, raw)
	}
	return json.Marshal(stored)
}

func messageFromStored(stored storedMessage) (Message, error) {
	m := Message{ID: stored.ID, Role: stored.Role, CreatedAt: stored.CreatedAt}
	for _, raw := range stored.Content {
		block, err := blockFromJSON(raw)
		if err != nil {
			return Message{}, err
		}
		m.Content = append(m.Content, block)
	}
	return m, nil
}
```

```go
// file: main.go

import (
	"encoding/json"
	"fmt"
)

func main() {
	original := NewMessage("assistant",
		TextBlock{Text: "我查一下。"},
		ToolUseBlock{ID: "call_1", Name: "get_time", Args: map[string]any{"city": "北京"}})

	raw, err := messageToJSON(original)
	if err != nil {
		fmt.Println("存不下来：", err)
		return
	}
	fmt.Println("存成这样：", string(raw))

	var stored storedMessage
	if err := json.Unmarshal(raw, &stored); err != nil {
		fmt.Println("读不回来：", err)
		return
	}
	restored, err := messageFromStored(stored)
	if err != nil {
		fmt.Println("解块失败：", err)
		return
	}

	fmt.Printf("块类型都对：")
	for _, b := range restored.Content {
		fmt.Printf(" %T", b)
	}
	fmt.Println()

	again, _ := messageToJSON(restored)
	fmt.Println("往返一致：", string(raw) == string(again))
}
```

## 第三个坑：写到一半崩了

进程在写第 N 条消息时被杀，文件尾就是半行。下次读到那一行，`json.Unmarshal` 会报错。

处理办法是**读的时候容忍，读完把文件截断**。截断这一步不能省：
不截断的话，下次追加会接在半行后面，产生一行「半条 + 一整条」的垃圾，那才是真的读不回来了。

```go
// file: store.go
package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"sort"
	"strings"
	"time"
)

type sessionHeader struct {
	Kind      string            `json:"kind"`
	Schema    int               `json:"schema"`
	ID        string            `json:"id"`
	Title     string            `json:"title"`
	CreatedAt time.Time         `json:"created_at"`
	Metadata  map[string]string `json:"metadata"`
}

type SessionStore struct{ Root string }

func NewSessionStore(root string) (*SessionStore, error) {
	if err := os.MkdirAll(root, 0o755); err != nil {
		return nil, err
	}
	return &SessionStore{Root: root}, nil
}

// path 会话标识直接当文件名，所以必须挡住路径穿越。
// 我们自己生成的 id 是安全的，但这个函数也会拿到外面传进来的值。
func (s *SessionStore) path(sessionID string) (string, error) {
	if sessionID == "" || strings.ContainsAny(sessionID, `/\`) || strings.HasPrefix(sessionID, ".") {
		return "", fmt.Errorf("不合法的会话标识：%q", sessionID)
	}
	return filepath.Join(s.Root, sessionID+".jsonl"), nil
}

func (s *SessionStore) Create(session *Session) error {
	path, err := s.path(session.ID)
	if err != nil {
		return err
	}
	header := sessionHeader{Kind: "session", Schema: SchemaVersion, ID: session.ID,
		Title: session.Title, CreatedAt: session.CreatedAt, Metadata: session.Metadata}
	raw, err := json.Marshal(header)
	if err != nil {
		return err
	}
	return os.WriteFile(path, append(raw, '\n'), 0o644)
}

// Append 追加几条消息。这是最常走的路径，所以它必须快。
func (s *SessionStore) Append(sessionID string, messages ...Message) error {
	if len(messages) == 0 {
		return nil
	}
	path, err := s.path(sessionID)
	if err != nil {
		return err
	}
	f, err := os.OpenFile(path, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o644)
	if err != nil {
		return err
	}
	defer f.Close()

	for _, m := range messages {
		raw, err := messageToJSON(m)
		if err != nil {
			return err
		}
		if _, err := f.Write(append(raw, '\n')); err != nil {
			return err
		}
	}
	// Write 只是把数据交给操作系统，还在内存缓冲里。
	// Sync 才是真的落到盘上。代价是每次几毫秒——对话场景值得，
	// 高频写日志的场景就不值得了。
	return f.Sync()
}

// Load 读回一个会话。第二个返回值是读的过程中发现的问题，不是错误。
//
// 为什么问题不当 error 返回：一条坏记录不该让整段对话打不开。
// 用户宁可看到「少了最后一句」，也不想看到「这个会话读不了」。
func (s *SessionStore) Load(sessionID string) (*Session, []string, error) {
	path, err := s.path(sessionID)
	if err != nil {
		return nil, nil, err
	}
	f, err := os.Open(path)
	if err != nil {
		return nil, nil, err
	}
	defer f.Close()

	var problems []string
	var session *Session
	var messages []Message
	goodBytes := int64(0) // 最后一条完整记录的结束位置

	scanner := bufio.NewScanner(f)
	// 默认单行上限是 64 KB。一条长回答很容易超过，超了 Scan 直接返回 false，
	// 表现是「会话莫名其妙少了后半截」。这一行不能省。
	scanner.Buffer(make([]byte, 0, 64*1024), 8*1024*1024)

	lineNo := 0
	for scanner.Scan() {
		lineNo++
		line := scanner.Bytes()
		lineSize := int64(len(line) + 1) // 加回被 Scanner 吃掉的换行符
		if len(strings.TrimSpace(string(line))) == 0 {
			goodBytes += lineSize
			continue
		}

		var probe struct {
			Kind string `json:"kind"`
		}
		if err := json.Unmarshal(line, &probe); err != nil {
			// 只有最后一行的半行才是崩溃留下的；中间出现坏行说明是别的问题。
			problems = append(problems, fmt.Sprintf("第 %d 行不是完整的 JSON，已跳过", lineNo))
			continue
		}

		switch probe.Kind {
		case "session":
			var header sessionHeader
			if err := json.Unmarshal(line, &header); err != nil {
				problems = append(problems, fmt.Sprintf("第 %d 行的会话头读不了，已跳过", lineNo))
				continue
			}
			session = &Session{ID: header.ID, Title: header.Title,
				CreatedAt: header.CreatedAt, Metadata: header.Metadata}
		case "message":
			var stored storedMessage
			if err := json.Unmarshal(line, &stored); err != nil {
				problems = append(problems, fmt.Sprintf("第 %d 行的消息读不了，已跳过", lineNo))
				continue
			}
			m, err := messageFromStored(stored)
			if err != nil {
				problems = append(problems, fmt.Sprintf("第 %d 行：%v，整条跳过", lineNo, err))
				continue
			}
			messages = append(messages, m)
		default:
			problems = append(problems, fmt.Sprintf("第 %d 行是不认识的记录类型，已跳过", lineNo))
			continue
		}
		goodBytes += lineSize
	}
	if err := scanner.Err(); err != nil {
		problems = append(problems, "读到一半出错："+err.Error())
	}
	if session == nil {
		return nil, problems, fmt.Errorf("%s 里没有会话头，这个文件不是会话文件", filepath.Base(path))
	}

	// 把尾巴上的垃圾截掉，否则下次追加会接在半行后面。
	if info, err := f.Stat(); err == nil && goodBytes < info.Size() {
		leftover := info.Size() - goodBytes
		if err := os.Truncate(path, goodBytes); err == nil {
			problems = append(problems, fmt.Sprintf("文件尾有 %d 字节残留，已截断", leftover))
		}
	}

	session.Messages = messages
	return session, problems, nil
}

type SessionMeta struct {
	ID        string
	Title     string
	CreatedAt time.Time
	Size      int64
}

// List 只读每个文件的第一行，不把整段对话读进来。
func (s *SessionStore) List() []SessionMeta {
	entries, err := os.ReadDir(s.Root)
	if err != nil {
		return nil
	}
	var out []SessionMeta
	for _, entry := range entries {
		if entry.IsDir() || !strings.HasSuffix(entry.Name(), ".jsonl") {
			continue
		}
		path := filepath.Join(s.Root, entry.Name())
		f, err := os.Open(path)
		if err != nil {
			continue
		}
		reader := bufio.NewReader(f)
		line, err := reader.ReadBytes('\n')
		f.Close()
		if err != nil && len(line) == 0 {
			continue
		}
		var header sessionHeader
		if json.Unmarshal(line, &header) != nil || header.Kind != "session" {
			continue
		}
		size := int64(0)
		if info, err := entry.Info(); err == nil {
			size = info.Size()
		}
		out = append(out, SessionMeta{ID: header.ID, Title: header.Title,
			CreatedAt: header.CreatedAt, Size: size})
	}
	sort.Slice(out, func(i, j int) bool { return out[i].CreatedAt.After(out[j].CreatedAt) })
	return out
}
```

## 装到 Agent 上

Go 没有继承，用**嵌入**。注意这和继承有个实打实的区别：
嵌入不会让 `Agent` 内部的调用转到 `PersistentAgent` 上——Go 没有虚函数表。
我们这里没问题，因为 `Send` 是外面显式调用的。

```go
// file: persist.go
package main

import "context"

// 存储目录放在这个文件里而不是 main.go 里。
//
// 原因是本章后面几格各有各的 main 函数，每一格都会把 main.go 整个覆盖掉——
// 写在 main.go 里的常量下一格就不存在了。跨格子要用的东西，放在不会被覆盖的文件里。
const storeDir = "/tmp/agent-course-go/sessions"

type PersistentAgent struct {
	*Agent
	Store *SessionStore
	known map[string]bool
}

func NewPersistentAgent(agent *Agent, store *SessionStore) *PersistentAgent {
	return &PersistentAgent{Agent: agent, Store: store, known: map[string]bool{}}
}

// Send 说完一轮就把新增的消息追加到磁盘。
func (p *PersistentAgent) Send(ctx context.Context, session *Session, userText string) (TurnResult, error) {
	if !p.known[session.ID] {
		if err := p.Store.Create(session); err != nil {
			return TurnResult{}, err
		}
		p.known[session.ID] = true
	}

	before := len(session.Messages)
	result := p.Agent.Send(ctx, session, userText)
	// 只写新增的那几条。这里能这么算，是因为 Agent.Send 只往后追加、不改动旧消息。
	if err := p.Store.Append(session.ID, session.Messages[before:]...); err != nil {
		return result, err
	}
	return result, nil
}
```

```go
func splitLines(text string) []string {
	var out []string
	start := 0
	for i := 0; i < len(text); i++ {
		if text[i] == '\n' {
			if i > start {
				out = append(out, text[start:i])
			}
			start = i + 1
		}
	}
	if start < len(text) {
		out = append(out, text[start:])
	}
	return out
}

func truncate(s string, n int) string {
	r := []rune(s)
	if len(r) <= n {
		return s
	}
	return string(r[:n]) + "…"
}
```

## 进程重启，接着聊

```go
// file: main.go
package main

import (
	"context"
	"fmt"
)

func main() {
	// 不删目录：上一格写下的东西还在，这一格扮演「重启后的新进程」。
	store, err := NewSessionStore(storeDir)
	if err != nil {
		fmt.Println("打不开存储目录：", err)
		return
	}

	metas := store.List()
	fmt.Println("磁盘上的会话：")
	for _, m := range metas {
		fmt.Printf("  %s  %s  %d 字节\n", m.ID, m.Title, m.Size)
	}

		fmt.Println("（先运行上一格）")
		return
	}

	restored, problems, err := store.Load(metas[0].ID)
	if err != nil {
		fmt.Println("读不回来：", err)
		return
	}
	fmt.Printf("\n读回 %d 条消息，问题：%v\n", len(restored.Messages), problems)

	agent := NewAgent(
		NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B"),

	persistent.known[restored.ID] = true // 这个会话已经在盘上了，别再写一次头

	fmt.Println("\n用户：我最喜欢的数字是多少？")

	fmt.Println("助手：", result.Text)
}
```

它答得出「17」，说明**跨进程的记忆真的接上了**。

手工把文件尾弄坏，看看能不能扛住：

```go
// file: main.go
package main

import (
	"fmt"
	"os"
	"path/filepath"
)

func main() {
	store, _ := NewSessionStore(storeDir)
	metas := store.List()
	if len(metas) == 0 {
		fmt.Println("（先运行前面两格）")
		return
	}
	path := filepath.Join(storeDir, metas[0].ID+".jsonl")
	info, _ := os.Stat(path)
	beforeSize := info.Size()

	// 模拟「写到一半进程被杀」：追加半行
	f, _ := os.OpenFile(path, os.O_APPEND|os.O_WRONLY, 0o644)
	f.WriteString(`{"kind":"message","id":"msg_broken","role":"assistant","cont`)
	f.Close()

	info, _ = os.Stat(path)
	fmt.Printf("文件被弄坏了：%d → %d 字节\n", beforeSize, info.Size())

	recovered, problems, err := store.Load(metas[0].ID)
	if err != nil {
		fmt.Println("读失败：", err)
		return
	}
	fmt.Printf("读回 %d 条消息\n", len(recovered.Messages))
	for _, p := range problems {
		fmt.Println("  ⚠", p)
	}
	info, _ = os.Stat(path)
	fmt.Printf("截断后：%d 字节（应该回到 %d）\n", info.Size(), beforeSize)

	// 再读一次，这次应该干干净净
	again, problems2, _ := store.Load(metas[0].ID)
	fmt.Printf("再读一次：%d 条消息，问题：%v\n", len(again.Messages), problems2)
}
```

**注意两件事**：一是坏行没有让整段对话打不开；二是读完就把尾巴截了，
所以第二次读干干净净。少了截断这一步，下次追加会接在半行后面，越坏越深。

## 第四个坑：格式以后要改

半年后你想加一种新的块类型。存量的文件怎么办？

```go
// file: main.go
package main

import (
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
)

func main() {
	store, _ := NewSessionStore(storeDir)
	metas := store.List()
	if len(metas) == 0 {
		fmt.Println("（先运行前面几格）")
		return
	}
	path := filepath.Join(storeDir, metas[0].ID+".jsonl")

	// 场景：一条来自更新版本的记录，里面有我们不认识的块类型
	future := `{"kind":"message","id":"msg_future","role":"assistant",` +
		`"created_at":"2026-01-01T00:00:00Z","content":[` +
		`{"type":"text","text":"正常部分"},` +
		`{"type":"thinking","text":"这是新版本才有的块"}]}`
	f, _ := os.OpenFile(path, os.O_APPEND|os.O_WRONLY, 0o644)
	f.WriteString(future + "\n")
	f.Close()

	session, problems, _ := store.Load(metas[0].ID)
	fmt.Println("读到未来版本的块：")
	for _, p := range problems {
		fmt.Println("  ⚠", p)
	}
	fmt.Printf("  消息数：%d（那条整体跳过了）\n\n", len(session.Messages))

	// 版本号是判断「新格式」还是「文件损坏」的唯一依据
	raw, _ := os.ReadFile(path)
	var header sessionHeader
	for i := 0; i < len(raw); i++ {
		if raw[i] == '\n' {
			json.Unmarshal(raw[:i], &header)
			break
		}
	}
	fmt.Printf("文件声明的格式版本：%d，本程序支持：%d\n", header.Schema, SchemaVersion)
	if header.Schema == SchemaVersion {
		fmt.Println("→ 版本一致，可以直接读")
	} else {
		fmt.Println("→ 版本不同，该走迁移流程")
	}
}
```

上面这个处置是「整条跳过」。够不够好取决于场景：

- **对话记录**：整条跳过是对的。一条消息缺了一半，拿给模型看会误导它。
- **审计日志**：应该原样保留那个不认识的块（`json.RawMessage` 正好能干这个），
  只是不解释它——将来升级了还能读懂。

不管选哪种，**前提都是文件里存了 `schema` 版本号**。没有版本号，你连「这是新格式」
都判断不出来，只能当成损坏。

## 本章小结

- **追加式日志，不要全量覆写**。覆写崩溃时丢的是整个会话，追加最多丢最后一条。
- **Go 没法把 JSON 解进接口**。标准解法是**两趟解码**：
  先用一个只有判别字段的匿名结构体探一下，再按结果解进具体类型。
  中间那个「先不解、留着」的类型是 `json.RawMessage`。
- 存的时候要**把判别字段写进去**——`TextBlock` 自己没有 `type` 字段，
  用匿名结构体现拼一个。
- `bufio.Scanner` 默认单行上限 **64 KB**，一条长回答就能超。
  不设 `Buffer` 的表现是「会话莫名其妙少了后半截」。
- `Write` 只是交给操作系统，`f.Sync()` 才是落盘。
- 一条坏记录**不该让整段对话打不开**。问题当返回值报告，不当 error 抛。
- 读完**要把尾巴上的残留截掉**，否则下次追加会接在半行后面。
- 会话标识会当文件名用，**必须挡住路径穿越**。
- **嵌入不是继承**：Go 没有虚函数表，`Agent` 内部的调用不会转到 `PersistentAgent` 上。
- 文件里要存**格式版本号**，否则「新格式」和「文件损坏」你分不开。
