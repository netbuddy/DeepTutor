# 出错的时候

前面十一章造出来的 agent，在一切顺利时能跑。这一章处理不顺利的情况。

错误分三层，处理办法完全不同：

| 层 | 例子 | 办法 |
|---|---|---|
| 传输层 | 连接被拒、服务返回 503、超时 | **重试**，退避着重试 |
| 工具层 | 文件不存在、除数为零、工具 panic | 变成结果回给模型（第 9 章已做完） |
| 模型行为层 | 参数一直填错、空回复、原地打转 | **干预**——这一章的主题 |

第三层是新东西。前两层是「东西坏了」，第三层是「东西没坏，但模型在犯病」。
它不返回 error，不 panic，程序看起来一切正常，只是**永远得不到结果**。

为了聚焦，这一章用顺序执行工具的主循环。把第 11 章的并发加回去是机械的，
不影响这一章的任何结论。

func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)
}

type Block interface{ BlockType() string }

type TextBlock struct{ Text string }

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string
	Name string
	Args map[string]any
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string
	Content   string
	IsError   bool
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

type Message struct {
	ID      string
	Role    string
	Content []Block
}

func (m Message) Text() string {
	out := ""
	for _, b := range m.Content {
		if t, ok := b.(TextBlock); ok {
			out += t.Text
		}
	}
	return out
}

func (m Message) ToolUses() []ToolUseBlock {
	var calls []ToolUseBlock
	for _, b := range m.Content {
		if c, ok := b.(ToolUseBlock); ok {
			calls = append(calls, c)
		}
	}
	return calls
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks}

func NewToolUse(name string, args map[string]any) ToolUseBlock {

// 做成一个类型而不是 fmt.Errorf 拼字符串：调用方要能判断「这个状态码值不值得重试」，

}

func toWireMessages(messages []Message) []wireMessage {
	var out []wireMessage
	for _, m := range messages {
		if m.Role == "tool" {
			for _, b := range m.Content {
				if r, ok := b.(ToolResultBlock); ok {
					content := r.Content
					out = append(out, wireMessage{Role: "tool", Content: &content, ToolCallID: r.ToolUseID})
				}
			}
			continue
		}
		text, calls := m.Text(), m.ToolUses()
		wm := wireMessage{Role: m.Role}
		if text != "" {
			wm.Content = &text
		}
		for _, c := range calls {
			args, err := json.Marshal(c.Args)
			if err != nil {
				args = []byte("{}")
			}
			wm.ToolCalls = append(wm.ToolCalls, wireToolCall{ID: c.ID, Type: "function",
				Function: wireFunction{Name: c.Name, Arguments: string(args)}})
		}
		out = append(out, wm)
	}
	return out
}

func toWireTools(specs []ToolSpec) []wireToolDef {
	var out []wireToolDef
	for _, s := range specs {
		var d wireToolDef
		d.Type = "function"
		d.Function.Name = s.Name
		d.Function.Description = s.Description
		if s.Parameters == nil {
			d.Function.Parameters = map[string]any{"type": "object", "properties": map[string]any{}}
		} else {
			d.Function.Parameters = s.Parameters
		}
		out = append(out, d)
	}
	return out
}

func parseArguments(raw string) map[string]any {
	if raw == "" {
		return map[string]any{}
	}
	var args map[string]any
	if err := json.Unmarshal([]byte(raw), &args); err != nil {
		return map[string]any{"_invalid_json": raw}
	}
	return args
}

type toolCallSlot struct{ id, name, arguments string }

type toolCallAccumulator struct{ slots map[int]*toolCallSlot }

func newToolCallAccumulator() *toolCallAccumulator {
	return &toolCallAccumulator{slots: map[int]*toolCallSlot{}}
}

func (a *toolCallAccumulator) Feed(calls []wireToolCall) {
	for _, item := range calls {
		slot, ok := a.slots[item.Index]
		if !ok {
			slot = &toolCallSlot{}
			a.slots[item.Index] = slot
		}
		if item.ID != "" {
			slot.id = item.ID
		}
		if item.Function.Name != "" {
			slot.name = item.Function.Name
		}
		if item.Function.Arguments != "" {
			slot.arguments += item.Function.Arguments
		}
	}
}

func (a *toolCallAccumulator) Finish() []ToolUseBlock {
	indexes := make([]int, 0, len(a.slots))
	for i := range a.slots {
		indexes = append(indexes, i)
	}
	sort.Ints(indexes)
	var blocks []ToolUseBlock
	for _, i := range indexes {
		slot := a.slots[i]
		id := slot.id
		if id == "" {
			id = newID("call")
		}
		blocks = append(blocks, ToolUseBlock{ID: id, Name: slot.name, Args: parseArguments(slot.arguments)})
	}
	return blocks
}

func parseSSELine(line string) (wireResponse, bool) {
	line = strings.TrimSpace(line)
	if line == "" || strings.HasPrefix(line, ":") || !strings.HasPrefix(line, "data:") {
		return wireResponse{}, false
	}
	payload := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
	if payload == "[DONE]" {
		return wireResponse{}, false
	}
	var parsed wireResponse
	if err := json.Unmarshal([]byte(payload), &parsed); err != nil {
		return wireResponse{}, false
	}
	return parsed, true
}

type OpenAICompatModel struct {
	BaseURL string
	ModelID string
	APIKey  string
	Timeout time.Duration
}

func NewOpenAICompatModel(baseURL, modelID string) *OpenAICompatModel {
	return &OpenAICompatModel{BaseURL: baseURL, ModelID: modelID, Timeout: 180 * time.Second}
}

func (m *OpenAICompatModel) Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta, 32)
	errs := make(chan error, 1)
	go func() {
		defer close(deltas)
		defer close(errs)
		send := func(d Delta) bool {
			select {
			case deltas <- d:
				return true
			case <-ctx.Done():
				return false
			}
		}
		body := wireRequest{Model: m.ModelID, Messages: toWireMessages(messages),
			Tools: toWireTools(tools), Stream: true}
		raw, err := json.Marshal(body)
		if err != nil {
			errs <- fmt.Errorf("请求体序列化失败：%w", err)
			return
		}
		req, err := http.NewRequestWithContext(ctx, "POST", m.BaseURL+"/chat/completions", bytes.NewReader(raw))
		if err != nil {
			errs <- fmt.Errorf("构造请求失败：%w", err)
			return
		}
		req.Header.Set("Content-Type", "application/json")
		if m.APIKey != "" {
			req.Header.Set("Authorization", "Bearer "+m.APIKey)
		}
		resp, err := (&http.Client{Timeout: m.Timeout}).Do(req)
		if err != nil {
			// %w 把原始错误包在里面，外层才能用 errors.As 认出「这是网络错误」。
			// 用 %v 的话链条就断了，重试判断也就无从做起。
			errs <- fmt.Errorf("请求失败：%w", err)
			return
		}
		defer resp.Body.Close()
		if resp.StatusCode != http.StatusOK {
			buf := make([]byte, 300)
			n, _ := resp.Body.Read(buf)
			errs <- &StatusError{Code: resp.StatusCode, Body: string(buf[:n])}
			return
		}
		acc := newToolCallAccumulator()
		scanner := bufio.NewScanner(resp.Body)
		scanner.Buffer(make([]byte, 0, 64*1024), 4*1024*1024)
		for scanner.Scan() {
			chunk, ok := parseSSELine(scanner.Text())
			if !ok || len(chunk.Choices) == 0 {
				continue
			}
			d := chunk.Choices[0].Delta
			if d.Content != nil && *d.Content != "" {
				if !send(Delta{Kind: "text", Text: *d.Content}) {
					errs <- ctx.Err()
					return
				}
			}
			if len(d.ToolCalls) > 0 {
				acc.Feed(d.ToolCalls)
			}
		}
		if err := scanner.Err(); err != nil {
			if ctx.Err() != nil {
				errs <- ctx.Err()
			} else {
				errs <- fmt.Errorf("读流中断：%w", err)
			}
			return
		}
		for _, block := range acc.Finish() {
			b := block
			if !send(Delta{Kind: "tool_use", Tool: &b}) {
				errs <- ctx.Err()
				return
			}
		}
		errs <- nil
	}()
	return deltas, errs
}

// file: tools.go
package main

import (
	"context"
	"fmt"
	"math"
	"sort"
	"strings"
	"sync"
	"time"
)

type ToolResult struct {
	Content       string
	IsError       bool
	IsInvalidArgs bool
}

type Tool interface {
	Spec() ToolSpec
	Run(ctx context.Context, args map[string]any) (string, error)
}

type FuncTool struct {
	spec ToolSpec
	fn   func(ctx context.Context, args map[string]any) (string, error)
}

func NewFuncTool(name, description string, parameters map[string]any,
	fn func(ctx context.Context, args map[string]any) (string, error)) *FuncTool {
	return &FuncTool{spec: ToolSpec{Name: name, Description: description, Parameters: parameters}, fn: fn}
}

func (t *FuncTool) Spec() ToolSpec { return t.spec }

func (t *FuncTool) Run(ctx context.Context, args map[string]any) (string, error) {
	return t.fn(ctx, args)
}

func argString(args map[string]any, key string) string {
	s, _ := args[key].(string)
	return s
}

func jsonTypeOf(value any) string {
	switch v := value.(type) {
	case string:
		return "string"
	case bool:
		return "boolean"
	case float64:
		if v == math.Trunc(v) {
			return "integer"
		}
		return "number"
	case int:
		return "integer"
	case []any:
		return "array"
	case map[string]any:
		return "object"
	case nil:
		return "null"
	}
	return "unknown"
}

func requiredNames(schema map[string]any) []string {
	switch v := schema["required"].(type) {
	case []string:
		return v
	case []any:
		var out []string
		for _, item := range v {
			if s, ok := item.(string); ok {
				out = append(out, s)
			}
		}
		return out
	}
	return nil
}

func sortedKeys(m map[string]any) []string {
	keys := make([]string, 0, len(m))
	for k := range m {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	return keys
}

func ValidateArgs(args map[string]any, schema map[string]any) string {
	if raw, bad := args["_invalid_json"]; bad {
		text := fmt.Sprint(raw)
		if len(text) > 120 {
			text = text[:120]
		}
		return "参数不是合法的 JSON：" + text
	}
	props, _ := schema["properties"].(map[string]any)
	if props == nil {
		props = map[string]any{}
	}
	for _, name := range requiredNames(schema) {
		if _, ok := args[name]; !ok {
			return "缺少必填参数 " + name
		}
	}
	for _, key := range sortedKeys(args) {
		spec, known := props[key].(map[string]any)
		if !known {
			return fmt.Sprintf("不认识的参数 %s；这个工具接受 %s", key, strings.Join(sortedKeys(props), "、"))
		}
		value := args[key]
		if want, _ := spec["type"].(string); want != "" {
			got := jsonTypeOf(value)
			if !(got == want || (want == "number" && got == "integer")) {
				return fmt.Sprintf("参数 %s 应该是 %s，收到的是 %s", key, want, got)
			}
		}
	}
	return ""
}

type ToolRegistry struct {
	tools map[string]Tool
	order []string
}

func NewToolRegistry(tools ...Tool) *ToolRegistry {
	reg := &ToolRegistry{tools: map[string]Tool{}}
	for _, t := range tools {
		reg.Add(t)
	}
	return reg
}

func (reg *ToolRegistry) Add(tool Tool) {
	name := tool.Spec().Name
	if _, exists := reg.tools[name]; !exists {
		reg.order = append(reg.order, name)
	}
	reg.tools[name] = tool
}

// SpecsExcept 报出可用工具，但跳过被停用的那些。这是这一章新增的参数。
func (reg *ToolRegistry) SpecsExcept(disabled map[string]bool) []ToolSpec {
	var out []ToolSpec
	for _, name := range reg.order {
		if disabled[name] {
			continue
		}
		out = append(out, reg.tools[name].Spec())
	}
	return out
}

func (reg *ToolRegistry) Execute(ctx context.Context, name string, args map[string]any) (result ToolResult) {
	tool, ok := reg.tools[name]
	if !ok {
		return ToolResult{Content: fmt.Sprintf("没有叫 %s 的工具。可用的：%s",
			name, strings.Join(reg.order, "、")), IsError: true}
	}
	if problem := ValidateArgs(args, tool.Spec().Parameters); problem != "" {
		return ToolResult{Content: "参数不合格：" + problem, IsError: true, IsInvalidArgs: true}
	}
	defer func() {
		if r := recover(); r != nil {
			result = ToolResult{Content: fmt.Sprintf("工具 %s 执行时崩溃了：%v", name, r), IsError: true}
		}
	}()
	output, err := tool.Run(ctx, args)
	if err != nil {
		return ToolResult{Content: err.Error(), IsError: true}
	}
	return ToolResult{Content: output}
}

type Event struct {
	Kind string
	Data map[string]any
}

type EventStream struct {
	mu   sync.RWMutex
	subs map[int]func(Event)
	next int
}

func NewEventStream() *EventStream { return &EventStream{subs: map[int]func(Event){}} }

func (s *EventStream) Subscribe(fn func(Event)) func() {
	s.mu.Lock()
	defer s.mu.Unlock()
	id := s.next
	s.next++
	s.subs[id] = fn
	return func() {
		s.mu.Lock()
		defer s.mu.Unlock()
		delete(s.subs, id)
	}
}

func (s *EventStream) Emit(kind string, data map[string]any) {
	s.mu.RLock()
	handlers := make([]func(Event), 0, len(s.subs))
	for _, fn := range s.subs {
		handlers = append(handlers, fn)
	}
	s.mu.RUnlock()
	event := Event{Kind: kind, Data: data}
	for _, fn := range handlers {
		fn(event)
	}
}

## 第一层：什么该重试，什么不该

重试的第一原则是**分清「重试有用」和「重试没用」**。
连接被拒、服务 503、限流 429——这些重试有用。请求体格式错了返回 400——重试一万次也是 400。

```go
// file: retry.go
package main

import (
	"context"
	"errors"
	"fmt"
	"math/rand"
	"net"
	"time"
)

// 这些状态码值得重试。共同点：问题出在服务端或链路上，请求本身没毛病。
var retryableStatus = map[int]bool{
	408: true, 409: true, 425: true, 429: true,
	500: true, 502: true, 503: true, 504: true,
}

type RetryPolicy struct {
	MaxAttempts int
	BaseDelay   time.Duration
	MaxDelay    time.Duration
}

func DefaultRetryPolicy() RetryPolicy {
	return RetryPolicy{MaxAttempts: 3, BaseDelay: 500 * time.Millisecond, MaxDelay: 8 * time.Second}
}

// ShouldRetry 判断一个错误值不值得重试。
//
// errors.As 会沿着 %w 包起来的链条一路往里找。
// 这就是为什么 core.go 里包错误必须用 %w 而不是 %v——用 %v 链条就断了。
func (p RetryPolicy) ShouldRetry(err error) bool {
	if err == nil {
		return false
	}
	if errors.Is(err, context.Canceled) || errors.Is(err, context.DeadlineExceeded) {
		return false // 用户主动取消的，不能替他重试
	}
	var statusErr *StatusError
	if errors.As(err, &statusErr) {
		return retryableStatus[statusErr.Code]
	}
	var netErr net.Error
	if errors.As(err, &netErr) {
		return true // 连不上、超时：链路问题，值得再试
	}
	return false // 其他（JSON 解析失败、我们自己的 bug）：重试没有意义
}

// DelayFor 指数退避 + 抖动。
//
// 指数退避：0.5、1、2、4 秒——服务端过载时，一群客户端一起猛敲会让它更起不来。
// 抖动：如果一百个客户端同时被限流、又用同样的退避曲线，它们会**同时**回来，
// 再次把服务打垮。加一点随机让它们错开。
func (p RetryPolicy) DelayFor(attempt int) time.Duration {
	delay := p.BaseDelay << (attempt - 1)
	if delay > p.MaxDelay {
		delay = p.MaxDelay
	}
	return time.Duration(float64(delay) * (0.5 + rand.Float64()*0.5))
}

// RetryingModel 包在任何 Model 外面，给它加上重试。
//
// 做成包装而不是改 OpenAICompatModel：重试策略和协议实现是两件事，
// 换模型不该重写重试，换重试策略也不该动协议代码。
// 它自己也满足 Model 接口，所以对调用方完全透明。
type RetryingModel struct {
	Inner  Model
	Policy RetryPolicy
	Events *EventStream
}

func NewRetryingModel(inner Model, policy RetryPolicy, events *EventStream) *RetryingModel {
	if events == nil {
		events = NewEventStream()
	}
	return &RetryingModel{Inner: inner, Policy: policy, Events: events}
}

func (m *RetryingModel) Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	out := make(chan Delta, 32)
	errs := make(chan error, 1)

	go func() {
		defer close(out)
		defer close(errs)

		for attempt := 1; attempt <= m.Policy.MaxAttempts; attempt++ {
			deltas, innerErrs := m.Inner.Stream(ctx, messages, tools)
			produced := false

			for d := range deltas {
				produced = true
				select {
				case out <- d:
				case <-ctx.Done():
					errs <- ctx.Err()
					return
				}
			}
			err := <-innerErrs
			if err == nil {
				errs <- nil
				return
			}

			// 已经吐出去的增量收不回来了。这时候重试会让同一段话说两遍。
			// 所以：只重试「还没产出任何内容」的失败。
			if produced || attempt >= m.Policy.MaxAttempts || !m.Policy.ShouldRetry(err) {
				errs <- err
				return
			}

			delay := m.Policy.DelayFor(attempt)
			m.Events.Emit("retry", map[string]any{
				"attempt": attempt, "delay": delay.Seconds(), "reason": err.Error()})

			select {
			case <-time.After(delay):
			case <-ctx.Done():
				errs <- ctx.Err()
				return
			}
		}
		errs <- fmt.Errorf("重试 %d 次都失败了", m.Policy.MaxAttempts)
	}()

	return out, errs
}
```

**那个「已经吐出内容就不重试」值得单说**：流式接口一旦开始吐内容，重试就不再安全——
用户已经看到半句话了，重来一遍会变成「半句话 + 完整的话」。
真正健壮的做法是断点续传，但那需要服务端支持，多数服务没有。

```go
// file: main.go
package main

import (
	"context"
	"errors"
	"fmt"
	"net"
	"time"
)

// FlakyModel 前 n 次连接失败，之后正常。用来验证重试逻辑，不联网。
type FlakyModel struct {
	FailTimes int
	calls     int
}

func (m *FlakyModel) Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta, 4)
	errs := make(chan error, 1)
	m.calls++
	failing := m.calls <= m.FailTimes
	go func() {
		defer close(deltas)
		defer close(errs)
		if failing {
			// net.OpError 满足 net.Error 接口，正好模拟「连不上」。
			errs <- fmt.Errorf("请求失败：%w", &net.OpError{
				Op: "dial", Net: "tcp", Err: errors.New("connection refused")})
			return
		}
		deltas <- Delta{Kind: "text", Text: "终于连上了。"}
		errs <- nil
	}()
	return deltas, errs
}

// BadRequestModel 永远返回 400。不该被重试。
type BadRequestModel struct{ calls int }

func (m *BadRequestModel) Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta)
	errs := make(chan error, 1)
	m.calls++
	close(deltas)
	errs <- &StatusError{Code: 400, Body: "Bad Request"}
	close(errs)
	return deltas, errs
}

func main() {
	events := NewEventStream()
	events.Subscribe(func(e Event) {
		if e.Kind == "retry" {
			fmt.Printf("  第 %v 次失败，%.2f 秒后重试（%v）\n",
				e.Data["attempt"], e.Data["delay"], e.Data["reason"])
		}
	})

	policy := DefaultRetryPolicy()
	policy.MaxAttempts = 4

	start := time.Now()
	model := NewRetryingModel(&FlakyModel{FailTimes: 2}, policy, events)
	deltas, errs := model.Stream(context.Background(), nil, nil)
	text := ""
	for d := range deltas {
		text += d.Text
	}
	fmt.Printf("结果：%s（err=%v）\n", text, <-errs)
	fmt.Printf("总共花了 %.1f 秒\n\n", time.Since(start).Seconds())

	bad := &BadRequestModel{}
	d2, e2 := NewRetryingModel(bad, policy, events).Stream(context.Background(), nil, nil)
	for range d2 {
	}
	fmt.Printf("400 错误：调了 %d 次内层模型，错误是 %v\n", bad.calls, <-e2)
	fmt.Println("→ 一次就放弃，重试它没有意义")
}
```

## 第二层：模型在犯病

这三种病症的共同点是**程序看起来一切正常**：没有 error，没有 panic，
只是循环一直转下去、或者永远得不到结果。

```go
// file: guard.go
package main

import (
	"encoding/json"
	"fmt"
)

// Symptom 一次诊断结果。Nudge 是要注入给模型的纠正提示。
type Symptom struct {
	Name   string
	Nudge  string
	GiveUp bool
}

// BehaviorGuard 看着模型的行为，发现它在犯病就干预。
//
// 三种病症：
//  1. 参数反复填错——同一个工具连续填错 N 次，把这个工具摘掉。
//  2. 空回复——既没说话也没调工具。这会让循环空转到预算耗尽。
//  3. 原地打转——反复用**完全相同的参数**调**同一个工具**。
type BehaviorGuard struct {
	MaxInvalidPerTool int
	MaxRepeat         int
	MaxEmpty          int

	invalidCounts map[string]int
	callCounts    map[string]int
	emptyReplies  int
	Disabled      map[string]bool
}

func NewBehaviorGuard() *BehaviorGuard {
	return &BehaviorGuard{
		MaxInvalidPerTool: 2, MaxRepeat: 3, MaxEmpty: 2,
		invalidCounts: map[string]int{}, callCounts: map[string]int{},
		Disabled: map[string]bool{},
	}
}

// fingerprint 一次调用的指纹。
//
// Go 这里比 Python 省一件事：encoding/json 序列化 map 时**键是自动排序的**，
// 所以 {"a":1,"b":2} 和 {"b":2,"a":1} 天然得到同一个字符串。
// Python 那边要显式写 sort_keys=True。
func fingerprint(call ToolUseBlock) string {
	raw, err := json.Marshal(call.Args)
	if err != nil {
		raw = []byte(fmt.Sprint(call.Args))
	}
	return call.Name + "|" + string(raw)
}

func (g *BehaviorGuard) CheckReply(reply Message) *Symptom {
	calls := reply.ToolUses()

	// 病症 2：空回复
	if len(calls) == 0 && reply.Text() == "" {
		g.emptyReplies++
		if g.emptyReplies >= g.MaxEmpty {
			return &Symptom{Name: "空回复", GiveUp: true}
		}
		return &Symptom{Name: "空回复", Nudge: "你刚才什么都没回。请直接给出答案，或者调用一个工具。"}
	}
	g.emptyReplies = 0

	// 病症 3：原地打转
	for _, call := range calls {
		fp := fingerprint(call)
		g.callCounts[fp]++
		if g.callCounts[fp] >= g.MaxRepeat {
			return &Symptom{Name: "原地打转", Nudge: fmt.Sprintf(
				"你已经用完全相同的参数调用了 %d 次 %s，结果不会变。"+
					"请换一个思路，或者基于已有的结果直接回答。", g.MaxRepeat, call.Name)}
		}
	}
	return nil
}

func (g *BehaviorGuard) CheckToolResult(call ToolUseBlock, result ToolResult) *Symptom {
	// 病症 1：参数反复填错
	if !result.IsInvalidArgs {
		delete(g.invalidCounts, call.Name) // 填对一次就清零
		return nil
	}
	g.invalidCounts[call.Name]++
	n := g.invalidCounts[call.Name]
	if n >= g.MaxInvalidPerTool {
		g.Disabled[call.Name] = true
		return &Symptom{Name: "参数反复填错", Nudge: fmt.Sprintf(
			"%s 的参数你连续填错了 %d 次，我把这个工具停掉了。"+
				"请用别的办法，或者直接告诉用户你做不到。", call.Name, n)}
	}
	return nil
}
```

**三个设计要点**：

- **纠正提示要具体**。「参数错误」模型改不动；「你已经用相同参数调了 3 次 get_time，结果不会变」
  它才知道该换招。这些话是写给模型看的，标准和写给人看的一样：说清楚发生了什么、该怎么办。
- **摘掉工具比一直纠正有效**。工具从 `tools` 列表里消失，模型就再也调不到它了——
  这比反复劝说可靠得多。**能用机制解决的，不要靠提示词解决。**
- **计数要能清零**。填对一次就把计数清掉，否则一次偶然的错误会一直记在账上。

## 装到主循环里

```go
// file: loop.go
package main

import "context"

type GuardedAgentLoop struct {
	Model     Model
	Registry  *ToolRegistry
	Events    *EventStream
	MaxRounds int
}

func NewGuardedAgentLoop(model Model, reg *ToolRegistry, events *EventStream) *GuardedAgentLoop {
	if events == nil {
		events = NewEventStream()
	}
	return &GuardedAgentLoop{
		Model:     NewRetryingModel(model, DefaultRetryPolicy(), events),
		Registry:  reg,
		Events:    events,
		MaxRounds: 8,
	}
}

func (l *GuardedAgentLoop) Run(ctx context.Context, history []Message) TurnResult {
	guard := NewBehaviorGuard() // 每一轮对话一个新的守卫，计数不跨轮累积
	var newMessages []Message

	for round := 1; round <= l.MaxRounds; round++ {
		reply, err := l.oneRound(ctx, append(append([]Message{}, history...), newMessages...), guard.Disabled)
		if err != nil {
			l.Events.Emit("error", map[string]any{"message": err.Error()})
			return TurnResult{StopReason: StopError, RoundsUsed: round, Messages: newMessages, Err: err}
		}
		newMessages = append(newMessages, reply)

		symptom := guard.CheckReply(reply)
		if symptom != nil {
			l.Events.Emit("symptom", map[string]any{"name": symptom.Name, "give_up": symptom.GiveUp})
			if symptom.GiveUp {
				return TurnResult{Text: reply.Text(), StopReason: StopGaveUp,
					RoundsUsed: round, Messages: newMessages}
			}
		}

		calls := reply.ToolUses()
		if len(calls) == 0 {
			if symptom == nil {
				return TurnResult{Text: reply.Text(), StopReason: StopCompleted,
					RoundsUsed: round, Messages: newMessages}
			}
			// 空回复：注入纠正提示，再来一轮
			newMessages = append(newMessages,
				NewMessage("user", TextBlock{Text: symptom.Nudge}))
			continue
		}

		var nudges []string
		if symptom != nil && symptom.Nudge != "" {
			nudges = append(nudges, symptom.Nudge)
		}
		var results []Block
		for _, call := range calls {
			r := l.Registry.Execute(ctx, call.Name, call.Args)
			l.Events.Emit("tool_end", map[string]any{
				"name": call.Name, "ok": !r.IsError, "output": r.Content})
			results = append(results, ToolResultBlock{
				ToolUseID: call.ID, Content: r.Content, IsError: r.IsError})
			if s := guard.CheckToolResult(call, r); s != nil {
				l.Events.Emit("symptom", map[string]any{"name": s.Name, "tool": call.Name})
				nudges = append(nudges, s.Nudge)
			}
		}
		newMessages = append(newMessages, NewMessage("tool", results...))

		// 纠正提示用 user 角色发。
		// 用 assistant 角色的话，模型会以为这是自己说的，纠正效果大打折扣；
		// 用 system 角色则很多服务只认开头那一条。
		if len(nudges) > 0 {
			text := ""
			for i, n := range nudges {
				if i > 0 {
					text += "\n"
				}
				text += n
			}
			newMessages = append(newMessages, NewMessage("user", TextBlock{Text: text}))
		}
	}

	return TurnResult{StopReason: StopBudget, RoundsUsed: l.MaxRounds, Messages: newMessages}
}

func (l *GuardedAgentLoop) oneRound(ctx context.Context, messages []Message, disabled map[string]bool) (Message, error) {
	deltas, errs := l.Model.Stream(ctx, messages, l.Registry.SpecsExcept(disabled))
	var textParts []string
	var blocks []Block
	flushText := func() {
		if len(textParts) > 0 {
			joined := ""
			for _, p := range textParts {
				joined += p
			}
			blocks = append(blocks, TextBlock{Text: joined})
			textParts = nil
		}
	}
	for d := range deltas {
		switch d.Kind {
		case "text":
			textParts = append(textParts, d.Text)
			l.Events.Emit("text_delta", map[string]any{"delta": d.Text})
		case "tool_use":
			if d.Tool != nil {
				flushText()
				blocks = append(blocks, *d.Tool)
			}
		}
	}
	flushText()
	if err := <-errs; err != nil {
		return Message{}, err
	}
	return NewMessage("assistant", blocks...), nil
}
```

## 让它犯病给你看

真模型不一定配合演出。用剧本模型精确复现三种病症：

```go
// file: main.go
package main

import (
	"context"
	"fmt"
)

type scriptTurn struct {
	text  string
	calls []ToolUseBlock
}

// ScriptedModel 按剧本出牌。RepeatLast=true 时剧本演完就一直重复最后一幕——
// 正好用来演「原地打转」。
type ScriptedModel struct {
	Script     []scriptTurn
	RepeatLast bool
	calls      int
}

func (m *ScriptedModel) Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta, 8)
	errs := make(chan error, 1)
	i := m.calls
	m.calls++
	if m.RepeatLast && i >= len(m.Script) {
		i = len(m.Script) - 1
	}
	go func() {
		defer close(deltas)
		defer close(errs)
		if i >= len(m.Script) {
			errs <- fmt.Errorf("剧本演完了")
			return
		}
		turn := m.Script[i]
		if turn.text != "" {
			deltas <- Delta{Kind: "text", Text: turn.text}
		}
		for _, c := range turn.calls {
			call := c
			deltas <- Delta{Kind: "tool_use", Tool: &call}
		}
		errs <- nil
	}()
	return deltas, errs
}

func watch(s *EventStream) {
	s.Subscribe(func(e Event) {
		if e.Kind != "symptom" {
			return
		}
		tail := ""
		if give, _ := e.Data["give_up"].(bool); give {
			tail = "，放弃"
		}
		fmt.Printf("  ⚠ 诊断出「%v」%s\n", e.Data["name"], tail)
	})
}

func main() {
	ctx := context.Background()
	reg := NewToolRegistry(GetTimeTool{})

	fmt.Println("── 病症 1：参数反复填错 ──")
	s := NewEventStream()
	watch(s)
	loop := NewGuardedAgentLoop(&ScriptedModel{Script: []scriptTurn{
		{text: "我查一下。", calls: []ToolUseBlock{NewToolUse("get_time", map[string]any{"town": "北京"})}},
		{text: "再试试。", calls: []ToolUseBlock{NewToolUse("get_time", map[string]any{"place": "北京"})}},
		{text: "算了，我直接说：现在是下午。"},
	}}, reg, s)
	r := loop.Run(ctx, []Message{NewMessage("user", TextBlock{Text: "北京几点"})})
	fmt.Printf("  结果：%s，答案「%s」\n\n", r.StopReason, r.Text)

	fmt.Println("── 病症 2：空回复 ──")
	s = NewEventStream()
	watch(s)
	loop = NewGuardedAgentLoop(&ScriptedModel{Script: []scriptTurn{{}, {}, {}}}, reg, s)
	r = loop.Run(ctx, []Message{NewMessage("user", TextBlock{Text: "你好"})})
	fmt.Printf("  结果：%s\n\n", r.StopReason)

	fmt.Println("── 病症 3：原地打转 ──")
	s = NewEventStream()
	watch(s)
	loop = NewGuardedAgentLoop(&ScriptedModel{RepeatLast: true, Script: []scriptTurn{
		{text: "查一下。", calls: []ToolUseBlock{NewToolUse("get_time", map[string]any{"city": "北京"})}},
	}}, reg, s)
	r = loop.Run(ctx, []Message{NewMessage("user", TextBlock{Text: "北京几点"})})
	fmt.Printf("  结果：%s，用了 %d 轮\n", r.StopReason, r.RoundsUsed)
	for _, m := range r.Messages {
		if m.Role == "user" {
			fmt.Printf("  注入的纠正提示：「%s」\n", m.Text())
			break
		}
	}
}
```

## 本章小结

- 错误分三层：**传输层重试、工具层变成结果、模型行为层干预**。
- 让错误**携带结构化信息**（`*StatusError` 而不是拼字符串），调用方才能判断该不该重试。
- 包错误用 `%w` 不用 `%v`，否则 `errors.As` 的链条断了，重试判断无从做起。
- 只重试「重试有用」的错误：连接失败、超时、429、5xx。400 重试一万次还是 400。
  用户主动取消的**绝不重试**。
- 退避要**指数 + 抖动**。没有抖动，一群被限流的客户端会同时回来，再把服务打垮。
- 流式接口**一旦开始吐内容就不能重试**了——用户已经看到半句话。
- 重试做成**包装类型**：它自己也满足 `Model` 接口，对调用方完全透明。
- 模型犯病不返回 error 也不 panic，程序看起来一切正常，只是永远得不到结果。
  三种常见病症：参数反复填错、空回复、原地打转。
- 打转指纹靠 `json.Marshal`——Go 序列化 map 时键自动排序，天然稳定。
- **能用机制解决的不要靠提示词**：把工具从列表里摘掉，比反复劝说可靠得多。
- 纠正提示用 **user 角色**注入，而且要**说清楚发生了什么、该怎么办**。

第三部分到这里结束。我们的 agent 现在有了心脏：能循环、能并发、能停下、能扛住模型犯病。
下一部分给它装上记忆——会话怎么存、怎么分叉、上下文满了怎么办。
