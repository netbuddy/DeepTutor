# 流式解析

上一章那次请求是「发出去，等它把话说完，一次性拿回来」。真实产品不能这样——
模型答一段长文要几十秒，用户盯着空白页面。

这一章把它改成流式：**边吐边收**。

## 服务端推送长什么样

OpenAI 兼容接口的流式响应是一串文本行：

```
data: {"choices":[{"delta":{"role":"assistant"}}]}

data: {"choices":[{"delta":{"content":"北京"}}]}

data: {"choices":[{"delta":{"content":"现在"}}]}

data: [DONE]
```

规矩：

- 每个事件之间用**空行**分隔
- 最后一行是 `data: [DONE]`，它**不是 JSON**，要单独判断
- 每个 `delta` 只带**变化的部分**，不是完整内容

纯文字很好处理。麻烦在工具调用。

## 工具调用是分片来的

模型要调工具时，`tool_calls` 也是一点点吐出来的，而且**参数是按字符流分片**的：

```
data: {"choices":[{"delta":{"tool_calls":[{"index":0,"id":"call_x","function":{"name":"get_time","arguments":""}}]}}]}
data: {"choices":[{"delta":{"tool_calls":[{"index":0,"function":{"arguments":"{\"ci"}}]}}]}
data: {"choices":[{"delta":{"tool_calls":[{"index":0,"function":{"arguments":"ty\": \"北"}}]}}]}
data: {"choices":[{"delta":{"tool_calls":[{"index":0,"function":{"arguments":"京\"}"}}]}}]}
```

三件事要注意：

1. **`index` 才是身份标识**，不是 `id`。同一轮里调三个工具就是 index 0、1、2，
   而 `id` 和 `name` 通常只在第一片里出现。
2. **`arguments` 要按 index 拼接**，拼完才是完整的 JSON 文本。
3. **拼到一半的 JSON 是不合法的**，只能等这个工具调用结束了才解析。

这是流式解析里最容易写错的地方。写成「每来一片就解析一次」必然失败。

## 本章起点

```go
// file: blocks.go
package main

import (
	"crypto/rand"
	"encoding/hex"
	"encoding/json"
)

func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)
}

type Block interface{ BlockType() string }

type TextBlock struct{ Text string }

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string
	Name string
	Args map[string]any
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string
	Content   string
	IsError   bool
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

type Message struct {
	ID      string
	Role    string
	Content []Block
}

func (m Message) Text() string {
	out := ""
	for _, b := range m.Content {
		if t, ok := b.(TextBlock); ok {
			out += t.Text
		}
	}
	return out
}

func (m Message) ToolUses() []ToolUseBlock {
	var calls []ToolUseBlock
	for _, b := range m.Content {
		if c, ok := b.(ToolUseBlock); ok {
			calls = append(calls, c)
		}
	}
	return calls
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks}
}

type ToolSpec struct {
	Name        string
	Description string
	Parameters  map[string]any
}

type Delta struct {
	Kind string // "text" 或 "tool_use"
	Text string
	Tool *ToolUseBlock
}

// ── 上一章的线上格式与翻译 ──

type wireFunction struct {
	Name      string `json:"name"`
	Arguments string `json:"arguments"`
}

type wireToolCall struct {
	Index    int          `json:"index,omitempty"`
	ID       string       `json:"id,omitempty"`
	Type     string       `json:"type,omitempty"`
	Function wireFunction `json:"function"`
}

type wireMessage struct {
	Role       string         `json:"role"`
	Content    *string        `json:"content"`
	ToolCallID string         `json:"tool_call_id,omitempty"`
	ToolCalls  []wireToolCall `json:"tool_calls,omitempty"`
}

type wireToolDef struct {
	Type     string `json:"type"`
	Function struct {
		Name        string         `json:"name"`
		Description string         `json:"description"`
		Parameters  map[string]any `json:"parameters"`
	} `json:"function"`
}

type wireRequest struct {
	Model    string        `json:"model"`
	Messages []wireMessage `json:"messages"`
	Tools    []wireToolDef `json:"tools,omitempty"`
	Stream   bool          `json:"stream,omitempty"`
}

type wireChoice struct {
	Delta        wireMessage `json:"delta"`
	Message      wireMessage `json:"message"`
	FinishReason string      `json:"finish_reason"`
}

type wireResponse struct {
	Choices []wireChoice `json:"choices"`
}

func toWireMessages(messages []Message) []wireMessage {
	var out []wireMessage
	for _, m := range messages {
		if m.Role == "tool" {
			for _, b := range m.Content {
				if r, ok := b.(ToolResultBlock); ok {
					content := r.Content
					out = append(out, wireMessage{Role: "tool", Content: &content, ToolCallID: r.ToolUseID})
				}
			}
			continue
		}
		text, calls := m.Text(), m.ToolUses()
		wm := wireMessage{Role: m.Role}
		if text != "" {
			wm.Content = &text
		}
		for _, c := range calls {
			args, err := json.Marshal(c.Args)
			if err != nil {
				args = []byte("{}")
			}
			wm.ToolCalls = append(wm.ToolCalls, wireToolCall{
				ID: c.ID, Type: "function",
				Function: wireFunction{Name: c.Name, Arguments: string(args)},
			})
		}
		out = append(out, wm)
	}
	return out
}

func toWireTools(specs []ToolSpec) []wireToolDef {
	var out []wireToolDef
	for _, s := range specs {
		var d wireToolDef
		d.Type = "function"
		d.Function.Name = s.Name
		d.Function.Description = s.Description
		if s.Parameters == nil {
			d.Function.Parameters = map[string]any{"type": "object", "properties": map[string]any{}}
		} else {
			d.Function.Parameters = s.Parameters
		}
		out = append(out, d)
	}
	return out
}

func parseArguments(raw string) map[string]any {
	if raw == "" {
		return map[string]any{}
	}
	var args map[string]any
	if err := json.Unmarshal([]byte(raw), &args); err != nil {
		return map[string]any{"_invalid_json": raw}
	}
	return args
}
```

## 分片累积器

先把「怎么把分片拼成完整工具调用」这件事单独写出来，它值得一个专门的类型。

```go
// file: accumulator.go
package main

import "sort"

// toolCallAccumulator 按 index 把分片攒成完整的工具调用。
//
// 为什么单独写一个类型：分片逻辑有状态（哪些 index 出现过、各自攒了多少），
// 混在解析循环里会让那个循环很难读，也很难单独测试。
type toolCallAccumulator struct {
	slots map[int]*toolCallSlot
}

type toolCallSlot struct {
	id        string
	name      string
	arguments string
}

func newToolCallAccumulator() *toolCallAccumulator {
	return &toolCallAccumulator{slots: map[int]*toolCallSlot{}}
}

func (a *toolCallAccumulator) Feed(calls []wireToolCall) {
	for _, item := range calls {
		slot, ok := a.slots[item.Index]
		if !ok {
			slot = &toolCallSlot{}
			a.slots[item.Index] = slot
		}
		// id 和 name 通常只在第一片出现，来了就记下，别被后面的空值覆盖
		if item.ID != "" {
			slot.id = item.ID
		}
		if item.Function.Name != "" {
			slot.name = item.Function.Name
		}
		if item.Function.Arguments != "" {
			slot.arguments += item.Function.Arguments // 关键：累加，不是覆盖
		}
	}
}

// Finish 全部收完之后再解析。中途的 arguments 是残缺的 JSON，解析必然失败。
func (a *toolCallAccumulator) Finish() []ToolUseBlock {
	indexes := make([]int, 0, len(a.slots))
	for i := range a.slots {
		indexes = append(indexes, i)
	}
	sort.Ints(indexes) // 按 index 排序，保证工具调用的顺序与模型给的一致

	var blocks []ToolUseBlock
	for _, i := range indexes {
		slot := a.slots[i]
		id := slot.id
		if id == "" {
			id = newID("call")
		}
		blocks = append(blocks, ToolUseBlock{
			ID: id, Name: slot.name, Args: parseArguments(slot.arguments),
		})
	}
	return blocks
}
```

	acc.Feed([]wireToolCall{{Index: 0, ID: "call_x", Function: wireFunction{Name: "get_time", Arguments: ""}}})
	acc.Feed([]wireToolCall{{Index: 0, Function: wireFunction{Arguments: `{"ci`}}})
	acc.Feed([]wireToolCall{{Index: 0, Function: wireFunction{Arguments: `ty": "北`}}})
	acc.Feed([]wireToolCall{{Index: 0, Function: wireFunction{Arguments: `京"}`}}})

	for _, c := range acc.Finish() {
		fmt.Printf("拼出来的工具调用：%s(%v)  编号 %s\n", c.Name, c.Args, c.ID)
	}
}

## 解析推送的行

```go
// file: sse.go
package main

import (
	"encoding/json"
	"strings"
)

// parseSSELine 把一行 `data: {...}` 解析成响应结构。
//
// 第二个返回值表示这一行要不要处理：空行（事件分隔）、注释行、结束标记都返回 false。
func parseSSELine(line string) (wireResponse, bool) {
	line = strings.TrimSpace(line)
	if line == "" || strings.HasPrefix(line, ":") || !strings.HasPrefix(line, "data:") {
		return wireResponse{}, false
	}
	payload := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
	if payload == "[DONE]" { // 结束标记，它不是 JSON
		return wireResponse{}, false
	}
	var parsed wireResponse
	if err := json.Unmarshal([]byte(payload), &parsed); err != nil {
		return wireResponse{}, false // 半行或者服务端抽风，跳过
	}
	return parsed, true
}
```

## 流式请求

```go
// file: stream.go
package main

import (
	"bufio"
	"bytes"
	"encoding/json"
	"fmt"
	"net/http"
	"time"
)

const (
	baseURL   = "http://192.168.213.116:8084/v1"
	modelName = "qwen3.6-27B"
)

// streamModel 一次流式请求。增量走一个通道，错误走另一个——
// 与第 5 章那个假模型的接口完全一致。
func streamModel(messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta, 32)
	errs := make(chan error, 1)

	go func() {
		defer close(deltas)
		defer close(errs)

		body := wireRequest{
			Model:    modelName,
			Messages: toWireMessages(messages),
			Tools:    toWireTools(tools),
			Stream:   true,
		}
		raw, err := json.Marshal(body)
		if err != nil {
			errs <- fmt.Errorf("请求体序列化失败：%w", err)
			return
		}

		req, err := http.NewRequest("POST", baseURL+"/chat/completions", bytes.NewReader(raw))
		if err != nil {
			errs <- fmt.Errorf("构造请求失败：%w", err)
			return
		}
		req.Header.Set("Content-Type", "application/json")

		client := &http.Client{Timeout: 180 * time.Second}
		resp, err := client.Do(req)
		if err != nil {
			errs <- fmt.Errorf("请求失败：%w", err)
			return
		}
		defer resp.Body.Close()

		if resp.StatusCode != http.StatusOK {
			buf := make([]byte, 300)
			n, _ := resp.Body.Read(buf)
			errs <- fmt.Errorf("模型服务返回 %d：%s", resp.StatusCode, string(buf[:n]))
			return
		}

		acc := newToolCallAccumulator()
		scanner := bufio.NewScanner(resp.Body)
		// 单个事件可能很长（一次吐出一大段文字），默认 64 KB 会不够
		scanner.Buffer(make([]byte, 0, 64*1024), 4*1024*1024)

		for scanner.Scan() {
			chunk, ok := parseSSELine(scanner.Text())
			if !ok || len(chunk.Choices) == 0 {
				continue
			}
			delta := chunk.Choices[0].Delta
			if delta.Content != nil && *delta.Content != "" {
				deltas <- Delta{Kind: "text", Text: *delta.Content}
			}
			if len(delta.ToolCalls) > 0 {
				acc.Feed(delta.ToolCalls) // 只攒，不吐
			}
		}
		if err := scanner.Err(); err != nil {
			errs <- fmt.Errorf("读流中断：%w", err)
			return
		}

		// 流结束了，这时工具调用才是完整的
		for _, block := range acc.Finish() {
			b := block
			deltas <- Delta{Kind: "tool_use", Tool: &b}
		}
		errs <- nil
	}()

	return deltas, errs
}

// Collect 把一串增量攒成一条完整消息。这个函数第 5 章就写好了，一个字没改。
func Collect(deltas <-chan Delta, errs <-chan error) (Message, error) {
	var textParts []string
	var blocks []Block

	flushText := func() {
		if len(textParts) > 0 {
			joined := ""
			for _, p := range textParts {
				joined += p
			}
			blocks = append(blocks, TextBlock{Text: joined})
			textParts = nil
		}
	}

	for d := range deltas {
		switch d.Kind {
		case "text":
			textParts = append(textParts, d.Text)
		case "tool_use":
			if d.Tool != nil {
				flushText()
				blocks = append(blocks, *d.Tool)
			}
		}
	}
	flushText()

	if err := <-errs; err != nil {
		return Message{}, err
	}
	return NewMessage("assistant", blocks...), nil
}
```

## 打字机效果

```go
// file: main.go
package main

import "fmt"

func main() {
	deltas, errs := streamModel([]Message{
		NewMessage("system", TextBlock{Text: "你是一个简洁的助教。"}),
		NewMessage("user", TextBlock{Text: "用两句话说明什么是闭包。"}),
	}, nil)

	fmt.Print("模型：")
	pieces := 0
	for d := range deltas {
		if d.Kind == "text" {
			pieces++
			fmt.Print(d.Text)
		}
	}
	if err := <-errs; err != nil {
		fmt.Println("\n出错：", err)
		return
	}
	fmt.Printf("\n\n（一共收到 %d 个文字增量）\n", pieces)
}
```

**这就是所有聊天界面打字机效果的全部原理**——没有别的魔法。

## 流式的工具调用

```go
// file: main.go
package main

import "fmt"

var timeTool = ToolSpec{
	Name:        "get_time",
	Description: "查询某个城市的当前时间",
	Parameters: map[string]any{
		"type": "object",
		"properties": map[string]any{
			"city": map[string]any{"type": "string", "description": "城市名"},
		},
		"required": []string{"city"},
	},
}

func main() {
	msg, err := Collect(streamModel([]Message{
		NewMessage("system", TextBlock{Text: "需要查时间时就调用工具，不要自己编。"}),
		NewMessage("user", TextBlock{Text: "北京现在几点？"}),
	}, []ToolSpec{timeTool}))
	if err != nil {
		fmt.Println("出错：", err)
		return
	}

	fmt.Println("文字：", msg.Text())
	for _, c := range msg.ToolUses() {
		_, bad := c.Args["_invalid_json"]
		fmt.Printf("工具调用：%s(%v)  参数解析成功：%v\n", c.Name, c.Args, !bad)
	}
	if len(msg.ToolUses()) == 0 {
		fmt.Println("这次模型没调工具——小模型有时会自己编答案，第 12 章会讲怎么应对。")
	}
}
```

注意 `Collect` 这个函数——**第 5 章为假模型写的，这里一个字没改就用在真模型上**。
这就是第 5 章那个接口设计的回报。

## 为什么工具调用要等流结束才吐

有人会想：拼到能解析的时候就吐出去，不是更早吗？

不行。`{"city": "北` 这样的残缺文本解析必然失败；而 `{"city": "北京"}` 和
`{"city": "北京", "tz": "…"}` 的前缀是一样的，你没法判断它到底完了没有。
唯一可靠的结束信号是**流本身结束**。

代价是工具调用比文字晚一点出现。真实产品里可以在收到第一片时就发一个「模型准备调用
工具」的提示事件，让界面先有反应。

## 本章小结

- 服务端推送就是一串 `data: {...}` 行，**`[DONE]` 不是 JSON**，要单独判断。
- 文字增量直接吐；**工具调用必须攒**。
- 分片的身份是 **`index` 不是 `id`**，`id` 和 `name` 通常只在第一片出现。
- `arguments` 是**按字符流分片的 JSON 文本**，要累加，**拼完才能解析**。
- `bufio.Scanner` 的默认单行上限 64 KB **对流式响应不够**，要手工调大。
- 第 5 章的 `Collect` 一个字没改就用上了。
