# 工具契约

前面几章的工具是一个字典：`{"get_time": get_time}`。模型传什么参数就直接展开调用。
这在演示里够用，真跑起来第一天就会崩：模型传了不存在的参数、少传了必填的、
类型不对、或者干脆传了一段自然语言。

这一章给工具定一份**契约**：怎么声明、参数怎么校验、结果怎么回。

## 一个工具要说清三件事

| 说什么 | 给谁看 | 不说清会怎样 |
|---|---|---|
| 名字和用途 | 模型 | 模型不知道什么时候该用它 |
| 参数的结构 | 模型 + 我们的校验代码 | 模型乱传，我们的代码崩 |
| 怎么执行 | 我们 | —— |

参数结构用 **JSON Schema** 描述。不是因为它优雅，是因为**模型接口就要这个格式**——
我们发过去的工具声明里，`parameters` 字段就是一份 JSON Schema。同一份声明既给模型看，
也用来校验它的回答，两边不会跑偏。

```python
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Callable, Iterator, Literal, Protocol
import json, urllib.request, uuid

def new_id(prefix: str) -> str:
    return f"{prefix}_{uuid.uuid4().hex[:12]}"

@dataclass
class TextBlock:
    text: str
    type: Literal["text"] = "text"

@dataclass
class ToolUseBlock:
    name: str
    args: dict[str, Any]
    id: str = field(default_factory=lambda: new_id("call"))
    type: Literal["tool_use"] = "tool_use"

@dataclass
class ToolResultBlock:
    tool_use_id: str
    content: str
    is_error: bool = False
    type: Literal["tool_result"] = "tool_result"

@dataclass
class Message:
    role: Literal["system", "user", "assistant", "tool"]
    content: list[Any]
    id: str = field(default_factory=lambda: new_id("msg"))
    def text(self) -> str:
        return "".join(b.text for b in self.content if isinstance(b, TextBlock))
    def tool_uses(self) -> list[ToolUseBlock]:
        return [b for b in self.content if isinstance(b, ToolUseBlock)]

@dataclass
class Delta:
    kind: Literal["text", "tool_use"]
    text: str = ""
    tool: ToolUseBlock | None = None

class Model(Protocol):
    def stream(self, messages: list[Message], tools: list[dict]) -> Iterator[Delta]: ...

def collect(deltas: Iterator[Delta]) -> Message:
    text_parts, blocks = [], []
    for d in deltas:
        if d.kind == "text":
            text_parts.append(d.text)
        elif d.tool is not None:
            if text_parts:
                blocks.append(TextBlock(text="".join(text_parts))); text_parts = []
            blocks.append(d.tool)
    if text_parts:
        blocks.append(TextBlock(text="".join(text_parts)))
    return Message(role="assistant", content=blocks)

def to_openai_messages(messages: list[Message]) -> list[dict]:
    out = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({"role": "assistant", "content": text or None, "tool_calls": [
                {"id": c.id, "type": "function",
                 "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                for c in calls]})
        else:
            out.append({"role": m.role, "content": text})
    return out

def to_openai_tools(tools: list[dict]) -> list[dict]:
    return [{"type": "function", "function": {
        "name": t["name"], "description": t.get("description", ""),
        "parameters": t.get("parameters", {"type": "object", "properties": {}})}} for t in tools]

class ToolCallAccumulator:
    def __init__(self): self._slots = {}
    def feed(self, raw_calls):
        for item in raw_calls:
            slot = self._slots.setdefault(int(item.get("index", 0)), {"id": "", "name": "", "arguments": ""})
            if item.get("id"): slot["id"] = item["id"]
            fn = item.get("function") or {}
            if fn.get("name"): slot["name"] = fn["name"]
            if fn.get("arguments"): slot["arguments"] += fn["arguments"]
    def finish(self):
        blocks = []
        for i in sorted(self._slots):
            s = self._slots[i]
            raw = s["arguments"] or "{}"
            try:
                args = json.loads(raw)
                if not isinstance(args, dict): args = {"_raw": raw}
            except json.JSONDecodeError:
                args = {"_invalid_json": raw}
            blocks.append(ToolUseBlock(id=s["id"] or new_id("call"), name=s["name"], args=args))
        return blocks

def parse_sse_line(line: str):
    line = line.strip()
    if not line or line.startswith(":") or not line.startswith("data:"): return None
    payload = line[5:].strip()
    if payload == "[DONE]": return None
    try: return json.loads(payload)
    except json.JSONDecodeError: return None

class OpenAICompatModel:
    def __init__(self, base_url: str, model: str, timeout: int = 180):
        self.base_url, self.model, self.timeout = base_url.rstrip("/"), model, timeout
    def stream(self, messages, tools):
        body = {"model": self.model, "messages": to_openai_messages(messages), "stream": True}
        if tools: body["tools"] = to_openai_tools(tools)
        req = urllib.request.Request(f"{self.base_url}/chat/completions",
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"}, method="POST")
        acc = ToolCallAccumulator()
        with urllib.request.urlopen(req, timeout=self.timeout) as resp:
            for raw in resp:
                chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
                if chunk is None: continue
                delta = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
                if delta.get("content"): yield Delta(kind="text", text=delta["content"])
                if delta.get("tool_calls"): acc.feed(delta["tool_calls"])
        for b in acc.finish(): yield Delta(kind="tool_use", tool=b)

real_model = OpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
print("前八章的成果已就位")
```

## 工具的定义

```python
@dataclass
class ToolResult:
    """一次工具执行的结果。

    分成三种情况而不是两种：成功、失败、以及**参数不合格**。
    第三种要单独区分，因为它的处理方式不同——参数错了应该把具体哪里错了告诉模型，
    让它重试；而工具执行失败通常是环境问题，重试也没用。
    """
    content: str
    is_error: bool = False
    is_invalid_args: bool = False


@dataclass
class Tool:
    name: str
    description: str
    parameters: dict[str, Any]              # JSON Schema
    run: Callable[..., str]

    def spec(self) -> dict:
        """给模型看的声明。同一份 parameters 也用于校验，两边不会跑偏。"""
        return {"name": self.name, "description": self.description, "parameters": self.parameters}


class ToolRegistry:
    """工具的登记处。

    存在的意义不只是"一个字典"：它还负责校验参数、把执行结果包成统一的形状、
    以及把任何异常挡在主循环之外。主循环不该关心某个工具会不会抛异常。
    """

    def __init__(self, tools: list[Tool] | None = None) -> None:
        self._tools: dict[str, Tool] = {t.name: t for t in (tools or [])}

    def add(self, tool: Tool) -> None:
        self._tools[tool.name] = tool

    def specs(self) -> list[dict]:
        return [t.spec() for t in self._tools.values()]

    def names(self) -> list[str]:
        return list(self._tools)

    def execute(self, name: str, args: dict[str, Any]) -> ToolResult:
        tool = self._tools.get(name)
        if tool is None:
            # 不存在的工具也是"结果"，不是异常。模型看到这条会改用别的办法。
            return ToolResult(
                content=f"没有叫 {name} 的工具。可用的工具：{'、'.join(self.names())}",
                is_error=True,
            )

        problem = validate_args(args, tool.parameters)
        if problem:
            return ToolResult(content=f"参数不合格：{problem}", is_error=True, is_invalid_args=True)

        try:
            return ToolResult(content=str(tool.run(**args)))
        except Exception as exc:
            # 工具的异常绝不能冒泡到主循环——一个写坏的工具不该让整轮对话崩掉。
            # 把异常变成结果，模型看到之后可以换个参数重试或者换条路。
            return ToolResult(content=f"{type(exc).__name__}: {exc}", is_error=True)


print("工具契约定义好了")
```

## 参数校验

不引入第三方库——JSON Schema 的完整实现很复杂，但**工具参数用到的那一小部分**
自己写二十行就够，而且看得见它在做什么。

```python
def validate_args(args: dict[str, Any], schema: dict[str, Any]) -> str:
    """按 schema 校验参数。返回空字符串表示合格，否则是一句给模型看的说明。

    只支持工具参数真正会用到的那一小部分 JSON Schema：对象、必填、基础类型、枚举。
    刻意不做全套——完整实现要处理引用、组合、条件，那些在工具声明里几乎用不到。
    """
    if not isinstance(args, dict):
        return "参数必须是一个对象"
    if "_invalid_json" in args:
        return f"参数不是合法的 JSON：{args['_invalid_json'][:120]}"

    properties = schema.get("properties") or {}
    required = schema.get("required") or []

    missing = [k for k in required if k not in args]
    if missing:
        return f"缺少必填参数 {'、'.join(missing)}"

    unknown = [k for k in args if k not in properties]
    if unknown:
        # 不是致命错误，但要告诉模型——它多半是记错了参数名
        return f"不认识的参数 {'、'.join(unknown)}；这个工具接受的参数是 {'、'.join(properties)}"

    type_checks = {
        "string": str, "number": (int, float), "integer": int,
        "boolean": bool, "array": list, "object": dict,
    }
    for key, value in args.items():
        spec = properties[key]
        expected = spec.get("type")
        python_type = type_checks.get(expected)
        if python_type and not isinstance(value, python_type):
            return f"参数 {key} 应该是 {expected}，收到的是 {type(value).__name__}"
        if spec.get("enum") and value not in spec["enum"]:
            return f"参数 {key} 只能是 {'、'.join(map(str, spec['enum']))} 之一，收到 {value!r}"
    return ""


SCHEMA = {
    "type": "object",
    "properties": {
        "city": {"type": "string", "description": "城市名"},
        "unit": {"type": "string", "enum": ["24h", "12h"], "description": "时间格式"},
    },
    "required": ["city"],
}

for args in [
    {"city": "北京"},
    {"city": "北京", "unit": "24h"},
    {},
    {"city": 123},
    {"city": "北京", "unit": "raw"},
    {"city": "北京", "citys": "上海"},
    {"_invalid_json": '{"city": "北'},
]:
    problem = validate_args(args, SCHEMA)
    print(f"{str(args):46} → {problem or '合格'}")
```

## 注册几个真工具

```python
import datetime

def get_time(city: str, unit: str = "24h") -> str:
    now = datetime.datetime.now()
    formatted = now.strftime("%H:%M") if unit == "24h" else now.strftime("%I:%M %p")
    return f"{city} 现在是 {formatted}"


def calculate(expression: str) -> str:
    """算一个算术表达式。

    注意这里没有用 eval——那等于让模型在我们的进程里执行任意代码。
    工具的输入来自模型，而模型的输入来自用户，所以**工具的输入始终是不可信的**。
    """
    allowed = set("0123456789+-*/(). ")
    if not set(expression) <= allowed:
        raise ValueError(f"表达式里有不允许的字符：{sorted(set(expression) - allowed)}")
    if not expression.strip():
        raise ValueError("表达式是空的")
    # 到这里只剩数字和四则运算符号了，用 eval 是安全的；但仍然限制内建函数。
    return str(eval(expression, {"__builtins__": {}}, {}))


registry = ToolRegistry([
    Tool(name="get_time", description="查询某个城市的当前时间", parameters=SCHEMA, run=get_time),
    Tool(
        name="calculate",
        description="计算一个算术表达式，比如 (3+5)*2",
        parameters={
            "type": "object",
            "properties": {"expression": {"type": "string", "description": "只含数字和 + - * / ( ) 的表达式"}},
            "required": ["expression"],
        },
        run=calculate,
    ),
])

print("登记的工具：", registry.names())
```

## 四种情况各演一遍

```python
cases = [
    ("正常调用", "get_time", {"city": "北京"}),
    ("工具不存在", "get_weather", {"city": "北京"}),
    ("参数不合格", "get_time", {}),
    ("工具执行时抛异常", "calculate", {"expression": "1/0"}),
    ("输入含危险字符", "calculate", {"expression": "__import__('os').system('ls')"}),
]

for label, name, args in cases:
    r = registry.execute(name, args)
    flag = "参数问题" if r.is_invalid_args else ("失败" if r.is_error else "成功")
    print(f"{label:18} [{flag}] {r.content[:70]}")
```

**注意最后两种都没有让程序崩溃。** 主循环拿到的永远是一个 `ToolResult`，
它不需要写 try。

## 接进主循环

```python
def run(user_text: str, model: Model, registry: ToolRegistry, max_rounds: int = 6) -> str:
    history = [
        Message(role="system", content=[TextBlock(text=
            "你可以使用工具。需要算数或查时间时调用对应的工具，不要自己心算。")]),
        Message(role="user", content=[TextBlock(text=user_text)]),
    ]
    print(f"用户：{user_text}\n")

    for round_no in range(1, max_rounds + 1):
        reply = collect(model.stream(history, registry.specs()))
        history.append(reply)
        if reply.text():
            print(f"模型：{reply.text()}")

        calls = reply.tool_uses()
        if not calls:
            return reply.text()

        results = []
        for call in calls:
            r = registry.execute(call.name, call.args)
            mark = "✗" if r.is_error else "✓"
            print(f"工具 {mark} {call.name}({call.args}) → {r.content[:60]}")
            results.append(ToolResultBlock(
                tool_use_id=call.id, content=r.content, is_error=r.is_error))
        history.append(Message(role="tool", content=results))

    return "循环次数用完了"


print("最终答案：", run("(128 + 47) * 3 等于多少？", real_model, registry))
```

模型自己不擅长算术，但它会调 `calculate`。**这就是工具存在的意义**——
把模型不擅长的事交给擅长的代码。

## 参数写错时模型会自己纠正

```python
print("最终答案：", run("北京现在几点？用 12 小时制。", real_model, registry))
```

如果模型第一次把 `unit` 写成了别的值，它会收到「只能是 24h、12h 之一」这条结果，
然后自己改对再调一次。**这就是为什么校验失败要回一条说明清楚的结果，而不是抛异常**——
说明写得越具体，模型自我纠正的成功率越高。

## 本章小结

- 工具要说清三件事：**名字用途**给模型看，**参数结构**两边都用，**执行体**我们用。
- 参数结构用 JSON Schema，因为**模型接口本来就要这个格式**，一份声明两处用。
- 校验只做工具参数用得到的那一小部分：对象、必填、基础类型、枚举。
- 结果分三种：成功、失败、**参数不合格**。第三种要单独区分，因为它值得让模型重试。
- **工具的异常绝不能冒泡到主循环**——包成结果回给模型。
- 工具的输入始终**不可信**：它来自模型，而模型的输入来自用户。
