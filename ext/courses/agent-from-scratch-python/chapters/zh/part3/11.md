# 并发与取消

上一章的主循环有两个明显的短板：

1. 同一轮里的多个工具调用是**排队执行**的。三个各要两秒的工具，用户等六秒。
2. **停不下来**。用户点了「停止」，循环还在跑，工具还在写文件。

这一章解决这两件事。第二件比第一件难得多，难点不在「怎么停」，而在**停完之后对话历史还能不能用**。

```python
def validate_args(args, schema) -> str:
    if not isinstance(args, dict): return "参数必须是一个对象"
    if "_invalid_json" in args: return f"参数不是合法的 JSON：{args['_invalid_json'][:120]}"
    props, required = schema.get("properties") or {}, schema.get("required") or []
    missing = [k for k in required if k not in args]
    if missing: return f"缺少必填参数 {'、'.join(missing)}"
    unknown = [k for k in args if k not in props]
    if unknown: return f"不认识的参数 {'、'.join(unknown)}；这个工具接受 {'、'.join(props)}"
    checks = {"string": str, "number": (int, float), "integer": int,
              "boolean": bool, "array": list, "object": dict}
    for k, v in args.items():
        spec = props[k]
        pt = checks.get(spec.get("type"))
        if pt and not isinstance(v, pt):
            return f"参数 {k} 应该是 {spec.get('type')}，收到的是 {type(v).__name__}"
        if spec.get("enum") and v not in spec["enum"]:
            return f"参数 {k} 只能是 {'、'.join(map(str, spec['enum']))} 之一"
    return ""

class ToolRegistry:
    def __init__(self, tools=None): self._tools = {t.name: t for t in (tools or [])}
    def add(self, tool): self._tools[tool.name] = tool
    def get(self, name): return self._tools.get(name)
    def specs(self): return [t.spec() for t in self._tools.values()]
    def names(self): return list(self._tools)
    def execute(self, name, args, **extra) -> ToolResult:
        tool = self._tools.get(name)
        if tool is None:
            return ToolResult(f"没有叫 {name} 的工具。可用的：{'、'.join(self.names())}", is_error=True)
        problem = validate_args(args, tool.parameters)
        if problem:
            return ToolResult(f"参数不合格：{problem}", is_error=True, is_invalid_args=True)
        try:
            return ToolResult(str(tool.run(**args, **extra)))
        except Exception as exc:
            return ToolResult(f"{type(exc).__name__}: {exc}", is_error=True)

def to_openai_messages(messages):
    out = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({"role": "assistant", "content": text or None, "tool_calls": [
                {"id": c.id, "type": "function",
                 "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                for c in calls]})
        else:
            out.append({"role": m.role, "content": text})
    return out

def to_openai_tools(tools):
    return [{"type": "function", "function": {"name": t["name"], "description": t.get("description", ""),
             "parameters": t.get("parameters", {"type": "object", "properties": {}})}} for t in tools]

class ToolCallAccumulator:
    def __init__(self): self._slots = {}
    def feed(self, raw_calls):
        for item in raw_calls:
            s = self._slots.setdefault(int(item.get("index", 0)), {"id": "", "name": "", "arguments": ""})
            if item.get("id"): s["id"] = item["id"]
            fn = item.get("function") or {}
            if fn.get("name"): s["name"] = fn["name"]
            if fn.get("arguments"): s["arguments"] += fn["arguments"]
    def finish(self):
        out = []
        for i in sorted(self._slots):
            s = self._slots[i]
            raw = s["arguments"] or "{}"
            try:
                args = json.loads(raw)
                if not isinstance(args, dict): args = {"_raw": raw}
            except json.JSONDecodeError:
                args = {"_invalid_json": raw}
            out.append(ToolUseBlock(id=s["id"] or new_id("call"), name=s["name"], args=args))
        return out

def parse_sse_line(line):
    line = line.strip()
    if not line or line.startswith(":") or not line.startswith("data:"): return None
    payload = line[5:].strip()
    if payload == "[DONE]": return None
    try: return json.loads(payload)
    except json.JSONDecodeError: return None

class OpenAICompatModel:
    def __init__(self, base_url, model, timeout=180):
        self.base_url, self.model, self.timeout = base_url.rstrip("/"), model, timeout
    def stream(self, messages, tools):
        body = {"model": self.model, "messages": to_openai_messages(messages), "stream": True}
        if tools: body["tools"] = to_openai_tools(tools)
        req = urllib.request.Request(f"{self.base_url}/chat/completions",
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"}, method="POST")
        acc = ToolCallAccumulator()
        with urllib.request.urlopen(req, timeout=self.timeout) as resp:
            for raw in resp:
                chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
                if chunk is None: continue
                d = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
                if d.get("content"): yield Delta(kind="text", text=d["content"])
                if d.get("tool_calls"): acc.feed(d["tool_calls"])
        for b in acc.finish(): yield Delta(kind="tool_use", tool=b)

@dataclass
class Event:
    kind: str
    data: dict

class EventStream:
    def __init__(self): self._subscribers = []
    def subscribe(self, fn):
        self._subscribers.append(fn)
        return lambda: self._subscribers.remove(fn) if fn in self._subscribers else None
    def emit(self, kind, **data):
        event = Event(kind=kind, data=data)
        for fn in list(self._subscribers):
            try: fn(event)
            except Exception as exc: print(f"  [订阅者出错，已忽略] {type(exc).__name__}: {exc}")
```

## 取消凭据

「取消」不是一个动作，是一个**状态**：调用方把它翻成 true，所有正在跑的东西自己去看。

```python
import threading


class Cancelled(Exception):
    """工具内部主动响应取消时抛这个。它不是错误，是一种正常的退出方式。"""


class CancelToken:
    """一个可以被翻起来的开关。

    为什么用 threading.Event 而不是一个布尔变量：
    Event 自带内存屏障和 wait()。布尔变量在多线程下的可见性没有保证，
    而且工具想「睡两秒，但被取消就立刻醒」时，wait(2) 一个调用就够了。
    """

    def __init__(self) -> None:
        self._event = threading.Event()

    def cancel(self) -> None:
        self._event.set()

    @property
    def cancelled(self) -> bool:
        return self._event.is_set()

    def raise_if_cancelled(self) -> None:
        if self._event.is_set():
            raise Cancelled("已取消")

    def sleep(self, seconds: float) -> None:
        """可被打断的等待。被取消时立刻抛 Cancelled，而不是睡满。"""
        if self._event.wait(seconds):
            raise Cancelled("已取消")


NEVER_CANCELLED = CancelToken()   # 一个永远不会被翻起来的，给不关心取消的调用方用
print("取消凭据就绪")
```

## 不是所有工具都能并发

并发执行工具之前，得先承认一件事：**有些工具不能并发跑**。

两个「写同一个文件」的调用同时执行，结果是坏文件。两个「查天气」的调用同时执行，没问题。
区别在于有没有副作用——这件事只有工具作者知道，所以让工具自己声明：

```python
@dataclass
class Tool2:
    """在第 9 章那个 Tool 上加两个声明位。"""
    name: str
    description: str
    parameters: dict[str, Any]
    run: Callable[..., str]

    # 能不能和别的工具同时跑。默认 True——只读工具是多数。
    # 写文件、改数据库、执行命令的工具要显式声明 False。
    concurrency_safe: bool = True

    # 要不要把取消凭据传给它。长时间运行的工具应当声明 True 并定期检查。
    wants_cancel: bool = False

    def spec(self):
        return {"name": self.name, "description": self.description, "parameters": self.parameters}


def upgrade(tool: Tool, **flags) -> Tool2:
    return Tool2(tool.name, tool.description, tool.parameters, tool.run, **flags)


import time

def slow_lookup(topic: str, seconds: float = 2.0, cancel: CancelToken = NEVER_CANCELLED) -> str:
    """一个假的慢工具，代表「查数据库」「调外部接口」这类要等的活。"""
    cancel.sleep(seconds)          # 可被打断的等待
    return f"关于「{topic}」的资料：（此处省略两千字）"


def append_note(text: str) -> str:
    """一个有副作用的工具：往同一个文件里追加。两个同时跑会写乱。"""
    with open("notes.txt", "a", encoding="utf-8") as f:
        f.write(text + "\n")
    return f"已记下：{text}"
```

## 并发执行

```python
from concurrent.futures import ThreadPoolExecutor


class ToolRunner:
    """执行一批工具调用。并发安全的并发跑，不安全的串行跑。"""

    def __init__(self, registry: ToolRegistry, max_workers: int = 4) -> None:
        self.registry = registry
        self.max_workers = max_workers

    def _execute_one(self, call: ToolUseBlock, cancel: CancelToken) -> ToolResultBlock:
        tool = self.registry.get(call.name)
        extra = {"cancel": cancel} if getattr(tool, "wants_cancel", False) else {}
        try:
            r = self.registry.execute(call.name, call.args, **extra)
        except Cancelled:
            r = ToolResult("用户取消了这次执行", is_error=True)
        return ToolResultBlock(tool_use_id=call.id, content=r.content, is_error=r.is_error)

    def run_batch(self, calls: list[ToolUseBlock], cancel: CancelToken) -> list[ToolResultBlock]:
        parallel = [c for c in calls if getattr(self.registry.get(c.name), "concurrency_safe", True)]
        serial = [c for c in calls if c not in parallel]

        done: dict[str, ToolResultBlock] = {}

        if parallel:
            with ThreadPoolExecutor(max_workers=self.max_workers) as pool:
                futures = {pool.submit(self._execute_one, c, cancel): c for c in parallel}
                for future, call in futures.items():
                    done[call.id] = future.result()

        for call in serial:
            if cancel.cancelled:
                break                      # 取消之后不再启动新的
            done[call.id] = self._execute_one(call, cancel)

        # 关键一步：按**原始顺序**输出，并且给没跑到的调用补一条结果。
        # 下一节解释为什么这一步不能省。
        return [
            done.get(c.id) or ToolResultBlock(
                tool_use_id=c.id, content="用户取消了这次执行", is_error=True)
            for c in calls
        ]


print("批量执行器就绪")
```

## 为什么必须给每个调用补一条结果

这是这一章最要紧的一句话：

> **模型发出的每一个 tool_use，历史里都必须有一条对应的 tool_result。**
> 少一条，下一次请求就会被服务端拒绝。

取消最容易违反这条：用户在三个工具跑到一半时点了停止，两个跑完了，一个没跑。
如果只把跑完的两条写进历史，这段对话就**永久坏掉了**——以后每次带上它都会报错。

亲眼看一次：

```python
calls = [
    ToolUseBlock(name="calculate", args={"expression": "1+1"}),
    ToolUseBlock(name="slow_lookup", args={"topic": "量子计算", "seconds": 5}),
]

assistant = Message(role="assistant", content=[TextBlock(text="我查一下。"), *calls])

# 坏的历史：第二个调用没有结果
broken = Message(role="tool", content=[
    ToolResultBlock(tool_use_id=calls[0].id, content="2")])

# 好的历史：两个都有，第二个标成已取消
fixed = Message(role="tool", content=[
    ToolResultBlock(tool_use_id=calls[0].id, content="2"),
    ToolResultBlock(tool_use_id=calls[1].id, content="用户取消了这次执行", is_error=True)])

for label, tool_msg in (("缺一条结果", broken), ("补齐了结果", fixed)):
    wire = to_openai_messages([assistant, tool_msg])
    n_calls = len(wire[0].get("tool_calls") or [])
    n_results = sum(1 for m in wire[1:] if m["role"] == "tool")
    print(f"{label}：{n_calls} 个调用 / {n_results} 条结果 → "
```

## 带取消的主循环

```python
class AgentLoop:
    def __init__(self, model: Model, registry: ToolRegistry, stream: EventStream | None = None,
                 max_rounds: int = 8, max_workers: int = 4) -> None:
        self.model = model
        self.registry = registry
        self.stream = stream or EventStream()
        self.max_rounds = max_rounds
        self.runner = ToolRunner(registry, max_workers=max_workers)

    def run(self, history: list[Message], cancel: CancelToken = NEVER_CANCELLED) -> TurnResult:
        turn_id = new_id("turn")
        new_messages: list[Message] = []
        self.stream.emit("turn_start", turn_id=turn_id)

        for round_no in range(1, self.max_rounds + 1):
            if cancel.cancelled:
                return self._cancelled(turn_id, round_no - 1, new_messages)

            try:
                reply, interrupted = self._one_round(history + new_messages, cancel)
            except Exception as exc:
                self.stream.emit("error", turn_id=turn_id, message=str(exc))
                return TurnResult("", StopReason.ERROR, round_no, new_messages,
                                  error=f"{type(exc).__name__}: {exc}")

            new_messages.append(reply)

            calls = reply.tool_uses()
            if calls:
                # 即使是被打断的那一轮，也要把已经解析出来的调用补上结果——
                # 否则历史就缺了对应关系。
                results = self.runner.run_batch(calls, cancel)
                new_messages.append(Message(role="tool", content=results))

            if interrupted or cancel.cancelled:
                return self._cancelled(turn_id, round_no, new_messages)

            if not calls:
                self.stream.emit("turn_done", turn_id=turn_id, reason=StopReason.COMPLETED.value)
                return TurnResult(reply.text(), StopReason.COMPLETED, round_no, new_messages)

        self.stream.emit("turn_done", turn_id=turn_id, reason=StopReason.BUDGET_EXHAUSTED.value)
        return TurnResult(new_messages[-2].text() if len(new_messages) >= 2 else "",
                          StopReason.BUDGET_EXHAUSTED, self.max_rounds, new_messages)

    def _cancelled(self, turn_id: str, rounds: int, messages: list[Message]) -> TurnResult:
        self.stream.emit("turn_done", turn_id=turn_id, reason=StopReason.CANCELLED.value)
        return TurnResult("", StopReason.CANCELLED, rounds, messages)

    def _one_round(self, messages: list[Message], cancel: CancelToken) -> tuple[Message, bool]:
        """返回 (消息, 是否被中途打断)。

        被打断时**已经收到的内容照样组装成消息**——那些字是模型真的说了的，
        丢掉它们会让历史与事实不符。
        """
        text_parts: list[str] = []
        blocks: list[Any] = []
        interrupted = False

        gen = self.model.stream(messages, self.registry.specs())
        try:
            for d in gen:
                if cancel.cancelled:
                    interrupted = True
                    break
                if d.kind == "text":
                    text_parts.append(d.text)
                    self.stream.emit("text_delta", delta=d.text)
                elif d.tool is not None:
                    if text_parts:
                        blocks.append(TextBlock(text="".join(text_parts)))
                        text_parts = []
                    blocks.append(d.tool)
        finally:
            # 生成器要显式关掉，否则那条 HTTP 连接会挂在那里直到垃圾回收。
            gen.close()

        if text_parts:
            blocks.append(TextBlock(text="".join(text_parts)))
        message = Message(role="assistant", content=blocks)
        self.stream.emit("message_done", message_id=message.id, text=message.text())
        return message, interrupted


print("带取消的主循环就绪")
```

**Python 的线程在这里是够用的**：工具干的多半是等待——等网络、等文件、等子进程。
等待时全局解释器锁是放开的，所以线程能真正重叠。
如果工具是纯计算（比如本地跑个模型推理），线程就没用了，那时候要换成进程池。

## 停完之后，历史还能用吗

这才是验收标准。数一遍调用和结果：

```python
def check_history(messages: list[Message]) -> None:
    call_ids, result_ids = [], []
    for m in messages:
        call_ids += [c.id for c in m.tool_uses()]
        result_ids += [r.tool_use_id for r in m.tool_results()]
    missing = [i for i in call_ids if i not in result_ids]
    orphan = [i for i in result_ids if i not in call_ids]
    print(f"工具调用 {len(call_ids)} 个，结果 {len(result_ids)} 条")
    print(f"缺结果的调用：{missing or '无'}")
    print(f"没有对应调用的结果：{orphan or '无'}")
    print("→ 这段历史" + ("可以继续用" if not missing and not orphan else "已经坏了"))

check_history(result.messages)

for m in result.messages:
    for r in m.tool_results():
        print(f"  结果：{r.content}")
```

取消之后历史自洽，用户就能接着说「算了，换个话题」，而不是被迫新开一个会话。

## 本章小结

- 取消是一个**状态**不是一个动作：翻起一个开关，所有正在跑的东西自己去看。
  用 `threading.Event` 而不是布尔变量——它自带可见性保证，还能做可打断的等待。
- 工具要自己声明**能不能并发**。只读工具默认能，有副作用的必须显式声明不能。
- 并发结果**必须按原始顺序输出**。
- **每一个 tool_use 都必须有一条对应的 tool_result**，包括被取消的那些。
  少一条，这段对话就永久坏掉，以后每次带上它都会被服务端拒绝。
- 被打断的那一轮，**已经收到的文字照样进历史**——那是模型真的说了的话。
- 生成器要显式 `close()`，否则 HTTP 连接会挂着不放。
- Python 线程对「等待型」工具有效，对「计算型」工具无效。
