# 上下文压缩

对话越长，发给模型的消息越多。模型的上下文是有上限的——超了，服务端直接报错，
这一轮什么都拿不到。

这一章解决「满了怎么办」。它有一条**绝对不能违反的约束**，和三个可选的策略。

## 先说那条约束

压缩就是扔东西。但有些东西扔了会让这段历史**直接坏掉**：

| 不能动的 | 为什么 |
|---|---|
| 系统提示 | 扔了 agent 就失忆了，连自己是干嘛的都不知道 |
| 成对的工具调用与结果 | 第 11 章讲过，少一条结果，服务端拒绝整个请求 |
| 最近几轮 | 用户刚说的话被扔了，对话就接不上了 |

第二条最容易犯：按「最早的 N 条消息」截断，正好切在助手的工具调用和工具结果中间。

所以压缩的单位不是**消息**，是**轮**——从一条用户消息开始，到助手给出最终回答为止。
一轮里面的东西，要么全留，要么全扔。

## 本章起点

```python
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Iterator, Literal, Protocol
import json, time, urllib.request, uuid

def new_id(prefix: str) -> str: return f"{prefix}_{uuid.uuid4().hex[:12]}"

@dataclass
class TextBlock:
    text: str
    type: Literal["text"] = "text"

@dataclass
class ToolUseBlock:
    name: str
    args: dict[str, Any]
    id: str = field(default_factory=lambda: new_id("call"))
    type: Literal["tool_use"] = "tool_use"

@dataclass
class ToolResultBlock:
    tool_use_id: str
    content: str
    is_error: bool = False
    type: Literal["tool_result"] = "tool_result"

@dataclass
class Message:
    role: Literal["system", "user", "assistant", "tool"]
    content: list[Any]
    id: str = field(default_factory=lambda: new_id("msg"))
    created_at: float = field(default_factory=time.time)
    # meta 放我们自己的标记，不发给模型。这一章用它记「这条是摘要」。
    meta: dict[str, Any] = field(default_factory=dict)
    def text(self): return "".join(b.text for b in self.content if isinstance(b, TextBlock))
    def tool_uses(self): return [b for b in self.content if isinstance(b, ToolUseBlock)]
    def tool_results(self): return [b for b in self.content if isinstance(b, ToolResultBlock)]

@dataclass
class Delta:
    kind: Literal["text", "tool_use"]
    text: str = ""
    tool: ToolUseBlock | None = None

class Model(Protocol):
    def stream(self, messages: list[Message], tools: list[dict]) -> Iterator[Delta]: ...

def to_openai_messages(messages):
    out = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({"role": "assistant", "content": text or None, "tool_calls": [
                {"id": c.id, "type": "function",
                 "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                for c in calls]})
        else:
            out.append({"role": m.role, "content": text})
    return out

def parse_sse_line(line):
    line = line.strip()
    if not line or line.startswith(":") or not line.startswith("data:"): return None
    payload = line[5:].strip()
    if payload == "[DONE]": return None
    try: return json.loads(payload)
    except json.JSONDecodeError: return None

class OpenAICompatModel:
    def __init__(self, base_url, model, timeout=180):
        self.base_url, self.model, self.timeout = base_url.rstrip("/"), model, timeout
    def stream(self, messages, tools=None):
        body = {"model": self.model, "messages": to_openai_messages(messages), "stream": True}
        req = urllib.request.Request(f"{self.base_url}/chat/completions",
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"}, method="POST")
        with urllib.request.urlopen(req, timeout=self.timeout) as resp:
            for raw in resp:
                chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
                if chunk is None: continue
                d = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
                if d.get("content"): yield Delta(kind="text", text=d["content"])

real_model = OpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
print("前十五章的成果已就位")
```

## 量：粗估就够

要知道「满没满」，得先能估算。精确算需要模型对应的分词器——一个几十兆的依赖，
而且换个模型就得换一个。**粗估的误差在 20% 以内，够用了**：真正的兜底是
服务端报错时的重试，不是估得准。

```python
def estimate_tokens(text: str) -> int:
    """粗估一段文本占多少 token。

    经验规律：中日韩文字大约一个字一个 token；拉丁字母大约四个字符一个 token。
    这个估法对中英混排的文本误差在 20% 以内，用来判断「要不要压缩」足够了。
    """
    cjk = sum(1 for ch in text if "一" <= ch <= "鿿" or "぀" <= ch <= "ヿ")
    return cjk + (len(text) - cjk) // 4


def message_tokens(m: Message) -> int:
    """一条消息的开销。除了正文，工具调用的参数和结果也要算进去。"""
    total = estimate_tokens(m.text())
    for c in m.tool_uses():
        total += estimate_tokens(c.name) + estimate_tokens(
            json.dumps(c.args, ensure_ascii=False))
    for r in m.tool_results():
        total += estimate_tokens(r.content)
    return total + 4     # 每条消息的角色、分隔符等固定开销


def total_tokens(messages: list[Message]) -> int:
    return sum(message_tokens(m) for m in messages)


sample = [
    Message(role="user", content=[TextBlock(text="帮我看看这段代码有什么问题")]),
    Message(role="assistant", content=[
        TextBlock(text="我读一下文件。"),
        ToolUseBlock(name="read_file", args={"path": "main.py", "start": 1, "end": 200})]),
    Message(role="tool", content=[
        ToolResultBlock(tool_use_id="call_1", content="def main():\n    pass\n" * 20)]),
]
for m in sample:
    print(f"{m.role:9} {message_tokens(m):5} token")
print(f"{'合计':9} {total_tokens(sample):5} token")
```

## 切成轮

```python
@dataclass
class Turn:
    """一轮：一条用户消息，加上它引出的全部助手与工具消息。

    压缩的最小单位。半轮是不能存在的东西。
    """
    messages: list[Message]

    @property
    def tokens(self) -> int:
        return total_tokens(self.messages)

    @property
    def user_text(self) -> str:
        return next((m.text() for m in self.messages if m.role == "user"), "")

    @property
    def is_summary(self) -> bool:
        return any(m.meta.get("summary") for m in self.messages)


def split_turns(history: list[Message]) -> list[Turn]:
    """按用户消息切分。用户开口 = 新的一轮开始。"""
    turns: list[Turn] = []
    current: list[Message] = []
    for m in history:
        if m.role == "user" and current:
            turns.append(Turn(current))
            current = []
        current.append(m)
    if current:
        turns.append(Turn(current))
    return turns


def check_pairing(messages: list[Message]) -> list[str]:
    calls = [c.id for m in messages for c in m.tool_uses()]
    results = [r.tool_use_id for m in messages for r in m.tool_results()]
    problems = [f"调用 {c} 没有结果" for c in calls if c not in results]
    problems += [f"结果 {r} 没有对应的调用" for r in results if r not in calls]
    return problems


print("轮的切分就绪")
```

## 造一段长历史

```python
import random

TOPICS = ["数据库连接池", "缓存穿透", "分布式锁", "消息重复消费", "慢查询",
          "内存泄漏", "线程池打满", "证书过期", "磁盘写满", "时钟漂移"]

def fake_history(n_turns: int) -> list[Message]:
    """造一段像模像样的长对话，其中一部分带工具调用。"""
    random.seed(42)
    history: list[Message] = []
    for i in range(n_turns):
        topic = TOPICS[i % len(TOPICS)]
        history.append(Message(role="user",
                               content=[TextBlock(text=f"第 {i + 1} 个问题：{topic}要怎么排查？")]))
        if i % 3 == 0:
            call = ToolUseBlock(name="search_docs", args={"q": topic, "limit": 5})
            history.append(Message(role="assistant",
                                   content=[TextBlock(text="我查一下资料。"), call]))
            history.append(Message(role="tool", content=[ToolResultBlock(
                tool_use_id=call.id,
                content=f"《{topic}排查手册》摘录：" + "这一段是资料正文。" * 12)]))
        history.append(Message(role="assistant", content=[TextBlock(
            text=f"关于{topic}，先看监控指标，再看日志，最后复现。" + "展开说明若干。" * 8)]))
    return history


history = fake_history(20)
turns = split_turns(history)
print(f"造了 {len(turns)} 轮、{len(history)} 条消息、约 {total_tokens(history)} token")
print(f"其中带工具调用的轮：{sum(1 for t in turns if any(m.tool_uses() for m in t.messages))}")
print(f"成对检查：{check_pairing(history) or '全部成对 ✓'}")
```

## 策略一：截断

最简单的办法——从最早的轮开始扔，扔到装得下为止。

```python
@dataclass
class CompactionResult:
    messages: list[Message]
    strategy: str
    turns_before: int
    turns_after: int
    tokens_before: int
    tokens_after: int

    def describe(self) -> str:
        return (f"{self.strategy}：{self.turns_before} 轮 → {self.turns_after} 轮，"
                f"{self.tokens_before} → {self.tokens_after} token "
                f"（省了 {100 - self.tokens_after * 100 // max(self.tokens_before, 1)}%）")


def truncate(history: list[Message], budget: int, keep_recent: int = 2) -> CompactionResult:
    """从最早的轮开始扔。

    keep_recent 是底线：最近这几轮无论如何都留着。
    没有这条底线，一段特别长的对话可能被压到只剩系统提示，用户会觉得 agent 突然失忆。
    """
    turns = split_turns(history)
    before_tokens = total_tokens(history)

    kept = list(turns)
    while len(kept) > keep_recent and total_tokens([m for t in kept for m in t.messages]) > budget:
        kept.pop(0)     # 扔最早的一整轮，绝不切开

    messages = [m for t in kept for m in t.messages]
    return CompactionResult(messages, "截断", len(turns), len(kept),
                            before_tokens, total_tokens(messages))


result = truncate(history, budget=1500)
print(result.describe())
print(f"成对检查：{check_pairing(result.messages) or '全部成对 ✓'}")
print(f"最早保留的一轮：{split_turns(result.messages)[0].user_text[:30]}")
```

截断的问题是**前面的信息彻底没了**。用户在第 3 轮说的「我的项目叫北极星」，
到第 20 轮就想不起来了。

## 策略二：摘要

让模型把要扔掉的那几轮总结成一段话，用这段话顶替它们。

```python
SUMMARY_PROMPT = """把下面这段对话压缩成一份要点清单，供后续对话参考。

必须保留：用户提到的事实（名字、数字、偏好、约束）、已经做出的决定、还没解决的问题。
可以丢弃：寒暄、重复的解释、工具返回的原始长文。

用中文，不超过 200 字，直接给清单，不要开场白。

对话：
{transcript}"""


def render_transcript(turns: list[Turn]) -> str:
    lines = []
    for t in turns:
        for m in t.messages:
            if m.role == "user":
                lines.append(f"用户：{m.text()}")
            elif m.role == "assistant" and m.text():
                lines.append(f"助手：{m.text()}")
            elif m.role == "tool":
                for r in m.tool_results():
                    lines.append(f"工具结果：{r.content[:120]}")
    return "\n".join(lines)


def summarize_turns(model: Model, turns: list[Turn]) -> str:
    prompt = SUMMARY_PROMPT.format(transcript=render_transcript(turns))
    parts = []
    for d in model.stream([Message(role="user", content=[TextBlock(text=prompt)])], []):
        if d.kind == "text":
            parts.append(d.text)
    return "".join(parts).strip()


def summarize(history: list[Message], budget: int, model: Model,
              keep_recent: int = 3) -> CompactionResult:
    turns = split_turns(history)
    before_tokens = total_tokens(history)

    if before_tokens <= budget or len(turns) <= keep_recent:
        return CompactionResult(history, "无需压缩", len(turns), len(turns),
                                before_tokens, before_tokens)

    # 已经是摘要的轮不再进摘要——否则每次压缩都把摘要再摘要一遍，
    # 信息一轮比一轮糊，几次之后什么都不剩了。
    head = [t for t in turns[:-keep_recent] if not t.is_summary]
    carried = [t for t in turns[:-keep_recent] if t.is_summary]
    tail = turns[-keep_recent:]

    if not head:
        return truncate(history, budget, keep_recent)

    text = summarize_turns(model, head)
    summary_msg = Message(role="user", meta={"summary": True, "covers": len(head)},
                          content=[TextBlock(text=f"【早前对话的要点，供你参考】\n{text}")])

    messages = [m for t in carried for m in t.messages] + [summary_msg] + \
               [m for t in tail for m in t.messages]
    return CompactionResult(messages, "摘要", len(turns), len(split_turns(messages)),
                            before_tokens, total_tokens(messages))


print("摘要压缩就绪（下一格真的调模型，会慢几秒）")
```

**摘要消息用 user 角色发**，理由和第 12 章那些纠正提示一样：
用 assistant 角色模型会当成自己说过的话，用 system 角色很多服务只认开头那一条。
文字前面加个方括号标注，让模型知道这是转述而不是用户新说的话。

```python
result = summarize(history, budget=1500, model=real_model)
print(result.describe())
print(f"成对检查：{check_pairing(result.messages) or '全部成对 ✓'}\n")

summary_msg = next(m for m in result.messages if m.meta.get("summary"))
print(f"摘要覆盖了 {summary_msg.meta['covers']} 轮，内容：")
print(summary_msg.text())
```

## 别把摘要再摘要一遍

对话继续变长，压缩会被再次触发。这时候必须**跳过已有的摘要**：

```python
# 在压缩过的历史后面再接十轮
extended = result.messages + fake_history(10)
print(f"接着聊之后：{len(split_turns(extended))} 轮，{total_tokens(extended)} token")

second = summarize(extended, budget=1500, model=real_model)
print(second.describe())

summaries = [m for m in second.messages if m.meta.get("summary")]
print(f"\n压缩两次之后，历史里有 {len(summaries)} 条摘要：")
for m in summaries:
    print(f"  覆盖 {m.meta['covers']} 轮：{m.text()[:60]}…")
print("\n注意第一条摘要**原样保留**了，没有被二次压缩——"
      "反复摘要会让信息一轮比一轮糊。")
```

## 策略三：分层——按预算自己选

实际用的是个组合策略：**没超就不动，超了先摘要，摘要还装不下就截断**。

```python
class ContextManager:
    """决定每次请求发多少历史。

    压缩不是每轮都做的事——它要调一次模型，慢而且花钱。
    只在超过阈值时做，而且阈值要留出余量给回复。
    """

    def __init__(self, model: Model, max_context: int = 4096,
                 reserve_for_reply: int = 1024, keep_recent: int = 3) -> None:
        self.model = model
        self.max_context = max_context
        self.reserve_for_reply = reserve_for_reply
        self.keep_recent = keep_recent

    @property
    def budget(self) -> int:
        """留给历史的额度 = 上下文上限 - 给回复留的位置。

        不留这个余量的后果：历史刚好塞满，模型一个字都吐不出来就报超限。
        """
        return self.max_context - self.reserve_for_reply

    def prepare(self, system_prompt: str, history: list[Message]) -> tuple[list[Message], str]:
        head = Message(role="system", content=[TextBlock(text=system_prompt)])
        head_tokens = message_tokens(head)
        room = self.budget - head_tokens      # 系统提示不参与压缩，先扣掉

        if total_tokens(history) <= room:
            return [head, *history], "无需压缩"

        result = summarize(history, room, self.model, self.keep_recent)
        if result.tokens_after > room:
            # 摘要之后还是装不下——比如最近三轮本身就特别长。截断兜底。
            result = truncate(result.messages, room, keep_recent=1)
            return [head, *result.messages], "摘要 + 截断"
        return [head, *result.messages], result.strategy


manager = ContextManager(real_model, max_context=3000, reserve_for_reply=800)
system_prompt = "你是一个运维排障助手。"

for label, hist in (("短对话", fake_history(2)), ("长对话", fake_history(20))):
    messages, strategy = manager.prepare(system_prompt, hist)
    print(f"{label}：{total_tokens(hist)} token → {total_tokens(messages)} token（{strategy}）")
    print(f"  额度 {manager.budget}，成对检查：{check_pairing(messages) or '✓'}")
```

## 验收：压缩之后还能接着聊

最实在的检验：在压缩过的历史里问一个只有早期才提到的事实。

```python
facts = [
    Message(role="user", content=[TextBlock(text="记住：我们的服务叫「北极星」，端口是 9187。")]),
    Message(role="assistant", content=[TextBlock(text="记住了。")]),
]
long_history = facts + fake_history(18)

messages, strategy = manager.prepare("你是一个运维排障助手，回答控制在一句话。", long_history)
print(f"压缩策略：{strategy}，发出去 {total_tokens(messages)} token")

messages.append(Message(role="user", content=[TextBlock(text="我们的服务叫什么？端口是多少？")]))
answer = "".join(d.text for d in real_model.stream(messages, []) if d.kind == "text")
print(f"\n问：我们的服务叫什么？端口是多少？")
print(f"答：{answer}")
print("\n→ 答得出「北极星 / 9187」，说明摘要把关键事实留住了；"
      "答不出说明摘要提示词还得再明确一点。")
```

这个检验也是**调摘要提示词的方法**：把「必须保留：用户提到的事实」这句话删掉再跑一次，
你会看到它答不出来了。摘要提示词写什么，直接决定 agent 长期记得什么。

## 本章小结

- 精确算 token 要分词器，**粗估误差 20% 以内就够**——真正的兜底是服务端报错时的重试。
- 压缩的单位是**轮**，不是消息。一轮里的东西要么全留要么全扔，
  切在工具调用和工具结果中间会让整段历史被服务端拒绝。
- 三样东西不能动：**系统提示、成对的调用与结果、最近几轮**。
- **截断**最简单，代价是早期信息彻底没了。
- **摘要**用模型把要扔的几轮压成一段话。摘要消息用 **user 角色**，
  文字里标注清楚这是转述。
- **已经是摘要的内容不要再摘要**。反复摘要会让信息一轮比一轮糊，几次之后什么都不剩。
- 预算要**给回复留余量**。历史刚好塞满的后果是模型一个字都吐不出来就报超限。
- 压缩要调一次模型，慢而且花钱，所以**只在超过阈值时做**。
- 验收方法：在压缩过的历史里问一个只有早期才提到的事实。
  这也是调摘要提示词的方法——**提示词写什么，直接决定 agent 长期记得什么**。

第四部分到这里结束。我们的 agent 现在有了记忆：会话、落盘、分叉、压缩。
下一部分给它装上真正干活的工具——读写文件、执行命令，以及随之而来的权限问题。
