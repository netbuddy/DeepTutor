# 会话树

到现在为止，历史是一条直线：消息一条接一条往后排。但用户会做三件事，直线撑不住：

1. **「这个回答不好，重新生成一个」**——旧回答要留着，用户可能想比较。
2. **「我刚才那句话说错了，改一下」**——改完之后，后面的对话全部作废，但不该消失。
3. **「先按方案 A 往下聊，回头我想试试方案 B」**——两条路都要保留，随时切换。

这三件事的共同形状是**分叉**。历史不是一条线，是**一棵树**：
每条消息记住自己的父亲，会话再记住「当前走到哪个叶子」。

发给模型的历史，就是**从当前叶子回溯到根的那条路径**。

## 本章起点

```python
from __future__ import annotations
from dataclasses import dataclass, field
from enum import Enum
from typing import Any, Callable, Iterator, Literal, Protocol
import datetime, json, time, urllib.request, uuid

def new_id(prefix: str) -> str: return f"{prefix}_{uuid.uuid4().hex[:12]}"

@dataclass
class TextBlock:
    text: str
    type: Literal["text"] = "text"

@dataclass
class ToolUseBlock:
    name: str
    args: dict[str, Any]
    id: str = field(default_factory=lambda: new_id("call"))
    type: Literal["tool_use"] = "tool_use"

@dataclass
class ToolResultBlock:
    tool_use_id: str
    content: str
    is_error: bool = False
    type: Literal["tool_result"] = "tool_result"

@dataclass
class Message:
    role: Literal["system", "user", "assistant", "tool"]
    content: list[Any]
    id: str = field(default_factory=lambda: new_id("msg"))
    created_at: float = field(default_factory=time.time)
    def text(self): return "".join(b.text for b in self.content if isinstance(b, TextBlock))
    def tool_uses(self): return [b for b in self.content if isinstance(b, ToolUseBlock)]
    def tool_results(self): return [b for b in self.content if isinstance(b, ToolResultBlock)]

@dataclass
class Delta:
    kind: Literal["text", "tool_use"]
    text: str = ""
    tool: ToolUseBlock | None = None

class Model(Protocol):
    def stream(self, messages: list[Message], tools: list[dict]) -> Iterator[Delta]: ...

@dataclass
class ToolResult:
    content: str
    is_error: bool = False

@dataclass
class Tool:
    name: str
    description: str
    parameters: dict[str, Any]
    run: Callable[..., str]
    def spec(self): return {"name": self.name, "description": self.description,
                            "parameters": self.parameters}

class ToolRegistry:
    def __init__(self, tools=None): self._tools = {t.name: t for t in (tools or [])}
    def specs(self): return [t.spec() for t in self._tools.values()]
    def execute(self, name, args) -> ToolResult:
        tool = self._tools.get(name)
        if tool is None: return ToolResult(f"没有叫 {name} 的工具", is_error=True)
        try: return ToolResult(str(tool.run(**args)))
        except Exception as exc: return ToolResult(f"{type(exc).__name__}: {exc}", is_error=True)

def to_openai_messages(messages):
    out = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({"role": "assistant", "content": text or None, "tool_calls": [
                {"id": c.id, "type": "function",
                 "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                for c in calls]})
        else:
            out.append({"role": m.role, "content": text})
    return out

def to_openai_tools(tools):
    return [{"type": "function", "function": {"name": t["name"], "description": t.get("description", ""),
             "parameters": t.get("parameters", {"type": "object", "properties": {}})}} for t in tools]

class ToolCallAccumulator:
    def __init__(self): self._slots = {}
    def feed(self, raw_calls):
        for item in raw_calls:
            s = self._slots.setdefault(int(item.get("index", 0)), {"id": "", "name": "", "arguments": ""})
            if item.get("id"): s["id"] = item["id"]
            fn = item.get("function") or {}
            if fn.get("name"): s["name"] = fn["name"]
            if fn.get("arguments"): s["arguments"] += fn["arguments"]
    def finish(self):
        out = []
        for i in sorted(self._slots):
            s = self._slots[i]
            try: args = json.loads(s["arguments"] or "{}")
            except json.JSONDecodeError: args = {}
            out.append(ToolUseBlock(id=s["id"] or new_id("call"), name=s["name"],
                                    args=args if isinstance(args, dict) else {}))
        return out

def parse_sse_line(line):
    line = line.strip()
    if not line or line.startswith(":") or not line.startswith("data:"): return None
    payload = line[5:].strip()
    if payload == "[DONE]": return None
    try: return json.loads(payload)
    except json.JSONDecodeError: return None

class OpenAICompatModel:
    def __init__(self, base_url, model, timeout=180, temperature=None):
        self.base_url, self.model, self.timeout = base_url.rstrip("/"), model, timeout
        self.temperature = temperature
    def stream(self, messages, tools):
        body = {"model": self.model, "messages": to_openai_messages(messages), "stream": True}
        if self.temperature is not None: body["temperature"] = self.temperature
        if tools: body["tools"] = to_openai_tools(tools)
        req = urllib.request.Request(f"{self.base_url}/chat/completions",
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"}, method="POST")
        acc = ToolCallAccumulator()
        with urllib.request.urlopen(req, timeout=self.timeout) as resp:
            for raw in resp:
                chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
                if chunk is None: continue
                d = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
                if d.get("content"): yield Delta(kind="text", text=d["content"])
                if d.get("tool_calls"): acc.feed(d["tool_calls"])
        for b in acc.finish(): yield Delta(kind="tool_use", tool=b)

def get_time(city: str) -> str:
    return f"{city} 现在是 {datetime.datetime.now().strftime('%H:%M')}"

registry = ToolRegistry([Tool("get_time", "查询某个城市的当前时间",
```

## 树

```python
@dataclass
class Node:
    """树上的一个位置。消息本身不变，变的是它挂在哪里。"""
    message: Message
    parent_id: str | None
    children: list[str] = field(default_factory=list)


class ConversationTree:
    """一棵会话树，外加一个「当前走到哪儿」的指针。

    和直线历史比，多出来的只有两样东西：每个节点记住父亲，会话记住当前叶子。
    别的复杂度都是从这两样推出来的。
    """

    def __init__(self, session_id: str = "", title: str = "") -> None:
        self.id = session_id or new_id("ses")
        self.title = title
        self.nodes: dict[str, Node] = {}
        self.roots: list[str] = []
        self.leaf_id: str | None = None    # 当前分支的末端

    # ── 写 ──────────────────────────────────────────────────────────────

    def add(self, message: Message, parent_id: str | None = None) -> str:
        """挂一条消息上去。不传父亲就挂在当前叶子后面。"""
        if parent_id is None:
            parent_id = self.leaf_id
        if parent_id is not None and parent_id not in self.nodes:
            raise KeyError(f"没有这个父节点：{parent_id}")

        node = Node(message=message, parent_id=parent_id)
        self.nodes[message.id] = node
        if parent_id is None:
            self.roots.append(message.id)
        else:
            self.nodes[parent_id].children.append(message.id)
        self.leaf_id = message.id
        return message.id

    # ── 读 ──────────────────────────────────────────────────────────────

    def path_to(self, node_id: str) -> list[Message]:
        """从根到这个节点的那条路径。这就是要发给模型的历史。"""
        chain: list[Message] = []
        cursor: str | None = node_id
        while cursor is not None:
            node = self.nodes[cursor]
            chain.append(node.message)
            cursor = node.parent_id
        chain.reverse()     # 回溯是从叶子往根走，发给模型要从根往叶子
        return chain

    def current_path(self) -> list[Message]:
        return [] if self.leaf_id is None else self.path_to(self.leaf_id)

    def siblings(self, node_id: str) -> list[str]:
        """同一个父亲下的所有兄弟，包括自己。「第 2 个回答，共 3 个」就是数这个。"""
        parent_id = self.nodes[node_id].parent_id
        return list(self.roots) if parent_id is None else list(self.nodes[parent_id].children)

    def deepest_leaf_under(self, node_id: str) -> str:
        """从某个节点往下一直走到底，每次选最新的那个孩子。

        切换分支时要用：用户点的是分叉点，但他想去的是那条分支的末端。
        """
        cursor = node_id
        while self.nodes[cursor].children:
            cursor = self.nodes[cursor].children[-1]
        return cursor

    def switch_to(self, node_id: str) -> None:
        if node_id not in self.nodes:
            raise KeyError(node_id)
        self.leaf_id = self.deepest_leaf_under(node_id)


print("会话树就绪")
```

## 一条硬规矩：只能在轮的边界分叉

不能随便挑一个节点分叉。**助手发出的工具调用和它的工具结果必须待在一起**——
第 11 章讲过，少一条结果，这段历史就会被服务端拒绝。

所以合法的分叉点只有两种：

- **用户消息**：在它的父亲下面挂一条新的用户消息 = 「改写我的提问」。
- **助手的最终回答**（不带工具调用的那条）：在它的父亲下面重新生成 = 「换一个回答」。

```python
def is_turn_boundary(tree: ConversationTree, node_id: str) -> bool:
    """这个节点能不能当分叉点。"""
    m = tree.nodes[node_id].message
    if m.role == "user":
        return True
    if m.role == "assistant" and not m.tool_uses():
        return True     # 最终回答，这一轮到此结束
    return False        # 带工具调用的助手消息、工具结果消息，都在轮的中间


def check_pairing(messages: list[Message]) -> list[str]:
    """检查一条路径上的调用与结果是否成对。这是分叉之后必须过的验收。"""
    calls = [c.id for m in messages for c in m.tool_uses()]
    results = [r.tool_use_id for m in messages for r in m.tool_results()]
    problems = []
    for cid in calls:
        if cid not in results:
            problems.append(f"调用 {cid} 没有结果")
    for rid in results:
        if rid not in calls:
            problems.append(f"结果 {rid} 没有对应的调用")
    return problems


print("分叉规矩就绪")
```

## 把 Agent 接到树上

```python
@dataclass
class AgentConfig:
    system_prompt: str = "你是一个简洁的助手，回答控制在两句话以内。"
    max_rounds: int = 6


class TreeAgent:
    """在树上跑的 agent。

    和上一章的区别只有一句话：历史不再是 session.messages，
    而是 tree.current_path()。别的全都一样。
    """

    def __init__(self, model: Model, registry: ToolRegistry, config: AgentConfig | None = None):
        self.model = model
        self.registry = registry
        self.config = config or AgentConfig()

    def _prepare(self, history: list[Message]) -> list[Message]:
        return [Message(role="system", content=[TextBlock(text=self.config.system_prompt)]),
                *history]

    def _generate(self, tree: ConversationTree) -> str:
        """从当前叶子往下生成，把新消息挂上去，返回最终回答的节点标识。"""
        for _ in range(self.config.max_rounds):
            history = self._prepare(tree.current_path())
            reply = self._one_round(history)
            node_id = tree.add(reply)

            calls = reply.tool_uses()
            if not calls:
                return node_id

            results = []
            for call in calls:
                r = self.registry.execute(call.name, call.args)
                results.append(ToolResultBlock(tool_use_id=call.id, content=r.content,
                                               is_error=r.is_error))
            tree.add(Message(role="tool", content=results))
        return tree.leaf_id or ""

    def send(self, tree: ConversationTree, user_text: str) -> str:
        tree.add(Message(role="user", content=[TextBlock(text=user_text)]))
        return self._generate(tree)

    def regenerate(self, tree: ConversationTree, node_id: str) -> str:
        """换一个回答。做法是：回到它的父亲，从那里重新生成。

        结果是一个**兄弟节点**——旧回答一个字没动，还挂在原处。
        这就是树相对于「直接覆盖」的全部价值。
        """
        node = tree.nodes[node_id]
        if node.message.role != "assistant" or node.message.tool_uses():
            raise ValueError("只能重新生成助手的最终回答")
        tree.leaf_id = node.parent_id
        return self._generate(tree)

    def edit_and_resend(self, tree: ConversationTree, node_id: str, new_text: str) -> str:
        """改写一条用户消息。原消息和它下面那一整条分支都留着。"""
        node = tree.nodes[node_id]
        if node.message.role != "user":
            raise ValueError("只能改写用户消息")
        tree.leaf_id = node.parent_id
        tree.add(Message(role="user", content=[TextBlock(text=new_text)]))
        return self._generate(tree)

    def _one_round(self, messages: list[Message]) -> Message:
        text_parts, blocks = [], []
        for d in self.model.stream(messages, self.registry.specs()):
            if d.kind == "text":
                text_parts.append(d.text)
            elif d.tool is not None:
                if text_parts:
                    blocks.append(TextBlock(text="".join(text_parts)))
                    text_parts = []
                blocks.append(d.tool)
        if text_parts:
            blocks.append(TextBlock(text="".join(text_parts)))
        return Message(role="assistant", content=blocks)


print("树上的 agent 就绪")
```

## 画出来

```python
def render(tree: ConversationTree, highlight: set[str] | None = None) -> None:
    """把树打出来。当前路径上的节点标一个箭头。"""
    highlight = highlight or set()
    labels = {"user": "用户", "assistant": "助手", "tool": "工具"}

    def walk(node_id: str, depth: int) -> None:
        node = tree.nodes[node_id]
        m = node.message
        text = m.text().replace("\n", " ")
        if not text:
            text = "、".join(f"调用 {c.name}" for c in m.tool_uses()) or \
                   "、".join(f"结果 {r.content[:14]}" for r in m.tool_results())
        text = text[:38] + "…" if len(text) > 38 else text
        mark = "→" if node_id in highlight else " "
        brothers = tree.siblings(node_id)
        tag = f" [{brothers.index(node_id) + 1}/{len(brothers)}]" if len(brothers) > 1 else ""
        print(f"{mark} {'  ' * depth}{labels[m.role]}{tag}：{text}")
        for child in node.children:
            walk(child, depth + 1)

    for root in tree.roots:
        walk(root, 0)


tree = ConversationTree(title="树的演示")
agent = TreeAgent(real_model, registry)

first = agent.send(tree, "用一句话介绍一下什么是递归。")
print("第一次回答：")
render(tree, highlight={m.id for m in tree.current_path()})
```

```python
second = agent.regenerate(tree, first)

render(tree, highlight=current)

print(f"\n两个回答都在树上：")
print(f"  第一个：{tree.nodes[first].message.text()[:50]}")
print(f"  第二个：{tree.nodes[second].message.text()[:50]}")
print(f"\n当前路径长度：{len(tree.current_path())} 条消息"
      f"（只包含第 {tree.siblings(second).index(second) + 1} 个回答）")
```

**这是关键**：树上有两个回答，但**发给模型的路径里只有一个**。
用户能看到两个、能来回切，模型只看到当前这条路径——它不会被另一个版本干扰。

## 验收：路径上的调用与结果成对

分叉最容易出的错，就是把一次工具调用和它的结果拆到两条分支上。检查一遍：

```python
tool_tree = ConversationTree(title="带工具的分叉")
tool_agent = TreeAgent(real_model, registry,
                       AgentConfig(system_prompt="需要查时间时调用 get_time 工具。"))

answer = tool_agent.send(tool_tree, "北京现在几点？")
again = tool_agent.regenerate(tool_tree, answer)

print("树的样子：")
render(tool_tree, highlight={m.id for m in tool_tree.current_path()})

for label, leaf in (("第一条分支", answer), ("第二条分支", again)):
    path = tool_tree.path_to(leaf)
    problems = check_pairing(path)
    calls = sum(len(m.tool_uses()) for m in path)
    print(f"\n{label}：{len(path)} 条消息，{calls} 次工具调用")
    print(f"  成对检查：{problems or '全部成对 ✓'}")

print("\n这棵树上哪些节点可以当分叉点：")
for node_id, node in tool_tree.nodes.items():
    m = node.message
    verdict = "可以" if is_turn_boundary(tool_tree, node_id) else "不行（在轮的中间）"
    detail = m.text()[:20] or f"{len(m.tool_uses())} 次调用 / {len(m.tool_results())} 条结果"
    print(f"  {m.role:9} {verdict:16} {detail}")
```

两条分支各自完整。它们**共享**分叉点之前的那段历史（包括工具调用与结果），
分叉之后各走各的——所以两边都是成对的。

## 存到磁盘上要改什么

几乎不用改。上一章那个追加式日志，每条消息多存一个 `parent_id` 就够了：

```python
def message_record(tree: ConversationTree, node_id: str) -> dict:
    node = tree.nodes[node_id]
    return {"kind": "message", "id": node.message.id, "parent_id": node.parent_id,
            "role": node.message.role, "created_at": node.message.created_at,
            "content": [b.__dict__ for b in node.message.content]}


sample = [message_record(tree, nid) for nid in list(tree.nodes)[:3]]
for record in sample:
    print(f"{record['id']}  父亲={record['parent_id']}  {record['role']}")

print("\n分支不需要额外的文件：树的形状全在 parent_id 里，读的时候重建就行。")
print(f"另外要存的只有一个数：当前叶子 = {tree.leaf_id}")
```

**追加式日志天然适合树**：分叉不修改任何旧记录，只是新增一条父亲指向别处的记录。
如果当初选的是「每次覆写整个会话」，这里就要重写整个文件。

## 本章小结

- 历史不是一条线，是**一棵树**。多出来的只有两样东西：
  每个节点记住父亲，会话记住当前叶子。
- 发给模型的历史 = **从当前叶子回溯到根的路径**（记得反转）。
- **只能在轮的边界分叉**：用户消息，或助手的最终回答。
  在带工具调用的消息中间分叉，会把调用和结果拆散，这段历史就废了。
- 「重新生成」= 回到父亲重新生成，结果是一个**兄弟节点**；旧回答一个字没动。
  **这是树相对于直接覆盖的全部价值。**
- 「改写提问」= 在原提问的父亲下面挂一条新提问；原提问和它下面整条分支都留着。
- 用户能看到全部分支，**模型只看到当前路径**——它不会被别的版本干扰。
- 切换分支时，用户点的是分叉点，但要去的是那条分支的**末端**。
- 存储几乎不用改：每条消息多存一个 `parent_id`，外加一个「当前叶子」。
  **追加式日志天然适合树**，因为分叉不修改任何旧记录。
