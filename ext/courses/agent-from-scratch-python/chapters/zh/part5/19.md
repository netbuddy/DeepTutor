# 技能

工具解决的是「agent **能做什么**」。这一章解决另一件事：「agent **怎么做**」。

举个具体的：你希望它写的 git 提交信息符合团队规范——首行不超过 50 字、用祈使句、
正文说清楚为什么而不是做了什么、破坏性变更要加脚注。这些话写哪儿？

写进系统提示是最直接的想法。但你还希望它会写变更日志、会排查慢查询、
会按公司模板出周报……**系统提示会膨胀到几千 token，而其中 95% 与当前这句话无关。**

膨胀的代价有三层，最后一层最要命：

1. 每一轮都要付这些 token 的钱。
2. 上下文被占掉，能装的对话历史变少。
3. **注意力被稀释**——真正重要的那三条规矩，淹没在两千字里。

办法是把「怎么做」搬到系统提示外面，做成**技能**：平时只留一行索引，
用得上的时候才把全文读进来。

## 技能和工具的区别

| | 工具 | 技能 |
|---|---|---|
| 是什么 | 一段代码 | 一段文字 |
| 提供什么 | 新的能力（读文件、发请求） | 做某件事的方法 |
| 谁执行 | 我们的程序 | 模型自己 |
| 加不加得动 | 要改代码、重启 | 放一个文件进目录就行 |

一句话：**工具扩展能力，技能扩展方法**。

```python
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Callable, Iterator, Literal, Protocol
import json, shutil, urllib.request, uuid
from pathlib import Path

def new_id(prefix: str) -> str: return f"{prefix}_{uuid.uuid4().hex[:12]}"

@dataclass
class TextBlock:
    text: str
    type: Literal["text"] = "text"

@dataclass
class ToolUseBlock:
    name: str
    args: dict[str, Any]
    id: str = field(default_factory=lambda: new_id("call"))
    type: Literal["tool_use"] = "tool_use"

@dataclass
class ToolResultBlock:
    tool_use_id: str
    content: str
    is_error: bool = False
    type: Literal["tool_result"] = "tool_result"

@dataclass
class Message:
    role: Literal["system", "user", "assistant", "tool"]
    content: list[Any]
    id: str = field(default_factory=lambda: new_id("msg"))
    def text(self): return "".join(b.text for b in self.content if isinstance(b, TextBlock))
    def tool_uses(self): return [b for b in self.content if isinstance(b, ToolUseBlock)]

@dataclass
class Delta:
    kind: Literal["text", "tool_use"]
    text: str = ""
    tool: ToolUseBlock | None = None

class Model(Protocol):
    def stream(self, messages: list[Message], tools: list[dict]) -> Iterator[Delta]: ...

@dataclass
class ToolResult:
    content: str
    is_error: bool = False

@dataclass
class Tool:
    name: str
    description: str
    parameters: dict[str, Any]
    run: Callable[..., str]
    def spec(self): return {"name": self.name, "description": self.description,
                            "parameters": self.parameters}

class ToolRegistry:
    def __init__(self, tools=None): self._tools = {t.name: t for t in (tools or [])}
    def add(self, tool): self._tools[tool.name] = tool
    def names(self): return list(self._tools)
    def specs(self): return [t.spec() for t in self._tools.values()]
    def execute(self, name, args) -> ToolResult:
        tool = self._tools.get(name)
        if tool is None:
            return ToolResult(f"没有叫 {name} 的工具", is_error=True)
        try:
            return ToolResult(str(tool.run(**args)))
        except Exception as exc:
            return ToolResult(f"{type(exc).__name__}: {exc}", is_error=True)

def to_openai_messages(messages):
    out = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({"role": "assistant", "content": text or None, "tool_calls": [
                {"id": c.id, "type": "function",
                 "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                for c in calls]})
        else:
            out.append({"role": m.role, "content": text})
    return out

def to_openai_tools(tools):
    return [{"type": "function", "function": {"name": t["name"], "description": t.get("description", ""),
             "parameters": t.get("parameters", {"type": "object", "properties": {}})}} for t in tools]

class ToolCallAccumulator:
    def __init__(self): self._slots = {}
    def feed(self, raw_calls):
        for item in raw_calls:
            s = self._slots.setdefault(int(item.get("index", 0)), {"id": "", "name": "", "arguments": ""})
            if item.get("id"): s["id"] = item["id"]
            fn = item.get("function") or {}
            if fn.get("name"): s["name"] = fn["name"]
            if fn.get("arguments"): s["arguments"] += fn["arguments"]
    def finish(self):
        out = []
        for i in sorted(self._slots):
            s = self._slots[i]
            try: args = json.loads(s["arguments"] or "{}")
            except json.JSONDecodeError: args = {}
            out.append(ToolUseBlock(id=s["id"] or new_id("call"), name=s["name"],
                                    args=args if isinstance(args, dict) else {}))
        return out

def parse_sse_line(line):
    line = line.strip()
    if not line or line.startswith(":") or not line.startswith("data:"): return None
    payload = line[5:].strip()
    if payload == "[DONE]": return None
    try: return json.loads(payload)
    except json.JSONDecodeError: return None

class OpenAICompatModel:
    def __init__(self, base_url, model, timeout=180):
        self.base_url, self.model, self.timeout = base_url.rstrip("/"), model, timeout
    def stream(self, messages, tools):
        body = {"model": self.model, "messages": to_openai_messages(messages), "stream": True}
        if tools: body["tools"] = to_openai_tools(tools)
        req = urllib.request.Request(f"{self.base_url}/chat/completions",
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"}, method="POST")
        acc = ToolCallAccumulator()
        with urllib.request.urlopen(req, timeout=self.timeout) as resp:
            for raw in resp:
                chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
                if chunk is None: continue
                d = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
                if d.get("content"): yield Delta(kind="text", text=d["content"])
                if d.get("tool_calls"): acc.feed(d["tool_calls"])
        for b in acc.finish(): yield Delta(kind="tool_use", tool=b)

def estimate_tokens(text: str) -> int:
    cjk = sum(1 for ch in text if "一" <= ch <= "鿿")
    return cjk + (len(text) - cjk) // 4

class Agent:
    def __init__(self, model, registry, system_prompt="", max_rounds=8, on_event=None):
        self.model, self.registry = model, registry
        self.system_prompt = system_prompt
        self.max_rounds, self.on_event = max_rounds, on_event or (lambda *a: None)
    def run(self, user_text):
        history = [Message(role="system", content=[TextBlock(text=self.system_prompt)]),
                   Message(role="user", content=[TextBlock(text=user_text)])]
        for _ in range(self.max_rounds):
            reply = self._one_round(history)
            history.append(reply)
            calls = reply.tool_uses()
            if not calls:
                return reply.text()
            results = []
            for c in calls:
                r = self.registry.execute(c.name, c.args)
                self.on_event(c.name, c.args, r)
                results.append(ToolResultBlock(tool_use_id=c.id, content=r.content,
                                               is_error=r.is_error))
            history.append(Message(role="tool", content=results))
        return "（轮数用完了）"
    def _one_round(self, messages):
        text_parts, blocks = [], []
        for d in self.model.stream(messages, self.registry.specs()):
            if d.kind == "text":
                text_parts.append(d.text)
            elif d.tool is not None:
                if text_parts:
                    blocks.append(TextBlock(text="".join(text_parts))); text_parts = []
                blocks.append(d.tool)
        if text_parts: blocks.append(TextBlock(text="".join(text_parts)))
        return Message(role="assistant", content=blocks)
```

一个技能就是一个 Markdown 文件：开头几行是元数据，正文是给模型看的做法。

```python
write_skill(
    "commit-message",
    # description 是给模型做检索用的。写「什么时候用」，不要写「这是什么」——
    # 模型要判断的是「现在该不该调它」，不是「它是个什么东西」。
    "用户要写 git 提交信息、要提交代码、或者问「这次改动怎么描述」时使用",
    """# 写 git 提交信息

## 格式

    <类型>: <首行摘要，祈使句，不超过 50 字，句末不加句号>

    <正文：说清楚**为什么**要改，而不是改了什么——改了什么看 diff 就知道了。
    每行不超过 72 字符。>

    <脚注：破坏性变更、关联的问题编号>

## 类型只有这七个

feat（新功能）、fix（修缺陷）、docs（只改文档）、refactor（不改行为的重构）、
test（只动测试）、perf（性能）、chore（构建与杂务）。

## 三条硬规矩

1. **首行不要用「优化」「完善」「更新」这种没有信息量的词**。
   反例：chore: 优化代码。正例：perf: 把 N+1 查询合并成一次 IN 查询。
2. **一次提交只做一件事**。改了两件事就分两次提交。
3. **破坏性变更必须在脚注写 BREAKING CHANGE:**，并说明调用方要怎么改。

## 例子

    fix: 用户取消后补齐缺失的工具结果

    取消发生在多个工具执行到一半时，只有跑完的那几个进了历史。
    下一次请求会因为「有 tool_call 没有对应 tool_result」被服务端拒绝，
    表现是这段对话彻底打不开。

    现在给没跑完的调用补一条「用户取消了这次执行」的结果。
""")

write_skill(
    "slow-query",
    "排查数据库慢查询、接口变慢、超时报警时使用",
    """# 排查慢查询

## 顺序不能乱

先量，再猜，最后改。跳过第一步直接猜原因，十次有九次猜错。

1. **量**：拿到具体的慢查询语句和它的耗时分布（p50 / p95 / p99）。
   只看平均值会漏掉长尾——平均 20ms 而 p99 是 8 秒，是很常见的情况。
2. **看执行计划**：`EXPLAIN ANALYZE`。重点看有没有全表扫描、
   预估行数和实际行数差多少（差一个数量级说明统计信息过期了）。
3. **看有没有 N+1**：一次请求里同一条 SQL 出现几十次，
   八成是在循环里查。这类问题改索引没用，要改代码。
4. **最后才考虑加索引**。加索引会拖慢写入，而且加错了没用。

## 常见原因，按出现频率排

- N+1 查询（最常见，也最容易改）
- 缺索引，或者索引没被用上（隐式类型转换会让索引失效）
- 一次取回太多行（缺 LIMIT，或者分页用 OFFSET 翻到很深的页）
- 锁等待（这时候 SQL 本身很快，慢的是等锁）
- 统计信息过期，优化器选错了计划

## 交付什么

一份结论要包含：慢在哪条语句、慢的原因（有执行计划佐证）、
改法、改完之后的耗时对比。**没有前后对比的优化不算做完。**
""")

write_skill(
    "changelog",
    "用户要写发版说明、变更日志、release notes 时使用",
    """# 写变更日志

## 读者是谁

**是使用者，不是开发者。** 使用者关心的是「我要不要升级、升级之后我要改什么」，
不关心你重构了哪个类。

## 分组，按对使用者的影响排

1. **破坏性变更**——放最前面，写清楚要改什么。
2. **新功能**——一句话说清楚能干什么，附一个最小例子。
3. **缺陷修复**——描述**症状**，不是描述代码。
   反例：「修复 NullPointerException」。正例：「导出空表格时不再报错」。
4. **其他**——性能、依赖升级之类，一行一条。

## 三条规矩

1. 每条都要能让没读过代码的人看懂。
2. 不要写「若干优化和修复」这种占位句。没有内容就删掉这一节。
3. 版本号和日期写在最前面。
""")

for path in sorted(SKILLS_DIR.glob("*.md")):
    print(f"  {path.name}  {estimate_tokens(path.read_text(encoding='utf-8'))} token")
```

## 技能库：只暴露索引

```python
@dataclass
class Skill:
    name: str
    description: str
    body: str
    path: Path

    @property
    def tokens(self) -> int:
        return estimate_tokens(self.body)


def parse_skill(path: Path) -> Skill:
    """解析技能文件。元数据用最朴素的三横线分隔，不引 YAML 库。

    格式简单是有意的：技能要让不写代码的人也能加一个。
    引入 YAML 就意味着「缩进写错了整个文件解析失败」，而报错信息还很难懂。
    """
    raw = path.read_text(encoding="utf-8")
    meta: dict[str, str] = {}
    body = raw
    if raw.startswith("---"):
        _, block, body = raw.split("---", 2)
        for line in block.strip().splitlines():
            if ":" in line:
                key, value = line.split(":", 1)
                meta[key.strip()] = value.strip()
    name = meta.get("name") or path.stem
    description = meta.get("description", "")
    if not description:
        raise ValueError(f"{path.name} 缺少 description。没有它模型不知道什么时候该用这个技能。")
    return Skill(name=name, description=description, body=body.strip(), path=path)


class SkillLibrary:
    """管一个目录的技能。

    只从这一个目录读，路径由我们拼——**不接受模型给的路径**。
    否则 load_skill("../../etc/passwd") 就变成了一个任意文件读取工具，
    而这个工具的输出会直接进上下文。
    """

    def __init__(self, root: Path) -> None:
        self.root = Path(root)
        self.skills: dict[str, Skill] = {}
        self.reload()

    def reload(self) -> None:
        self.skills = {}
        for path in sorted(self.root.glob("*.md")):
            try:
                skill = parse_skill(path)
            except ValueError as exc:
                print(f"  ⚠ 跳过 {path.name}：{exc}")
                continue
            self.skills[skill.name] = skill

    def index_text(self) -> str:
        """给系统提示用的索引。每个技能一行：名字 + 什么时候用。"""
        lines = [f"- {s.name}：{s.description}" for s in self.skills.values()]
        return "\n".join(lines)

    def get(self, name: str) -> Skill:
        skill = self.skills.get(name)
        if skill is None:
            raise KeyError(f"没有叫 {name} 的技能。现有的：{'、'.join(self.skills)}")
        return skill


library = SkillLibrary(SKILLS_DIR)
index = library.index_text()

print("索引（每轮都要发）：")
print(index)
print(f"\n索引 {estimate_tokens(index)} token")
print(f"全部技能正文加起来 {sum(s.tokens for s in library.skills.values())} token")
print(f"→ 省下 {100 - estimate_tokens(index) * 100 // sum(s.tokens for s in library.skills.values())}%，"
      f"而且技能越多省得越多")
```

## 加载技能的工具

```python
class SkillLoader:
    """把技能装成一个工具，并记住这一轮已经加载过哪些。"""

    def __init__(self, library: SkillLibrary) -> None:
        self.library = library
        self.loaded: set[str] = set()

    def reset(self) -> None:
        self.loaded = set()

    def load(self, name: str) -> str:
        skill = self.library.get(name)
        if name in self.loaded:
            # 重复加载不再返回全文：模型有时会为了「确认一下」再调一次，
            # 那会让同样的两千字在上下文里躺两份。
            return f"技能「{name}」上文已经加载过了，直接照着做就行。"
        self.loaded.add(name)
        return f"=== 技能：{skill.name} ===\n{skill.body}"

    def as_tool(self) -> Tool:
        return Tool(
            "load_skill",
            "加载一个技能的完整说明。当手头的任务与某个技能的适用场景相符时调用它，"
            "然后严格按照技能里的要求去做。",
            {"type": "object",
             "properties": {"name": {"type": "string",
                                     "enum": list(self.library.skills),
                                     "description": "技能名字"}},
             "required": ["name"]},
            self.load)


loader = SkillLoader(library)
print(loader.load("commit-message")[:200], "…")
print()
print(loader.load("commit-message"))
```

**注意那个 `enum`**：把可选值写进参数结构，模型就不会去猜技能名。
这比在描述里写「可选值有 xxx」可靠——第 9 章的参数校验会真的把它挡住。

```python
def build_system_prompt(library: SkillLibrary) -> str:
    """系统提示 = 短短的角色说明 + 技能索引 + 一条使用规则。"""
    return f"""你是一个工程助手。

你有一批技能，每个技能是一份详细的做法说明。下面是它们的名字和适用场景：

{library.index_text()}

规则：
- 手头的任务与某个技能相符时，**先调用 load_skill 读到它的完整说明，再动手**。
  不要凭印象做——技能里的规矩比你的直觉更具体。
- 任务和所有技能都无关时，直接回答，不要硬套。
- 一次只加载真正用得上的技能。"""


def trace(name, args, result):
    print(f"  ⚙ {name}({args}) → {result.content.splitlines()[0][:60]}")


loader = SkillLoader(library)
registry = ToolRegistry([loader.as_tool()])
system_prompt = build_system_prompt(library)

print(f"系统提示 {estimate_tokens(system_prompt)} token\n")
```

## 它挑对了吗

```python
cases = [
    ("帮我写这次发版的变更日志，这版加了会话树、修了导出乱码。", "changelog"),
    ("我们的订单列表接口从 200ms 变成了 3 秒，怎么查？", "slow-query"),
    ("这次改动怎么写提交信息？就是给 Session 加了个 title 字段。", "commit-message"),
    ("帮我把这句英文翻译成中文：the quick brown fox", None),   # 和所有技能都无关
]

for question, expected in cases:
    loader.reset()
    used: list[str] = []
    probe = Agent(real_model, ToolRegistry([loader.as_tool()]), system_prompt, max_rounds=5,
                  on_event=lambda name, args, r: used.append(args.get("name", "")))
    probe.run(question)
    got = used[0] if used else None
    mark = "✓" if got == expected else "✗"
    print(f"  {mark} {question[:26]:28} 期望 {str(expected):16} 实际 {str(got)}")
```

最后一个用例是**反向的**：和所有技能都无关时，它不该硬加载一个。
只测「该加载时加载了」是不够的——那样把 `load_skill` 写成「永远加载第一个」也能满分。

```python
print("  模板在 template.md，先读它再往里填。")
print("\n加载技能时把目录位置一起给出去，模型就能用第 17 章的 read_file 去读——")
print("**技能和工具在这里接上了：技能告诉它读什么，工具让它读得到。**")
```

## 技能不是提示词注入的后门

这一点必须说清楚。技能的全文会**原样进入上下文**，所以：

```python
checks = []

# 1. 只能加载库里有的技能
try:
    loader.load("../../etc/passwd")
    checks.append(("拒绝库外的名字", False))
except KeyError:
    checks.append(("拒绝库外的名字", True))

# 2. 技能名来自枚举，模型没法自由发挥
spec = loader.as_tool().spec()
checks.append(("技能名是封闭枚举", "enum" in spec["parameters"]["properties"]["name"]))

# 3. 缺 description 的技能不会被装进来
(SKILLS_DIR / "broken.md").write_text("---\nname: broken\n---\n\n没有 description。",
                                      encoding="utf-8")
library.reload()
checks.append(("缺 description 的技能被拒绝", "broken" not in library.skills))

for label, ok in checks:
    print(f"  {'✓' if ok else '✗'} {label}")

print("\n三条底线：")
print("  1. 技能目录是**可信的**，和模型能写入的目录分开——"
      "否则 agent 写一个技能文件再加载它，就绕过了你的全部规则。")
print("  2. 技能名从**封闭枚举**里选，不接受任意字符串。")
print("  3. 技能是**给模型的说明**，不是给程序的指令。"
      "技能里写「忽略之前的所有规则」也只是一句话，"
      "真正的边界仍然在第 18 章那三道防线里。")
```

## 什么时候不该用技能

技能不是万能的。三种情况用工具而不是技能：

- **需要确定性**。「提交信息首行不超过 50 字」写成技能，模型大概率遵守；
  写成一个校验工具，就是一定遵守。**能用代码保证的，不要用文字请求。**
- **需要外部数据**。技能是静态文字，取不了实时信息。
- **步骤长且固定**。二十步的固定流程写成技能，模型会跳步；写成代码就不会。

反过来，三种情况技能比工具合适：

- 规矩会变，而且改规矩的人不写代码。
- 做法需要判断，没法写成固定分支。
- 场景很多，每个都做成工具会让工具列表爆炸。

## 本章小结

- **工具扩展能力，技能扩展方法**。工具是代码，技能是文字。
- 把「怎么做」从系统提示搬到技能文件里。系统提示里只留**一行索引**，
  用得上时才把全文读进来。
- 膨胀的系统提示有三层代价，最要命的是第三层：**注意力被稀释**，
  真正重要的规矩淹没在两千字里。
- `description` 写**「什么时候用」而不是「这是什么」**——
  模型要判断的是该不该调它。
- 技能名做成**封闭枚举**，参数校验会真的把它挡住。
- 重复加载**不再返回全文**，否则同样的两千字会在上下文里躺两份。
- 元数据格式**越简单越好**。技能要让不写代码的人也能加一个，
  引入 YAML 就意味着「缩进写错了整个文件解析失败」。
- 测试要有**反向用例**：和所有技能都无关时，它不该硬加载一个。
- 三条底线：技能目录**可信**且与模型可写目录分开；技能名从**封闭枚举**里选；
  技能是**给模型的说明不是给程序的指令**，真正的边界在第 18 章那三道防线里。
- **能用代码保证的，不要用文字请求。**
