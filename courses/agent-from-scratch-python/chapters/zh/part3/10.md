# 主循环

前面九章的 `run` 函数一直是个草稿：一个 for 循环，跑满次数就返回一句「循环次数用完了」。
这一章把它写成正经的东西。

主循环是 agent 的心脏，它要回答四个问题：

1. **什么时候停？** 模型不再要求调工具就停——但还有别的停法。
2. **停不下来怎么办？** 模型可能一直要求调工具，得有预算。
3. **一轮里多个工具调用怎么办？** 顺序执行还是并发？
4. **每一步要让外界知道什么？** 第 3 章那条事件流该派上用场了。

## 本章起点

```python
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Callable, Iterator, Literal, Protocol
import json, urllib.request, uuid

def new_id(prefix: str) -> str:
    return f"{prefix}_{uuid.uuid4().hex[:12]}"

@dataclass
class TextBlock:
    text: str
    type: Literal["text"] = "text"

@dataclass
class ToolUseBlock:
    name: str
    args: dict[str, Any]
    id: str = field(default_factory=lambda: new_id("call"))
    type: Literal["tool_use"] = "tool_use"

@dataclass
class ToolResultBlock:
    tool_use_id: str
    content: str
    is_error: bool = False
    type: Literal["tool_result"] = "tool_result"

@dataclass
class Message:
    role: Literal["system", "user", "assistant", "tool"]
    content: list[Any]
    id: str = field(default_factory=lambda: new_id("msg"))
    def text(self) -> str:
        return "".join(b.text for b in self.content if isinstance(b, TextBlock))
    def tool_uses(self) -> list[ToolUseBlock]:
        return [b for b in self.content if isinstance(b, ToolUseBlock)]

@dataclass
class Delta:
    kind: Literal["text", "tool_use"]
    text: str = ""
    tool: ToolUseBlock | None = None

class Model(Protocol):
    def stream(self, messages: list[Message], tools: list[dict]) -> Iterator[Delta]: ...

@dataclass
class ToolResult:
    content: str
    is_error: bool = False
    is_invalid_args: bool = False

@dataclass
class Tool:
    name: str
    description: str
    parameters: dict[str, Any]
    run: Callable[..., str]
    def spec(self): return {"name": self.name, "description": self.description,
                            "parameters": self.parameters}

def validate_args(args, schema) -> str:
    if not isinstance(args, dict): return "参数必须是一个对象"
    if "_invalid_json" in args: return f"参数不是合法的 JSON：{args['_invalid_json'][:120]}"
    props, required = schema.get("properties") or {}, schema.get("required") or []
    missing = [k for k in required if k not in args]
    if missing: return f"缺少必填参数 {'、'.join(missing)}"
    unknown = [k for k in args if k not in props]
    if unknown: return f"不认识的参数 {'、'.join(unknown)}；这个工具接受 {'、'.join(props)}"
    checks = {"string": str, "number": (int, float), "integer": int,
              "boolean": bool, "array": list, "object": dict}
    for k, v in args.items():
        spec = props[k]
        pt = checks.get(spec.get("type"))
        if pt and not isinstance(v, pt):
            return f"参数 {k} 应该是 {spec.get('type')}，收到的是 {type(v).__name__}"
        if spec.get("enum") and v not in spec["enum"]:
            return f"参数 {k} 只能是 {'、'.join(map(str, spec['enum']))} 之一"
    return ""

class ToolRegistry:
    def __init__(self, tools=None): self._tools = {t.name: t for t in (tools or [])}
    def add(self, tool): self._tools[tool.name] = tool
    def specs(self): return [t.spec() for t in self._tools.values()]
    def names(self): return list(self._tools)
    def execute(self, name, args) -> ToolResult:
        tool = self._tools.get(name)
        if tool is None:
            return ToolResult(f"没有叫 {name} 的工具。可用的：{'、'.join(self.names())}", is_error=True)
        problem = validate_args(args, tool.parameters)
        if problem:
            return ToolResult(f"参数不合格：{problem}", is_error=True, is_invalid_args=True)
        try:
            return ToolResult(str(tool.run(**args)))
        except Exception as exc:
            return ToolResult(f"{type(exc).__name__}: {exc}", is_error=True)

def to_openai_messages(messages):
    out = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({"role": "assistant", "content": text or None, "tool_calls": [
                {"id": c.id, "type": "function",
                 "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                for c in calls]})
        else:
            out.append({"role": m.role, "content": text})
    return out

def to_openai_tools(tools):
    return [{"type": "function", "function": {"name": t["name"],
             "description": t.get("description", ""),
             "parameters": t.get("parameters", {"type": "object", "properties": {}})}} for t in tools]

class ToolCallAccumulator:
    def __init__(self): self._slots = {}
    def feed(self, raw_calls):
        for item in raw_calls:
            s = self._slots.setdefault(int(item.get("index", 0)), {"id": "", "name": "", "arguments": ""})
            if item.get("id"): s["id"] = item["id"]
            fn = item.get("function") or {}
            if fn.get("name"): s["name"] = fn["name"]
            if fn.get("arguments"): s["arguments"] += fn["arguments"]
    def finish(self):
        out = []
        for i in sorted(self._slots):
            s = self._slots[i]
            raw = s["arguments"] or "{}"
            try:
                args = json.loads(raw)
                if not isinstance(args, dict): args = {"_raw": raw}
            except json.JSONDecodeError:
                args = {"_invalid_json": raw}
            out.append(ToolUseBlock(id=s["id"] or new_id("call"), name=s["name"], args=args))
        return out

def parse_sse_line(line):
    line = line.strip()
    if not line or line.startswith(":") or not line.startswith("data:"): return None
    payload = line[5:].strip()
    if payload == "[DONE]": return None
    try: return json.loads(payload)
    except json.JSONDecodeError: return None

class OpenAICompatModel:
    def __init__(self, base_url, model, timeout=180):
        self.base_url, self.model, self.timeout = base_url.rstrip("/"), model, timeout
    def stream(self, messages, tools):
        body = {"model": self.model, "messages": to_openai_messages(messages), "stream": True}
        if tools: body["tools"] = to_openai_tools(tools)
        req = urllib.request.Request(f"{self.base_url}/chat/completions",
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"}, method="POST")
        acc = ToolCallAccumulator()
        with urllib.request.urlopen(req, timeout=self.timeout) as resp:
            for raw in resp:
                chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
                if chunk is None: continue
                d = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
                if d.get("content"): yield Delta(kind="text", text=d["content"])
                if d.get("tool_calls"): acc.feed(d["tool_calls"])
        for b in acc.finish(): yield Delta(kind="tool_use", tool=b)

import datetime
def get_time(city: str, unit: str = "24h") -> str:
    now = datetime.datetime.now()
    return f"{city} 现在是 {now.strftime('%H:%M') if unit == '24h' else now.strftime('%I:%M %p')}"

def calculate(expression: str) -> str:
    allowed = set("0123456789+-*/(). ")
    if not set(expression) <= allowed:
        raise ValueError(f"表达式里有不允许的字符：{sorted(set(expression) - allowed)}")
    return str(eval(expression, {"__builtins__": {}}, {}))

registry = ToolRegistry([
    Tool("get_time", "查询某个城市的当前时间",
         {"type": "object", "properties": {
             "city": {"type": "string", "description": "城市名"},
             "unit": {"type": "string", "enum": ["24h", "12h"]}},
          "required": ["city"]}, get_time),
    Tool("calculate", "计算一个算术表达式，比如 (3+5)*2",
         {"type": "object", "properties": {
             "expression": {"type": "string", "description": "只含数字和 + - * / ( )"}},
          "required": ["expression"]}, calculate),
])

real_model = OpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
print("前九章的成果已就位")
```

## 事件流回来了

第 3 章造的事件流，到这一章才真正派上用场——主循环的每一步都该让外界知道。

```python
@dataclass
class Event:
    kind: str
    data: dict


class EventStream:
    def __init__(self) -> None:
        self._subscribers: list[Callable[[Event], None]] = []

    def subscribe(self, fn):
        self._subscribers.append(fn)
        return lambda: self._subscribers.remove(fn) if fn in self._subscribers else None

    def emit(self, kind: str, **data) -> None:
        event = Event(kind=kind, data=data)
        for fn in list(self._subscribers):
            try:
                fn(event)
            except Exception as exc:
                print(f"  [订阅者出错，已忽略] {type(exc).__name__}: {exc}")


print("事件流就绪")
```

## 停止的四种理由

草稿版只有一种停法：模型不调工具了。真实的主循环有四种，**而且要让调用方知道是哪一种**。

```python
from enum import Enum

class StopReason(str, Enum):
    COMPLETED = "completed"          # 模型给出了最终回答，正常结束
    BUDGET_EXHAUSTED = "budget"      # 循环预算用完了，它还想接着调工具
    CANCELLED = "cancelled"          # 外部要求停下（下一章讲）
    ERROR = "error"                  # 模型调用本身失败了


@dataclass
class TurnResult:
    """一轮的结果。

    为什么不只返回一个字符串：调用方需要区分「答完了」和「预算用完了」——
    前者可以直接展示，后者应该提示用户「它还没做完，要不要继续」。
    只返回文本的话这个区别就丢了。
    """
    text: str
    stop_reason: StopReason
    rounds_used: int
    messages: list[Message] = field(default_factory=list)   # 这一轮新增的消息
    error: str = ""

    @property
    def completed(self) -> bool:
        return self.stop_reason is StopReason.COMPLETED
```

## 主循环

```python
class AgentLoop:
    """一轮对话的主循环。

    职责边界：它只管「问模型 → 执行工具 → 再问模型」这个循环本身，
    不管历史存哪、不管会话怎么组织——那些是第 13、14 章的事。
    """

    def __init__(self, model: Model, registry: ToolRegistry, stream: EventStream | None = None,
                 max_rounds: int = 8) -> None:
        self.model = model
        self.registry = registry
        self.stream = stream or EventStream()
        self.max_rounds = max_rounds

    def run(self, history: list[Message]) -> TurnResult:
        turn_id = new_id("turn")
        new_messages: list[Message] = []
        self.stream.emit("turn_start", turn_id=turn_id)

        for round_no in range(1, self.max_rounds + 1):
            self.stream.emit("round_start", turn_id=turn_id, round=round_no)

            try:
                reply = self._one_round(history + new_messages)
            except Exception as exc:
                # 模型调用失败：网络断了、服务返回 500、响应格式不对。
                # 这类错误不该让调用方拿到一个异常——它拿到的应该是一个"这轮失败了"的结果。
                self.stream.emit("error", turn_id=turn_id, message=str(exc))
                return TurnResult(text="", stop_reason=StopReason.ERROR,
                                  rounds_used=round_no, messages=new_messages,
                                  error=f"{type(exc).__name__}: {exc}")

            new_messages.append(reply)
            calls = reply.tool_uses()

            if not calls:
                # 模型不再要求调工具 = 它认为答完了
                self.stream.emit("turn_done", turn_id=turn_id, reason=StopReason.COMPLETED.value)
                return TurnResult(text=reply.text(), stop_reason=StopReason.COMPLETED,
                                  rounds_used=round_no, messages=new_messages)

            results = self._run_tools(turn_id, calls)
            new_messages.append(Message(role="tool", content=results))

        # 预算用完了，而模型还想接着调工具
        self.stream.emit("turn_done", turn_id=turn_id, reason=StopReason.BUDGET_EXHAUSTED.value)
        return TurnResult(
            text=new_messages[-2].text() if len(new_messages) >= 2 else "",
            stop_reason=StopReason.BUDGET_EXHAUSTED,
            rounds_used=self.max_rounds, messages=new_messages)

    def _one_round(self, messages: list[Message]) -> Message:
        """问一次模型，把增量转成事件，返回攒好的消息。"""
        text_parts: list[str] = []
        blocks: list[Any] = []
        for d in self.model.stream(messages, self.registry.specs()):
            if d.kind == "text":
                text_parts.append(d.text)
                self.stream.emit("text_delta", delta=d.text)
            elif d.tool is not None:
                if text_parts:
                    blocks.append(TextBlock(text="".join(text_parts)))
                    text_parts = []
                blocks.append(d.tool)
        if text_parts:
            blocks.append(TextBlock(text="".join(text_parts)))
        message = Message(role="assistant", content=blocks)
        self.stream.emit("message_done", message_id=message.id, text=message.text())
        return message

    def _run_tools(self, turn_id: str, calls: list[ToolUseBlock]) -> list[ToolResultBlock]:
        """顺序执行这一轮的所有工具调用。并发版本在下一章。"""
        results = []
        for call in calls:
            self.stream.emit("tool_start", turn_id=turn_id, name=call.name,
                             args=call.args, call_id=call.id)
            r = self.registry.execute(call.name, call.args)
            self.stream.emit("tool_end", turn_id=turn_id, name=call.name,
                             call_id=call.id, ok=not r.is_error, output=r.content)
            results.append(ToolResultBlock(tool_use_id=call.id, content=r.content,
                                           is_error=r.is_error))
        return results


print("主循环就绪")
```

## 跑一次，看事件流

```python
stream = EventStream()

def show(e: Event) -> None:
    if e.kind == "text_delta":
        print(e.data["delta"], end="", flush=True)
    elif e.kind == "tool_start":
        print(f"\n  ⚙ 开始执行 {e.data['name']}({e.data['args']})")
    elif e.kind == "tool_end":
        mark = "✓" if e.data["ok"] else "✗"
        print(f"  {mark} {e.data['name']} → {e.data['output'][:60]}")
    elif e.kind == "turn_done":
        print(f"\n  ── 这一轮结束，原因：{e.data['reason']}")

stream.subscribe(show)

loop = AgentLoop(real_model, registry, stream)
history = [
    Message(role="system", content=[TextBlock(text="你可以使用工具。算数一律调用工具，不要心算。")]),
    Message(role="user", content=[TextBlock(text="(128 + 47) * 3 等于多少？再告诉我北京现在几点。")]),
]
result = loop.run(history)

print(f"\n停止原因：{result.stop_reason.value}")
print(f"用了 {result.rounds_used} 轮，新增 {len(result.messages)} 条消息")
print(f"最终答案：{result.text}")
```

**注意事件流里的顺序**：文字增量、工具开始、工具结束、下一轮的文字增量……
这正是一个 agent 界面需要显示的全部信息。

## 预算用完是什么样

给一个它做不完的任务，把预算压到 2 轮：

```python
tight = AgentLoop(real_model, registry, EventStream(), max_rounds=2)
result = tight.run([
    Message(role="system", content=[TextBlock(text="你可以使用工具。每次只算一步，不要一次算完。")]),
    Message(role="user", content=[TextBlock(text=
        "依次算出：1+1、然后结果乘 3、然后再加 10、然后再乘 2。每一步都要调用工具。")]),
])
print(f"停止原因：{result.stop_reason.value}")
print(f"用了 {result.rounds_used} 轮")
if result.stop_reason is StopReason.BUDGET_EXHAUSTED:
    print("→ 界面应当提示：它还没做完，要不要让它继续？")
```

**为什么必须有预算**：模型可能陷入循环——反复调同一个工具、或者两个工具互相触发。
没有预算的 agent 会一直烧钱直到你发现。

## 一轮里多个工具调用

上面那个问题里，模型可能在**同一轮**里同时要求算数和查时间。看看它是怎么做的：

```python
counts = []
probe = EventStream()
probe.subscribe(lambda e: counts.append(e.data["name"]) if e.kind == "tool_start" else None)

loop2 = AgentLoop(real_model, registry, probe)
result = loop2.run([
    Message(role="system", content=[TextBlock(text="你可以使用工具。算数一律调用工具。")]),
    Message(role="user", content=[TextBlock(text="算一下 7*8，同时告诉我上海现在几点。")]),
])
print(f"一共调了 {len(counts)} 次工具：{counts}")
print(f"用了 {result.rounds_used} 轮 —— 如果是 1 轮，说明模型把两个调用放在了同一轮里")
print("答案：", result.text)
```

同一轮里的多个调用现在是**顺序执行**的。如果每个工具要跑几秒，用户就得等它们排队。

## 本章小结

- 主循环的职责边界很窄：**只管问模型、执行工具、再问模型**。历史怎么存、会话怎么组织
  是别的章的事。
- 停止有四种理由，**调用方需要能区分**——尤其是「答完了」和「预算用完了」。
- **必须有循环预算**。模型会陷入循环，没有预算的 agent 会一直烧钱。
- 模型调用失败要变成「这轮失败了」的结果，**不要把异常抛给调用方**。
- 每一步都往事件流上发东西，界面需要的信息全在那里。
