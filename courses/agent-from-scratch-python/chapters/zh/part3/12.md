# 出错的时候

前面十一章造出来的 agent，在一切顺利时能跑。这一章处理不顺利的情况。

错误分三层，处理办法完全不同：

| 层 | 例子 | 办法 |
|---|---|---|
| 传输层 | 连接被拒、服务返回 503、超时 | **重试**，退避着重试 |
| 工具层 | 文件不存在、除数为零、工具崩了 | 变成结果回给模型（第 9 章已做完） |
| 模型行为层 | 参数一直填错、空回复、原地打转 | **干预**——这一章的主题 |

第三层是新东西。前两层是「东西坏了」，第三层是「东西没坏，但模型在犯病」。
它不会抛异常，不会返回错误码，程序看起来一切正常，只是**永远得不到结果**。

为了聚焦，这一章用第 10 章那个顺序版主循环。把第 11 章的并发与取消加回去是机械的，
不影响这一章的任何结论。

```python
def validate_args(args, schema) -> str:
    if not isinstance(args, dict): return "参数必须是一个对象"
    if "_invalid_json" in args: return f"参数不是合法的 JSON：{args['_invalid_json'][:120]}"
    props, required = schema.get("properties") or {}, schema.get("required") or []
    missing = [k for k in required if k not in args]
    if missing: return f"缺少必填参数 {'、'.join(missing)}"
    unknown = [k for k in args if k not in props]
    if unknown: return f"不认识的参数 {'、'.join(unknown)}；这个工具接受 {'、'.join(props)}"
    checks = {"string": str, "number": (int, float), "integer": int,
              "boolean": bool, "array": list, "object": dict}
    for k, v in args.items():
        spec = props[k]
        pt = checks.get(spec.get("type"))
        if pt and not isinstance(v, pt):
            return f"参数 {k} 应该是 {spec.get('type')}，收到的是 {type(v).__name__}"
        if spec.get("enum") and v not in spec["enum"]:
            return f"参数 {k} 只能是 {'、'.join(map(str, spec['enum']))} 之一"
    return ""

class ToolRegistry:
    def __init__(self, tools=None): self._tools = {t.name: t for t in (tools or [])}
    def add(self, tool): self._tools[tool.name] = tool
    def get(self, name): return self._tools.get(name)
    def names(self): return list(self._tools)
    def specs(self, exclude=()): 
        return [t.spec() for n, t in self._tools.items() if n not in exclude]
    def execute(self, name, args) -> ToolResult:
        tool = self._tools.get(name)
        if tool is None:
            return ToolResult(f"没有叫 {name} 的工具。可用的：{'、'.join(self.names())}", is_error=True)
        problem = validate_args(args, tool.parameters)
        if problem:
            return ToolResult(f"参数不合格：{problem}", is_error=True, is_invalid_args=True)
        try:
            return ToolResult(str(tool.run(**args)))
        except Exception as exc:
            return ToolResult(f"{type(exc).__name__}: {exc}", is_error=True)

def to_openai_messages(messages):
    out = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({"role": "assistant", "content": text or None, "tool_calls": [
                {"id": c.id, "type": "function",
                 "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                for c in calls]})
        else:
            out.append({"role": m.role, "content": text})
    return out

def to_openai_tools(tools):
    return [{"type": "function", "function": {"name": t["name"], "description": t.get("description", ""),
             "parameters": t.get("parameters", {"type": "object", "properties": {}})}} for t in tools]

class ToolCallAccumulator:
    def __init__(self): self._slots = {}
    def feed(self, raw_calls):
        for item in raw_calls:
            s = self._slots.setdefault(int(item.get("index", 0)), {"id": "", "name": "", "arguments": ""})
            if item.get("id"): s["id"] = item["id"]
            fn = item.get("function") or {}
            if fn.get("name"): s["name"] = fn["name"]
            if fn.get("arguments"): s["arguments"] += fn["arguments"]
    def finish(self):
        out = []
        for i in sorted(self._slots):
            s = self._slots[i]
            raw = s["arguments"] or "{}"
            try:
                args = json.loads(raw)
                if not isinstance(args, dict): args = {"_raw": raw}
            except json.JSONDecodeError:
                args = {"_invalid_json": raw}
            out.append(ToolUseBlock(id=s["id"] or new_id("call"), name=s["name"], args=args))
        return out

def parse_sse_line(line):
    line = line.strip()
    if not line or line.startswith(":") or not line.startswith("data:"): return None
    payload = line[5:].strip()
    if payload == "[DONE]": return None
    try: return json.loads(payload)
    except json.JSONDecodeError: return None

class OpenAICompatModel:
    def __init__(self, base_url, model, timeout=180):
        self.base_url, self.model, self.timeout = base_url.rstrip("/"), model, timeout
    def stream(self, messages, tools):
        body = {"model": self.model, "messages": to_openai_messages(messages), "stream": True}
        if tools: body["tools"] = to_openai_tools(tools)
        req = urllib.request.Request(f"{self.base_url}/chat/completions",
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"}, method="POST")
        acc = ToolCallAccumulator()
        with urllib.request.urlopen(req, timeout=self.timeout) as resp:
            for raw in resp:
                chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
                if chunk is None: continue
                d = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
                if d.get("content"): yield Delta(kind="text", text=d["content"])
                if d.get("tool_calls"): acc.feed(d["tool_calls"])
        for b in acc.finish(): yield Delta(kind="tool_use", tool=b)

@dataclass
class Event:

    data: dict

class EventStream:
    def __init__(self): self._subscribers = []
    def subscribe(self, fn):
        self._subscribers.append(fn)
        return lambda: self._subscribers.remove(fn) if fn in self._subscribers else None
    def emit(self, kind, **data):
        event = Event(kind=kind, data=data)
        for fn in list(self._subscribers):
            try: fn(event)
            except Exception as exc: print(f"  [订阅者出错，已忽略] {type(exc).__name__}: {exc}")
```

## 第一层：什么该重试，什么不该

重试的第一原则是**分清「重试有用」和「重试没用」**。
连接被拒、服务 503、限流 429——这些重试有用。请求体格式错了返回 400——重试一万次也是 400，
只会浪费时间和额度。

```python
# 这些状态码值得重试。共同点：问题出在服务端或链路上，请求本身没毛病。
RETRYABLE_STATUS = {408, 409, 425, 429, 500, 502, 503, 504}


class RetryPolicy:
    def __init__(self, max_attempts: int = 3, base_delay: float = 0.5,
                 max_delay: float = 8.0) -> None:
        self.max_attempts = max_attempts
        self.base_delay = base_delay
        self.max_delay = max_delay

    def should_retry(self, exc: Exception) -> bool:
        if isinstance(exc, urllib.error.HTTPError):
            return exc.code in RETRYABLE_STATUS
        if isinstance(exc, (urllib.error.URLError, TimeoutError, ConnectionError)):
            return True     # 连不上、超时：链路问题，值得再试
        return False        # 其他（JSON 解析失败、我们自己的 bug）：重试没有意义

    def delay_for(self, attempt: int) -> float:
        """指数退避 + 抖动。

        指数退避：0.5、1、2、4 秒——服务端过载时，一群客户端一起猛敲会让它更起不来。
        抖动：如果一百个客户端同时被限流、又用同样的退避曲线，它们会**同时**回来，
        再次把服务打垮。加一点随机让它们错开。
        """
        delay = min(self.base_delay * (2 ** (attempt - 1)), self.max_delay)
        return delay * (0.5 + random.random() * 0.5)


class RetryingModel:
    """包在任何 Model 外面，给它加上重试。

    做成包装而不是改 OpenAICompatModel：重试策略和协议实现是两件事，
    换模型不该重写重试，换重试策略也不该动协议代码。
    """

    def __init__(self, inner: Model, policy: RetryPolicy | None = None,
                 events: EventStream | None = None) -> None:
        self.inner = inner
        self.policy = policy or RetryPolicy()
        # 这个字段叫 events 而不是 stream 是有原因的：
        # 下面有个方法也叫 stream，实例属性会把同名方法盖掉，
        # 调用 model.stream(...) 就会变成「调用一个 EventStream 对象」。
        # Python 的属性和方法住在同一个命名空间里，这个坑很常见。
        self.events = events or EventStream()

    def stream(self, messages, tools):
        for attempt in range(1, self.policy.max_attempts + 1):
            produced = False
            try:
                for delta in self.inner.stream(messages, tools):
                    produced = True
                    yield delta
                return
            except Exception as exc:
                if produced:
                    # 已经吐出去的增量收不回来了。这时候重试会让同一段话说两遍。
                    # 所以：只重试「还没产出任何内容」的失败。
                    raise
                if attempt >= self.policy.max_attempts or not self.policy.should_retry(exc):
                    raise
                delay = self.policy.delay_for(attempt)
                self.events.emit("retry", attempt=attempt, delay=round(delay, 2),
                                 reason=f"{type(exc).__name__}: {exc}")
                time.sleep(delay)


print("重试层就绪")
```

**那个坑值得单说**：流式接口一旦开始吐内容，重试就不再安全——用户已经看到半句话了，
重来一遍会变成「半句话 + 完整的话」。所以只重试**还没产出任何东西**的失败。
真正健壮的做法是断点续传，但那需要服务端支持，多数服务没有。

看看它是怎么退避的：

```python
class FlakyModel:
    """前 n 次连接失败，之后正常。用来验证重试逻辑，不联网。"""
    def __init__(self, fail_times: int):
        self.fail_times, self.calls = fail_times, 0
    def stream(self, messages, tools):
        self.calls += 1
        if self.calls <= self.fail_times:
            raise urllib.error.URLError("连接被拒绝")
        yield Delta(kind="text", text="终于连上了。")

log = EventStream()
log.subscribe(lambda e: print(f"  第 {e.data['attempt']} 次失败，{e.data['delay']} 秒后重试"
                              f"（{e.data['reason']}）") if e.kind == "retry" else None)

start = time.time()
model = RetryingModel(FlakyModel(fail_times=2), RetryPolicy(max_attempts=4), log)
print("结果：", "".join(d.text for d in model.stream([], [])))
print(f"总共花了 {time.time() - start:.1f} 秒")

# 不该重试的错误：一次就放弃
class BadRequestModel:
    def stream(self, messages, tools):
        raise urllib.error.HTTPError("url", 400, "Bad Request", {}, None)
        yield  # 让它是个生成器

try:
    list(RetryingModel(BadRequestModel()).stream([], []))
except urllib.error.HTTPError as exc:
    print(f"\n400 错误没有重试，直接抛出：{exc.code} —— 重试它没有意义")
```

## 第二层：模型在犯病

这三种病症的共同点是**程序看起来一切正常**：没有异常，没有错误码，
只是循环一直转下去、或者永远得不到结果。

```python
@dataclass
class Symptom:
    """一次诊断结果。nudge 是要注入给模型的纠正提示。"""
    name: str
    nudge: str = ""
    give_up: bool = False


class BehaviorGuard:
    """看着模型的行为，发现它在犯病就干预。

    三种病症：
      1. 参数反复填错——同一个工具连续填错 N 次，把这个工具摘掉。
      2. 空回复——既没说话也没调工具。这会让循环空转到预算耗尽。
      3. 原地打转——反复用**完全相同的参数**调**同一个工具**。
    """

    def __init__(self, max_invalid_per_tool: int = 2, max_repeat: int = 3,
                 max_empty: int = 2) -> None:
        self.max_invalid_per_tool = max_invalid_per_tool
        self.max_repeat = max_repeat
        self.max_empty = max_empty
        self.invalid_counts: dict[str, int] = {}
        self.call_counts: dict[str, int] = {}
        self.empty_replies = 0
        self.disabled: set[str] = set()

    @staticmethod
    def fingerprint(call: ToolUseBlock) -> str:
        """一次调用的指纹。参数要排序后序列化——{"a":1,"b":2} 和 {"b":2,"a":1} 是同一次调用。"""
        return call.name + "|" + json.dumps(call.args, sort_keys=True, ensure_ascii=False)

    def check_reply(self, reply: Message) -> Symptom | None:
        calls = reply.tool_uses()

        # 病症 2：空回复
        if not calls and not reply.text().strip():
            self.empty_replies += 1
            if self.empty_replies >= self.max_empty:
                return Symptom("空回复", give_up=True)
            return Symptom("空回复", nudge="你刚才什么都没回。请直接给出答案，"
                                          "或者调用一个工具。")
        self.empty_replies = 0

        # 病症 3：原地打转
        for call in calls:
            fp = self.fingerprint(call)
            self.call_counts[fp] = self.call_counts.get(fp, 0) + 1
            if self.call_counts[fp] >= self.max_repeat:
                return Symptom("原地打转", nudge=(
                    f"你已经用完全相同的参数调用了 {self.max_repeat} 次 {call.name}，"
                    f"结果不会变。请换一个思路，或者基于已有的结果直接回答。"))
        return None

    def check_tool_result(self, call: ToolUseBlock, result: ToolResult) -> Symptom | None:
        # 病症 1：参数反复填错
        if not result.is_invalid_args:
            self.invalid_counts.pop(call.name, None)   # 填对一次就清零
            return None
        n = self.invalid_counts.get(call.name, 0) + 1
        self.invalid_counts[call.name] = n
        if n >= self.max_invalid_per_tool:
            self.disabled.add(call.name)
            return Symptom("参数反复填错", nudge=(
                f"{call.name} 的参数你连续填错了 {n} 次，我把这个工具停掉了。"
                f"请用别的办法，或者直接告诉用户你做不到。"))
        return None


print("行为守卫就绪")
```

**三个设计要点**：

- **纠正提示要具体**。「参数错误」模型改不动；「你已经用相同参数调了 3 次 calculate，结果不会变」
  它才知道该换招。这些话是写给模型看的，标准和写给人看的一样：说清楚发生了什么、该怎么办。
- **摘掉工具比一直纠正有效**。工具从 `tools` 列表里消失，模型就再也调不到它了——
  这比反复劝说可靠得多。**能用机制解决的，不要靠提示词解决。**
- **计数要能清零**。填对一次就把计数清掉，否则一次偶然的错误会一直记在账上。

## 装到主循环里

```python
class GuardedAgentLoop:
    def __init__(self, model: Model, registry: ToolRegistry, stream: EventStream | None = None,
                 max_rounds: int = 8, retry: RetryPolicy | None = None) -> None:
        self.stream = stream or EventStream()
        self.model = RetryingModel(model, retry or RetryPolicy(), events=self.stream)
        self.registry = registry
        self.max_rounds = max_rounds

    def run(self, history: list[Message]) -> TurnResult:
        turn_id = new_id("turn")
        guard = BehaviorGuard()          # 每一轮对话一个新的守卫，计数不跨轮累积
        new_messages: list[Message] = []
        self.stream.emit("turn_start", turn_id=turn_id)

        for round_no in range(1, self.max_rounds + 1):
            try:
                reply = self._one_round(history + new_messages, guard.disabled)
            except Exception as exc:
                self.stream.emit("error", turn_id=turn_id, message=str(exc))
                return TurnResult("", StopReason.ERROR, round_no, new_messages,
                                  error=f"{type(exc).__name__}: {exc}")
            new_messages.append(reply)

            symptom = guard.check_reply(reply)
            if symptom is not None:
                self.stream.emit("symptom", name=symptom.name, give_up=symptom.give_up)
                if symptom.give_up:
                    return TurnResult(reply.text(), StopReason.GAVE_UP, round_no, new_messages,
                                      error=f"模型持续{symptom.name}")

            calls = reply.tool_uses()
            if not calls:
                if symptom is None:
                    self.stream.emit("turn_done", turn_id=turn_id, reason=StopReason.COMPLETED.value)
                    return TurnResult(reply.text(), StopReason.COMPLETED, round_no, new_messages)
                # 空回复：注入纠正提示，再来一轮
                new_messages.append(Message(role="user", content=[TextBlock(text=symptom.nudge)]))
                continue

            results = []
            nudges = [symptom.nudge] if symptom and symptom.nudge else []
            for call in calls:
                r = self.registry.execute(call.name, call.args)
                self.stream.emit("tool_end", name=call.name, ok=not r.is_error, output=r.content)
                results.append(ToolResultBlock(tool_use_id=call.id, content=r.content,
                                               is_error=r.is_error))
                s = guard.check_tool_result(call, r)
                if s is not None:
                    self.stream.emit("symptom", name=s.name, tool=call.name)
                    nudges.append(s.nudge)
            new_messages.append(Message(role="tool", content=results))

            # 纠正提示用 user 角色发。
            # 用 assistant 角色的话，模型会以为这是自己说的，纠正效果大打折扣；
            # 用 system 角色则很多服务只认开头那一条。
            if nudges:
                new_messages.append(Message(role="user",
                                            content=[TextBlock(text="\n".join(nudges))]))

        self.stream.emit("turn_done", turn_id=turn_id, reason=StopReason.BUDGET_EXHAUSTED.value)
        return TurnResult("", StopReason.BUDGET_EXHAUSTED, self.max_rounds, new_messages)

    def _one_round(self, messages: list[Message], disabled: set[str]) -> Message:
        text_parts, blocks = [], []
        for d in self.model.stream(messages, self.registry.specs(exclude=disabled)):
            if d.kind == "text":
                text_parts.append(d.text)
                self.stream.emit("text_delta", delta=d.text)
            elif d.tool is not None:
                if text_parts:
                    blocks.append(TextBlock(text="".join(text_parts)))
                    text_parts = []
                blocks.append(d.tool)
        if text_parts:
            blocks.append(TextBlock(text="".join(text_parts)))
        return Message(role="assistant", content=blocks)


print("带守卫的主循环就绪")
```

```python
class ScriptedModel:
    """按剧本出牌。剧本用完就一直重复最后一幕——正好用来演「原地打转」。"""
    def __init__(self, script, repeat_last=False):
        self.script, self.repeat_last, self.calls = script, repeat_last, 0
    def stream(self, messages, tools):
        i = min(self.calls, len(self.script) - 1) if self.repeat_last else self.calls
        self.calls += 1
        if i >= len(self.script):
            raise RuntimeError("剧本演完了")
        text, calls = self.script[i]
        if text:
            yield Delta(kind="text", text=text)
        for name, args in calls:
            yield Delta(kind="tool_use", tool=ToolUseBlock(name=name, args=args))
```

## 本章小结

- 错误分三层：**传输层重试、工具层变成结果、模型行为层干预**。
- 只重试「重试有用」的错误：连接失败、超时、429、5xx。400 重试一万次还是 400。
- 退避要**指数 + 抖动**。没有抖动，一群被限流的客户端会同时回来，再把服务打垮。
- 流式接口**一旦开始吐内容就不能重试**了——用户已经看到半句话。
- 模型犯病不会抛异常，程序看起来一切正常，只是永远得不到结果。三种常见病症：
  参数反复填错、空回复、原地打转。
- 「原地打转」的指纹要**把参数排序后序列化**，否则键顺序一变就认不出来。
- **能用机制解决的不要靠提示词**：把工具从列表里摘掉，比反复劝说可靠得多。
- 纠正提示用 **user 角色**注入，而且要**说清楚发生了什么、该怎么办**。
- 计数要能清零，否则一次偶然错误会一直记在账上。

第三部分到这里结束。我们的 agent 现在有了心脏：能循环、能并发、能停下、能扛住模型犯病。
