# 供应商适配

前五章一直在用假模型。这一章接上真的。

真模型的接口是别人定的，跟我们第 2 章定的内部协议长得不一样。所以需要一层**适配器**：
把内部协议翻译成对方要的格式，再把对方的回答翻译回内部协议。

这一章只做非流式（一次请求，等它把话说完），流式留到下一章。

## 为什么只对接一家

pi 的 `pi-ai` 包有两万多行，绝大部分在处理各家供应商的差异——Anthropic 的
`tool_use` 块、OpenAI 的 `tool_calls` 数组、Gemini 的 `functionCall`，还有各家不同的
流式分帧、不同的推理内容字段、不同的错误码。

这些差异对**理解 agent 原理毫无帮助**。所以这门课只对接一种：**OpenAI 兼容接口**。
它是事实上的通用格式，本地模型（llama.cpp、vLLM、Ollama）和大多数云服务都提供它。
理解了这一家，再看别家的适配器就是查文档的事。

## 两边的格式差在哪

| 我们的内部协议 | OpenAI 兼容接口 |
|---|---|
| `Message(role, content=[块…])` | `{"role": …, "content": "字符串"}` |
| `ToolUseBlock(id, name, args)` | 助手消息里的 `tool_calls: [{id, function:{name, arguments}}]` |
| `ToolResultBlock(tool_use_id, content)` | 单独一条 `{"role":"tool", "tool_call_id": …, "content": …}` |
| 工具声明 | `tools: [{"type":"function","function":{name, description, parameters}}]` |

三个要留意的地方：

1. **参数是字符串不是对象。** `function.arguments` 是一段 JSON **文本**，要自己解析。
   模型偶尔会吐出不合法的 JSON，得容错。
2. **工具结果是独立的消息。** 我们把一轮里的多个结果放在一条 `tool` 消息里，
   发过去时要**拆成多条**，每条带自己的 `tool_call_id`。
3. **一条助手消息可能既有文字又有工具调用**，翻译时两边都要带上。

## 本章起点

```python
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Iterator, Literal, Protocol
import uuid

def new_id(prefix: str) -> str:
    return f"{prefix}_{uuid.uuid4().hex[:12]}"

@dataclass
class TextBlock:
    text: str
    type: Literal["text"] = "text"

@dataclass
class ToolUseBlock:
    name: str
    args: dict[str, Any]
    id: str = field(default_factory=lambda: new_id("call"))
    type: Literal["tool_use"] = "tool_use"

@dataclass
class ToolResultBlock:
    tool_use_id: str
    content: str
    is_error: bool = False
    type: Literal["tool_result"] = "tool_result"

@dataclass
class Message:
    role: Literal["system", "user", "assistant", "tool"]
    content: list[Any]
    id: str = field(default_factory=lambda: new_id("msg"))
    def text(self) -> str:
        return "".join(b.text for b in self.content if isinstance(b, TextBlock))
    def tool_uses(self) -> list[ToolUseBlock]:
        return [b for b in self.content if isinstance(b, ToolUseBlock)]

@dataclass
class Delta:
    kind: Literal["text", "tool_use"]
    text: str = ""
    tool: ToolUseBlock | None = None

print("内部协议已就位")
```

## 出去的方向：内部协议 → 接口格式

```python
import json

def to_openai_messages(messages: list[Message]) -> list[dict]:
    """把内部消息翻译成接口要的格式。

    注意工具结果那一段：我们把一轮里的多个结果放在一条 tool 消息里，
    但接口要求一个结果一条消息。所以这里是一对多的展开。
    """
    out: list[dict] = []
    for m in messages:
        if m.role == "tool":
            for block in m.content:
                if isinstance(block, ToolResultBlock):
                    out.append({
                        "role": "tool",
                        "tool_call_id": block.tool_use_id,
                        "content": block.content,
                    })
            continue

        text = m.text()
        calls = m.tool_uses()

        if m.role == "assistant" and calls:
            entry: dict[str, Any] = {"role": "assistant", "content": text or None}
            entry["tool_calls"] = [
                {
                    "id": c.id,
                    "type": "function",
                    "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)},
                }
                for c in calls
            ]
            out.append(entry)
        else:
            out.append({"role": m.role, "content": text})
    return out


def to_openai_tools(tools: list[dict]) -> list[dict]:
    """工具声明的格式转换。我们内部用扁平的 {name, description, parameters}。"""
    return [
        {
            "type": "function",
            "function": {
                "name": t["name"],
                "description": t.get("description", ""),
                "parameters": t.get("parameters", {"type": "object", "properties": {}}),
            },
        }
        for t in tools
    ]


# 看一眼翻译结果
call = ToolUseBlock(name="get_time", args={"city": "北京"})
demo = [
    Message(role="user", content=[TextBlock(text="北京现在几点？")]),
    Message(role="assistant", content=[TextBlock(text="我查一下。"), call]),
    Message(role="tool", content=[ToolResultBlock(tool_use_id=call.id, content="14:32")]),
]
print(json.dumps(to_openai_messages(demo), ensure_ascii=False, indent=2))
```

跑一下能看到：**那条同时有文字和工具调用的助手消息**，翻译成了一个带 `tool_calls`
数组的对象；而工具结果被拆成了独立的一条 `role: "tool"` 消息。

## 回来的方向：接口格式 → 内部协议

```python
def parse_openai_response(payload: dict) -> Message:
    """把接口的回答翻译回内部消息。"""
    choice = (payload.get("choices") or [{}])[0]
    raw = choice.get("message") or {}

    blocks: list[Any] = []
    text = raw.get("content")
    if text:
        blocks.append(TextBlock(text=text))

    for tc in raw.get("tool_calls") or []:
        fn = tc.get("function") or {}
        # arguments 是一段 JSON 文本，不是对象。模型偶尔会吐出不合法的 JSON，
        # 这时不能让整轮崩掉——把原文留下，让上层决定怎么办。
        raw_args = fn.get("arguments") or "{}"
        try:
            args = json.loads(raw_args)
            if not isinstance(args, dict):
                args = {"_raw": raw_args}
        except json.JSONDecodeError:
            args = {"_invalid_json": raw_args}
        blocks.append(ToolUseBlock(
            id=tc.get("id") or new_id("call"),
            name=fn.get("name") or "",
            args=args,
        ))

    return Message(role="assistant", content=blocks)


# 拿一份典型的响应试试
sample = {
    "choices": [{
        "message": {
            "role": "assistant",
            "content": "我查一下。",
            "tool_calls": [{
                "id": "call_abc123",
                "type": "function",
                "function": {"name": "get_time", "arguments": '{"city": "北京"}'},
            }],
        },
        "finish_reason": "tool_calls",
    }]
}
msg = parse_openai_response(sample)
print("块类型：", [b.type for b in msg.content])
print("文字：", msg.text())
for c in msg.tool_uses():
    print(f"工具调用：{c.name}({c.args})  编号 {c.id}")
```

## 参数不是合法 JSON 时不能崩

```python
broken = {
    "choices": [{"message": {"role": "assistant", "content": None, "tool_calls": [
        {"id": "call_x", "type": "function",
         "function": {"name": "get_time", "arguments": '{"city": "北京"'}},   # 少一个右括号
    ]}}]
}
msg = parse_openai_response(broken)
print("没有崩，参数被原样保留下来了：", msg.tool_uses()[0].args)
```

这种事真的会发生，尤其是小模型或者被截断的响应。**适配层的职责是如实转达，不是替模型
遮丑**——把原文留着，第 12 章会讲怎么把这个错误告诉模型让它重试。

## 真的调一次

```python
import urllib.request

BASE_URL = "http://192.168.213.116:8084/v1"
MODEL = "qwen3.6-27B"


def call_model(messages: list[Message], tools: list[dict] | None = None, timeout: int = 120) -> Message:
    """一次非流式请求。"""
    body: dict[str, Any] = {
        "model": MODEL,
        "messages": to_openai_messages(messages),
    }
    if tools:
        body["tools"] = to_openai_tools(tools)

    req = urllib.request.Request(
        f"{BASE_URL}/chat/completions",
        data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
        headers={"Content-Type": "application/json"},
        method="POST",
    )
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        payload = json.load(resp)
    return parse_openai_response(payload)


answer = call_model([
    Message(role="system", content=[TextBlock(text="你是一个简洁的助教，只用一句话回答。")]),
    Message(role="user", content=[TextBlock(text="用一句话说明什么是尾递归。")]),
])
print("模型答：", answer.text())
```

**这是这门课第一次真的联网。** 前五章的代码一行没改——它们只依赖内部协议，
不知道底下接的是谁。

## 让它调工具

```python
tools = [{
    "name": "get_time",
    "description": "查询某个城市的当前时间",
    "parameters": {
        "type": "object",
        "properties": {"city": {"type": "string", "description": "城市名"}},
        "required": ["city"],
    },
}]

reply = call_model([
    Message(role="system", content=[TextBlock(text="需要查时间时就调用工具，不要自己编。")]),
    Message(role="user", content=[TextBlock(text="北京现在几点？")]),
], tools=tools)

print("文字：", reply.text() or "（没有文字）")
for c in reply.tool_uses():
    print(f"模型要求调用：{c.name}({c.args})")
if not reply.tool_uses():
    print("这次模型没调工具——小模型有时会自己编答案，第 12 章会讲怎么应对。")
```

## 把结果回填，让它给最终答案

```python
history = [
    Message(role="system", content=[TextBlock(text="需要查时间时就调用工具，不要自己编。")]),
    Message(role="user", content=[TextBlock(text="北京现在几点？")]),
]
reply = call_model(history, tools=tools)
history.append(reply)

calls = reply.tool_uses()
if calls:
    results = [ToolResultBlock(tool_use_id=c.id, content="14:32") for c in calls]
    history.append(Message(role="tool", content=results))
    final = call_model(history, tools=tools)
    print("最终回答：", final.text())
else:
    print("模型直接回答了：", reply.text())
```

一轮完整的往返跑通了——**这就是第 1 章那张图的真实版本**。

## 本章小结

- 适配层的职责是**双向翻译**：内部协议 ↔ 接口格式。
- 三个容易踩的地方：**参数是 JSON 文本不是对象**、**工具结果要拆成多条消息**、
  **一条助手消息可能既有文字又有工具调用**。
- 参数不是合法 JSON 时**不要崩，把原文留着**——适配层如实转达，不替模型遮丑。
- 只对接 OpenAI 兼容一家，因为各家差异对理解原理没有帮助。
- **前五章的代码一行没改**，它们不知道底下接的是谁。
