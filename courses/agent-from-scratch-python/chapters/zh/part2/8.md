```python
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Iterator, Literal, Protocol
import json
import urllib.request
import uuid

def new_id(prefix: str) -> str:
    return f"{prefix}_{uuid.uuid4().hex[:12]}"

# ── 第 2 章：消息与内容块 ──

@dataclass
class TextBlock:
    text: str
    type: Literal["text"] = "text"

@dataclass
class ToolUseBlock:
    name: str
    args: dict[str, Any]
    id: str = field(default_factory=lambda: new_id("call"))
    type: Literal["tool_use"] = "tool_use"

@dataclass
class ToolResultBlock:
    tool_use_id: str
    content: str
    is_error: bool = False
    type: Literal["tool_result"] = "tool_result"

@dataclass
class Message:
    role: Literal["system", "user", "assistant", "tool"]
    content: list[Any]
    id: str = field(default_factory=lambda: new_id("msg"))
    def text(self) -> str:
        return "".join(b.text for b in self.content if isinstance(b, TextBlock))
    def tool_uses(self) -> list[ToolUseBlock]:
        return [b for b in self.content if isinstance(b, ToolUseBlock)]

# ── 第 5 章：模型接口 ──

@dataclass
class Delta:
    kind: Literal["text", "tool_use"]
    text: str = ""
    tool: ToolUseBlock | None = None

class Model(Protocol):
    def stream(self, messages: list[Message], tools: list[dict]) -> Iterator[Delta]: ...

def collect(deltas: Iterator[Delta]) -> Message:
    text_parts: list[str] = []
    blocks: list[Any] = []
    for d in deltas:
        if d.kind == "text":
            text_parts.append(d.text)
        elif d.tool is not None:
            if text_parts:
                blocks.append(TextBlock(text="".join(text_parts)))
                text_parts = []
            blocks.append(d.tool)
    if text_parts:
        blocks.append(TextBlock(text="".join(text_parts)))
    return Message(role="assistant", content=blocks)

# ── 第 5 章：脚本化模型 ──

@dataclass
class ScriptedTurn:
    text: str = ""
    tool_calls: list[tuple[str, dict]] = field(default_factory=list)

class ScriptedModel:
    def __init__(self, script: list[ScriptedTurn], chunk_size: int = 4) -> None:
        self.script, self.chunk_size, self.call_count = script, chunk_size, 0

    def stream(self, messages: list[Message], tools: list[dict]) -> Iterator[Delta]:
        if self.call_count >= len(self.script):
            raise RuntimeError(f"剧本只有 {len(self.script)} 幕，模型被调用了第 {self.call_count + 1} 次")
        turn = self.script[self.call_count]
        self.call_count += 1
        for i in range(0, len(turn.text), self.chunk_size):
            yield Delta(kind="text", text=turn.text[i : i + self.chunk_size])
        for name, args in turn.tool_calls:
            yield Delta(kind="tool_use", tool=ToolUseBlock(name=name, args=args))

# ── 第 6 章：格式翻译 ──

def to_openai_messages(messages: list[Message]) -> list[dict]:
    out: list[dict] = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({
                "role": "assistant",
                "content": text or None,
                "tool_calls": [
                    {"id": c.id, "type": "function",
                     "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                    for c in calls
                ],
            })
        else:
            out.append({"role": m.role, "content": text})
    return out

def to_openai_tools(tools: list[dict]) -> list[dict]:
    return [{"type": "function", "function": {
        "name": t["name"], "description": t.get("description", ""),
        "parameters": t.get("parameters", {"type": "object", "properties": {}})}} for t in tools]

# ── 第 7 章：分片累积与推送解析 ──

class ToolCallAccumulator:
    def __init__(self) -> None:
        self._slots: dict[int, dict[str, str]] = {}

    def feed(self, raw_calls: list[dict]) -> None:
        for item in raw_calls:
            slot = self._slots.setdefault(int(item.get("index", 0)),
                                          {"id": "", "name": "", "arguments": ""})
            if item.get("id"):
                slot["id"] = item["id"]
            fn = item.get("function") or {}
            if fn.get("name"):
                slot["name"] = fn["name"]
            if fn.get("arguments"):
                slot["arguments"] += fn["arguments"]

    def finish(self) -> list[ToolUseBlock]:
        blocks = []
        for index in sorted(self._slots):
            slot = self._slots[index]
            raw = slot["arguments"] or "{}"
            try:
                args = json.loads(raw)
                if not isinstance(args, dict):
                    args = {"_raw": raw}
            except json.JSONDecodeError:
                args = {"_invalid_json": raw}
            blocks.append(ToolUseBlock(id=slot["id"] or new_id("call"),
                                       name=slot["name"], args=args))
        return blocks

def parse_sse_line(line: str) -> dict | None:
    line = line.strip()
    if not line or line.startswith(":") or not line.startswith("data:"):
        return None
    payload = line[5:].strip()
    if payload == "[DONE]":
        return None
    try:
        return json.loads(payload)
    except json.JSONDecodeError:
        return None

print("第 2、5、6、7 章的成果已就位")
```

## 真模型

```python
class OpenAICompatModel:
    """接 OpenAI 兼容接口的模型。

    它实现的是第 5 章那个 Model 协议——注意它并不继承任何东西，
    也不知道 Model 这个协议存在。Python 的结构化子类型（Protocol）就是这么用的：
    **只要方法签名对得上就算实现了**，被依赖的一方不必知道依赖它的人是谁。
    """

    def __init__(self, base_url: str, model: str, api_key: str = "", timeout: int = 180) -> None:
        self.base_url = base_url.rstrip("/")
        self.model = model
        self.api_key = api_key
        self.timeout = timeout

    def stream(self, messages: list[Message], tools: list[dict]) -> Iterator[Delta]:
        body: dict[str, Any] = {
            "model": self.model,
            "messages": to_openai_messages(messages),
            "stream": True,
        }
        if tools:
            body["tools"] = to_openai_tools(tools)

        headers = {"Content-Type": "application/json"}
        if self.api_key:
            headers["Authorization"] = f"Bearer {self.api_key}"

        req = urllib.request.Request(
            f"{self.base_url}/chat/completions",
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers=headers,
            method="POST",
        )

        acc = ToolCallAccumulator()
        with urllib.request.urlopen(req, timeout=self.timeout) as resp:
            for raw in resp:
                chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
                if chunk is None:
                    continue
                delta = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
                if delta.get("content"):
                    yield Delta(kind="text", text=delta["content"])
                if delta.get("tool_calls"):
                    acc.feed(delta["tool_calls"])
        for block in acc.finish():
            yield Delta(kind="tool_use", tool=block)


real_model = OpenAICompatModel(base_url="http://192.168.213.116:8084/v1", model="qwen3.6-27B")
print("真模型就绪：", real_model.model)
```

## 第 5 章那个 run 函数，一个字没改

```python
def get_time(city: str) -> str:
    return "14:32"

TOOLS = {"get_time": get_time}

TOOL_SPECS = [{
    "name": "get_time",
    "description": "查询某个城市的当前时间",
    "parameters": {
        "type": "object",
        "properties": {"city": {"type": "string", "description": "城市名"}},
        "required": ["city"],
    },
}]


def run(user_text: str, model: Model, max_rounds: int = 5) -> str:
    """这个函数在第 5 章就是这样，没有为真模型改动任何一处。

    它只依赖 Model 协议、Message 类型和 collect 函数——
    底下接的是剧本还是真模型，它不知道也不需要知道。
    """
    history = [
        Message(role="system", content=[TextBlock(text="需要查时间时就调用工具，不要自己编。")]),
        Message(role="user", content=[TextBlock(text=user_text)]),
    ]
    print(f"用户：{user_text}\n")

    for round_no in range(1, max_rounds + 1):
        reply = collect(model.stream(history, TOOL_SPECS))
        history.append(reply)

        if reply.text():
            print(f"模型：{reply.text()}")

        calls = reply.tool_uses()
        if not calls:
            return reply.text()

        results = []
        for call in calls:
            fn = TOOLS.get(call.name)
            if fn is None:
                print(f"工具：{call.name} 不存在")
                results.append(ToolResultBlock(
                    tool_use_id=call.id, content=f"没有叫 {call.name} 的工具", is_error=True))
                continue
            output = fn(**call.args)
            print(f"工具：{call.name}({call.args}) → {output}")
            results.append(ToolResultBlock(tool_use_id=call.id, content=output))
        history.append(Message(role="tool", content=results))

    return "循环次数用完了"
```

## 边跑边显示

`run` 用的是 `collect`，它会等一整条消息攒完。要打字机效果就别用 `collect`，
直接消费增量：

```python
def run_streaming(user_text: str, model: Model, max_rounds: int = 5) -> str:
    history = [
        Message(role="system", content=[TextBlock(text="需要查时间时就调用工具，不要自己编。")]),
        Message(role="user", content=[TextBlock(text=user_text)]),
    ]
    print(f"用户：{user_text}\n")

    for round_no in range(1, max_rounds + 1):
        text_parts: list[str] = []
        blocks: list[Any] = []
        printed_prefix = False

        for d in model.stream(history, TOOL_SPECS):
            if d.kind == "text":
                if not printed_prefix:
                    print("模型：", end="", flush=True)
                    printed_prefix = True
                print(d.text, end="", flush=True)      # 边收边显示
                text_parts.append(d.text)
            elif d.tool is not None:
                if text_parts:
                    blocks.append(TextBlock(text="".join(text_parts)))
                    text_parts = []
                blocks.append(d.tool)
        if text_parts:
            blocks.append(TextBlock(text="".join(text_parts)))
        if printed_prefix:
            print()

        reply = Message(role="assistant", content=blocks)
        history.append(reply)

        calls = reply.tool_uses()
        if not calls:
            return reply.text()

        results = []
        for call in calls:
            fn = TOOLS.get(call.name)
            output = fn(**call.args) if fn else f"没有叫 {call.name} 的工具"
            print(f"工具：{call.name}({call.args}) → {output}")
            results.append(ToolResultBlock(tool_use_id=call.id, content=output,
                                           is_error=fn is None))
        history.append(Message(role="tool", content=results))

    return "循环次数用完了"


print("最终答案：", run_streaming("北京现在几点？", real_model))
```

## 小模型有时会不听话

跑几次你可能会发现：它有时不调工具，直接编一个时间；或者调了工具但参数名写错。
这不是你的代码有问题，是**小模型的指令遵循能力有限**。

真实系统里的应对办法有几种，都留到第 12 章讲：把工具描述写得更明确、参数校验失败时
把错误回给模型让它重试、必要时强制它必须调某个工具。

现在先确认一件事——**即使模型不听话，我们的代码也不能崩**：

```python
# 故意给一个模型很可能答错的问题：它没有这个工具，只能说自己做不到
print("最终答案：", run("帮我订一张明天去上海的机票。", real_model))
```

不管模型怎么回答，`run` 都能正常结束——因为工具不存在时我们回了一条错误结果，
而不是抛异常。

## 本章小结

- 真模型只是 `Model` 协议的另一个实现，**调用它的代码一行不改**。
- Python 的 `Protocol` 是结构化子类型：**只要方法签名对得上就算实现**，
  实现方不必知道协议存在。
- 要打字机效果就**别用 `collect`**，直接消费增量流。
- 小模型不听话是常态，代码要能扛住——**工具不存在时回一条错误结果，不要抛异常**。

第二部分到这里结束。我们已经有了一个能跑真模型、能调真工具的最小 agent。
