# 流式解析

上一章那次请求是「发出去，等它把话说完，一次性拿回来」。真实产品不能这样——
模型答一段长文要几十秒，用户盯着空白页面。

这一章把它改成流式：**边吐边收**。

## 服务端推送长什么样

OpenAI 兼容接口的流式响应是一串文本行，每行以 `data: ` 开头，后面跟一个 JSON：

```
data: {"choices":[{"delta":{"role":"assistant"}}]}

data: {"choices":[{"delta":{"content":"北京"}}]}

data: {"choices":[{"delta":{"content":"现在"}}]}

data: [DONE]
```

规矩：

- 每个事件之间用**空行**分隔
- 最后一行是 `data: [DONE]`，它**不是 JSON**，要单独判断
- 每个 `delta` 只带**变化的部分**，不是完整内容

纯文字很好处理。麻烦在工具调用。

## 工具调用是分片来的

模型要调工具时，`tool_calls` 也是一点点吐出来的，而且**参数是按字符流分片**的：

```
data: {"choices":[{"delta":{"tool_calls":[{"index":0,"id":"call_x","function":{"name":"get_time","arguments":""}}]}}]}
data: {"choices":[{"delta":{"tool_calls":[{"index":0,"function":{"arguments":"{\"ci"}}]}}]}
data: {"choices":[{"delta":{"tool_calls":[{"index":0,"function":{"arguments":"ty\": \"北"}}]}}]}
data: {"choices":[{"delta":{"tool_calls":[{"index":0,"function":{"arguments":"京\"}"}}]}}]}
```

三件事要注意：

1. **`index` 才是身份标识**，不是 `id`。同一轮里调三个工具就是 index 0、1、2，
   而 `id` 和 `name` 通常只在第一片里出现。
2. **`arguments` 要按 index 拼接**，拼完才是完整的 JSON 文本。
3. **拼到一半的 JSON 是不合法的**，只能等这个工具调用结束了才解析。

这是流式解析里最容易写错的地方。写成「每来一片就 `json.loads` 一次」必然崩。

## 本章起点

```python
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Iterator, Literal
import json
import uuid

def new_id(prefix: str) -> str:
    return f"{prefix}_{uuid.uuid4().hex[:12]}"

@dataclass
class TextBlock:
    text: str
    type: Literal["text"] = "text"

@dataclass
class ToolUseBlock:
    name: str
    args: dict[str, Any]
    id: str = field(default_factory=lambda: new_id("call"))
    type: Literal["tool_use"] = "tool_use"

@dataclass
class ToolResultBlock:
    tool_use_id: str
    content: str
    is_error: bool = False
    type: Literal["tool_result"] = "tool_result"

@dataclass
class Message:
    role: Literal["system", "user", "assistant", "tool"]
    content: list[Any]
    id: str = field(default_factory=lambda: new_id("msg"))
    def text(self) -> str:
        return "".join(b.text for b in self.content if isinstance(b, TextBlock))
    def tool_uses(self) -> list[ToolUseBlock]:
        return [b for b in self.content if isinstance(b, ToolUseBlock)]

@dataclass
class Delta:
    kind: Literal["text", "tool_use"]
    text: str = ""
    tool: ToolUseBlock | None = None

def to_openai_messages(messages: list[Message]) -> list[dict]:
    out: list[dict] = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({
                "role": "assistant",
                "content": text or None,
                "tool_calls": [
                    {"id": c.id, "type": "function",
                     "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                    for c in calls
                ],
            })
        else:
            out.append({"role": m.role, "content": text})
    return out

def to_openai_tools(tools: list[dict]) -> list[dict]:
    return [{"type": "function", "function": {
        "name": t["name"], "description": t.get("description", ""),
        "parameters": t.get("parameters", {"type": "object", "properties": {}})}} for t in tools]

print("上一章的适配器已就位")
```

## 分片累积器

先把「怎么把分片拼成完整工具调用」这件事单独写出来，它值得一个专门的类。

```python
class ToolCallAccumulator:
    """按 index 把分片攒成完整的工具调用。

    为什么单独写一个类：分片逻辑有状态（哪些 index 出现过、各自攒了多少），
    混在解析循环里会让那个循环很难读，也很难单独测试。
    """

    def __init__(self) -> None:
        # index -> {"id": …, "name": …, "arguments": 已攒的文本}
        self._slots: dict[int, dict[str, str]] = {}

    def feed(self, raw_calls: list[dict]) -> None:
        for item in raw_calls:
            index = int(item.get("index", 0))
            slot = self._slots.setdefault(index, {"id": "", "name": "", "arguments": ""})
            # id 和 name 通常只在第一片出现，来了就记下，别被后面的空值覆盖
            if item.get("id"):
                slot["id"] = item["id"]
            fn = item.get("function") or {}
            if fn.get("name"):
                slot["name"] = fn["name"]
            if fn.get("arguments"):
                slot["arguments"] += fn["arguments"]      # 关键：累加，不是覆盖

    def finish(self) -> list[ToolUseBlock]:
        """全部收完之后再解析。中途的 arguments 是残缺的 JSON，解析必然失败。"""
        blocks = []
        for index in sorted(self._slots):
            slot = self._slots[index]
            raw = slot["arguments"] or "{}"
            try:
                args = json.loads(raw)
                if not isinstance(args, dict):
                    args = {"_raw": raw}
            except json.JSONDecodeError:
                args = {"_invalid_json": raw}
            blocks.append(ToolUseBlock(
                id=slot["id"] or new_id("call"),
                name=slot["name"],
                args=args,
            ))
        return blocks


# 用上面那串分片试一下
acc = ToolCallAccumulator()
acc.feed([{"index": 0, "id": "call_x", "function": {"name": "get_time", "arguments": ""}}])
acc.feed([{"index": 0, "function": {"arguments": '{"ci'}}])
acc.feed([{"index": 0, "function": {"arguments": 'ty": "北'}}])
acc.feed([{"index": 0, "function": {"arguments": '京"}'}}])

for c in acc.finish():
    print(f"拼出来的工具调用：{c.name}({c.args})  编号 {c.id}")
```

## 解析服务端推送的行

```python
def parse_sse_line(line: str) -> dict | None:
    """把一行 `data: {...}` 解析成字典。

    返回 None 表示这一行不用处理：空行（事件分隔）、注释行、或者结束标记。
    """
    line = line.strip()
    if not line or line.startswith(":"):       # 空行或注释
        return None
    if not line.startswith("data:"):
        return None
    payload = line[5:].strip()
    if payload == "[DONE]":                    # 结束标记，它不是 JSON
        return None
    try:
        return json.loads(payload)
    except json.JSONDecodeError:
        return None                            # 半行或者服务端抽风，跳过


for line in [
    "data: {\"choices\":[{\"delta\":{\"content\":\"北京\"}}]}",
    "",
    ": 这是注释",
    "data: [DONE]",
    "data: {不是合法 JSON",
]:
    print(f"{line[:40]!r:45} → {parse_sse_line(line)}")
```

## 流式请求

```python
import urllib.request

BASE_URL = "http://192.168.213.116:8084/v1"
MODEL = "qwen3.6-27B"


def stream_model(
    messages: list[Message], tools: list[dict] | None = None, timeout: int = 180
) -> Iterator[Delta]:
    """一次流式请求，边收边吐增量。"""
    body: dict[str, Any] = {
        "model": MODEL,
        "messages": to_openai_messages(messages),
        "stream": True,
    }
    if tools:
        body["tools"] = to_openai_tools(tools)

    req = urllib.request.Request(
        f"{BASE_URL}/chat/completions",
        data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
        headers={"Content-Type": "application/json"},
        method="POST",
    )

    acc = ToolCallAccumulator()
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        for raw in resp:
            chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
            if chunk is None:
                continue
            delta = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
            if delta.get("content"):
                yield Delta(kind="text", text=delta["content"])
            if delta.get("tool_calls"):
                acc.feed(delta["tool_calls"])       # 只攒，不吐

    # 流结束了，这时工具调用才是完整的
    for block in acc.finish():
        yield Delta(kind="tool_use", tool=block)


def collect(deltas: Iterator[Delta]) -> Message:
    """把增量攒成一条完整消息。这个函数第 5 章就写好了，一个字没改。"""
    text_parts: list[str] = []
    blocks: list[Any] = []
    for d in deltas:
        if d.kind == "text":
            text_parts.append(d.text)
        elif d.tool is not None:
            if text_parts:
                blocks.append(TextBlock(text="".join(text_parts)))
                text_parts = []
            blocks.append(d.tool)
    if text_parts:
        blocks.append(TextBlock(text="".join(text_parts)))
    return Message(role="assistant", content=blocks)


print("流式请求就绪")
```

## 打字机效果

```python
import sys

print("模型：", end="", flush=True)
pieces = 0
for d in stream_model([
    Message(role="system", content=[TextBlock(text="你是一个简洁的助教。")]),
    Message(role="user", content=[TextBlock(text="用两句话说明什么是闭包。")]),
]):
    if d.kind == "text":
        pieces += 1
        print(d.text, end="", flush=True)
print(f"\n\n（一共收到 {pieces} 个文字增量）")
```

**这就是所有聊天界面打字机效果的全部原理**——没有别的魔法。

## 流式的工具调用

```python
tools = [{
    "name": "get_time",
    "description": "查询某个城市的当前时间",
    "parameters": {
        "type": "object",
        "properties": {"city": {"type": "string", "description": "城市名"}},
        "required": ["city"],
    },
}]

msg = collect(stream_model([
    Message(role="system", content=[TextBlock(text="需要查时间时就调用工具，不要自己编。")]),
    Message(role="user", content=[TextBlock(text="北京现在几点？")]),
], tools=tools))

print("块类型：", [b.type for b in msg.content])
print("文字：", msg.text() or "（没有文字）")
for c in msg.tool_uses():
    print(f"工具调用：{c.name}({c.args})  参数解析成功：{'_invalid_json' not in c.args}")
```

注意 `collect` 这个函数——**第 5 章为假模型写的，这里一个字没改就用在真模型上**。
这就是第 5 章那个接口设计的回报。

## 为什么工具调用要等流结束才吐

有人会想：拼到能解析的时候就吐出去，不是更早吗？

不行。`{"city": "北` 这样的残缺文本，`json.loads` 只会报错；而 `{"city": "北京"}`
和 `{"city": "北京", "tz": "…"}` 的前缀是一样的，你没法判断它到底完了没有。
唯一可靠的结束信号是**流本身结束**。

代价是工具调用比文字晚一点出现。真实产品里可以在收到第一片时就发一个「模型准备调用
工具」的提示事件，让界面先有反应，参数等完整了再显示。

## 本章小结

- 服务端推送就是一串 `data: {...}` 行，**`[DONE]` 不是 JSON**，要单独判断。
- 文字增量直接吐；**工具调用必须攒**。
- 分片的身份是 **`index` 不是 `id`**，`id` 和 `name` 通常只在第一片出现。
- `arguments` 是**按字符流分片的 JSON 文本**，要累加，**拼完才能解析**。
- 分片累积单独写成一个类，因为它有状态，混在解析循环里既难读又难测。
- 第 5 章的 `collect` 一个字没改就用上了。
