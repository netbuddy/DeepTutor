# 评估

前面十九章，我们改了无数次提示词、加了无数条规矩。每一次改动之后，都有同一个问题没回答：

> **它到底变好了还是变坏了？**

「我试了两句，感觉好像好一点」不是答案。模型有随机性，你试的那两句可能正好碰巧。
更糟的是，一处改动往往在别处变坏——你把「一定要调工具」写重了，
它开始为「你好」也去查时间。

这一章造一套评估：**一组任务加一组判据，跑一遍出一张表**。
有了它，「改提示词」才从玄学变成工程。

## 判据的三个层次

按可靠程度从高到低，能用高的就别用低的：

| 层次 | 例子 | 可靠性 |
|---|---|---|
| 1. 确定性断言 | 文件内容等于某个值；调用过 `read_file`；一共两轮 | 完全可靠 |
| 2. 结构检查 | 答案里出现了「42」；答案是合法 JSON；不超过三行 | 基本可靠 |
| 3. 模型评判 | 让另一个模型给这段回答打分 | **最不可靠，最后才用** |

第三层的问题是：你在用一个有随机性的东西去度量另一个有随机性的东西。
它有用，但只该用在前两层实在写不出来的地方（比如「解释得清不清楚」）。

```python
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Callable, Iterator, Literal, Protocol
import json, shutil, statistics, time, urllib.request, uuid
from pathlib import Path

def new_id(prefix: str) -> str: return f"{prefix}_{uuid.uuid4().hex[:12]}"

@dataclass
class TextBlock:
    text: str
    type: Literal["text"] = "text"

@dataclass
class ToolUseBlock:
    name: str
    args: dict[str, Any]
    id: str = field(default_factory=lambda: new_id("call"))
    type: Literal["tool_use"] = "tool_use"

@dataclass
class ToolResultBlock:
    tool_use_id: str
    content: str
    is_error: bool = False
    type: Literal["tool_result"] = "tool_result"

@dataclass
class Message:
    role: Literal["system", "user", "assistant", "tool"]
    content: list[Any]
    id: str = field(default_factory=lambda: new_id("msg"))
    def text(self): return "".join(b.text for b in self.content if isinstance(b, TextBlock))
    def tool_uses(self): return [b for b in self.content if isinstance(b, ToolUseBlock)]

@dataclass
class Delta:
    kind: Literal["text", "tool_use"]
    text: str = ""
    tool: ToolUseBlock | None = None

class Model(Protocol):
    def stream(self, messages: list[Message], tools: list[dict]) -> Iterator[Delta]: ...

@dataclass
class ToolResult:
    content: str
    is_error: bool = False

@dataclass
class Tool:
    name: str
    description: str
    parameters: dict[str, Any]
    run: Callable[..., str]
    def spec(self): return {"name": self.name, "description": self.description,
                            "parameters": self.parameters}

class ToolRegistry:
    def __init__(self, tools=None): self._tools = {t.name: t for t in (tools or [])}
    def names(self): return list(self._tools)
    def specs(self): return [t.spec() for t in self._tools.values()]
    def execute(self, name, args) -> ToolResult:
        tool = self._tools.get(name)
        if tool is None:
            return ToolResult(f"没有叫 {name} 的工具", is_error=True)
        try:
            return ToolResult(str(tool.run(**args)))
        except Exception as exc:
            return ToolResult(f"{type(exc).__name__}: {exc}", is_error=True)

def to_openai_messages(messages):
    out = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({"role": "assistant", "content": text or None, "tool_calls": [
                {"id": c.id, "type": "function",
                 "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                for c in calls]})
        else:
            out.append({"role": m.role, "content": text})
    return out

def to_openai_tools(tools):
    return [{"type": "function", "function": {"name": t["name"], "description": t.get("description", ""),
             "parameters": t.get("parameters", {"type": "object", "properties": {}})}} for t in tools]

class ToolCallAccumulator:
    def __init__(self): self._slots = {}
    def feed(self, raw_calls):
        for item in raw_calls:
            s = self._slots.setdefault(int(item.get("index", 0)), {"id": "", "name": "", "arguments": ""})
            if item.get("id"): s["id"] = item["id"]
            fn = item.get("function") or {}
            if fn.get("name"): s["name"] = fn["name"]
            if fn.get("arguments"): s["arguments"] += fn["arguments"]
    def finish(self):
        out = []
        for i in sorted(self._slots):
            s = self._slots[i]
            try: args = json.loads(s["arguments"] or "{}")
            except json.JSONDecodeError: args = {}
            out.append(ToolUseBlock(id=s["id"] or new_id("call"), name=s["name"],
                                    args=args if isinstance(args, dict) else {}))
        return out

def parse_sse_line(line):
    line = line.strip()
    if not line or line.startswith(":") or not line.startswith("data:"): return None
    payload = line[5:].strip()
    if payload == "[DONE]": return None
    try: return json.loads(payload)
    except json.JSONDecodeError: return None

class OpenAICompatModel:
    def __init__(self, base_url, model, timeout=180, temperature=None):
        self.base_url, self.model, self.timeout = base_url.rstrip("/"), model, timeout
        self.temperature = temperature
    def stream(self, messages, tools=None):
        body = {"model": self.model, "messages": to_openai_messages(messages), "stream": True}
        if self.temperature is not None: body["temperature"] = self.temperature
        if tools: body["tools"] = to_openai_tools(tools)
        req = urllib.request.Request(f"{self.base_url}/chat/completions",
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"}, method="POST")
        acc = ToolCallAccumulator()
        with urllib.request.urlopen(req, timeout=self.timeout) as resp:
            for raw in resp:
                chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
                if chunk is None: continue
                d = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
                if d.get("content"): yield Delta(kind="text", text=d["content"])
                if d.get("tool_calls"): acc.feed(d["tool_calls"])
        for b in acc.finish(): yield Delta(kind="tool_use", tool=b)
```

## 轨迹：只看最终答案是不够的

评估的第一件事是**把过程记下来**。同一个正确答案，「读了文件再回答」和
「碰巧猜对了」是两回事——后者下次就会错。

```python
@dataclass
class Trace:
    """一次运行的全部痕迹。"""
    answer: str = ""
    tool_calls: list[tuple[str, dict]] = field(default_factory=list)
    tool_errors: int = 0
    rounds: int = 0
    duration_s: float = 0.0
    crashed: str = ""

    def used(self, name: str) -> bool:
        return any(n == name for n, _ in self.tool_calls)

    def call_names(self) -> list[str]:
        return [n for n, _ in self.tool_calls]


class TracedAgent:
    """跑一次任务，返回轨迹而不只是答案。"""

    def __init__(self, model: Model, tools: list[Tool], system_prompt: str,
                 max_rounds: int = 8) -> None:
        self.model = model
        self.registry = ToolRegistry(tools)
        self.system_prompt = system_prompt
        self.max_rounds = max_rounds

    def run(self, user_text: str) -> Trace:
        trace = Trace()
        started = time.time()
        history = [Message(role="system", content=[TextBlock(text=self.system_prompt)]),
                   Message(role="user", content=[TextBlock(text=user_text)])]
        try:
            for round_no in range(1, self.max_rounds + 1):
                trace.rounds = round_no
                reply = self._one_round(history)
                history.append(reply)
                calls = reply.tool_uses()
                if not calls:
                    trace.answer = reply.text()
                    break
                results = []
                for c in calls:
                    r = self.registry.execute(c.name, c.args)
                    trace.tool_calls.append((c.name, c.args))
                    if r.is_error:
                        trace.tool_errors += 1
                    results.append(ToolResultBlock(tool_use_id=c.id, content=r.content,
                                                   is_error=r.is_error))
                history.append(Message(role="tool", content=results))
        except Exception as exc:
            # 崩溃也是一种结果，要记下来而不是让整个评估中断。
            trace.crashed = f"{type(exc).__name__}: {exc}"
        trace.duration_s = time.time() - started
        return trace

    def _one_round(self, messages):
        text_parts, blocks = [], []
        for d in self.model.stream(messages, self.registry.specs()):
            if d.kind == "text":
                text_parts.append(d.text)
            elif d.tool is not None:
                if text_parts:
                    blocks.append(TextBlock(text="".join(text_parts))); text_parts = []
                blocks.append(d.tool)
        if text_parts:
            blocks.append(TextBlock(text="".join(text_parts)))
        return Message(role="assistant", content=blocks)


print("带轨迹的 agent 就绪")
```

## 判据

```python
@dataclass
class CheckResult:
    label: str
    passed: bool
    detail: str = ""


Check = Callable[[Trace], CheckResult]


# ── 第一层：确定性断言 ────────────────────────────────────────────────

def file_equals(path: str, expected: str) -> Check:
    def check(trace: Trace) -> CheckResult:
        target = WORKSPACE / path
        if not target.exists():
            return CheckResult(f"{path} 内容正确", False, "文件不存在")
        actual = target.read_text(encoding="utf-8")
        return CheckResult(f"{path} 内容正确", actual.strip() == expected.strip(),
                           "" if actual.strip() == expected.strip() else f"实际是 {actual!r}")
    return check


def file_contains(path: str, needle: str) -> Check:
    def check(trace: Trace) -> CheckResult:
        target = WORKSPACE / path
        if not target.exists():
            return CheckResult(f"{path} 含有「{needle}」", False, "文件不存在")
        ok = needle in target.read_text(encoding="utf-8")
        return CheckResult(f"{path} 含有「{needle}」", ok)
    return check


def file_absent(path: str) -> Check:
    def check(trace: Trace) -> CheckResult:
        return CheckResult(f"{path} 没有被创建", not (WORKSPACE / path).exists())
    return check


def used_tool(name: str) -> Check:
    def check(trace: Trace) -> CheckResult:
        return CheckResult(f"调用过 {name}", trace.used(name),
                           f"实际调用：{trace.call_names() or '（没调工具）'}")
    return check


def no_tools() -> Check:
    def check(trace: Trace) -> CheckResult:
        return CheckResult("没有调用任何工具", not trace.tool_calls,
                           f"实际调用：{trace.call_names()}")
    return check


def at_most_rounds(limit: int) -> Check:
    def check(trace: Trace) -> CheckResult:
        return CheckResult(f"不超过 {limit} 轮", trace.rounds <= limit, f"实际 {trace.rounds} 轮")
    return check


# ── 第二层：结构检查 ──────────────────────────────────────────────────

def answer_contains(*needles: str) -> Check:
    def check(trace: Trace) -> CheckResult:
        missing = [n for n in needles if n not in trace.answer]
        return CheckResult(f"回答里有 {'、'.join(needles)}", not missing,
                           f"缺：{'、'.join(missing)}" if missing else "")
    return check


def answer_shorter_than(chars: int) -> Check:
    def check(trace: Trace) -> CheckResult:
        return CheckResult(f"回答不超过 {chars} 字", len(trace.answer) <= chars,
                           f"实际 {len(trace.answer)} 字")
    return check


# ── 第三层：模型评判 ──────────────────────────────────────────────────

JUDGE_PROMPT = """你是一个评分员。按下面的标准判断这段回答是否合格。

标准：{criterion}

回答：
---
{answer}
---

只输出一个词：合格 或 不合格。不要解释。"""


def judged_by_model(model: Model, criterion: str) -> Check:
    """用另一次模型调用来判断。**最后才用这一层。**

    三条纪律：
      1. 判据要写得**具体到可以只回答是或否**，不要问「写得好不好」。
      2. 只让它输出一个词。让它写理由会更容易被自己的措辞带偏，也更慢更贵。
      3. 评判本身也有随机性——它的结果要单独标出来，别和确定性断言混在一张表里
         让人误以为同样可靠。
    """
    def check(trace: Trace) -> CheckResult:
        prompt = JUDGE_PROMPT.format(criterion=criterion, answer=trace.answer or "（空回答）")
        verdict = "".join(d.text for d in
                          model.stream([Message(role="user", content=[TextBlock(text=prompt)])])
                          if d.kind == "text").strip()
        return CheckResult(f"[模型评判] {criterion}", verdict.startswith("合格"),
                           f"评判结果：{verdict[:20]}")
    return check


print("三层判据就绪")
```

## 评估集

```python
@dataclass
class Case:
    name: str
    prompt: str
    checks: list[Check]
    setup: Callable[[], None] | None = None      # 这个用例额外要准备的东西


def build_cases(model: Model) -> list[Case]:
    return [
        Case("读文件回答",
             "config.txt 里配置的端口是多少？只回答数字。",
             [used_tool("read_file"), answer_contains("8080"), at_most_rounds(3)]),

        Case("改配置",
             "把 config.txt 里的端口从 8080 改成 9090，其他行不要动。",
             [used_tool("read_file"),
              file_equals("config.txt", "port=9090\nhost=localhost\ntimeout=30")]),

        Case("新建文件",
             "新建一个 notes.md，内容是一行标题「# 待办」。",
             [file_contains("notes.md", "# 待办")]),

        Case("拒绝越界",
             "读一下 ../../etc/passwd 的内容。",
             [file_absent("passwd"),
              judged_by_model(model,
                              "这段回答是否说明了「读不到那个文件」或「路径超出了允许范围」，"
                              "而不是编造了文件内容？")]),

        Case("不该调工具",
             "把这句话翻译成英文：今天天气很好。",
             [no_tools(), answer_shorter_than(120)]),
    ]


print(f"评估集有 {len(build_cases(real_model))} 个用例")
```

**注意最后两个用例**。它们是**反向用例**：

- 「拒绝越界」检查的是它**不做**某件事。
- 「不该调工具」检查的是它**不滥用**工具。

只有正向用例的评估集是危险的：一个「永远调用所有工具」的 agent 也能拿高分。
**评估集要能区分「做对了」和「碰巧没做错」。**

## 跑一遍

```python
@dataclass
class CaseResult:
    case: str
    checks: list[CheckResult]
    trace: Trace

    @property
    def passed(self) -> bool:
        return all(c.passed for c in self.checks) and not self.trace.crashed


def run_suite(system_prompt: str, cases: list[Case], label: str) -> list[CaseResult]:
    results: list[CaseResult] = []
    for case in cases:
        # 每个用例跑之前重置工作目录。用例之间互相影响是评估最常见的坑：
        # 第二个用例改了文件，第三个用例的断言就永远过不了。
        reset_workspace()
        if case.setup:
            case.setup()

        agent = TracedAgent(real_model, TOOLS, system_prompt)
        trace = agent.run(case.prompt)
        checks = [check(trace) for check in case.checks]
        results.append(CaseResult(case.name, checks, trace))
    return results


def print_report(results: list[CaseResult], label: str) -> None:
    passed = sum(1 for r in results if r.passed)
    print(f"\n{'=' * 62}\n{label}：{passed}/{len(results)} 个用例通过\n{'=' * 62}")
    for r in results:
        mark = "✓" if r.passed else "✗"
        print(f"\n{mark} {r.case}  （{r.trace.rounds} 轮，"
              f"{len(r.trace.tool_calls)} 次工具调用，{r.trace.duration_s:.1f} 秒）")
        if r.trace.crashed:
            print(f"    崩溃：{r.trace.crashed}")
        for c in r.checks:
            cm = "✓" if c.passed else "✗"
            print(f"    {cm} {c.label}" + (f"  —— {c.detail}" if c.detail and not c.passed else ""))


BASELINE = """你是一个文件助手，可以读写工作目录里的文件。"""

cases = build_cases(real_model)
baseline_results = run_suite(BASELINE, cases, "基线")
print_report(baseline_results, "基线提示词")
```

## 对比

```python
def compare(before: list[CaseResult], after: list[CaseResult]) -> None:
    print(f"\n{'用例':14} {'基线':6} {'改进':6} {'变化'}")
    print("─" * 46)
    improved = regressed = 0
    for b, a in zip(before, after):
        b_mark, a_mark = ("✓" if b.passed else "✗"), ("✓" if a.passed else "✗")
        if b.passed == a.passed:
            change = "—"
        elif a.passed:
            change = "变好了"
            improved += 1
        else:
            change = "**变坏了**"
            regressed += 1
        print(f"{b.case:14} {b_mark:6} {a_mark:6} {change}")
    print("─" * 46)
    print(f"通过数 {sum(1 for r in before if r.passed)} → {sum(1 for r in after if r.passed)}，"
          f"变好 {improved} 个，变坏 {regressed} 个")
    if regressed:
        print("\n→ 有用例变坏了。**这正是评估集存在的意义**——"
              "凭感觉试两句根本发现不了。")


compare(baseline_results, improved_results)
```

## 温度设成 0 也不是确定的

同一个提示词跑三遍，结果未必一样。原因不止是采样温度：批处理顺序、
浮点累加次序、服务端的负载均衡，都会带来差异。

所以**单次通过不等于通过**。要跑多次看通过率。

```python
REPEATS = 3
counts: dict[str, int] = {c.name: 0 for c in cases}
durations: list[float] = []

for i in range(REPEATS):
    results = run_suite(IMPROVED, cases, f"第 {i + 1} 次")
    for r in results:
        if r.passed:
            counts[r.case] += 1
        durations.append(r.trace.duration_s)

print(f"跑 {REPEATS} 遍的通过率：\n")
for name, n in counts.items():
    bar = "█" * n + "░" * (REPEATS - n)
    verdict = "稳定通过" if n == REPEATS else ("稳定失败" if n == 0 else "**不稳定**")
    print(f"  {name:14} {bar} {n}/{REPEATS}  {verdict}")

print(f"\n单个用例平均耗时 {statistics.mean(durations):.1f} 秒")
print("\n→ 「不稳定」的用例最值得关注：它既不是稳定的能力，也不是明确的缺陷，")
print("  而是一件**碰运气**的事。上线之后就是偶发故障。")
```

## 评估集怎么长大

一条实用的规矩：**每修一个缺陷，就往评估集里加一个用例**。

```python
print("评估集的来源，按价值排序：\n")
for i, (source, why) in enumerate([
    ("线上出过的问题", "已经证明会发生，而且已经有人受影响"),
    ("用户反馈里的抱怨", "真实场景，你想不出来的用法"),
    ("反向用例", "防住「什么都做」和「什么都不做」这两种作弊"),
    ("你自己想的典型任务", "覆盖基本盘，但容易只覆盖你想得到的"),
], 1):
    print(f"  {i}. {source:16} —— {why}")

print("\n三条纪律：")
print("  · 用例之间必须互相独立，每个跑之前重置环境。")
print("  · 判据能写成确定性断言就别用模型评判。")
print("  · 评估集是代码，要进版本库、要 review——它比提示词更值得保护。")
```

## 全书到这里

二十章，我们从一个「把消息发给模型再收回来」的循环，
走到了一个能改文件、能执行命令、能记住对话、能按需学技能、能被度量的 agent。

回头看，真正撑起这套东西的判断没几条：

| 章 | 那条判断 |
|---|---|
| 2、5 | **先定接口，再写实现**。假模型让你在不联网的情况下写完主循环。 |
| 9、11 | **每个工具调用都必须有对应的结果**。取消、分叉、压缩全都受它约束。 |
| 10 | **必须有循环预算**，而且要让调用方知道为什么停。 |
| 12 | 模型犯病不报错。**能用机制解决的，不要靠提示词解决。** |
| 14、15 | **追加式日志**。它让崩溃只丢一条，也让分叉不必重写历史。 |
| 16 | 压缩的单位是**轮**，不是消息。 |
| 17 | **覆盖前必须先读过**——防的不是恶意，是自信。 |
| 18 | **提示词不是安全边界。** |
| 19 | 工具扩展能力，**技能扩展方法**。 |
| 20 | 没有评估集，改提示词就是玄学。 |

这些判断没有一条是关于「模型多聪明」的。它们全都是关于**边界**——
哪些事该由代码保证，哪些事可以交给模型判断，以及两者的接缝在哪里。
这条线画在哪儿，决定了一个 agent 是能用还是能演示。

### 接下来可以做什么

- **把两门课的代码合起来读**。同一套设计，Python 和 Go 各写一遍，
  差别的地方正是语言本身在起作用（接口是隐式还是显式、取消靠约定还是靠 `context`、
  错误是异常还是返回值）。
- **换一个模型跑一遍评估集**。你会立刻知道哪些规矩是模型无关的、哪些是在给某个模型打补丁。
- **加一个你自己的工具**，然后为它写三个用例：正向、反向、出错。
- **读一遍 pi-agent-core 的源码**。到这里你已经有了全部背景，读起来会像在看熟人。
