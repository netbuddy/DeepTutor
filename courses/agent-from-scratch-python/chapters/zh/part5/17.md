# 编码工具

前面十六章的工具都是玩具：查时间、算算术。这一章做真正干活的四个工具——
**读文件、写文件、改文件、列目录**。它们是编码 agent 的全部基础。

这四个工具看着简单，但每一个都有一处不做就会出事的设计。先把它们摆出来：

| 工具 | 那处设计 | 不做的后果 |
|---|---|---|
| 读文件 | 带行号返回，并且能只读一段 | 大文件一次读完撑爆上下文；没有行号，模型没法精确指认位置 |
| 写文件 | 覆盖前必须先读过 | 模型凭想象重写整个文件，把没看过的内容删掉 |
| 改文件 | 按**唯一的原文片段**替换，不按行号 | 行号在上一次编辑后就失效了，改错位置 |
| 列目录 | 有深度和数量上限 | 一个 node_modules 就把上下文吃光 |

```python
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Callable, Iterator, Literal, Protocol
import json, shutil, time, urllib.request, uuid
from pathlib import Path

def new_id(prefix: str) -> str: return f"{prefix}_{uuid.uuid4().hex[:12]}"

@dataclass
class TextBlock:
    text: str
    type: Literal["text"] = "text"

@dataclass
class ToolUseBlock:
    name: str
    args: dict[str, Any]
    id: str = field(default_factory=lambda: new_id("call"))
    type: Literal["tool_use"] = "tool_use"

@dataclass
class ToolResultBlock:
    tool_use_id: str
    content: str
    is_error: bool = False
    type: Literal["tool_result"] = "tool_result"

@dataclass
class Message:
    role: Literal["system", "user", "assistant", "tool"]
    content: list[Any]
    id: str = field(default_factory=lambda: new_id("msg"))
    def text(self): return "".join(b.text for b in self.content if isinstance(b, TextBlock))
    def tool_uses(self): return [b for b in self.content if isinstance(b, ToolUseBlock)]

@dataclass
class Delta:
    kind: Literal["text", "tool_use"]
    text: str = ""
    tool: ToolUseBlock | None = None

class Model(Protocol):
    def stream(self, messages: list[Message], tools: list[dict]) -> Iterator[Delta]: ...

@dataclass
class ToolResult:
    content: str
    is_error: bool = False
    is_invalid_args: bool = False

@dataclass
class Tool:
    name: str
    description: str
    parameters: dict[str, Any]
    run: Callable[..., str]
    def spec(self): return {"name": self.name, "description": self.description,
                            "parameters": self.parameters}

def validate_args(args, schema) -> str:
    props, required = schema.get("properties") or {}, schema.get("required") or []
    missing = [k for k in required if k not in args]
    if missing: return f"缺少必填参数 {'、'.join(missing)}"
    unknown = [k for k in args if k not in props]
    if unknown: return f"不认识的参数 {'、'.join(unknown)}；这个工具接受 {'、'.join(props)}"
    return ""

class ToolRegistry:
    def __init__(self, tools=None): self._tools = {t.name: t for t in (tools or [])}
    def add(self, tool): self._tools[tool.name] = tool
    def names(self): return list(self._tools)
    def specs(self): return [t.spec() for t in self._tools.values()]
    def execute(self, name, args) -> ToolResult:
        tool = self._tools.get(name)
        if tool is None:
            return ToolResult(f"没有叫 {name} 的工具。可用的：{'、'.join(self.names())}", is_error=True)
        problem = validate_args(args, tool.parameters)
        if problem:
            return ToolResult(f"参数不合格：{problem}", is_error=True, is_invalid_args=True)
        try:
            return ToolResult(str(tool.run(**args)))
        except Exception as exc:
            return ToolResult(f"{type(exc).__name__}: {exc}", is_error=True)

def to_openai_messages(messages):
    out = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({"role": "assistant", "content": text or None, "tool_calls": [
                {"id": c.id, "type": "function",
                 "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                for c in calls]})
        else:
            out.append({"role": m.role, "content": text})
    return out

def to_openai_tools(tools):
    return [{"type": "function", "function": {"name": t["name"], "description": t.get("description", ""),
             "parameters": t.get("parameters", {"type": "object", "properties": {}})}} for t in tools]

class ToolCallAccumulator:
    def __init__(self): self._slots = {}
    def feed(self, raw_calls):
        for item in raw_calls:
            s = self._slots.setdefault(int(item.get("index", 0)), {"id": "", "name": "", "arguments": ""})
            if item.get("id"): s["id"] = item["id"]
            fn = item.get("function") or {}
            if fn.get("name"): s["name"] = fn["name"]
            if fn.get("arguments"): s["arguments"] += fn["arguments"]
    def finish(self):
        out = []
        for i in sorted(self._slots):
            s = self._slots[i]
            try: args = json.loads(s["arguments"] or "{}")
            except json.JSONDecodeError: args = {}
            out.append(ToolUseBlock(id=s["id"] or new_id("call"), name=s["name"],
                                    args=args if isinstance(args, dict) else {}))
        return out

def parse_sse_line(line):
    line = line.strip()
    if not line or line.startswith(":") or not line.startswith("data:"): return None
    payload = line[5:].strip()
    if payload == "[DONE]": return None
    try: return json.loads(payload)
    except json.JSONDecodeError: return None

class OpenAICompatModel:
    def __init__(self, base_url, model, timeout=180):
        self.base_url, self.model, self.timeout = base_url.rstrip("/"), model, timeout
    def stream(self, messages, tools):
        body = {"model": self.model, "messages": to_openai_messages(messages), "stream": True}
        if tools: body["tools"] = to_openai_tools(tools)
        req = urllib.request.Request(f"{self.base_url}/chat/completions",
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"}, method="POST")
        acc = ToolCallAccumulator()
        with urllib.request.urlopen(req, timeout=self.timeout) as resp:
            for raw in resp:
                chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
                if chunk is None: continue
                d = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
                if d.get("content"): yield Delta(kind="text", text=d["content"])
                if d.get("tool_calls"): acc.feed(d["tool_calls"])
        for b in acc.finish(): yield Delta(kind="tool_use", tool=b)

class Agent:
    def __init__(self, model, registry, system_prompt="", max_rounds=10, on_event=None):
        self.model, self.registry = model, registry
        self.system_prompt = system_prompt or "你是一个编码助手。"
        self.max_rounds, self.on_event = max_rounds, on_event or (lambda *a: None)
    def run(self, user_text):
        history = [Message(role="system", content=[TextBlock(text=self.system_prompt)]),
                   Message(role="user", content=[TextBlock(text=user_text)])]
        for _ in range(self.max_rounds):
            reply = self._one_round(history)
            history.append(reply)
            calls = reply.tool_uses()
            if not calls:
                return reply.text()
            results = []
            for c in calls:
                r = self.registry.execute(c.name, c.args)
                self.on_event(c.name, c.args, r)
                results.append(ToolResultBlock(tool_use_id=c.id, content=r.content,
                                               is_error=r.is_error))
            history.append(Message(role="tool", content=results))
        return "（轮数用完了）"
    def _one_round(self, messages):
        text_parts, blocks = [], []
        for d in self.model.stream(messages, self.registry.specs()):
            if d.kind == "text":
                text_parts.append(d.text)
            elif d.tool is not None:
                if text_parts:
                    blocks.append(TextBlock(text="".join(text_parts))); text_parts = []
                blocks.append(d.tool)
        if text_parts: blocks.append(TextBlock(text="".join(text_parts)))
        return Message(role="assistant", content=blocks)
```

## 先把路径管住

所有文件工具的第一件事，都是**把模型给的路径钉在工作目录里**。
模型会写出 `../../etc/passwd`，不是因为它有恶意，而是因为它在猜项目结构。

```python
class PathError(Exception):
    """路径不合法。单独一个类型，因为它的处置和别的错误不同——
    这是要拦住的越界，不是可以重试的失败。"""


def resolve_in_workspace(root: Path, raw: str) -> Path:
    """把模型给的路径解析成一个确定在 root 里面的绝对路径。

    三步都不能省：
      1. 拒绝绝对路径——模型给 /etc/passwd 时，root/'/etc/passwd' 在 Python 里
         就等于 /etc/passwd，拼接根本不起作用。
      2. resolve() 展开 .. 和符号链接。只做字符串检查挡不住符号链接：
         工作目录里放一个指向 /etc 的链接，字符串上看它还在工作目录里。
      3. 用 is_relative_to 做最终判断，不要用 str.startswith——
         /tmp/workspace-evil 是以 /tmp/workspace 开头的。
    """
    if not raw or raw.strip() == "":
        raise PathError("路径不能为空")
    candidate = Path(raw)
    if candidate.is_absolute():
        raise PathError(f"只接受相对路径，收到的是绝对路径：{raw}")

    resolved = (root / candidate).resolve()
    root_resolved = root.resolve()
    if not resolved.is_relative_to(root_resolved):
        raise PathError(f"路径越出了工作目录：{raw}")
    return resolved


root = WORKSPACE
for raw in ["app/main.py", "./README.md", "../../etc/passwd", "/etc/passwd", "app/../README.md"]:
    try:
        resolved = resolve_in_workspace(root, raw)
        print(f"  ✓ {raw:20} → {resolved.relative_to(root.resolve())}")
    except PathError as exc:
        print(f"  ✗ {raw:20} → {exc}")
```

## 读文件：带行号，能读一段

```python
MAX_READ_LINES = 400          # 一次最多给模型这么多行
MAX_LINE_CHARS = 300          # 单行超长就截断（压缩过的前端代码能有几万字符一行）


def read_file(path: str, start_line: int = 1, end_line: int = 0) -> str:
    """读文件，返回带行号的内容。

    行号不是为了好看：模型要用它来说「第 12 行有问题」，
    也要用它来决定下一步读哪一段。没有行号，它只能整文件重读。
    """
    target = resolve_in_workspace(WORKSPACE, path)
    if not target.exists():
        # 报错要给出下一步能怎么办。「文件不存在」是死路，
        # 「文件不存在，同目录下有这些」是线索。
        siblings = [p.name for p in target.parent.iterdir()][:10] if target.parent.exists() else []
        hint = f"，同目录下有：{'、'.join(siblings)}" if siblings else ""
        raise FileNotFoundError(f"{path} 不存在{hint}")
    if target.is_dir():
        raise IsADirectoryError(f"{path} 是目录，不是文件。列目录请用 list_dir")

    lines = target.read_text(encoding="utf-8", errors="replace").splitlines()
    total = len(lines)
    start = max(1, start_line)
    end = total if end_line <= 0 else min(end_line, total)
    if start > total:
        return f"（{path} 只有 {total} 行，第 {start} 行之后没有内容）"

    chunk = lines[start - 1:end]
    truncated_tail = False
    if len(chunk) > MAX_READ_LINES:
        chunk = chunk[:MAX_READ_LINES]
        end = start + MAX_READ_LINES - 1
        truncated_tail = True

    width = len(str(end))
    body = "\n".join(
        f"{start + i:>{width}}│{line[:MAX_LINE_CHARS]}{'…（本行已截断）' if len(line) > MAX_LINE_CHARS else ''}"
        for i, line in enumerate(chunk))

    header = f"{path}（共 {total} 行，显示 {start}-{end}）"
    footer = (f"\n…还有 {total - end} 行没显示，接着读请传 start_line={end + 1}"
              if truncated_tail or end < total else "")
    return f"{header}\n{body}{footer}"


print(read_file("app/main.py"))
print()
print(read_file("app/main.py", start_line=4, end_line=5))
```

## 写文件：覆盖前必须先读过

这是这一章最重要的一条规矩。

模型很爱干一件事：你让它「在文件里加个函数」，它直接 `write_file` 写一份**它想象中的完整文件**——
把它没读过的部分全删了。防住这件事只有一个可靠办法：**记住哪些文件被读过，没读过就不许覆盖**。

```python
class ReadTracker:
    """记住哪些文件被读过，以及读的时候是什么样子。

    这不是「安全机制」，是「防止模型自信地覆盖它没看过的内容」。
    两者的区别在于：安全机制防的是恶意，这个防的是自信。
    """

    def __init__(self) -> None:
        self._seen: dict[Path, float] = {}     # 路径 → 读的时候文件的修改时间

    def mark_read(self, path: Path) -> None:
        self._seen[path] = path.stat().st_mtime if path.exists() else 0.0

    def check_can_overwrite(self, path: Path, display: str) -> None:
        if not path.exists():
            return                              # 新建文件不需要先读
        if path not in self._seen:
            raise PermissionError(
                f"{display} 已经存在，但你还没读过它。请先用 read_file 读一遍，"
                f"确认要保留哪些内容再写。")
        if path.stat().st_mtime > self._seen[path]:
            # 别人改过了。直接覆盖会把那个人的改动无声地抹掉。
            raise PermissionError(
                f"{display} 在你读过之后被改动过。请重新 read_file 再写。")


tracker = ReadTracker()


def write_file(path: str, content: str) -> str:
    """整文件覆盖。新建文件用它；改已有文件优先用 edit_file。"""
    target = resolve_in_workspace(WORKSPACE, path)
    tracker.check_can_overwrite(target, path)

    target.parent.mkdir(parents=True, exist_ok=True)
    existed = target.exists()
    old_lines = len(target.read_text(encoding="utf-8").splitlines()) if existed else 0

    target.write_text(content, encoding="utf-8")
    tracker.mark_read(target)                   # 刚写完就是最新的，不必再读一遍

    new_lines = len(content.splitlines())
    if existed:
        return f"已覆盖 {path}：{old_lines} 行 → {new_lines} 行"
    return f"已新建 {path}：{new_lines} 行"


# 演示这条规矩
try:
    write_file("app/main.py", "print('全没了')\n")
except PermissionError as exc:
    print("✗ 拦住了：", exc)

read_file("app/main.py")            # 但注意：这里还没登记
print("\n直接调 read_file 不会登记——登记发生在工具被执行的时候（见下一节）")
```

## 改文件：按原文片段，不按行号

```python
def edit_file(path: str, old_text: str, new_text: str, expect_count: int = 1) -> str:
    """把 old_text 换成 new_text。

    为什么按原文片段而不按行号：**行号会失效**。
    模型读文件时看到第 12 行，改完第 5 行之后，原来的第 12 行已经变成第 13 行了。
    多改几处，行号全乱。原文片段没有这个问题——它锚定的是内容本身。

    为什么要求唯一：`return None` 在一个文件里可能出现二十次。
    替换错地方比不替换糟糕得多，所以数量对不上就拒绝，让模型带更多上下文重来。
    """
    target = resolve_in_workspace(WORKSPACE, path)
    if not target.exists():
        raise FileNotFoundError(f"{path} 不存在")
    if not old_text:
        raise ValueError("old_text 不能为空。要整文件覆盖请用 write_file")

    content = target.read_text(encoding="utf-8")
    count = content.count(old_text)

    if count == 0:
        raise ValueError(
            f"在 {path} 里找不到这段内容。请先 read_file 确认原文，"
            f"注意缩进和空格必须一模一样。")
    if count != expect_count:
        raise ValueError(
            f"这段内容在 {path} 里出现了 {count} 次，不是期望的 {expect_count} 次。"
            f"请带上前后几行让它唯一，或者把 expect_count 改成 {count}。")

    target.write_text(content.replace(old_text, new_text), encoding="utf-8")
    tracker.mark_read(target)

    # 返回改动的样子，让模型（和人）能确认改对了
    return (f"已修改 {path}（替换 {count} 处）\n"
            f"- {old_text.splitlines()[0][:60]}\n"
            f"+ {new_text.splitlines()[0][:60] if new_text.splitlines() else '（删除）'}")


print(edit_file("app/main.py", "'Hello, ' + name", "f'Hello, {name}!'"))
print()
print(read_file("app/main.py", start_line=1, end_line=2))

# 不唯一的情况
try:
    edit_file("app/main.py", "    ", "  ")
except ValueError as exc:
    print("\n✗ 拦住了：", exc)
```

## 列目录：有上限

```python
SKIP_DIRS = {".git", "node_modules", "__pycache__", ".venv", "venv", "dist", "build", ".idea"}
MAX_ENTRIES = 200


def list_dir(path: str = ".", max_depth: int = 2) -> str:
    """列出目录树。

    两个上限都是必须的：
      - 跳过 .git、node_modules 之类：一个 node_modules 有几万个文件，
        列出来就把上下文吃光了，而且对模型毫无用处。
      - 条目总数封顶：防住「上面那个清单漏了某个巨大目录」这种情况。
    """
    root_dir = resolve_in_workspace(WORKSPACE, path) if path not in (".", "") else WORKSPACE.resolve()
    if not root_dir.is_dir():
        raise NotADirectoryError(f"{path} 不是目录")

    lines: list[str] = []
    skipped: list[str] = []
    truncated = False

    def walk(directory: Path, depth: int, prefix: str) -> None:
        nonlocal truncated
        if depth > max_depth or truncated:
            return
        try:
            entries = sorted(directory.iterdir(), key=lambda p: (p.is_file(), p.name))
        except PermissionError:
            lines.append(f"{prefix}（没有读取权限）")
            return
        for entry in entries:
            if len(lines) >= MAX_ENTRIES:
                truncated = True
                return
            if entry.is_dir():
                if entry.name in SKIP_DIRS:
                    skipped.append(entry.name)
                    continue
                lines.append(f"{prefix}{entry.name}/")
                walk(entry, depth + 1, prefix + "  ")
            else:
                size = entry.stat().st_size
                unit = f"{size}B" if size < 1024 else f"{size // 1024}KB"
                lines.append(f"{prefix}{entry.name}  ({unit})")

    walk(root_dir, 1, "")
    out = "\n".join(lines) or "（空目录）"
    if skipped:
        out += f"\n（跳过了 {'、'.join(sorted(set(skipped)))}）"
    if truncated:
        out += f"\n（超过 {MAX_ENTRIES} 项，只显示了前面这些）"
    return out


print(list_dir("."))
```

## 装成工具

```python
def make_coding_tools() -> list[Tool]:
    """把上面四个函数包成工具。

    读文件的登记放在这一层而不是 read_file 里：登记的语义是「模型看过了」，
    而只有经由工具调用的那次读，模型才真的看见了内容。
    我们自己在讲义里调 read_file 看看内容，不该算模型看过。
    """

    def tool_read(path: str, start_line: int = 1, end_line: int = 0) -> str:
        out = read_file(path, start_line, end_line)
        target = resolve_in_workspace(WORKSPACE, path)
        if end_line <= 0 and start_line <= 1:
            tracker.mark_read(target)       # 只有读了全文才算「看过」
        return out

    return [
        Tool("read_file", "读一个文件，返回带行号的内容。大文件可以只读一段。",
             {"type": "object", "properties": {
                 "path": {"type": "string", "description": "相对于工作目录的路径"},
                 "start_line": {"type": "integer", "description": "起始行号，从 1 开始"},
                 "end_line": {"type": "integer", "description": "结束行号，0 表示到文件末尾"}},
              "required": ["path"]}, tool_read),

        Tool("write_file", "整文件写入。新建文件用它；修改已有文件请优先用 edit_file。"
                           "覆盖已有文件之前必须先 read_file 读过。",
             {"type": "object", "properties": {
                 "path": {"type": "string"},
                 "content": {"type": "string", "description": "文件的完整新内容"}},
              "required": ["path", "content"]}, write_file),

        Tool("edit_file", "把文件里的一段原文替换成新内容。old_text 必须与原文完全一致"
                          "（含缩进），并且在文件里唯一。",
             {"type": "object", "properties": {
                 "path": {"type": "string"},
                 "old_text": {"type": "string", "description": "要被替换的原文，必须唯一"},
                 "new_text": {"type": "string", "description": "替换成什么，空字符串表示删除"},

        Tool("list_dir", "列出目录树。",
             {"type": "object", "properties": {
                 "path": {"type": "string", "description": "相对路径，默认当前目录"},

tracker = ReadTracker()          # 重新开一份，前面几格的手工调用不算数
```

## 让它真的改一次代码

```python
SYSTEM = """你是一个编码助手，在一个小项目里工作。

工作方式：
1. 先用 list_dir 看看有什么文件。
2. 用 read_file 读你要改的文件。
3. 用 edit_file 做小改动，用 write_file 新建文件。
4. 改完简短说明你做了什么。

规矩：修改已有文件之前必须先读过它。edit_file 的 old_text 必须和原文一模一样。"""


def trace(name, args, result):
    short = {k: (v[:40] + "…" if isinstance(v, str) and len(v) > 40 else v)
             for k, v in args.items()}
    mark = "✗" if result.is_error else "✓"
    print(f"  {mark} {name}({short})")
    print(f"      → {result.content.splitlines()[0][:80] if result.content else ''}")


agent = Agent(real_model, coding_registry, SYSTEM, max_rounds=12, on_event=trace)

print("任务：给 greet 函数加一个 greeting 参数，默认值 'Hello'\n")
answer = agent.run("给 app/main.py 里的 greet 函数加一个 greeting 参数，"
                   "默认值是 'Hello'，并且用它拼接问候语。改完把文件读一遍确认。")
print(f"\n助手：{answer}")
print("\n改完之后的文件：")
print(read_file("app/main.py"))
```

## 验收：那四条规矩真的生效了吗

```python
checks = []

# 1. 路径越界被拦住
r = coding_registry.execute("read_file", {"path": "../../etc/passwd"})
checks.append(("路径越界被拦住", r.is_error and "越出" in r.content))

# 2. 没读过就覆盖被拦住（重开一份读取记录，忘掉刚才那轮读过什么）
tracker = ReadTracker()
fresh_registry = ToolRegistry(make_coding_tools())
r = fresh_registry.execute("write_file", {"path": "README.md", "content": "没了"})
checks.append(("没读过就覆盖被拦住", r.is_error and "还没读过" in r.content))

# 3. 读过之后可以覆盖
fresh_registry.execute("read_file", {"path": "README.md"})
r = fresh_registry.execute("write_file", {"path": "README.md",
                                          "content": "# 示例项目\n\n改过了。\n"})
checks.append(("读过之后可以覆盖", not r.is_error))

# 4. 不唯一的替换被拒绝
r = fresh_registry.execute("edit_file", {"path": "app/main.py",
                                         "old_text": "\n", "new_text": ""})
checks.append(("不唯一的替换被拒绝", r.is_error and "出现了" in r.content))

# 5. 新建文件不需要先读
r = fresh_registry.execute("write_file", {"path": "app/util.py", "content": "PI = 3.14\n"})
checks.append(("新建文件不需要先读", not r.is_error))

for label, ok in checks:
    print(f"  {'✓' if ok else '✗'} {label}")
print(f"\n{sum(1 for _, ok in checks if ok)}/{len(checks)} 条规矩生效")
```

## 本章小结

- 每个文件工具的第一件事是**把路径钉在工作目录里**。三步都不能省：
  拒绝绝对路径（拼接对它无效）、`resolve()` 展开符号链接、
  用 `is_relative_to` 判断（`startswith` 会被 `workspace-evil` 骗过）。
- 读文件要**带行号**，模型靠它精确指认位置；要**能读一段**，否则大文件撑爆上下文；
  单行也要有长度上限。
- **覆盖已有文件之前必须先读过**。这不是安全机制，是防止模型自信地重写它没看过的内容。
  顺带还能挡住「你读完之后别人改了」的情况。
- 改文件**按唯一的原文片段，不按行号**。行号在上一次编辑之后就失效了。
  片段不唯一时**拒绝执行**——替换错地方比不替换糟糕得多。
- 列目录要跳过 `.git`、`node_modules` 之类，并且**条目总数封顶**。
- 报错要给出**下一步能怎么办**：「文件不存在」是死路，
  「文件不存在，同目录下有这些」是线索。
