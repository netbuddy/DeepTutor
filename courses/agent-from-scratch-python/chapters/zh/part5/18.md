# 权限与沙箱

上一章的 agent 能改文件了。这一章给它加最后一个、也是最危险的工具：**执行命令**。

危险在哪儿？编译、跑测试、看 git 状态，这些都要执行命令，agent 没有它几乎干不了活。
但同一个工具也能执行 `rm -rf`。

这一章的主线是一句话：

> **提示词不是安全边界。**

在系统提示里写「不要删除文件」，挡不住任何事情——模型可能误解、可能被输入里的指令带偏、
也可能只是判断失误。真正的边界必须是**代码里的机制**。

我们建三道防线，从内到外：

| 防线 | 拦住什么 | 拦不住什么 |
|---|---|---|
| 1. 不给 shell | 命令注入 | 命令本身就是危险的 |
| 2. 审批门 | 危险操作没经人同意就跑 | 人点了同意之后 |
| 3. 隔离 | 跑起来之后的破坏范围 | 隔离本身的漏洞 |

@dataclass
class TextBlock:
    text: str
    type: Literal["text"] = "text"

@dataclass
class ToolUseBlock:
    name: str
    args: dict[str, Any]
    id: str = field(default_factory=lambda: new_id("call"))

@dataclass

    tool_use_id: str
    content: str
    is_error: bool = False

@dataclass
class Message:
    role: Literal["system", "user", "assistant", "tool"]
    content: list[Any]
    id: str = field(default_factory=lambda: new_id("msg"))

    def tool_uses(self): return [b for b in self.content if isinstance(b, ToolUseBlock)]

@dataclass

    kind: Literal["text", "tool_use"]

    tool: ToolUseBlock | None = None

class Model(Protocol):
    def stream(self, messages: list[Message], tools: list[dict]) -> Iterator[Delta]: ...

@dataclass

    is_error: bool = False

@dataclass
class Tool:
    name: str
    description: str
    parameters: dict[str, Any]
    run: Callable[..., str]
    def spec(self): return {"name": self.name, "description": self.description,

class ToolRegistry:
    def __init__(self, tools=None): self._tools = {t.name: t for t in (tools or [])}
    def names(self): return list(self._tools)
    def specs(self): return [t.spec() for t in self._tools.values()]
    def execute(self, name, args) -> ToolResult:
        tool = self._tools.get(name)
        if tool is None:
            return ToolResult(f"没有叫 {name} 的工具", is_error=True)
        try:
            return ToolResult(str(tool.run(**args)))
        except Exception as exc:
            return ToolResult(f"{type(exc).__name__}: {exc}", is_error=True)

def to_openai_messages(messages):
    out = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({"role": "assistant", "content": text or None, "tool_calls": [
                {"id": c.id, "type": "function",
                 "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                for c in calls]})
        else:
            out.append({"role": m.role, "content": text})
    return out

def to_openai_tools(tools):
    return [{"type": "function", "function": {"name": t["name"], "description": t.get("description", ""),
             "parameters": t.get("parameters", {"type": "object", "properties": {}})}} for t in tools]

class ToolCallAccumulator:
    def __init__(self): self._slots = {}
    def feed(self, raw_calls):
        for item in raw_calls:
            s = self._slots.setdefault(int(item.get("index", 0)), {"id": "", "name": "", "arguments": ""})
            if item.get("id"): s["id"] = item["id"]
            fn = item.get("function") or {}
            if fn.get("name"): s["name"] = fn["name"]
            if fn.get("arguments"): s["arguments"] += fn["arguments"]
    def finish(self):
        out = []
        for i in sorted(self._slots):
            s = self._slots[i]
            try: args = json.loads(s["arguments"] or "{}")
            except json.JSONDecodeError: args = {}
            out.append(ToolUseBlock(id=s["id"] or new_id("call"), name=s["name"],
                                    args=args if isinstance(args, dict) else {}))
        return out

def parse_sse_line(line):
    line = line.strip()
    if not line or line.startswith(":") or not line.startswith("data:"): return None
    payload = line[5:].strip()
    if payload == "[DONE]": return None
    try: return json.loads(payload)
    except json.JSONDecodeError: return None

class OpenAICompatModel:
    def __init__(self, base_url, model, timeout=180):
        self.base_url, self.model, self.timeout = base_url.rstrip("/"), model, timeout
    def stream(self, messages, tools):
        body = {"model": self.model, "messages": to_openai_messages(messages), "stream": True}
        if tools: body["tools"] = to_openai_tools(tools)
        req = urllib.request.Request(f"{self.base_url}/chat/completions",
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"}, method="POST")
        acc = ToolCallAccumulator()
        with urllib.request.urlopen(req, timeout=self.timeout) as resp:
            for raw in resp:
                chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
                if chunk is None: continue
                d = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
                if d.get("content"): yield Delta(kind="text", text=d["content"])
                if d.get("tool_calls"): acc.feed(d["tool_calls"])
        for b in acc.finish(): yield Delta(kind="tool_use", tool=b)

class Agent:
    def __init__(self, model, registry, system_prompt="", max_rounds=10, on_event=None):
        self.model, self.registry = model, registry
        self.system_prompt = system_prompt or "你是一个编码助手。"
        self.max_rounds, self.on_event = max_rounds, on_event or (lambda *a: None)
    def run(self, user_text):
        history = [Message(role="system", content=[TextBlock(text=self.system_prompt)]),
                   Message(role="user", content=[TextBlock(text=user_text)])]
        for _ in range(self.max_rounds):
            reply = self._one_round(history)
            history.append(reply)
            calls = reply.tool_uses()
            if not calls:
                return reply.text()
            results = []
            for c in calls:
                r = self.registry.execute(c.name, c.args)
                self.on_event(c.name, c.args, r)
                results.append(ToolResultBlock(tool_use_id=c.id, content=r.content,
                                               is_error=r.is_error))
            history.append(Message(role="tool", content=results))
        return "（轮数用完了）"
    def _one_round(self, messages):
        text_parts, blocks = [], []
        for d in self.model.stream(messages, self.registry.specs()):
            if d.kind == "text":
                text_parts.append(d.text)
            elif d.tool is not None:
                if text_parts:
                    blocks.append(TextBlock(text="".join(text_parts))); text_parts = []
                blocks.append(d.tool)
        if text_parts: blocks.append(TextBlock(text="".join(text_parts)))
        return Message(role="assistant", content=blocks)

## 第一道防线：不给 shell

最省事的写法是把模型给的字符串直接丢给 shell。**这是这一章最不该做的事。**

```python
untrusted = "notes.txt; echo 我是被注入的命令"

print("── 交给 shell 执行（错误做法）──")
proc = subprocess.run(f"cat {untrusted}", shell=True, cwd=WORKSPACE,
                      capture_output=True, text=True)
print(proc.stdout.strip())
print("→ 分号后面那条也执行了。模型只是想读一个文件，参数里的分号却变成了第二条命令。")

print("\n── 用参数数组执行（正确做法）──")
proc = subprocess.run(["cat", untrusted], cwd=WORKSPACE, capture_output=True, text=True)
print(f"stdout: {proc.stdout.strip()!r}")
print(f"stderr: {proc.stderr.strip()}")
print("→ 整个字符串被当成一个文件名。没有 shell，就没有 shell 注入。")
```

这不是「模型会写恶意命令」的问题。更常见的情况是：模型把一段**从网页或文件里读来的文本**
当参数传进来，而那段文本里正好有分号、反引号或者 `$(...)`。
不给 shell，这类问题**从根上不存在**。

代价是失去了管道和重定向。这个代价值得付：真要用管道，让 agent 分两步做，
或者提供一个专门的工具。

## 第二道防线：审批门

命令分三类。分类的依据不是「危不危险」，而是**「出了事能不能撤回」**。

```python
class Decision(str, Enum):
    ALLOW = "allow"      # 自动放行
    ASK = "ask"          # 要人点头
    DENY = "deny"        # 直接拒绝，不问


@dataclass
class Verdict:
    decision: Decision
    reason: str


# 只读、无副作用、跑错了也不心疼的命令。(程序, 子命令) 两级匹配——
# 只看程序名不够：git status 是只读的，git push 不是。
AUTO_ALLOW: set[tuple[str, str | None]] = {
    ("ls", None), ("cat", None), ("head", None), ("tail", None),
    ("wc", None), ("grep", None), ("find", None), ("file", None),
    ("git", "status"), ("git", "diff"), ("git", "log"), ("git", "show"),
    ("go", "build"), ("go", "vet"), ("go", "test"), ("go", "fmt"),
    ("python3", "-c"), ("python3", "-V"), ("pytest", None),
}

# 这些直接拒绝。
#
# 说清楚一件事：**这不是安全边界**。绕过一张名单的办法太多了
# （改名、软链接、用别的程序做同一件事）。它的作用是「明显不该由 agent 自己决定的事，
# 连问都不问」——省掉一次无意义的审批打扰。真正的边界是第三道防线。
NEVER = {"rm", "rmdir", "dd", "mkfs", "shutdown", "reboot", "halt", "poweroff",
         "sudo", "su", "chmod", "chown", "kill", "pkill", "killall",
         "curl", "wget", "ssh", "scp", "nc", "systemctl"}


def classify(argv: list[str]) -> Verdict:
    if not argv:
        return Verdict(Decision.DENY, "命令是空的")
    program = os.path.basename(argv[0])
    sub = argv[1] if len(argv) > 1 else None

    if program in NEVER:
        return Verdict(Decision.DENY, f"{program} 不在这个 agent 的能力范围内，请人工执行")
    if (program, None) in AUTO_ALLOW:
        return Verdict(Decision.ALLOW, f"{program} 是只读命令")
    if (program, sub) in AUTO_ALLOW:
        return Verdict(Decision.ALLOW, f"{program} {sub} 是只读命令")
    return Verdict(Decision.ASK, f"{program} 可能有副作用，需要你确认")


for argv in (["ls", "-la"], ["git", "status"], ["git", "push"], ["go", "test", "./..."],
             ["rm", "-rf", "/"], ["make", "install"]):
    v = classify(argv)
    mark = {"allow": "自动放行", "ask": "要人确认", "deny": "直接拒绝"}[v.decision.value]
    print(f"  {' '.join(argv):24} {mark:6} （{v.reason}）")
```

### 记住用户的选择

每跑一次 `make build` 都问一遍，用户会疯掉，然后开始无脑点同意——
**审批疲劳会让审批形同虚设**。所以要能记住。

```python
class ApprovalStore:
    """记住用户在这次会话里同意过什么。

    两种粒度：
      - 这一条命令（含全部参数）——最保守。
      - 这个程序的这个子命令（不含其余参数）——好用，范围仍然可控。

    没有「永久同意一切」这一档：那等于把审批门关掉，
    留着一个关不掉的门比留一个默认敞开的门诚实。
    """

    def __init__(self) -> None:
        self.exact: set[str] = set()
        self.prefix: set[tuple[str, str | None]] = set()

    @staticmethod
    def key(argv: list[str]) -> str:
        return "\x00".join(argv)

    def remembered(self, argv: list[str]) -> bool:
        if self.key(argv) in self.exact:
            return True
        program = os.path.basename(argv[0])
        sub = argv[1] if len(argv) > 1 else None
        return (program, sub) in self.prefix

    def remember(self, argv: list[str], scope: str) -> None:
        if scope == "exact":
            self.exact.add(self.key(argv))
        elif scope == "prefix":
            program = os.path.basename(argv[0])
            sub = argv[1] if len(argv) > 1 else None
            self.prefix.add((program, sub))


# 演示用的自动审批人：把「用户会点什么」写成一个函数。
# 真实程序里这里是一个界面弹窗；接口一样，行为由人决定。
def make_approver(script: dict[str, str]):
    """script: 命令首词 → 'no' / 'once' / 'always'"""
    def approve(argv: list[str], reason: str) -> str:
        answer = script.get(os.path.basename(argv[0]), "no")
        label = {"no": "拒绝", "once": "同意这一次", "always": "这个命令以后都同意"}[answer]
        print(f"    [审批] {' '.join(argv)} —— {reason} → 用户选择：{label}")
        return answer
    return approve


print("审批门就绪")
```

## 第三道防线：隔离

前两道防线管的是**要不要跑**。第三道管的是**跑起来之后能碰到什么**。

```python
# 传给子进程的环境变量白名单。
#
# 这一条很容易被忽略，但它是最实在的一条：默认情况下 os.environ 会原样继承给子进程，
# 里面通常有 API 密钥、令牌、数据库密码。一条 `env` 命令就能把它们全打印出来，
# 而输出会原样进对话历史、再发给模型服务。
ENV_ALLOWLIST = ("PATH", "HOME", "LANG", "LC_ALL", "TZ", "TERM")

MAX_OUTPUT_CHARS = 4000
DEFAULT_TIMEOUT_S = 30


def build_env() -> dict[str, str]:
    env = {k: os.environ[k] for k in ENV_ALLOWLIST if k in os.environ}
    env.setdefault("PATH", "/usr/local/bin:/usr/bin:/bin")
    # 明确告诉子进程「这里没有交互」，免得它弹出编辑器或分页器把执行卡死
    env["GIT_PAGER"] = "cat"
    env["PAGER"] = "cat"
    env["NO_COLOR"] = "1"
    return env


def clip(text: str, limit: int = MAX_OUTPUT_CHARS) -> str:
    """输出太长时保留头尾，中间省略。

    保留头尾而不是只留开头：报错信息经常在末尾（栈回溯、测试汇总），
    只截前面等于把最有用的部分扔了。
    """
    if len(text) <= limit:
        return text
    head, tail = text[: limit * 2 // 3], text[-limit // 3:]
    return f"{head}\n…（中间省略 {len(text) - len(head) - len(tail)} 字符）…\n{tail}"


@dataclass
class CommandOutcome:
    argv: list[str]
    exit_code: int
    stdout: str
    stderr: str
    duration_s: float
    timed_out: bool = False

    def render(self) -> str:
        parts = [f"$ {' '.join(self.argv)}"]
        if self.timed_out:
            parts.append(f"（超时，已终止，跑了 {self.duration_s:.1f} 秒）")
        else:
            parts.append(f"退出码 {self.exit_code}，用时 {self.duration_s:.1f} 秒")
        if self.stdout.strip():
            parts.append("stdout:\n" + clip(self.stdout))
        if self.stderr.strip():
            parts.append("stderr:\n" + clip(self.stderr))
        if not self.stdout.strip() and not self.stderr.strip():
            parts.append("（没有输出）")
        return "\n".join(parts)


def run_isolated(argv: list[str], cwd: Path, timeout_s: int = DEFAULT_TIMEOUT_S) -> CommandOutcome:
    """在受限环境里执行一条命令。

    这里做到的隔离是**进程级**的：工作目录、环境变量、超时、输出上限。
    做不到的是文件系统隔离和网络隔离——子进程照样能读整个磁盘、照样能联网。
    真要那两样，得靠操作系统级的手段（容器、命名空间、bwrap 之类），
    那已经超出这门课的范围。**知道自己的隔离到哪儿为止，比以为自己隔离了更重要。**
    """
    started = time.time()
    try:
        proc = subprocess.run(
            argv,
            cwd=str(cwd),
            env=build_env(),
            capture_output=True,
            text=True,
            timeout=timeout_s,
            stdin=subprocess.DEVNULL,   # 别让它等着读输入，那会一直挂到超时
            shell=False,                # 写出来是为了让读代码的人一眼确认
        )
    except subprocess.TimeoutExpired as exc:
        return CommandOutcome(argv, -1, exc.stdout or "", exc.stderr or "",
                              time.time() - started, timed_out=True)
    except FileNotFoundError:
        return CommandOutcome(argv, -1, "", f"找不到程序 {argv[0]}", time.time() - started)
    return CommandOutcome(argv, proc.returncode, proc.stdout, proc.stderr,
                          time.time() - started)


print(run_isolated(["python3", "hello.py"], WORKSPACE).render())
print()
print(run_isolated(["python3", "-c", "import time; time.sleep(10)"], WORKSPACE, timeout_s=2).render())
```

## 三道防线合起来

```python
class CommandRunner:
    """把三道防线串成一个工具。"""

    def __init__(self, workspace: Path, approver: Callable[[list[str], str], str],
                 store: ApprovalStore | None = None) -> None:
        self.workspace = workspace
        self.approver = approver
        self.store = store or ApprovalStore()
        self.audit: list[dict[str, Any]] = []

    def run(self, command: list[str], timeout_s: int = DEFAULT_TIMEOUT_S) -> str:
        if not isinstance(command, list) or not all(isinstance(x, str) for x in command):
            raise ValueError("command 必须是一个字符串数组，比如 [\"go\", \"test\", \"./...\"]。"
                             "不要传一整行 shell 命令。")

        verdict = classify(command)
        record = {"argv": command, "decision": verdict.decision.value, "reason": verdict.reason}

        if verdict.decision is Decision.DENY:
            self.audit.append({**record, "outcome": "拒绝"})
            raise PermissionError(f"{verdict.reason}。这条命令没有执行。")

        if verdict.decision is Decision.ASK and not self.store.remembered(command):
            answer = self.approver(command, verdict.reason)
            if answer == "no":
                self.audit.append({**record, "outcome": "用户拒绝"})
                raise PermissionError("用户拒绝了这条命令。换个办法，或者问问他想怎么做。")
            if answer == "always":
                self.store.remember(command, "prefix")
            record["approved_by"] = "用户"

        outcome = run_isolated(command, self.workspace, timeout_s)
        self.audit.append({**record, "outcome": f"退出码 {outcome.exit_code}",
                           "duration_s": round(outcome.duration_s, 2)})
        return outcome.render()


approver = make_approver({"make": "always", "git": "once", "npm": "no"})
runner = CommandRunner(WORKSPACE, approver)

print("── 自动放行 ──")

except PermissionError as exc:
    print("被拒：", exc)

print("\n── 同一条命令再来一次（记住了，不再打扰）──")

    print(runner.run(["make", "--version"]).splitlines()[0])
except PermissionError as exc:
    print("被拒：", exc)

print("\n── 用户拒绝 ──")

    runner.run(["npm", "install"])
except PermissionError as exc:
    print("被拒：", exc)

    print("被拒：", exc)

def make_command_tool(runner: CommandRunner) -> Tool:

        "run_command",
        "在工作目录里执行一条命令。command 必须是参数数组，例如 [\"go\", \"test\", \"./...\"]。"
        "不支持管道、重定向和 shell 语法——需要的话请分成多步。",
        {"type": "object", "properties": {
            "command": {"type": "array", "items": {"type": "string"},
                        "description": "程序名和各个参数，每个一项"},
            "timeout_s": {"type": "integer", "description": "超时秒数，默认 30"}},
         "required": ["command"]},
        lambda command, timeout_s=DEFAULT_TIMEOUT_S: runner.run(command, timeout_s))


runner = CommandRunner(WORKSPACE, make_approver({"python3": "always"}))
registry = ToolRegistry([make_command_tool(runner)])

SYSTEM = """你是一个能执行命令的助手，工作目录里有一些文件。

只能用 run_command，command 必须是参数数组。不支持管道和 shell 语法。
命令被拒绝时不要换个写法重试同一件事，直接告诉用户你需要什么。"""


def trace(name, args, result):
    print(f"  {'✗' if result.is_error else '✓'} {name}({args})")

agent = Agent(real_model, registry, SYSTEM, max_rounds=8, on_event=trace)
print("任务：数一数工作目录里有几个文件\n")
print("助手：", agent.run("工作目录里有几个文件？列出来。"))
```

## 验收

```python
checks = []

# 1. 参数里的分号不会变成第二条命令
out = run_isolated(["echo", "a; echo b"], WORKSPACE)
checks.append(("参数里的分号不被解释", out.stdout.strip() == "a; echo b"))

# 2. 环境变量不外泄
out = run_isolated(["python3", "-c", "import os; print(os.environ.get('FAKE_API_KEY', '（没有）'))"],
                   WORKSPACE)
checks.append(("敏感环境变量没传下去", "（没有）" in out.stdout))

# 3. 超时能终止
out = run_isolated(["python3", "-c", "import time; time.sleep(10)"], WORKSPACE, timeout_s=1)
checks.append(("超时能终止", out.timed_out))

# 4. 输出有上限
out = run_isolated(["python3", "-c", "print('x' * 100000)"], WORKSPACE)
checks.append(("输出被截断", len(out.render()) < 10000))

# 5. 拒绝名单生效
strict = CommandRunner(WORKSPACE, make_approver({}))
try:
    strict.run(["rm", "-rf", "."])
    checks.append(("拒绝名单生效", False))
except PermissionError:
    checks.append(("拒绝名单生效", True))

# 6. 一整行 shell 命令会被工具本身挡回去
try:
    strict.run("ls -la | grep py")
    checks.append(("拒绝整行 shell 命令", False))
except ValueError:
    checks.append(("拒绝整行 shell 命令", True))

# 7. 没被同意的命令不会执行
marker = WORKSPACE / "should-not-exist.txt"
try:
    strict.run(["touch", str(marker)])
except PermissionError:
    pass
checks.append(("没被同意的命令没有执行", not marker.exists()))

for label, ok in checks:
    print(f"  {'✓' if ok else '✗'} {label}")
print(f"\n{sum(1 for _, ok in checks if ok)}/{len(checks)} 项通过")
```

## 说清楚防线到哪儿为止

这三道防线**不是**一个安全沙箱。诚实地列一下它们挡不住什么：

- **文件系统**：子进程照样能读你整个家目录。工作目录只是它的起始位置，不是牢笼。
- **网络**：进程级限制管不了网络。子进程能联网，就能把读到的东西发出去。
- **资源**：内存和 CPU 没有上限，一个死循环能把机器拖垮。

要挡住这三样，得上**操作系统级隔离**：容器、命名空间、seccomp，或者干脆一台隔离的机器。
那是另一个话题，但有一条现在就该记住：

> **在你没有操作系统级隔离的环境里，不要让 agent 自动执行未经审批的命令。**
> 审批门此时不是「防打扰」，它是唯一真正起作用的那道防线。

## 本章小结

- **提示词不是安全边界**。系统提示里写「不要删文件」挡不住任何事，
  真正的边界必须是代码里的机制。
- **不给 shell**。用参数数组执行，命令注入这类问题从根上不存在。
  代价是失去管道和重定向，这个代价值得付。
- 命令分三类的依据是**「出了事能不能撤回」**，不是「危不危险」。
  分类要看（程序，子命令）两级——`git status` 只读，`git push` 不是。
- **拒绝名单不是安全边界**。它的作用是省掉无意义的审批打扰，绕过它的办法太多了。
- **审批疲劳会让审批形同虚设**。要能记住用户的选择，但**不要提供「永久同意一切」**——
  留一个关不掉的门比留一个默认敞开的门诚实。
- 传给子进程的**环境变量要走白名单**。默认继承会把 API 密钥交出去，
  而一条 `env` 的输出会原样进对话历史、再发给模型服务。
- 输出截断要**保留头尾**：报错信息经常在末尾。
- 子进程的标准输入要接到 `DEVNULL`，否则等待输入的程序会一直挂到超时。
- 审批和执行都要**留痕**：出了事，「谁在什么时候同意了什么」是唯一能查的东西。
- **知道自己的隔离到哪儿为止，比以为自己隔离了更重要**。进程级隔离管不了
  文件系统、网络和资源——那三样要靠操作系统级的手段。
