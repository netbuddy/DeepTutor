# 有状态的 agent

到目前为止，我们的入口一直是一个函数：`run(history)`。历史从外面传进来，跑完就完事。
这足够跑通一轮，但**没法做多轮对话**——第二次调用怎么知道第一次说了什么？

这一章把 agent 从「一个函数」变成「一个对象」，并且回答一个设计问题：
**状态到底放在谁身上。**

## 三样东西，三个归属

| 东西 | 例子 | 谁持有 | 为什么 |
|---|---|---|---|
| 配置 | 用哪个模型、几轮预算、系统提示 | **Agent** | 一个 agent 的性格是固定的 |
| 对话 | 消息列表、这轮对话的标识 | **Session** | 同一个 agent 可以同时开几段对话 |
| 一次执行 | 这一轮用了几轮、为什么停 | **TurnResult** | 用完就丢 |

把对话放在 Agent 身上是最常见的设计错误。那样一个 agent 只能有一段对话，
换个话题就得重建整个对象，工具和配置全都白搭一遍。

## 本章起点

第 1 到 12 章的成果。这一章不需要重试和守卫，所以那两层没带进来——
它们是包在外面的，随时可以套回去。

from __future__ import annotations
from dataclasses import dataclass, field
from enum import Enum
from typing import Any, Callable, Iterator, Literal, Protocol
import datetime, json, time, urllib.request, uuid

def new_id(prefix: str) -> str: return f"{prefix}_{uuid.uuid4().hex[:12]}"

@dataclass
class TextBlock:
    text: str
    type: Literal["text"] = "text"

@dataclass
class ToolUseBlock:
    name: str
    args: dict[str, Any]
    id: str = field(default_factory=lambda: new_id("call"))
    type: Literal["tool_use"] = "tool_use"

@dataclass
class ToolResultBlock:
    tool_use_id: str
    content: str
    is_error: bool = False
    type: Literal["tool_result"] = "tool_result"

@dataclass
class Message:
    role: Literal["system", "user", "assistant", "tool"]
    content: list[Any]
    id: str = field(default_factory=lambda: new_id("msg"))
    created_at: float = field(default_factory=time.time)
    def text(self): return "".join(b.text for b in self.content if isinstance(b, TextBlock))
    def tool_uses(self): return [b for b in self.content if isinstance(b, ToolUseBlock)]
    def tool_results(self): return [b for b in self.content if isinstance(b, ToolResultBlock)]

@dataclass
class Delta:
    kind: Literal["text", "tool_use"]
    text: str = ""
    tool: ToolUseBlock | None = None

class Model(Protocol):
    def stream(self, messages: list[Message], tools: list[dict]) -> Iterator[Delta]: ...

@dataclass
class ToolResult:
    content: str
    is_error: bool = False
    is_invalid_args: bool = False

@dataclass
class Tool:
    name: str
    description: str
    parameters: dict[str, Any]
    run: Callable[..., str]
    def spec(self): return {"name": self.name, "description": self.description,
                            "parameters": self.parameters}

def validate_args(args, schema) -> str:
    if not isinstance(args, dict): return "参数必须是一个对象"
    if "_invalid_json" in args: return f"参数不是合法的 JSON：{args['_invalid_json'][:120]}"
    props, required = schema.get("properties") or {}, schema.get("required") or []
    missing = [k for k in required if k not in args]
    if missing: return f"缺少必填参数 {'、'.join(missing)}"
    unknown = [k for k in args if k not in props]
    if unknown: return f"不认识的参数 {'、'.join(unknown)}；这个工具接受 {'、'.join(props)}"
    checks = {"string": str, "number": (int, float), "integer": int,
              "boolean": bool, "array": list, "object": dict}
    for k, v in args.items():
        spec = props[k]
        pt = checks.get(spec.get("type"))
        if pt and not isinstance(v, pt):
            return f"参数 {k} 应该是 {spec.get('type')}，收到的是 {type(v).__name__}"
        if spec.get("enum") and v not in spec["enum"]:
            return f"参数 {k} 只能是 {'、'.join(map(str, spec['enum']))} 之一"
    return ""

class ToolRegistry:
    def __init__(self, tools=None): self._tools = {t.name: t for t in (tools or [])}
    def add(self, tool): self._tools[tool.name] = tool
    def names(self): return list(self._tools)
    def specs(self): return [t.spec() for t in self._tools.values()]
    def execute(self, name, args) -> ToolResult:
        tool = self._tools.get(name)
        if tool is None:
            return ToolResult(f"没有叫 {name} 的工具。可用的：{'、'.join(self.names())}", is_error=True)
        problem = validate_args(args, tool.parameters)
        if problem:
            return ToolResult(f"参数不合格：{problem}", is_error=True, is_invalid_args=True)
        try:
            return ToolResult(str(tool.run(**args)))
        except Exception as exc:
            return ToolResult(f"{type(exc).__name__}: {exc}", is_error=True)

def to_openai_messages(messages):
    out = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({"role": "assistant", "content": text or None, "tool_calls": [
                {"id": c.id, "type": "function",
                 "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                for c in calls]})
        else:
            out.append({"role": m.role, "content": text})
    return out

def to_openai_tools(tools):
    return [{"type": "function", "function": {"name": t["name"], "description": t.get("description", ""),
             "parameters": t.get("parameters", {"type": "object", "properties": {}})}} for t in tools]

class ToolCallAccumulator:
    def __init__(self): self._slots = {}
    def feed(self, raw_calls):
        for item in raw_calls:
            s = self._slots.setdefault(int(item.get("index", 0)), {"id": "", "name": "", "arguments": ""})
            if item.get("id"): s["id"] = item["id"]
            fn = item.get("function") or {}
            if fn.get("name"): s["name"] = fn["name"]
            if fn.get("arguments"): s["arguments"] += fn["arguments"]
    def finish(self):
        out = []
        for i in sorted(self._slots):
            s = self._slots[i]
            raw = s["arguments"] or "{}"
            try:
                args = json.loads(raw)
                if not isinstance(args, dict): args = {"_raw": raw}
            except json.JSONDecodeError:
                args = {"_invalid_json": raw}
            out.append(ToolUseBlock(id=s["id"] or new_id("call"), name=s["name"], args=args))
        return out

def parse_sse_line(line):
    line = line.strip()
    if not line or line.startswith(":") or not line.startswith("data:"): return None
    payload = line[5:].strip()
    if payload == "[DONE]": return None
    try: return json.loads(payload)
    except json.JSONDecodeError: return None

class OpenAICompatModel:
    def __init__(self, base_url, model, timeout=180):
        self.base_url, self.model, self.timeout = base_url.rstrip("/"), model, timeout
    def stream(self, messages, tools):
        body = {"model": self.model, "messages": to_openai_messages(messages), "stream": True}
        if tools: body["tools"] = to_openai_tools(tools)
        req = urllib.request.Request(f"{self.base_url}/chat/completions",
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"}, method="POST")
        acc = ToolCallAccumulator()
        with urllib.request.urlopen(req, timeout=self.timeout) as resp:
            for raw in resp:
                chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
                if chunk is None: continue
                d = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
                if d.get("content"): yield Delta(kind="text", text=d["content"])
                if d.get("tool_calls"): acc.feed(d["tool_calls"])
        for b in acc.finish(): yield Delta(kind="tool_use", tool=b)

@dataclass
class Event:
    kind: str
    data: dict

class EventStream:
    def __init__(self): self._subscribers = []
    def subscribe(self, fn):
        self._subscribers.append(fn)
        return lambda: self._subscribers.remove(fn) if fn in self._subscribers else None
    def emit(self, kind, **data):
        event = Event(kind=kind, data=data)
        for fn in list(self._subscribers):
            try: fn(event)
            except Exception as exc: print(f"  [订阅者出错，已忽略] {type(exc).__name__}: {exc}")

class StopReason(str, Enum):
    COMPLETED = "completed"
    BUDGET_EXHAUSTED = "budget"
    CANCELLED = "cancelled"
    ERROR = "error"

@dataclass
class TurnResult:
    text: str
    stop_reason: StopReason
    rounds_used: int
    messages: list[Message] = field(default_factory=list)
    error: str = ""

def get_time(city: str) -> str:
    return f"{city} 现在是 {datetime.datetime.now().strftime('%H:%M')}"

def calculate(expression: str) -> str:
    allowed = set("0123456789+-*/(). ")
    if not set(expression) <= allowed:
        raise ValueError(f"表达式里有不允许的字符：{sorted(set(expression) - allowed)}")
    return str(eval(expression, {"__builtins__": {}}, {}))

registry = ToolRegistry([
    Tool("get_time", "查询某个城市的当前时间",
         {"type": "object", "properties": {"city": {"type": "string", "description": "城市名"}},
          "required": ["city"]}, get_time),
    Tool("calculate", "计算一个算术表达式，比如 (3+5)*2",
         {"type": "object", "properties": {
             "expression": {"type": "string", "description": "只含数字和 + - * / ( )"}},
          "required": ["expression"]}, calculate),
])

real_model = OpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
print("前十二章的成果已就位")

## Session：一段对话

```python
@dataclass
class Session:
    """一段对话。

    只装消息，不装模型也不装工具——那些是 Agent 的事。
    这样同一个 agent 能同时开好几段对话，而一段对话也能换个 agent 接着聊。
    """
    id: str = field(default_factory=lambda: new_id("ses"))
    title: str = ""
    messages: list[Message] = field(default_factory=list)
    created_at: float = field(default_factory=time.time)
    metadata: dict[str, Any] = field(default_factory=dict)

    def append(self, *messages: Message) -> None:
        self.messages.extend(messages)

    def user_turns(self) -> int:
        return sum(1 for m in self.messages if m.role == "user")

    def last_assistant_text(self) -> str:
        for m in reversed(self.messages):
            if m.role == "assistant" and m.text():
                return m.text()
        return ""

    def summary_line(self) -> str:
        first = next((m.text() for m in self.messages if m.role == "user"), "")
        return self.title or (first[:24] + "…" if len(first) > 24 else first) or "（空会话）"


print("会话对象就绪")
```

## 系统提示：只有一条，永远在最前

系统提示不放在 `Session.messages` 里，而是 Agent 的配置。原因有三条：

1. 它属于 agent 的性格，不属于这段对话。
2. 多数服务只认**开头那一条** system 消息，混在中间会被忽略。
3. 第 16 章要压缩上下文，系统提示是唯一**绝对不能压掉**的部分。
   把它单独放着，压缩逻辑就不用小心翼翼地绕开它。

```python
@dataclass
class AgentConfig:
    system_prompt: str = "你是一个helpful的助手。需要用工具时就调用，不要凭空编造。"
    max_rounds: int = 8
    # 每次请求都把系统提示放在最前面重新拼一遍，而不是往会话里塞一条。
    # 好处：改了系统提示，历史会话立刻跟着改，不用回去改数据。


class Agent:
    """一个 agent：一套配置、一个模型、一组工具。

    注意它**不持有会话**。会话是参数，每次调用传进来。
    这样一个 agent 实例能服务很多段对话——服务端上这一点是刚需，
    不然每来一个用户就要重建一遍工具表。
    """

    def __init__(self, model: Model, registry: ToolRegistry,
                 config: AgentConfig | None = None,
                 stream: EventStream | None = None) -> None:
        self.model = model
        self.registry = registry
        self.config = config or AgentConfig()
        self.stream = stream or EventStream()

    def _prepare(self, session: Session) -> list[Message]:
        """拼出这一次要发给模型的消息。系统提示在最前，然后是会话历史。"""
        head = Message(role="system", content=[TextBlock(text=self.config.system_prompt)])
        return [head, *session.messages]

    def send(self, session: Session, user_text: str) -> TurnResult:
        """用户说一句话，agent 回一句。会话历史就地更新。"""
        user_msg = Message(role="user", content=[TextBlock(text=user_text)])
        session.append(user_msg)

        result = self._loop(session)

        # 这一轮产生的消息全部并进会话——包括中间那些工具调用与结果。
        # 不能只留最终答案：模型下一轮需要看到它自己调过什么、拿到了什么。
        session.append(*result.messages)
        return result

    def _loop(self, session: Session) -> TurnResult:
        turn_id = new_id("turn")
        new_messages: list[Message] = []
        self.stream.emit("turn_start", turn_id=turn_id, session_id=session.id)

        for round_no in range(1, self.config.max_rounds + 1):
            base = self._prepare(session) + new_messages
            try:
                reply = self._one_round(base)
            except Exception as exc:
                return TurnResult("", StopReason.ERROR, round_no, new_messages,
                                  error=f"{type(exc).__name__}: {exc}")
            new_messages.append(reply)

            calls = reply.tool_uses()
            if not calls:
                self.stream.emit("turn_done", turn_id=turn_id, reason=StopReason.COMPLETED.value)
                return TurnResult(reply.text(), StopReason.COMPLETED, round_no, new_messages)

            results = []
            for call in calls:
                r = self.registry.execute(call.name, call.args)
                self.stream.emit("tool_end", name=call.name, ok=not r.is_error, output=r.content)
                results.append(ToolResultBlock(tool_use_id=call.id, content=r.content,
                                               is_error=r.is_error))
            new_messages.append(Message(role="tool", content=results))

        return TurnResult("", StopReason.BUDGET_EXHAUSTED, self.config.max_rounds, new_messages)

    def _one_round(self, messages: list[Message]) -> Message:
        text_parts, blocks = [], []
        for d in self.model.stream(messages, self.registry.specs()):
            if d.kind == "text":
                text_parts.append(d.text)
                self.stream.emit("text_delta", delta=d.text)
            elif d.tool is not None:
                if text_parts:
                    blocks.append(TextBlock(text="".join(text_parts)))
                    text_parts = []
                blocks.append(d.tool)
        if text_parts:
            blocks.append(TextBlock(text="".join(text_parts)))
        return Message(role="assistant", content=blocks)


print("Agent 就绪")
```

## 多轮对话：指代能被消解，说明历史真的带上了

```python
agent = Agent(real_model, registry, AgentConfig(
    system_prompt="你是一个简洁的助手。算数一律调用 calculate 工具，不要心算。"))
session = Session(title="随手算几笔")

for question in ["帮我算 128 * 7",
                 "再乘以 3 呢？",          # 「再」——它得知道上一次的结果
                 "刚才我一共问了你几个数学问题？"]:
    print(f"\n用户：{question}")
    result = agent.send(session, question)
    print(f"助手：{result.text}")

print(f"\n会话 {session.id} 现在有 {len(session.messages)} 条消息，"
      f"{session.user_turns()} 轮用户提问")
```

第二个问题里的「再乘以 3」没有主语。它答对了，就说明**历史确实被带上了**——
这是有状态和无状态的分界线。

## 一个 agent，多段对话

```python
work = Session(title="工作")
life = Session(title="生活")

agent.send(work, "记住：我的项目代号叫「北极星」。")
agent.send(life, "记住：我养的猫叫「馒头」。")

print("在工作会话里问项目代号：")
print(" ", agent.send(work, "我的项目代号是什么？").text)

print("在生活会话里问项目代号（它不该知道）：")
print(" ", agent.send(life, "我的项目代号是什么？").text)

for s in (work, life):
    print(f"  {s.summary_line()}：{len(s.messages)} 条消息")
```

两段对话互不串味。**这正是把消息放在 Session 而不是 Agent 上的回报**——
如果历史挂在 agent 身上，上面这段代码要建两个 agent，工具表和配置都得重复一份。

## 会话里到底存了什么

```python
demo = Session(title="看一眼内部结构")
agent.send(demo, "北京现在几点？")

for i, m in enumerate(demo.messages, 1):
    kinds = []
    if m.text():
        kinds.append(f"文字({len(m.text())}字)")
    for c in m.tool_uses():
        kinds.append(f"调用 {c.name}")
    for r in m.tool_results():
        kinds.append(f"结果 {'✗' if r.is_error else '✓'}")
    print(f"{i}. {m.role:9} {' + '.join(kinds)}")
```

注意**中间过程也在会话里**：用户消息、助手的工具调用、工具结果、助手的最终回答。
界面上通常只显示第一条和最后一条，但发给模型的必须是全部——
少了工具调用与结果的对应关系，下一轮就会被服务端拒绝（第 11 章讲过）。

## 本章小结

- 三样东西三个归属：**配置归 Agent，对话归 Session，一次执行的结果用完就丢**。
- **Agent 不持有会话**。会话是参数，一个 agent 实例服务很多段对话——
  否则每来一个用户就要重建一遍工具表和配置。
- **系统提示不进会话**，每次请求时拼在最前面。三个理由：它属于 agent 不属于对话；
  多数服务只认开头那一条；第 16 章压缩上下文时它是唯一不能动的部分。
- 一轮里产生的**全部**消息都要并进会话，不能只留最终答案——
  模型下一轮需要看到自己调过什么、拿到了什么。
- 验证「历史真的带上了」的最简单办法：问一个**带指代**的问题（「再乘以 3 呢？」），
  它答对了才算。
