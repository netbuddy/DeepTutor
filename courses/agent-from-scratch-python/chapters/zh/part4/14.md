# 会话持久化

上一章的会话活在内存里，进程一退全没了。这一章把它写到磁盘上。

看起来这是个「序列化一下就完了」的小事。其实有四个坑：

1. **怎么写**——每次全量覆写，还是追加？答案影响的是崩溃时你丢多少东西。
2. **怎么读回来**——`{"type": "text", ...}` 要变回 `TextBlock`，靠什么判断？
3. **写到一半崩了**——文件尾多半行，下次读会炸。
4. **格式以后要改**——存了半年的会话，加一个字段就读不了了？

```python
class ToolRegistry:
    def __init__(self, tools=None): self._tools = {t.name: t for t in (tools or [])}
    def names(self): return list(self._tools)
    def specs(self): return [t.spec() for t in self._tools.values()]

        tool = self._tools.get(name)
        if tool is None:
            return ToolResult(f"没有叫 {name} 的工具", is_error=True)
        try:
            return ToolResult(str(tool.run(**args)))
        except Exception as exc:
            return ToolResult(f"{type(exc).__name__}: {exc}", is_error=True)

def to_openai_messages(messages):
    out = []
    for m in messages:
        if m.role == "tool":
            for b in m.content:
                if isinstance(b, ToolResultBlock):
                    out.append({"role": "tool", "tool_call_id": b.tool_use_id, "content": b.content})
            continue
        text, calls = m.text(), m.tool_uses()
        if m.role == "assistant" and calls:
            out.append({"role": "assistant", "content": text or None, "tool_calls": [
                {"id": c.id, "type": "function",
                 "function": {"name": c.name, "arguments": json.dumps(c.args, ensure_ascii=False)}}
                for c in calls]})
        else:
            out.append({"role": m.role, "content": text})
    return out

def to_openai_tools(tools):
    return [{"type": "function", "function": {"name": t["name"], "description": t.get("description", ""),
             "parameters": t.get("parameters", {"type": "object", "properties": {}})}} for t in tools]

class ToolCallAccumulator:
    def __init__(self): self._slots = {}
    def feed(self, raw_calls):
        for item in raw_calls:
            s = self._slots.setdefault(int(item.get("index", 0)), {"id": "", "name": "", "arguments": ""})
            if item.get("id"): s["id"] = item["id"]
            fn = item.get("function") or {}
            if fn.get("name"): s["name"] = fn["name"]
            if fn.get("arguments"): s["arguments"] += fn["arguments"]
    def finish(self):
        out = []
        for i in sorted(self._slots):
            s = self._slots[i]
            try: args = json.loads(s["arguments"] or "{}")
            except json.JSONDecodeError: args = {"_invalid_json": s["arguments"]}
            out.append(ToolUseBlock(id=s["id"] or new_id("call"), name=s["name"],
                                    args=args if isinstance(args, dict) else {}))
        return out

def parse_sse_line(line):
    line = line.strip()
    if not line or line.startswith(":") or not line.startswith("data:"): return None
    payload = line[5:].strip()
    if payload == "[DONE]": return None
    try: return json.loads(payload)
    except json.JSONDecodeError: return None

class OpenAICompatModel:
    def __init__(self, base_url, model, timeout=180):
        self.base_url, self.model, self.timeout = base_url.rstrip("/"), model, timeout
    def stream(self, messages, tools):
        body = {"model": self.model, "messages": to_openai_messages(messages), "stream": True}
        if tools: body["tools"] = to_openai_tools(tools)
        req = urllib.request.Request(f"{self.base_url}/chat/completions",
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"}, method="POST")
        acc = ToolCallAccumulator()
        with urllib.request.urlopen(req, timeout=self.timeout) as resp:
            for raw in resp:
                chunk = parse_sse_line(raw.decode("utf-8", errors="replace"))
                if chunk is None: continue
                d = ((chunk.get("choices") or [{}])[0]).get("delta") or {}
                if d.get("content"): yield Delta(kind="text", text=d["content"])
                if d.get("tool_calls"): acc.feed(d["tool_calls"])
        for b in acc.finish(): yield Delta(kind="tool_use", tool=b)

class StopReason(str, Enum):
    COMPLETED = "completed"; BUDGET_EXHAUSTED = "budget"; ERROR = "error"

@dataclass
class TurnResult:
    text: str
    stop_reason: StopReason
    rounds_used: int
    messages: list[Message] = field(default_factory=list)
    error: str = ""

@dataclass
class AgentConfig:
    system_prompt: str = "你是一个简洁的助手。需要用工具时就调用。"
    max_rounds: int = 6

class Agent:
    def __init__(self, model, registry, config=None):
        self.model, self.registry = model, registry
        self.config = config or AgentConfig()
    def _prepare(self, session):
        return [Message(role="system", content=[TextBlock(text=self.config.system_prompt)]),
                *session.messages]
    def send(self, session: Session, user_text: str) -> TurnResult:
        session.append(Message(role="user", content=[TextBlock(text=user_text)]))
        result = self._loop(session)
        session.append(*result.messages)
        return result
    def _loop(self, session):
        new_messages = []
        for round_no in range(1, self.config.max_rounds + 1):
            try:
                reply = self._one_round(self._prepare(session) + new_messages)
            except Exception as exc:
                return TurnResult("", StopReason.ERROR, round_no, new_messages, error=str(exc))
            new_messages.append(reply)
            calls = reply.tool_uses()
            if not calls:
                return TurnResult(reply.text(), StopReason.COMPLETED, round_no, new_messages)
            results = [ToolResultBlock(tool_use_id=c.id,
                                       content=(r := self.registry.execute(c.name, c.args)).content,
                                       is_error=r.is_error) for c in calls]
            new_messages.append(Message(role="tool", content=results))
        return TurnResult("", StopReason.BUDGET_EXHAUSTED, self.config.max_rounds, new_messages)
    def _one_round(self, messages):
        text_parts, blocks = [], []
        for d in self.model.stream(messages, self.registry.specs()):
            if d.kind == "text":
                text_parts.append(d.text)
            elif d.tool is not None:
                if text_parts:
                    blocks.append(TextBlock(text="".join(text_parts))); text_parts = []
                blocks.append(d.tool)
        if text_parts: blocks.append(TextBlock(text="".join(text_parts)))
        return Message(role="assistant", content=blocks)

    return f"{city} 现在是 {datetime.datetime.now().strftime('%H:%M')}"
```

## 第一个坑：怎么写

两种写法：

| 写法 | 崩溃时丢什么 | 一次对话的写入量 |
|---|---|---|
| 每次把整个会话覆写一遍 | **整个会话**（写到一半停电，文件是半个 JSON） | 全部消息 |
| 只追加新增的那几条 | 最多最后一条 | 新增的几条 |

选追加。文件格式是 **JSONL**——每行一个独立的 JSON 对象。
第一行是会话本身的信息，后面每行一条消息。

这个选择还有个附带好处：**追加不需要先读**。一段有两千条消息的会话，
每加一句话都要先把两千条读进来再全写回去，是很荒唐的。

## 第二个坑：读回来怎么知道它是什么

存进去的是 `{"type": "text", "text": "你好"}`，读回来要变成 `TextBlock`。
靠什么判断？**靠 `type` 这个判别字段**，而不是靠「有 text 字段就是文本块」这种猜测——
猜测在字段重名时会出错，而且新增块类型时会静默走错分支。

```python
SCHEMA_VERSION = 1

# 判别字段 → 类型。新增块类型时在这里加一行，别的地方都不用改。
_BLOCK_TYPES = {
    "text": TextBlock,
    "tool_use": ToolUseBlock,
    "tool_result": ToolResultBlock,
}


def block_to_dict(block: Any) -> dict:
    if isinstance(block, TextBlock):
        return {"type": "text", "text": block.text}
    if isinstance(block, ToolUseBlock):
        return {"type": "tool_use", "id": block.id, "name": block.name, "args": block.args}
    if isinstance(block, ToolResultBlock):
        return {"type": "tool_result", "tool_use_id": block.tool_use_id,
                "content": block.content, "is_error": block.is_error}
    raise TypeError(f"不知道怎么存这种块：{type(block).__name__}")


class UnknownBlock(Exception):
    """读到一个本版本不认识的块类型。"""


def block_from_dict(raw: dict) -> Any:
    kind = raw.get("type", "")
    cls = _BLOCK_TYPES.get(kind)
    if cls is None:
        # 这不是「格式坏了」，而是「这条记录来自一个更新的版本」。
        # 两者的处置完全不同，所以要能区分开。
        raise UnknownBlock(kind or "（没有 type 字段）")
    fields = {k: v for k, v in raw.items() if k != "type"}
    return cls(**fields)


def message_to_dict(message: Message) -> dict:
    return {"id": message.id, "role": message.role, "created_at": message.created_at,
            "content": [block_to_dict(b) for b in message.content]}


def message_from_dict(raw: dict) -> Message:
    return Message(id=raw["id"], role=raw["role"], created_at=raw.get("created_at", 0.0),
                   content=[block_from_dict(b) for b in raw.get("content", [])])


# 自测一遍：存进去再读回来，应该一模一样
original = Message(role="assistant", content=[
    TextBlock(text="我查一下。"),
    ToolUseBlock(name="get_time", args={"city": "北京"}),
])
restored = message_from_dict(json.loads(json.dumps(message_to_dict(original))))
print("往返一致：", message_to_dict(original) == message_to_dict(restored))
print("块类型都对：", [type(b).__name__ for b in restored.content])
```

## 第三个坑：写到一半崩了

进程在写第 N 条消息时被杀，文件尾就是半行。下次读到那一行，`json.loads` 会抛异常。

处理办法是**读的时候容忍，读完把文件截断**。截断这一步不能省：
不截断的话，下次追加会接在半行后面，产生一行「半条 + 一整条」的垃圾，那才是真的读不回来了。

```python
class SessionStore:
    """一个目录一堆会话，一个会话一个 JSONL 文件。"""

    def __init__(self, root: Path) -> None:
        self.root = Path(root)
        self.root.mkdir(parents=True, exist_ok=True)

    def _path(self, session_id: str) -> Path:
        # 会话标识直接当文件名，所以必须挡住路径穿越。
        # 我们自己生成的 id 是安全的，但这个函数也会拿到外面传进来的值。
        if "/" in session_id or "\\" in session_id or session_id.startswith("."):
            raise ValueError(f"不合法的会话标识：{session_id!r}")
        return self.root / f"{session_id}.jsonl"

    # ── 写 ──────────────────────────────────────────────────────────────

    def create(self, session: Session) -> None:
        """写下会话头。只在会话第一次落盘时调用。"""
        path = self._path(session.id)
        header = {"kind": "session", "schema": SCHEMA_VERSION, "id": session.id,
                  "title": session.title, "created_at": session.created_at,
                  "metadata": session.metadata}
        with path.open("w", encoding="utf-8") as f:
            f.write(json.dumps(header, ensure_ascii=False) + "\n")

    def append(self, session_id: str, *messages: Message) -> None:
        """追加几条消息。这是最常走的路径，所以它必须快。"""
        if not messages:
            return
        path = self._path(session_id)
        with path.open("a", encoding="utf-8") as f:
            for m in messages:
                f.write(json.dumps({"kind": "message", **message_to_dict(m)},
                                   ensure_ascii=False) + "\n")
            f.flush()
            # flush 只是把数据交给操作系统，还在内存缓冲里。
            # fsync 才是真的落到盘上。代价是每次几毫秒——对话场景值得，
            # 高频写日志的场景就不值得了。
            os.fsync(f.fileno())

    # ── 读 ──────────────────────────────────────────────────────────────

    def load(self, session_id: str) -> tuple[Session, list[str]]:
        """读回一个会话。第二个返回值是读的过程中发现的问题，不是异常。

        为什么不抛异常：一条坏记录不该让整段对话打不开。
        用户宁可看到「少了最后一句」，也不想看到「这个会话读不了」。
        """
        path = self._path(session_id)
        problems: list[str] = []
        session: Session | None = None
        messages: list[Message] = []
        good_bytes = 0    # 最后一条完整记录的结束位置

        with path.open("r", encoding="utf-8") as f:
            for line_no, line in enumerate(f, 1):
                raw_len = len(line.encode("utf-8"))
                stripped = line.strip()
                if not stripped:
                    good_bytes += raw_len
                    continue
                try:
                    record = json.loads(stripped)
                except json.JSONDecodeError:
                    # 只有最后一行的半行才是崩溃留下的；中间出现坏行说明是别的问题。
                    problems.append(f"第 {line_no} 行不是完整的 JSON，已跳过")
                    continue
                try:
                    if record.get("kind") == "session":
                        session = Session(id=record["id"], title=record.get("title", ""),
                                          created_at=record.get("created_at", 0.0),
                                          metadata=record.get("metadata") or {})
                    elif record.get("kind") == "message":
                        messages.append(message_from_dict(record))
                    else:
                        problems.append(f"第 {line_no} 行是不认识的记录类型，已跳过")
                        continue
                except UnknownBlock as exc:
                    problems.append(f"第 {line_no} 行含有本版本不认识的块「{exc}」，已跳过")
                    continue
                except (KeyError, TypeError) as exc:
                    problems.append(f"第 {line_no} 行字段不全（{exc}），已跳过")
                    continue
                good_bytes += raw_len

        if session is None:
            raise ValueError(f"{path.name} 里没有会话头，这个文件不是会话文件")

        # 把尾巴上的垃圾截掉，否则下次追加会接在半行后面。
        if good_bytes < path.stat().st_size:
            with path.open("r+", encoding="utf-8") as f:
                f.truncate(good_bytes)
            problems.append(f"文件尾有 {path.stat().st_size - good_bytes} 字节残留，已截断")

        session.messages = messages
        return session, problems

    # ── 列表 ────────────────────────────────────────────────────────────

    def list_sessions(self) -> list[dict]:
        """只读每个文件的第一行，不把整段对话读进来。"""
        out = []
        for path in sorted(self.root.glob("*.jsonl")):
            try:
                with path.open("r", encoding="utf-8") as f:
                    header = json.loads(f.readline())
                if header.get("kind") != "session":
                    continue
                out.append({"id": header["id"], "title": header.get("title", ""),
                            "created_at": header.get("created_at", 0.0),
                            "size": path.stat().st_size})
            except (OSError, json.JSONDecodeError):
                continue
        return sorted(out, key=lambda s: s["created_at"], reverse=True)


print("会话存储就绪")
```

## 装到 Agent 上

```python
class PersistentAgent(Agent):
    """每说完一轮就把新增的消息追加到磁盘。

    继承而不是改 Agent：存储是可选的能力，不该让不需要它的人也背上。
    """

    def __init__(self, model, registry, store: SessionStore, config=None):
        super().__init__(model, registry, config)
        self.store = store
        self._known: set[str] = set()

    def send(self, session: Session, user_text: str) -> TurnResult:
        if session.id not in self._known:
            if not self.store._path(session.id).exists():
                self.store.create(session)
            self._known.add(session.id)

        before = len(session.messages)
        result = super().send(session, user_text)
        # 只写新增的那几条。这里能这么算，是因为 Agent.send 只往后追加、不改动旧消息。
        self.store.append(session.id, *session.messages[before:])
        return result


store = SessionStore(STORE_DIR)
agent = PersistentAgent(real_model, registry, store,
                        AgentConfig(system_prompt="你是一个简洁的助手，回答控制在一句话。"))

session = Session(title="持久化演示")
for q in ["记住：我最喜欢的数字是 17。", "北京现在几点？"]:
    print(f"用户：{q}")
    print(f"助手：{agent.send(session, q).text}\n")

print("文件内容（每行一条记录）：")
for i, line in enumerate((STORE_DIR / f"{session.id}.jsonl").read_text(encoding="utf-8").splitlines(), 1):
    record = json.loads(line)
    label = record.get("kind")
    detail = record.get("title") if label == "session" else f"{record['role']}：{str(record['content'])[:50]}"
    print(f"  {i}. [{label}] {detail}")
```

```python
    print(f"  ⚠ {p}")
```

**注意两件事**：一是坏行没有让整段对话打不开；二是读完就把尾巴截了，
所以第二次读干干净净。少了截断这一步，下次追加会接在半行后面，越坏越深。

## 第四个坑：格式以后要改

半年后你想给 `Message` 加个字段，或者加一种新的块类型。存量的文件怎么办？

```python
# 场景一：读到一个更新版本写的块类型
future = {"kind": "message", "id": "msg_future", "role": "assistant", "created_at": 0,
          "content": [{"type": "text", "text": "正常部分"},
                      {"type": "thinking", "text": "这是新版本才有的块"}]}
with path.open("a", encoding="utf-8") as f:
    f.write(json.dumps(future, ensure_ascii=False) + "\n")

session_v, problems = fresh_store.load(restored.id)
print("读到未来版本的块：")
for p in problems:
    print(f"  ⚠ {p}")
print(f"  消息数：{len(session_v.messages)}（那条整体跳过了）")
```

上面这个处置是「整条跳过」。够不够好取决于场景：

- **对话记录**：整条跳过是对的。一条消息缺了一半，拿给模型看会误导它。
- **审计日志**：应该原样保留那个不认识的块，只是不解释它——将来升级了还能读懂。

不管选哪种，**前提都是文件里存了 `schema` 版本号**。没有版本号，你连「这是新格式」
都判断不出来，只能当成损坏。会话头第一行那个 `"schema": 1` 就是为这个留的。

```python
with (STORE_DIR / f"{restored.id}.jsonl").open() as f:
    header = json.loads(f.readline())
print(f"文件声明的格式版本：{header['schema']}，本程序支持：{SCHEMA_VERSION}")
print("→ 版本一致，可以直接读" if header["schema"] == SCHEMA_VERSION
      else "→ 版本不同，该走迁移流程")
```

## 本章小结

- **追加式日志，不要全量覆写**。覆写崩溃时丢的是整个会话，追加最多丢最后一条；
  而且追加不需要先把全部历史读进来。
- 格式选 **JSONL**：一行一条记录，坏一行不影响其他行。
- 反序列化靠**判别字段**（`type`），不靠猜。新增块类型时只改一张表。
- **`flush` 不等于落盘**，`os.fsync` 才是。对话场景值得付这几毫秒。
- 一条坏记录**不该让整段对话打不开**。读的时候容忍，把问题作为返回值报告，
  而不是抛异常。
- 读完**要把尾巴上的残留截掉**，否则下次追加会接在半行后面，越坏越深。
- 会话标识会当文件名用，**必须挡住路径穿越**。
- 文件里要存**格式版本号**，否则「新格式」和「文件损坏」你分不开。
