# 脚本化模型

第 1 章那个假模型是写死的。这一章把它做成**能按剧本演的假模型**：给它一份剧本，
它就照着吐。有了这个东西，整条链路可以在完全不联网的情况下跑，而且**每次输出都一模
一样**——这是写测试的前提，也是这门课前八章都不需要联网的原因。

pi 里这个东西叫 `ScriptedModel`，几乎所有 agent 框架都有对应物。

## 先定接口

在写假模型之前，得先定义**真模型和假模型共同的接口**。这个接口定得好不好，决定了
第 8 章换成真模型时要改多少代码——理想情况是一行不改。

一次模型调用的输入输出：

- **输入**：消息历史、可用的工具声明
- **输出**：一串增量（文字一段段来、工具调用一个个来）

Go 里表达「一串陆续产生的东西」最自然的还是通道。

```go
// file: blocks.go
package main

import (
	"crypto/rand"
	"encoding/hex"
)

func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)
}

type Block interface{ BlockType() string }

type TextBlock struct{ Text string }

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string
	Name string
	Args map[string]any
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string
	Content   string
	IsError   bool
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

type Message struct {
	ID      string
	Role    string
	Content []Block
}

func (m Message) Text() string {
	out := ""
	for _, b := range m.Content {
		if t, ok := b.(TextBlock); ok {
			out += t.Text
		}
	}
	return out
}

func (m Message) ToolUses() []ToolUseBlock {
	var calls []ToolUseBlock
	for _, b := range m.Content {
		if c, ok := b.(ToolUseBlock); ok {
			calls = append(calls, c)
		}
	}
	return calls
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks}
}

func NewToolUse(name string, args map[string]any) ToolUseBlock {
	return ToolUseBlock{ID: newID("call"), Name: name, Args: args}
}
```

```go
// file: model.go
package main

// Delta 模型吐出来的一小段。真模型的流式响应最终都会被翻译成这个。
type Delta struct {
	Kind string // "text" 或 "tool_use"
	Text string
	Tool *ToolUseBlock
}

// Model 所有模型都长这样。假模型和真模型的唯一区别是这个方法怎么实现。
//
// 返回两个通道：一个吐增量，一个报错。分开是因为 Go 没有「通道里带错误」的惯例，
// 而把错误塞进 Delta 会让每个消费者都得判断一次。
type Model interface {
	Stream(messages []Message, tools []map[string]any) (<-chan Delta, <-chan error)
}

// Collect 把一串增量攒成一条完整的助手消息。真模型那边也用这个函数。
func Collect(deltas <-chan Delta, errs <-chan error) (Message, error) {
	var textParts []string
	var blocks []Block

	flushText := func() {
		if len(textParts) > 0 {
			joined := ""
			for _, p := range textParts {
				joined += p
			}
			blocks = append(blocks, TextBlock{Text: joined})
			textParts = nil
		}
	}

	for d := range deltas {
		switch d.Kind {
		case "text":
			textParts = append(textParts, d.Text)
		case "tool_use":
			if d.Tool != nil {
				flushText() // 文字块要排在它后面的工具调用之前
				blocks = append(blocks, *d.Tool)
			}
		}
	}
	flushText()

	// 增量通道关了之后再看有没有错误
	if err := <-errs; err != nil {
		return Message{}, err
	}
	return NewMessage("assistant", blocks...), nil
}
```

## 剧本怎么写

```go
// file: scripted.go
package main

import "fmt"

// ScriptedCall 剧本里的一次工具调用。
type ScriptedCall struct {
	Name string
	Args map[string]any
}

// ScriptedTurn 剧本里的一幕：模型这一次要说什么、要调什么工具。
type ScriptedTurn struct {
	Text  string
	Calls []ScriptedCall
}

// ScriptedModel 按剧本演的假模型。
//
// 它完全不看输入的消息历史——这是有意的。假模型的价值在于**输出可预测**，
// 一旦让它根据历史做判断，它自己就变成一个需要单独测试的东西了。
// 需要按历史分叉的场景，写两份剧本分别跑。
type ScriptedModel struct {
	Script    []ScriptedTurn
	ChunkSize int // 文字每次吐几个字，模拟真实分块
	calls     int // 演到第几幕了
}

func NewScriptedModel(script []ScriptedTurn) *ScriptedModel {
	return &ScriptedModel{Script: script, ChunkSize: 4}
}

func (m *ScriptedModel) Stream(messages []Message, tools []map[string]any) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta, 8)
	errs := make(chan error, 1)

	if m.calls >= len(m.Script) {
		close(deltas)
		errs <- fmt.Errorf(
			"剧本只有 %d 幕，但模型被调用了第 %d 次。多半是主循环没有正常结束——去看看退出条件",
			len(m.Script), m.calls+1)
		close(errs)
		return deltas, errs
	}

	turn := m.Script[m.calls]
	m.calls++

	go func() {
		defer close(deltas)
		defer close(errs)

		runes := []rune(turn.Text)
		for i := 0; i < len(runes); i += m.ChunkSize {
			end := i + m.ChunkSize
			if end > len(runes) {
				end = len(runes)
			}
			deltas <- Delta{Kind: "text", Text: string(runes[i:end])}
		}
		for _, c := range turn.Calls {
			call := NewToolUse(c.Name, c.Args)
			deltas <- Delta{Kind: "tool_use", Tool: &call}
		}
		errs <- nil // 明确表示这一趟没出错
	}()

	return deltas, errs
}
```

## 演一遍

```go
// file: main.go
package main

import "fmt"

var script = []ScriptedTurn{
	{Text: "我查一下时间。", Calls: []ScriptedCall{{Name: "get_time", Args: map[string]any{"city": "北京"}}}},
	{Text: "北京现在是下午 2 点 32 分。"},
}

func main() {
	model := NewScriptedModel(script)

	fmt.Println("── 第一次调用（模型要求调工具）──")
	deltas, errs := model.Stream(nil, nil)
	for d := range deltas {
		if d.Kind == "text" {
			fmt.Printf("  文字增量：%q\n", d.Text)
		} else {
			fmt.Printf("  工具调用：%s(%v)  编号 %s\n", d.Tool.Name, d.Tool.Args, d.Tool.ID)
		}
	}
	<-errs

	fmt.Println("\n── 第二次调用（模型给最终回答）──")
	msg, err := Collect(model.Stream(nil, nil))
	if err != nil {
		fmt.Println("出错：", err)
		return
	}
	fmt.Printf("  收集成的消息有 %d 个块\n", len(msg.Content))
	fmt.Println("  文字：", msg.Text())
}
```

## 剧本用完了会明确报错

新手最常见的 bug 是主循环不退出——模型明明已经给了最终回答，循环还在继续问。
脚本化模型让这种 bug **立刻暴露**：

```go
// file: exhausted.go
package main

import "fmt"

func demoExhausted() {
	m := NewScriptedModel([]ScriptedTurn{{Text: "只有一幕"}})

	if _, err := Collect(m.Stream(nil, nil)); err != nil { // 第一次，正常
		fmt.Println("第一次不该出错：", err)
		return
	}
	if _, err := Collect(m.Stream(nil, nil)); err != nil { // 第二次，剧本没了
		fmt.Println("报错了，而且说清了原因：")
		fmt.Println(" ", err)
	}
}
```

换成真模型，这里只会多花几秒钱、多一条无意义的回复，你可能根本发现不了。

## 第 1 章那个流程，现在用真接口跑一遍

```go
// file: loop.go
package main

import "fmt"

func getTime(args map[string]any) string { return "14:32" }

var tools = map[string]func(map[string]any) string{"get_time": getTime}

func run(userText string, model Model, maxRounds int) (string, error) {
	history := []Message{NewMessage("user", TextBlock{Text: userText})}
	fmt.Printf("用户：%s\n\n", userText)

	for round := 1; round <= maxRounds; round++ {
		reply, err := Collect(model.Stream(history, nil))
		if err != nil {
			return "", fmt.Errorf("第 %d 轮模型出错：%w", round, err)
		}
		history = append(history, reply)

		if t := reply.Text(); t != "" {
			fmt.Printf("模型：%s\n", t)
		}

		calls := reply.ToolUses()
		if len(calls) == 0 { // 没有工具调用了，这一轮结束
			return reply.Text(), nil
		}

		var results []Block
		for _, call := range calls {
			fn, ok := tools[call.Name]
			if !ok {
				return "", fmt.Errorf("模型要调一个不存在的工具：%s", call.Name)
			}
			output := fn(call.Args)
			fmt.Printf("工具：%s → %s\n", call.Name, output)
			results = append(results, ToolResultBlock{ToolUseID: call.ID, Content: output})
		}
		history = append(history, NewMessage("tool", results...))
	}
	return "", fmt.Errorf("循环了 %d 轮还没结束", maxRounds)
}
```

把 `main.go` 改成这样，跑完整流程：

```go
// file: main.go
package main

import "fmt"

var script = []ScriptedTurn{
	{Text: "我查一下时间。", Calls: []ScriptedCall{{Name: "get_time", Args: map[string]any{"city": "北京"}}}},
	{Text: "北京现在是下午 2 点 32 分。"},
}

func main() {
	answer, err := run("北京现在几点？", NewScriptedModel(script), 5)
	if err != nil {
		fmt.Println("出错：", err)
		return
	}
	fmt.Println("\n最终答案：", answer)

	fmt.Println()
	demoExhausted()
}
```

对比第 1 章那段桩代码：**流程完全一样，但每个部件都换成了正经的类型**。
第 8 章我们把 `ScriptedModel` 换成真模型时，`run` 这个函数一个字都不用改——
它只依赖 `Model` 这个接口。

## 本章小结

- 先定**模型接口**，再实现假模型和真模型，这样换实现时上层不用改。
- 接口的输出是**增量流**（文字一段段、工具调用一个个），因为真模型就是这么吐的。
- Go 里用两个通道：一个吐增量，一个报错。**错误不塞进数据通道**。
- 脚本化模型的价值是**确定性**：能写断言、能复现、不花钱、不联网。
- **剧本用完要明确报错**——它能立刻暴露主循环不退出这类 bug。

第一部分到这里结束。我们手上有了：结构化的消息、事件流、能落盘的历史、可预测的假模型。
下一部分接上真模型——而且前面这些代码一行都不会改。
