# 消息存储

历史现在还在内存里的一个切片中，程序一退就没了。这一章把它存到磁盘上。

看起来是个小问题，但存储格式的选择会影响后面好几章：**会话能不能续上**（第 14 章）、
**能不能从中间分叉**（第 15 章）、**上下文压缩后原始记录还在不在**（第 16 章）。

## 为什么是 JSONL 而不是一个大 JSON

最直觉的做法是把整个历史序列化成一个 JSON 数组写进文件。别这么做：

| 做法 | 追加一条消息 | 进程被杀 | 读取 |
|---|---|---|---|
| 一个大 JSON 数组 | 要读出整个文件、改、再整个写回 | 写到一半 → **整个文件废了** | 必须整个读进内存 |
| 一行一条（JSONL） | 直接往文件尾追加一行 | 写到一半 → **只有最后一行不完整** | 可以边读边处理 |

JSONL 就是「每行一个 JSON 对象」。pi 用的是这个。**追加只写不改**这个性质，
是它比数据库和大 JSON 都更适合会话历史的原因——历史本来就是只增不改的。

## 本章起点

```go
// file: blocks.go
package main

import (
	"crypto/rand"
	"encoding/hex"
	"encoding/json"
	"fmt"
)

func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)
}

type Block interface{ BlockType() string }

type TextBlock struct {
	Text string `json:"text"`
}

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string         `json:"id"`
	Name string         `json:"name"`
	Args map[string]any `json:"args"`
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string `json:"tool_use_id"`
	Content   string `json:"content"`
	IsError   bool   `json:"is_error"`
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

type Message struct {
	ID      string
	Role    string
	Content []Block
}

func (m Message) Text() string {
	out := ""
	for _, b := range m.Content {
		if t, ok := b.(TextBlock); ok {
			out += t.Text
		}
	}
	return out
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks}
}

func NewToolUse(name string, args map[string]any) ToolUseBlock {
	return ToolUseBlock{ID: newID("call"), Name: name, Args: args}
}

// ── 上一章的信封式序列化 ──

type blockEnvelope struct {
	Type string          `json:"type"`
	Data json.RawMessage `json:"data"`
}

type messageEnvelope struct {
	ID      string          `json:"id"`
	Role    string          `json:"role"`
	Content []blockEnvelope `json:"content"`
}

func encodeMessage(m Message) ([]byte, error) {
	env := messageEnvelope{ID: m.ID, Role: m.Role}
	for _, b := range m.Content {
		data, err := json.Marshal(b)
		if err != nil {
			return nil, fmt.Errorf("块 %s 序列化失败：%w", b.BlockType(), err)
		}
		env.Content = append(env.Content, blockEnvelope{Type: b.BlockType(), Data: data})
	}
	return json.Marshal(env)
}

func decodeMessage(raw []byte) (Message, error) {
	var env messageEnvelope
	if err := json.Unmarshal(raw, &env); err != nil {
		return Message{}, fmt.Errorf("消息读不出来：%w", err)
	}
	m := Message{ID: env.ID, Role: env.Role}
	for _, item := range env.Content {
		var b Block
		switch item.Type {
		case "text":
			var v TextBlock
			if err := json.Unmarshal(item.Data, &v); err != nil {
				return Message{}, err
			}
			b = v
		case "tool_use":
			var v ToolUseBlock
			if err := json.Unmarshal(item.Data, &v); err != nil {
				return Message{}, err
			}
			b = v
		case "tool_result":
			var v ToolResultBlock
			if err := json.Unmarshal(item.Data, &v); err != nil {
				return Message{}, err
			}
			b = v
		default:
			continue // 不认识的块类型跳过，别让整条消息读不出来
		}
		m.Content = append(m.Content, b)
	}
	return m, nil
}
```

## 存储实现

```go
// file: store.go
package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
)

// JsonlStore 一行一条消息的会话存储。
//
// 只提供两个动作：追加、读全部。刻意不提供"改某一条"——历史是只增不改的，
// 需要修正时应当追加一条新的。这条约束让崩溃恢复变得简单：最坏情况是最后一行
// 写了一半，丢掉它就行。
type JsonlStore struct {
	path string
}

func NewJsonlStore(path string) (*JsonlStore, error) {
	if err := os.MkdirAll(filepath.Dir(path), 0o755); err != nil {
		return nil, fmt.Errorf("建目录失败：%w", err)
	}
	return &JsonlStore{path: path}, nil
}

func (s *JsonlStore) Append(m Message) error {
	raw, err := encodeMessage(m)
	if err != nil {
		return err
	}
	f, err := os.OpenFile(s.path, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o644)
	if err != nil {
		return fmt.Errorf("打开存储文件失败：%w", err)
	}
	defer f.Close()

	if _, err := f.Write(append(raw, '\n')); err != nil {
		return fmt.Errorf("写入失败：%w", err)
	}
	// 真落盘。不调 Sync 的话，进程被强杀时数据还在操作系统的缓冲区里，
	// 用户会看到"明明回答过，重启就没了"。
	return f.Sync()
}

func (s *JsonlStore) Load() ([]Message, int, error) {
	f, err := os.Open(s.path)
	if os.IsNotExist(err) {
		return nil, 0, nil // 还没存过，不是错误
	}
	if err != nil {
		return nil, 0, fmt.Errorf("打开存储文件失败：%w", err)
	}
	defer f.Close()

	var messages []Message
	broken := 0
	scanner := bufio.NewScanner(f)
	scanner.Buffer(make([]byte, 0, 64*1024), 8*1024*1024) // 单条消息可能很长

	for scanner.Scan() {
		line := scanner.Bytes()
		if len(line) == 0 {
			continue
		}
		m, err := decodeMessage(line)
		if err != nil {
			broken++ // 只可能是最后一行写了一半，跳过它
			continue
		}
		messages = append(messages, m)
	}
	if err := scanner.Err(); err != nil {
		return messages, broken, fmt.Errorf("读取中断：%w", err)
	}
	return messages, broken, nil
}

var _ = json.Marshal
```

## 存一轮对话再读回来

```go
// file: main.go
package main

import (
	"fmt"
	"os"
)

func main() {
	path := "/tmp/agent-course-go/session-demo.jsonl"
	os.RemoveAll("/tmp/agent-course-go") // 每次运行先清空，免得反复运行越积越多

	store, err := NewJsonlStore(path)
	if err != nil {
		fmt.Println("建存储失败：", err)
		return
	}

	call := NewToolUse("get_time", map[string]any{"city": "北京"})
	history := []Message{
		NewMessage("user", TextBlock{Text: "北京现在几点？"}),
		NewMessage("assistant", TextBlock{Text: "我查一下。"}, call),
		NewMessage("tool", ToolResultBlock{ToolUseID: call.ID, Content: "14:32"}),
		NewMessage("assistant", TextBlock{Text: "北京现在是 14:32。"}),
	}
	for _, m := range history {
		if err := store.Append(m); err != nil {
			fmt.Println("写入失败：", err)
			return
		}
	}

	raw, _ := os.ReadFile(path)
	fmt.Println("文件原文：")
	fmt.Println(string(raw))

	loaded, broken, err := store.Load()
	if err != nil {
		fmt.Println("读取出错：", err)
		return
	}
	fmt.Printf("读回来 %d 条（跳过 %d 条读不出来的）：\n", len(loaded), broken)
	for _, m := range loaded {
		text := m.Text()
		if text == "" {
			text = "（无文字）"
		}
		fmt.Printf("  %-10s %s\n", m.Role, text)
	}
}
```

注意看文件原文——**每一行都是一个完整的 JSON**，用文本工具就能查看和统计。

## 崩溃恢复：故意写坏最后一行

```go
// file: crash.go
package main

import (
	"fmt"
	"os"
)

func simulateCrash(path string) {
	// 模拟进程在写第五条消息时被杀：文件尾多了半行
	f, err := os.OpenFile(path, os.O_APPEND|os.O_WRONLY, 0o644)
	if err != nil {
		fmt.Println("打不开文件：", err)
		return
	}
	f.WriteString(`{"id":"msg_broken","role":"assist`) // 故意不写完，也不换行
	f.Close()

	store, _ := NewJsonlStore(path)
	loaded, broken, _ := store.Load()
	fmt.Printf("\n写坏最后一行之后：仍然读出 %d 条，跳过 %d 条。\n", len(loaded), broken)
	fmt.Println("前面的记录一条没丢：")
	for _, m := range loaded {
		text := m.Text()
		if text == "" {
			text = "（无文字）"
		}
		fmt.Printf("  %-10s %s\n", m.Role, text)
	}
}
```

在 `main` 末尾加一行 `simulateCrash(path)` 再运行，就能看到效果。这就是 JSONL 的关键
性质：**坏也只坏最后一行**。换成一个大 JSON 数组，这个文件现在已经完全读不出来了。

## 关于 Sync 的代价

`Append` 里调了 `f.Sync()`。不调的话写入快得多，但进程被强杀时最后几条会丢。

对会话历史来说，「用户明明看到助手回答了，重启后却不见了」是很糟糕的体验，值得为此
付出每次写入的一点开销。真实系统里会做批量刷盘来平衡，但**默认应该是安全的那一边**。

## 本章小结

- 会话历史用 **JSONL**：一行一条，只追加不修改。
- 只追加换来两件事：**写入快**，**坏只坏最后一行**。
- 读取时要**容忍读不出来的行**，跳过而不是让整个会话失败。
- `Write` 之后要 `Sync`，否则进程被杀时数据还在缓冲区里。
- Go 的 `bufio.Scanner` 默认单行上限 64 KB，**消息可能超过它**，要手工调大。
