# 技能

工具解决的是「agent **能做什么**」。这一章解决另一件事：「agent **怎么做**」。

举个具体的：你希望它写的 git 提交信息符合团队规范——首行不超过 50 字、用祈使句、
正文说清楚为什么而不是做了什么、破坏性变更要加脚注。这些话写哪儿？

写进系统提示是最直接的想法。但你还希望它会写变更日志、会排查慢查询、
会按公司模板出周报……**系统提示会膨胀到几千 token，而其中 95% 与当前这句话无关。**

膨胀的代价有三层，最后一层最要命：

1. 每一轮都要付这些 token 的钱。
2. 上下文被占掉，能装的对话历史变少。
3. **注意力被稀释**——真正重要的那三条规矩，淹没在两千字里。

办法是把「怎么做」搬到系统提示外面，做成**技能**：平时只留一行索引，
用得上的时候才把全文读进来。

## 技能和工具的区别

| | 工具 | 技能 |
|---|---|---|
| 是什么 | 一段代码 | 一段文字 |
| 提供什么 | 新的能力（读文件、发请求） | 做某件事的方法 |
| 谁执行 | 我们的程序 | 模型自己 |
| 加不加得动 | 要改代码、重新编译 | 放一个文件进目录就行 |

一句话：**工具扩展能力，技能扩展方法**。Go 是编译型语言，第四行的差别比在 Python 里更大——
加一个工具要重新编译发版，加一个技能只要往目录里丢个文件。

```go
// file: core.go
package main

import (
	"bufio"
	"bytes"
	"context"
	"crypto/rand"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"net/http"
	"sort"
	"strings"
	"time"
	"unicode"
)

func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)
}

type Block interface{ BlockType() string }

type TextBlock struct{ Text string }

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string
	Name string
	Args map[string]any
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string
	Content   string
	IsError   bool
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

type Message struct {
	ID      string
	Role    string
	Content []Block
}

func (m Message) Text() string {
	var b strings.Builder
	for _, block := range m.Content {
		if t, ok := block.(TextBlock); ok {
			b.WriteString(t.Text)
		}
	}
	return b.String()
}

func (m Message) ToolUses() []ToolUseBlock {
	var calls []ToolUseBlock
	for _, b := range m.Content {
		if c, ok := b.(ToolUseBlock); ok {
			calls = append(calls, c)
		}
	}
	return calls
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks}
}

type ToolSpec struct {
	Name        string
	Description string
	Parameters  map[string]any
}

type Delta struct {
	Kind string
	Text string
	Tool *ToolUseBlock
}

type Model interface {
	Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error)
}

type wireFunction struct {
	Name      string `json:"name"`
	Arguments string `json:"arguments"`
}

type wireToolCall struct {
	Index    int          `json:"index,omitempty"`
	ID       string       `json:"id,omitempty"`
	Type     string       `json:"type,omitempty"`
	Function wireFunction `json:"function"`
}

type wireMessage struct {
	Role       string         `json:"role"`
	Content    *string        `json:"content"`
	ToolCallID string         `json:"tool_call_id,omitempty"`
	ToolCalls  []wireToolCall `json:"tool_calls,omitempty"`
}

type wireToolDef struct {
	Type     string `json:"type"`
	Function struct {
		Name        string         `json:"name"`
		Description string         `json:"description"`
		Parameters  map[string]any `json:"parameters"`
	} `json:"function"`
}

type wireRequest struct {
	Model    string        `json:"model"`
	Messages []wireMessage `json:"messages"`
	Tools    []wireToolDef `json:"tools,omitempty"`
	Stream   bool          `json:"stream,omitempty"`
}

type wireChoice struct {
	Delta wireMessage `json:"delta"`
}

type wireResponse struct {
	Choices []wireChoice `json:"choices"`
}

func toWireMessages(messages []Message) []wireMessage {
	var out []wireMessage
	for _, m := range messages {
		if m.Role == "tool" {
			for _, b := range m.Content {
				if r, ok := b.(ToolResultBlock); ok {
					content := r.Content
					out = append(out, wireMessage{Role: "tool", Content: &content, ToolCallID: r.ToolUseID})
				}
			}
			continue
		}
		text, calls := m.Text(), m.ToolUses()
		wm := wireMessage{Role: m.Role}
		if text != "" {
			wm.Content = &text
		}
		for _, c := range calls {
			args, err := json.Marshal(c.Args)
			if err != nil {
				args = []byte("{}")
			}
			wm.ToolCalls = append(wm.ToolCalls, wireToolCall{ID: c.ID, Type: "function",
				Function: wireFunction{Name: c.Name, Arguments: string(args)}})
		}
		out = append(out, wm)
	}
	return out
}

func toWireTools(specs []ToolSpec) []wireToolDef {
	var out []wireToolDef
	for _, s := range specs {
		var d wireToolDef
		d.Type = "function"
		d.Function.Name = s.Name
		d.Function.Description = s.Description
		if s.Parameters == nil {
			d.Function.Parameters = map[string]any{"type": "object", "properties": map[string]any{}}
		} else {
			d.Function.Parameters = s.Parameters
		}
		out = append(out, d)
	}
	return out
}

func parseArguments(raw string) map[string]any {
	if raw == "" {
		return map[string]any{}
	}
	var args map[string]any
	if err := json.Unmarshal([]byte(raw), &args); err != nil {
		return map[string]any{"_invalid_json": raw}
	}
	return args
}

type toolCallSlot struct{ id, name, arguments string }

type toolCallAccumulator struct{ slots map[int]*toolCallSlot }

func newToolCallAccumulator() *toolCallAccumulator {
	return &toolCallAccumulator{slots: map[int]*toolCallSlot{}}
}

func (a *toolCallAccumulator) Feed(calls []wireToolCall) {
	for _, item := range calls {
		slot, ok := a.slots[item.Index]
		if !ok {
			slot = &toolCallSlot{}
			a.slots[item.Index] = slot
		}
		if item.ID != "" {
			slot.id = item.ID
		}
		if item.Function.Name != "" {
			slot.name = item.Function.Name
		}
		if item.Function.Arguments != "" {
			slot.arguments += item.Function.Arguments
		}
	}
}

func (a *toolCallAccumulator) Finish() []ToolUseBlock {
	indexes := make([]int, 0, len(a.slots))
	for i := range a.slots {
		indexes = append(indexes, i)
	}
	sort.Ints(indexes)
	var blocks []ToolUseBlock
	for _, i := range indexes {
		slot := a.slots[i]
		id := slot.id
		if id == "" {
			id = newID("call")
		}
		blocks = append(blocks, ToolUseBlock{ID: id, Name: slot.name, Args: parseArguments(slot.arguments)})
	}
	return blocks
}

func parseSSELine(line string) (wireResponse, bool) {
	line = strings.TrimSpace(line)
	if line == "" || strings.HasPrefix(line, ":") || !strings.HasPrefix(line, "data:") {
		return wireResponse{}, false
	}
	payload := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
	if payload == "[DONE]" {
		return wireResponse{}, false
	}
	var parsed wireResponse
	if err := json.Unmarshal([]byte(payload), &parsed); err != nil {
		return wireResponse{}, false
	}
	return parsed, true
}

type OpenAICompatModel struct {
	BaseURL string
	ModelID string
	Timeout time.Duration
}

func NewOpenAICompatModel(baseURL, modelID string) *OpenAICompatModel {
	return &OpenAICompatModel{BaseURL: baseURL, ModelID: modelID, Timeout: 180 * time.Second}
}

func (m *OpenAICompatModel) Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta, 32)
	errs := make(chan error, 1)
	go func() {
		defer close(deltas)
		defer close(errs)
		body := wireRequest{Model: m.ModelID, Messages: toWireMessages(messages),
			Tools: toWireTools(tools), Stream: true}
		raw, err := json.Marshal(body)
		if err != nil {
			errs <- fmt.Errorf("请求体序列化失败：%w", err)
			return
		}
		req, err := http.NewRequestWithContext(ctx, "POST", m.BaseURL+"/chat/completions", bytes.NewReader(raw))
		if err != nil {
			errs <- fmt.Errorf("构造请求失败：%w", err)
			return
		}
		req.Header.Set("Content-Type", "application/json")
		resp, err := (&http.Client{Timeout: m.Timeout}).Do(req)
		if err != nil {
			errs <- fmt.Errorf("请求失败：%w", err)
			return
		}
		defer resp.Body.Close()
		if resp.StatusCode != http.StatusOK {
			buf := make([]byte, 300)
			n, _ := resp.Body.Read(buf)
			errs <- fmt.Errorf("模型服务返回 %d：%s", resp.StatusCode, string(buf[:n]))
			return
		}
		acc := newToolCallAccumulator()
		scanner := bufio.NewScanner(resp.Body)
		scanner.Buffer(make([]byte, 0, 64*1024), 4*1024*1024)
		for scanner.Scan() {
			chunk, ok := parseSSELine(scanner.Text())
			if !ok || len(chunk.Choices) == 0 {
				continue
			}
			d := chunk.Choices[0].Delta
			if d.Content != nil && *d.Content != "" {
				deltas <- Delta{Kind: "text", Text: *d.Content}
			}
			if len(d.ToolCalls) > 0 {
				acc.Feed(d.ToolCalls)
			}
		}
		if err := scanner.Err(); err != nil {
			errs <- fmt.Errorf("读流中断：%w", err)
			return
		}
		for _, block := range acc.Finish() {
			b := block
			deltas <- Delta{Kind: "tool_use", Tool: &b}
		}
		errs <- nil
	}()
	return deltas, errs
}

type ToolResult struct {
	Content string
	IsError bool
}

type Tool interface {
	Spec() ToolSpec
	Run(ctx context.Context, args map[string]any) (string, error)
}

type FuncTool struct {
	spec ToolSpec
	fn   func(ctx context.Context, args map[string]any) (string, error)
}

func NewFuncTool(name, description string, parameters map[string]any,
	fn func(ctx context.Context, args map[string]any) (string, error)) *FuncTool {
	return &FuncTool{spec: ToolSpec{Name: name, Description: description, Parameters: parameters}, fn: fn}
}

func (t *FuncTool) Spec() ToolSpec { return t.spec }

func (t *FuncTool) Run(ctx context.Context, args map[string]any) (string, error) {
	return t.fn(ctx, args)
}

type ToolRegistry struct {
	tools map[string]Tool
	order []string
}

func NewToolRegistry(tools ...Tool) *ToolRegistry {
	reg := &ToolRegistry{tools: map[string]Tool{}}
	for _, t := range tools {
		reg.tools[t.Spec().Name] = t
		reg.order = append(reg.order, t.Spec().Name)
	}
	return reg
}

func (reg *ToolRegistry) Specs() []ToolSpec {
	var out []ToolSpec
	for _, name := range reg.order {
		out = append(out, reg.tools[name].Spec())
	}
	return out
}

func (reg *ToolRegistry) Execute(ctx context.Context, name string, args map[string]any) (result ToolResult) {
	tool, ok := reg.tools[name]
	if !ok {
		return ToolResult{Content: "没有叫 " + name + " 的工具", IsError: true}
	}
	defer func() {
		if r := recover(); r != nil {
			result = ToolResult{Content: fmt.Sprintf("工具崩溃：%v", r), IsError: true}
		}
	}()
	output, err := tool.Run(ctx, args)
	if err != nil {
		return ToolResult{Content: err.Error(), IsError: true}
	}
	return ToolResult{Content: output}
}

func EstimateTokens(text string) int {
	cjk, other := 0, 0
	for _, r := range text {
		if unicode.Is(unicode.Han, r) {
			cjk++
		} else {
			other++
		}
	}
	return cjk + other/4
}

type Agent struct {
	Model        Model
	Registry     *ToolRegistry
	SystemPrompt string
	MaxRounds    int
	OnEvent      func(name string, args map[string]any, result ToolResult)
}

func (a *Agent) Run(ctx context.Context, userText string) (string, error) {
	history := []Message{
		NewMessage("system", TextBlock{Text: a.SystemPrompt}),
		NewMessage("user", TextBlock{Text: userText}),
	}
	for round := 0; round < a.MaxRounds; round++ {
		reply, err := a.oneRound(ctx, history)
		if err != nil {
			return "", err
		}
		history = append(history, reply)
		calls := reply.ToolUses()
		if len(calls) == 0 {
			return reply.Text(), nil
		}
		var results []Block
		for _, c := range calls {
			r := a.Registry.Execute(ctx, c.Name, c.Args)
			if a.OnEvent != nil {
				a.OnEvent(c.Name, c.Args, r)
			}
			results = append(results, ToolResultBlock{
				ToolUseID: c.ID, Content: r.Content, IsError: r.IsError})
		}
		history = append(history, NewMessage("tool", results...))
	}
	return "（轮数用完了）", nil
}

func (a *Agent) oneRound(ctx context.Context, messages []Message) (Message, error) {
	deltas, errs := a.Model.Stream(ctx, messages, a.Registry.Specs())
	var textParts []string
	var blocks []Block
	flushText := func() {
		if len(textParts) > 0 {
			joined := ""
			for _, p := range textParts {
				joined += p
			}
			blocks = append(blocks, TextBlock{Text: joined})
			textParts = nil
		}
	}
	for d := range deltas {
		switch d.Kind {
		case "text":
			textParts = append(textParts, d.Text)
		case "tool_use":
			if d.Tool != nil {
				flushText()
				blocks = append(blocks, *d.Tool)
			}
		}
	}
	flushText()
	if err := <-errs; err != nil {
		return Message{}, err
	}
	return NewMessage("assistant", blocks...), nil
}
```

## 技能长什么样

一个技能就是一个 Markdown 文件：开头几行是元数据，正文是给模型看的做法。

```go
// file: seed.go
package main

import (
	"os"
	"path/filepath"
)

func writeSkill(name, description, body string) error {
	content := "---\nname: " + name + "\ndescription: " + description + "\n---\n\n" + body
	return os.WriteFile(filepath.Join(skillsDir, name+".md"), []byte(content), 0o644)
}

//

// 模型要判断的是「现在该不该调它」，不是「它是个什么东西」。
func SeedSkills() error {
	os.RemoveAll(skillsDir)
	if err := os.MkdirAll(skillsDir, 0o755); err != nil {
		return err
	}

	if err := writeSkill("commit-message",
		"用户要写 git 提交信息、要提交代码、或者问「这次改动怎么描述」时使用",
		`# 写 git 提交信息

## 格式

    <类型>: <首行摘要，祈使句，不超过 50 字，句末不加句号>

    <正文：说清楚**为什么**要改，而不是改了什么——改了什么看 diff 就知道了。
    每行不超过 72 字符。>

    <脚注：破坏性变更、关联的问题编号>

## 类型只有这七个

feat（新功能）、fix（修缺陷）、docs（只改文档）、refactor（不改行为的重构）、
test（只动测试）、perf（性能）、chore（构建与杂务）。

## 三条硬规矩

1. **首行不要用「优化」「完善」「更新」这种没有信息量的词**。
   反例：chore: 优化代码。正例：perf: 把 N+1 查询合并成一次 IN 查询。
2. **一次提交只做一件事**。改了两件事就分两次提交。
3. **破坏性变更必须在脚注写 BREAKING CHANGE:**，并说明调用方要怎么改。

## 例子

    fix: 用户取消后补齐缺失的工具结果

    取消发生在多个工具执行到一半时，只有跑完的那几个进了历史。
    下一次请求会因为「有 tool_call 没有对应 tool_result」被服务端拒绝，
    表现是这段对话彻底打不开。

    现在给没跑完的调用补一条「用户取消了这次执行」的结果。
`); err != nil {
		return err
	}

	if err := writeSkill("slow-query",
		"排查数据库慢查询、接口变慢、超时报警时使用",
		`# 排查慢查询

## 顺序不能乱

先量，再猜，最后改。跳过第一步直接猜原因，十次有九次猜错。

1. **量**：拿到具体的慢查询语句和它的耗时分布（p50 / p95 / p99）。
   只看平均值会漏掉长尾——平均 20ms 而 p99 是 8 秒，是很常见的情况。
2. **看执行计划**：EXPLAIN ANALYZE。重点看有没有全表扫描、
   预估行数和实际行数差多少（差一个数量级说明统计信息过期了）。
3. **看有没有 N+1**：一次请求里同一条 SQL 出现几十次，
   八成是在循环里查。这类问题改索引没用，要改代码。
4. **最后才考虑加索引**。加索引会拖慢写入，而且加错了没用。

## 常见原因，按出现频率排

- N+1 查询（最常见，也最容易改）
- 缺索引，或者索引没被用上（隐式类型转换会让索引失效）
- 一次取回太多行（缺 LIMIT，或者分页用 OFFSET 翻到很深的页）
- 锁等待（这时候 SQL 本身很快，慢的是等锁）
- 统计信息过期，优化器选错了计划

## 交付什么

一份结论要包含：慢在哪条语句、慢的原因（有执行计划佐证）、
改法、改完之后的耗时对比。**没有前后对比的优化不算做完。**
`); err != nil {
		return err
	}

	return writeSkill("changelog",
		"用户要写发版说明、变更日志、release notes 时使用",
		`# 写变更日志

## 读者是谁

**是使用者，不是开发者。** 使用者关心的是「我要不要升级、升级之后我要改什么」，
不关心你重构了哪个类。

## 分组，按对使用者的影响排

1. **破坏性变更**——放最前面，写清楚要改什么。
2. **新功能**——一句话说清楚能干什么，附一个最小例子。
3. **缺陷修复**——描述**症状**，不是描述代码。
   反例：「修复空指针异常」。正例：「导出空表格时不再报错」。
4. **其他**——性能、依赖升级之类，一行一条。

## 三条规矩

1. 每条都要能让没读过代码的人看懂。
2. 不要写「若干优化和修复」这种占位句。没有内容就删掉这一节。
3. 版本号和日期写在最前面。
`)
```

## 技能库：只暴露索引

```go
// file: skills.go
package main

import (
	"fmt"
	"os"
	"path/filepath"
	"sort"
	"strings"
)

type Skill struct {
	Name        string
	Description string
	Body        string
	Path        string
}

func (s Skill) Tokens() int { return EstimateTokens(s.Body) }

// ParseSkill 解析技能文件。元数据用最朴素的三横线分隔，不引 YAML 库。
//
// 格式简单是有意的：技能要让不写代码的人也能加一个。
// 引入 YAML 就意味着「缩进写错了整个文件解析失败」，而报错信息还很难懂。
func ParseSkill(path string) (Skill, error) {
	raw, err := os.ReadFile(path)
	if err != nil {
		return Skill{}, err
	}
	text := string(raw)
	meta := map[string]string{}
	body := text

	if strings.HasPrefix(text, "---") {
		parts := strings.SplitN(text, "---", 3)
		if len(parts) == 3 {
			for _, line := range strings.Split(strings.TrimSpace(parts[1]), "\n") {
				if key, value, ok := strings.Cut(line, ":"); ok {
					meta[strings.TrimSpace(key)] = strings.TrimSpace(value)
				}
			}
			body = parts[2]
		}
	}

	name := meta["name"]
	if name == "" {
		name = strings.TrimSuffix(filepath.Base(path), ".md")
	}
	if meta["description"] == "" {
		return Skill{}, fmt.Errorf("%s 缺少 description。没有它模型不知道什么时候该用这个技能",
			filepath.Base(path))
	}
	return Skill{Name: name, Description: meta["description"],
		Body: strings.TrimSpace(body), Path: path}, nil
}

// SkillLibrary 管一个目录的技能。
//
// 只从这一个目录读，路径由我们拼——**不接受模型给的路径**。
// 否则 load_skill("../../etc/passwd") 就变成了一个任意文件读取工具，
// 而这个工具的输出会直接进上下文。
type SkillLibrary struct {
	Root   string
	skills map[string]Skill
	order  []string // 稳定顺序，见 Reload 里的说明
}

func NewSkillLibrary(root string) (*SkillLibrary, error) {
	lib := &SkillLibrary{Root: root}
	return lib, lib.Reload()
}

func (l *SkillLibrary) Reload() error {
	l.skills = map[string]Skill{}
	l.order = nil

	entries, err := os.ReadDir(l.Root)
	if err != nil {
		return err
	}
	for _, entry := range entries {
		if entry.IsDir() || !strings.HasSuffix(entry.Name(), ".md") {
			continue
		}
		skill, err := ParseSkill(filepath.Join(l.Root, entry.Name()))
		if err != nil {
			fmt.Println("  ⚠ 跳过", entry.Name()+"：", err)
			continue
		}
		l.skills[skill.Name] = skill
		l.order = append(l.order, skill.Name)
	}
	// 排序是必须的，不是为了好看。
	//
	// Go 的 map 遍历顺序是**随机的**——每次进程启动都不一样。
	// 不排序的话，同样一批技能生成出来的系统提示每次都不同，
	// 上游的提示词缓存全部失效，而且两次运行没法对比。
	sort.Strings(l.order)
	return nil
}

func (l *SkillLibrary) Names() []string { return l.order }

// IndexText 给系统提示用的索引。每个技能一行：名字 + 什么时候用。
func (l *SkillLibrary) IndexText() string {
	var b strings.Builder
	for _, name := range l.order {
		b.WriteString("- " + name + "：" + l.skills[name].Description + "\n")
	}
	return strings.TrimRight(b.String(), "\n")
}

func (l *SkillLibrary) Get(name string) (Skill, error) {
	skill, ok := l.skills[name]
	if !ok {
		return Skill{}, fmt.Errorf("没有叫 %s 的技能。现有的：%s", name, strings.Join(l.order, "、"))
	}
	return skill, nil
}

func (l *SkillLibrary) TotalBodyTokens() int {
	total := 0
	for _, name := range l.order {
		total += l.skills[name].Tokens()
	}
	return total
}
```

```go
// file: loader.go
package main

import (
	"context"
	"fmt"
)

// SkillLoader 把技能装成一个工具，并记住这一轮已经加载过哪些。
type SkillLoader struct {
	Library *SkillLibrary
	loaded  map[string]bool
}

func NewSkillLoader(lib *SkillLibrary) *SkillLoader {
	return &SkillLoader{Library: lib, loaded: map[string]bool{}}
}

func (l *SkillLoader) Reset() { l.loaded = map[string]bool{} }

func (l *SkillLoader) Load(name string) (string, error) {
	skill, err := l.Library.Get(name)
	if err != nil {
		return "", err
	}
	if l.loaded[name] {
		// 重复加载不再返回全文：模型有时会为了「确认一下」再调一次，
		// 那会让同样的两千字在上下文里躺两份。
		return fmt.Sprintf("技能「%s」上文已经加载过了，直接照着做就行。", name), nil
	}
	l.loaded[name] = true
	return "=== 技能：" + skill.Name + " ===\n" + skill.Body, nil
}

func (l *SkillLoader) AsTool() Tool {
	// 把可选值写进参数结构，模型就不会去猜技能名。
	// 这比在描述里写「可选值有 xxx」可靠——第 9 章的参数校验会真的把它挡住。
	names := make([]any, 0, len(l.Library.Names()))
	for _, n := range l.Library.Names() {
		names = append(names, n)
	}
	return NewFuncTool("load_skill",
		"加载一个技能的完整说明。当手头的任务与某个技能的适用场景相符时调用它，"+
			"然后严格按照技能里的要求去做。",
		map[string]any{"type": "object", "properties": map[string]any{
			"name": map[string]any{"type": "string", "enum": names, "description": "技能名字"}},
			"required": []string{"name"}},
		func(ctx context.Context, args map[string]any) (string, error) {
			name, _ := args["name"].(string)
			return l.Load(name)
		})
}

// BuildSystemPrompt 系统提示 = 短短的角色说明 + 技能索引 + 一条使用规则。
func BuildSystemPrompt(lib *SkillLibrary) string {
	return `你是一个工程助手。

你有一批技能，每个技能是一份详细的做法说明。下面是它们的名字和适用场景：

` + lib.IndexText() + `

规则：
- 手头的任务与某个技能相符时，**先调用 load_skill 读到它的完整说明，再动手**。
  不要凭印象做——技能里的规矩比你的直觉更具体。
- 任务和所有技能都无关时，直接回答，不要硬套。
- 一次只加载真正用得上的技能。`
}
```

```go
package main

import (
	"context"
	"fmt"
	"sync"

func main() {
	SeedSkills()

	systemPrompt := BuildSystemPrompt(lib)
	model := NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
	ctx := context.Background()

	cases := []struct {
		question string
		expected string // 空串表示「不该加载任何技能」
	}{
		{"帮我写这次发版的变更日志，这版加了会话树、修了导出乱码。", "changelog"},
		{"我们的订单列表接口从 200ms 变成了 3 秒，怎么查？", "slow-query"},
		{"这次改动怎么写提交信息？就是给 Session 加了个 title 字段。", "commit-message"},
		{"帮我把这句英文翻译成中文：the quick brown fox", ""}, // 和所有技能都无关
	}

	for _, c := range cases {
		loader := NewSkillLoader(lib)
		var mu sync.Mutex
		var used []string
		agent := &Agent{
			Model: model, Registry: NewToolRegistry(loader.AsTool()),
			SystemPrompt: systemPrompt, MaxRounds: 5,
			OnEvent: func(name string, args map[string]any, r ToolResult) {
				mu.Lock()
				defer mu.Unlock()
				if s, ok := args["name"].(string); ok {
					used = append(used, s)
				}
			},
		}
		agent.Run(ctx, c.question)

		got := ""
		if len(used) > 0 {
			got = used[0]
		}
		mark := "✗"
		if got == c.expected {
			mark = "✓"
		}
		question := []rune(c.question)
		if len(question) > 26 {
			question = question[:26]
		}
		fmt.Printf("  %s %-28s 期望 %-16s 实际 %s\n", mark, string(question),
			display(c.expected), display(got))
	}
}

func display(s string) string {
	if s == "" {
		return "（不加载）"
	}
	return s
}
```

## 技能不是提示词注入的后门

技能的全文会**原样进入上下文**，所以要划清三条底线。

```go
// file: main.go
package main

import (
	"fmt"
	"os"
	"path/filepath"
)

func main() {
	SeedSkills()
	lib, _ := NewSkillLibrary(skillsDir)
	loader := NewSkillLoader(lib)

	type check struct {
		label string
		ok    bool
	}
	var checks []check

	// 1. 只能加载库里有的技能
	_, err := loader.Load("../../etc/passwd")
	checks = append(checks, check{"拒绝库外的名字", err != nil})

	// 2. 技能名来自枚举，模型没法自由发挥
	spec := loader.AsTool().Spec()
	props, _ := spec.Parameters["properties"].(map[string]any)
	nameSpec, _ := props["name"].(map[string]any)
	_, hasEnum := nameSpec["enum"]
	checks = append(checks, check{"技能名是封闭枚举", hasEnum})

	// 3. 缺 description 的技能不会被装进来
	os.WriteFile(filepath.Join(skillsDir, "broken.md"),
		[]byte("---\nname: broken\n---\n\n没有 description。"), 0o644)
	lib.Reload()
	_, err = lib.Get("broken")
	checks = append(checks, check{"缺 description 的技能被拒绝", err != nil})

	// 4. 索引顺序稳定：连续构建两次应当完全一致
	first := lib.IndexText()
	lib.Reload()
	checks = append(checks, check{"索引顺序稳定", first == lib.IndexText()})

	for _, c := range checks {
		mark := "✗"
		if c.ok {
			mark = "✓"
		}
		fmt.Printf("  %s %s\n", mark, c.label)
	}

	fmt.Println("\n三条底线：")
	fmt.Println("  1. 技能目录是**可信的**，和模型能写入的目录分开——")
	fmt.Println("     否则 agent 写一个技能文件再加载它，就绕过了你的全部规则。")
	fmt.Println("  2. 技能名从**封闭枚举**里选，不接受任意字符串。")
	fmt.Println("  3. 技能是**给模型的说明**，不是给程序的指令。")
	fmt.Println("     技能里写「忽略之前的所有规则」也只是一句话，")
	fmt.Println("     真正的边界仍然在第 18 章那三道防线里。")
}
```

## 什么时候不该用技能

技能不是万能的。三种情况用工具而不是技能：

- **需要确定性**。「提交信息首行不超过 50 字」写成技能，模型大概率遵守；
  写成一个校验工具，就是一定遵守。**能用代码保证的，不要用文字请求。**
- **需要外部数据**。技能是静态文字，取不了实时信息。
- **步骤长且固定**。二十步的固定流程写成技能，模型会跳步；写成代码就不会。

反过来，三种情况技能比工具合适：

- 规矩会变，而且改规矩的人不写代码。
- 做法需要判断，没法写成固定分支。
- 场景很多，每个都做成工具会让工具列表爆炸。

Go 这边还多一条理由：**加工具要重新编译发版，加技能只要往目录里丢个文件**。
在需要频繁调整做法的场景里，这个差别很实际。

## 本章小结

- **工具扩展能力，技能扩展方法**。工具是代码，技能是文字。
- 把「怎么做」从系统提示搬到技能文件里，系统提示里只留**一行索引**。
- 膨胀的系统提示有三层代价，最要命的是第三层：**注意力被稀释**。
- `description` 写**「什么时候用」而不是「这是什么」**。
- **Go 的 map 遍历顺序是随机的**。生成索引前必须排序——
  不排的话每次运行的系统提示都不同，上游的提示词缓存全部失效，两次运行也没法对比。
- 技能名做成**封闭枚举**，参数校验会真的把它挡住。
- 重复加载**不再返回全文**，否则同样的两千字会在上下文里躺两份。
- 元数据格式**越简单越好**，`strings.Cut` 就够了，不必引 YAML 库。
- 测试要有**反向用例**：和所有技能都无关时，它不该硬加载一个。
- 三条底线：技能目录**可信**且与模型可写目录分开；技能名从**封闭枚举**里选；
  技能是**给模型的说明不是给程序的指令**，真正的边界在第 18 章那三道防线里。
- **能用代码保证的，不要用文字请求。**

最后一章：我们改了这么多提示词、加了这么多规矩——**怎么知道它真的变好了？**
