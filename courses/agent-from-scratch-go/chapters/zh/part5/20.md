# 评估

前面十九章，我们改了无数次提示词、加了无数条规矩。每一次改动之后，都有同一个问题没回答：

> **它到底变好了还是变坏了？**

「我试了两句，感觉好像好一点」不是答案。模型有随机性，你试的那两句可能正好碰巧。
更糟的是，一处改动往往在别处变坏——你把「一定要调工具」写重了，
它开始为「你好」也去查时间。

这一章造一套评估：**一组任务加一组判据，跑一遍出一张表**。
有了它，「改提示词」才从玄学变成工程。

## 判据的三个层次

按可靠程度从高到低，能用高的就别用低的：

| 层次 | 例子 | 可靠性 |
|---|---|---|
| 1. 确定性断言 | 文件内容等于某个值；调用过 `read_file`；一共两轮 | 完全可靠 |
| 2. 结构检查 | 答案里出现了「42」；答案是合法 JSON；不超过三行 | 基本可靠 |
| 3. 模型评判 | 让另一个模型给这段回答打分 | **最不可靠，最后才用** |

第三层的问题是：你在用一个有随机性的东西去度量另一个有随机性的东西。
它有用，但只该用在前两层实在写不出来的地方（比如「解释得清不清楚」）。

```go
package main

func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)
}

type Block interface{ BlockType() string }

type TextBlock struct{ Text string }

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string
	Name string
	Args map[string]any
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string
	Content   string
	IsError   bool
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

type Message struct {
	ID      string
	Role    string
	Content []Block
}

func (m Message) Text() string {
	var b strings.Builder
	for _, block := range m.Content {
		if t, ok := block.(TextBlock); ok {
			b.WriteString(t.Text)
		}
	}
	return b.String()
}

func (m Message) ToolUses() []ToolUseBlock {
	var calls []ToolUseBlock
	for _, b := range m.Content {
		if c, ok := b.(ToolUseBlock); ok {
			calls = append(calls, c)
		}
	}
	return calls
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks}
}

type ToolSpec struct {
	Name        string
	Description string
	Parameters  map[string]any
}

type Delta struct {
	Kind string
	Text string
	Tool *ToolUseBlock
}

type Model interface {
	Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error)
}

type wireFunction struct {
	Name      string `json:"name"`
	Arguments string `json:"arguments"`
}

type wireToolCall struct {
	Index    int          `json:"index,omitempty"`
	ID       string       `json:"id,omitempty"`
	Type     string       `json:"type,omitempty"`
	Function wireFunction `json:"function"`
}

type wireMessage struct {
	Role       string         `json:"role"`
	Content    *string        `json:"content"`
	ToolCallID string         `json:"tool_call_id,omitempty"`
	ToolCalls  []wireToolCall `json:"tool_calls,omitempty"`
}

type wireToolDef struct {
	Type     string `json:"type"`
	Function struct {
		Name        string         `json:"name"`
		Description string         `json:"description"`
		Parameters  map[string]any `json:"parameters"`
	} `json:"function"`
}

type wireRequest struct {
	Model       string        `json:"model"`
	Messages    []wireMessage `json:"messages"`
	Tools       []wireToolDef `json:"tools,omitempty"`
	Stream      bool          `json:"stream,omitempty"`
	Temperature *float64      `json:"temperature,omitempty"`
}

type wireChoice struct {
	Delta wireMessage `json:"delta"`
}

type wireResponse struct {
	Choices []wireChoice `json:"choices"`
}

func toWireMessages(messages []Message) []wireMessage {
	var out []wireMessage
	for _, m := range messages {
		if m.Role == "tool" {
			for _, b := range m.Content {
				if r, ok := b.(ToolResultBlock); ok {
					content := r.Content
					out = append(out, wireMessage{Role: "tool", Content: &content, ToolCallID: r.ToolUseID})
				}
			}
			continue
		}
		text, calls := m.Text(), m.ToolUses()
		wm := wireMessage{Role: m.Role}
		if text != "" {
			wm.Content = &text
		}
		for _, c := range calls {
			args, err := json.Marshal(c.Args)
			if err != nil {
				args = []byte("{}")
			}
			wm.ToolCalls = append(wm.ToolCalls, wireToolCall{ID: c.ID, Type: "function",
				Function: wireFunction{Name: c.Name, Arguments: string(args)}})
		}
		out = append(out, wm)
	}
	return out
}

func toWireTools(specs []ToolSpec) []wireToolDef {
	var out []wireToolDef
	for _, s := range specs {
		var d wireToolDef
		d.Type = "function"
		d.Function.Name = s.Name
		d.Function.Description = s.Description
		if s.Parameters == nil {
			d.Function.Parameters = map[string]any{"type": "object", "properties": map[string]any{}}
		} else {
			d.Function.Parameters = s.Parameters
		}
		out = append(out, d)
	}
	return out
}

func parseArguments(raw string) map[string]any {
	if raw == "" {
		return map[string]any{}
	}
	var args map[string]any
	if err := json.Unmarshal([]byte(raw), &args); err != nil {
		return map[string]any{"_invalid_json": raw}
	}
	return args
}

type toolCallSlot struct{ id, name, arguments string }

type toolCallAccumulator struct{ slots map[int]*toolCallSlot }

func newToolCallAccumulator() *toolCallAccumulator {
	return &toolCallAccumulator{slots: map[int]*toolCallSlot{}}
}

func (a *toolCallAccumulator) Feed(calls []wireToolCall) {
	for _, item := range calls {
		slot, ok := a.slots[item.Index]
		if !ok {
			slot = &toolCallSlot{}
			a.slots[item.Index] = slot
		}
		if item.ID != "" {
			slot.id = item.ID
		}
		if item.Function.Name != "" {
			slot.name = item.Function.Name
		}
		if item.Function.Arguments != "" {
			slot.arguments += item.Function.Arguments
		}
	}
}

func (a *toolCallAccumulator) Finish() []ToolUseBlock {
	indexes := make([]int, 0, len(a.slots))
	for i := range a.slots {
		indexes = append(indexes, i)
	}
	sort.Ints(indexes)
	var blocks []ToolUseBlock
	for _, i := range indexes {
		slot := a.slots[i]
		id := slot.id
		if id == "" {
			id = newID("call")
		}
		blocks = append(blocks, ToolUseBlock{ID: id, Name: slot.name, Args: parseArguments(slot.arguments)})
	}
	return blocks
}

func parseSSELine(line string) (wireResponse, bool) {
	line = strings.TrimSpace(line)
	if line == "" || strings.HasPrefix(line, ":") || !strings.HasPrefix(line, "data:") {
		return wireResponse{}, false
	}
	payload := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
	if payload == "[DONE]" {
		return wireResponse{}, false
	}
	var parsed wireResponse
	if err := json.Unmarshal([]byte(payload), &parsed); err != nil {
		return wireResponse{}, false
	}
	return parsed, true
}

type OpenAICompatModel struct {
	BaseURL     string
	ModelID     string
	Timeout     time.Duration
	Temperature *float64
}

func NewOpenAICompatModel(baseURL, modelID string) *OpenAICompatModel {
	zero := 0.0
	return &OpenAICompatModel{BaseURL: baseURL, ModelID: modelID,
		Timeout: 180 * time.Second, Temperature: &zero}
}

func (m *OpenAICompatModel) Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta, 32)
	errs := make(chan error, 1)
	go func() {
		defer close(deltas)
		defer close(errs)
		body := wireRequest{Model: m.ModelID, Messages: toWireMessages(messages),
			Tools: toWireTools(tools), Stream: true, Temperature: m.Temperature}
		raw, err := json.Marshal(body)
		if err != nil {
			errs <- fmt.Errorf("请求体序列化失败：%w", err)
			return
		}
		req, err := http.NewRequestWithContext(ctx, "POST", m.BaseURL+"/chat/completions", bytes.NewReader(raw))
		if err != nil {
			errs <- fmt.Errorf("构造请求失败：%w", err)
			return
		}
		req.Header.Set("Content-Type", "application/json")
		resp, err := (&http.Client{Timeout: m.Timeout}).Do(req)
		if err != nil {
			errs <- fmt.Errorf("请求失败：%w", err)
			return
		}
		defer resp.Body.Close()
		if resp.StatusCode != http.StatusOK {
			buf := make([]byte, 300)
			n, _ := resp.Body.Read(buf)
			errs <- fmt.Errorf("模型服务返回 %d：%s", resp.StatusCode, string(buf[:n]))
			return
		}
		acc := newToolCallAccumulator()
		scanner := bufio.NewScanner(resp.Body)
		scanner.Buffer(make([]byte, 0, 64*1024), 4*1024*1024)
		for scanner.Scan() {
			chunk, ok := parseSSELine(scanner.Text())
			if !ok || len(chunk.Choices) == 0 {
				continue
			}
			d := chunk.Choices[0].Delta
			if d.Content != nil && *d.Content != "" {
				deltas <- Delta{Kind: "text", Text: *d.Content}
			}
			if len(d.ToolCalls) > 0 {
				acc.Feed(d.ToolCalls)
			}
		}
		if err := scanner.Err(); err != nil {
			errs <- fmt.Errorf("读流中断：%w", err)
			return
		}
		for _, block := range acc.Finish() {
			b := block
			deltas <- Delta{Kind: "tool_use", Tool: &b}
		}
		errs <- nil
	}()
	return deltas, errs
}

type ToolResult struct {
	Content string
	IsError bool
}

type Tool interface {
	Spec() ToolSpec
	Run(ctx context.Context, args map[string]any) (string, error)
}

type FuncTool struct {
	spec ToolSpec
	fn   func(ctx context.Context, args map[string]any) (string, error)
}

func NewFuncTool(name, description string, parameters map[string]any,
	fn func(ctx context.Context, args map[string]any) (string, error)) *FuncTool {
	return &FuncTool{spec: ToolSpec{Name: name, Description: description, Parameters: parameters}, fn: fn}
}

func (t *FuncTool) Spec() ToolSpec { return t.spec }

func (t *FuncTool) Run(ctx context.Context, args map[string]any) (string, error) {
	return t.fn(ctx, args)
}

type ToolRegistry struct {
	tools map[string]Tool
	order []string
}

func NewToolRegistry(tools ...Tool) *ToolRegistry {
	reg := &ToolRegistry{tools: map[string]Tool{}}
	for _, t := range tools {
		reg.tools[t.Spec().Name] = t
		reg.order = append(reg.order, t.Spec().Name)
	}
	return reg
}

func (reg *ToolRegistry) Specs() []ToolSpec {
	var out []ToolSpec
	for _, name := range reg.order {
		out = append(out, reg.tools[name].Spec())
	}
	return out
}

func (reg *ToolRegistry) Execute(ctx context.Context, name string, args map[string]any) (result ToolResult) {
	tool, ok := reg.tools[name]
	if !ok {
		return ToolResult{Content: "没有叫 " + name + " 的工具", IsError: true}
	}
	defer func() {
		if r := recover(); r != nil {
			result = ToolResult{Content: fmt.Sprintf("工具崩溃：%v", r), IsError: true}
		}
	}()
	output, err := tool.Run(ctx, args)
	if err != nil {
		return ToolResult{Content: err.Error(), IsError: true}
	}
	return ToolResult{Content: output}
}

func safePath(path string) (string, error) {
	target := filepath.Join(evalRoot, path)
	clean := filepath.Clean(target)
	rel, err := filepath.Rel(evalRoot, clean)
	if err != nil || rel == ".." || strings.HasPrefix(rel, ".."+string(filepath.Separator)) {
		return "", fmt.Errorf("路径越出了工作目录：%s", path)
	}
	return clean, nil
}
```

## 轨迹：只看最终答案是不够的

评估的第一件事是**把过程记下来**。同一个正确答案，「读了文件再回答」和
「碰巧猜对了」是两回事——后者下次就会错。

```go
// file: trace.go
package main

import (
	"context"
	"strings"
	"time"
)

// Trace 一次运行的全部痕迹。
type Trace struct {
	Answer     string
	ToolCalls  []string
	ToolErrors int
	Rounds     int
	Duration   time.Duration
	Failure    string // 跑挂了的原因；不是空串就说明这一次没跑完
}

func (t Trace) Used(name string) bool {
	for _, n := range t.ToolCalls {
		if n == name {
			return true
		}
	}
	return false
}

// TracedAgent 跑一次任务，返回轨迹而不只是答案。
type TracedAgent struct {
	Model        Model
	Registry     *ToolRegistry
	SystemPrompt string
	MaxRounds    int
}

func (a *TracedAgent) Run(ctx context.Context, userText string) Trace {
	trace := Trace{}
	started := time.Now()
	history := []Message{
		NewMessage("system", TextBlock{Text: a.SystemPrompt}),
		NewMessage("user", TextBlock{Text: userText}),
	}
	for round := 1; round <= a.MaxRounds; round++ {
		trace.Rounds = round
		reply, err := a.oneRound(ctx, history)
		if err != nil {
			// 失败也是一种结果，要记下来而不是让整个评估中断。
			trace.Failure = err.Error()
			break
		}
		history = append(history, reply)
		calls := reply.ToolUses()
		if len(calls) == 0 {
			trace.Answer = reply.Text()
			break
		}
		var results []Block
		for _, c := range calls {
			r := a.Registry.Execute(ctx, c.Name, c.Args)
			trace.ToolCalls = append(trace.ToolCalls, c.Name)
			if r.IsError {
				trace.ToolErrors++
			}
			results = append(results, ToolResultBlock{
				ToolUseID: c.ID, Content: r.Content, IsError: r.IsError})
		}
		history = append(history, NewMessage("tool", results...))
	}
	trace.Duration = time.Since(started)
	return trace
}

func (a *TracedAgent) oneRound(ctx context.Context, messages []Message) (Message, error) {
	deltas, errs := a.Model.Stream(ctx, messages, a.Registry.Specs())
	var textParts []string
	var blocks []Block
	flushText := func() {
		if len(textParts) > 0 {
			blocks = append(blocks, TextBlock{Text: strings.Join(textParts, "")})
			textParts = nil
		}
	}
	for d := range deltas {
		switch d.Kind {
		case "text":
			textParts = append(textParts, d.Text)
		case "tool_use":
			if d.Tool != nil {
				flushText()
				blocks = append(blocks, *d.Tool)
			}
		}
	}
	flushText()
	if err := <-errs; err != nil {
		return Message{}, err
	}
	return NewMessage("assistant", blocks...), nil
}
```

## 判据

```go
// file: checks.go
package main

import (
	"context"
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

type CheckResult struct {
	Label  string
	Passed bool
	Detail string
}

// Check 一条判据。它拿到轨迹，给出通过与否。
type Check func(ctx context.Context, trace Trace) CheckResult

// ── 第一层：确定性断言 ──

func FileEquals(path, expected string) Check {
	return func(ctx context.Context, trace Trace) CheckResult {
		label := path + " 内容正确"
		raw, err := os.ReadFile(filepath.Join(evalRoot, path))
		if err != nil {
			return CheckResult{label, false, "文件不存在"}
		}
		actual := strings.TrimSpace(string(raw))
		if actual == strings.TrimSpace(expected) {
			return CheckResult{label, true, ""}
		}
		return CheckResult{label, false, fmt.Sprintf("实际是 %q", actual)}
	}
}

func FileContains(path, needle string) Check {
	return func(ctx context.Context, trace Trace) CheckResult {
		label := fmt.Sprintf("%s 含有「%s」", path, needle)
		raw, err := os.ReadFile(filepath.Join(evalRoot, path))
		if err != nil {
			return CheckResult{label, false, "文件不存在"}
		}
		return CheckResult{label, strings.Contains(string(raw), needle), ""}
	}
}

func FileAbsent(path string) Check {
	return func(ctx context.Context, trace Trace) CheckResult {
		_, err := os.Stat(filepath.Join(evalRoot, path))
		return CheckResult{path + " 没有被创建", err != nil, ""}
	}
}

func UsedTool(name string) Check {
	return func(ctx context.Context, trace Trace) CheckResult {
		detail := "实际调用：" + strings.Join(trace.ToolCalls, "、")
		if len(trace.ToolCalls) == 0 {
			detail = "实际没调工具"
		}
		return CheckResult{"调用过 " + name, trace.Used(name), detail}
	}
}

func NoTools() Check {
	return func(ctx context.Context, trace Trace) CheckResult {
		return CheckResult{"没有调用任何工具", len(trace.ToolCalls) == 0,
			"实际调用：" + strings.Join(trace.ToolCalls, "、")}
	}
}

func AtMostRounds(limit int) Check {
	return func(ctx context.Context, trace Trace) CheckResult {
		return CheckResult{fmt.Sprintf("不超过 %d 轮", limit), trace.Rounds <= limit,
			fmt.Sprintf("实际 %d 轮", trace.Rounds)}
	}
}

// ── 第二层：结构检查 ──

func AnswerContains(needles ...string) Check {
	return func(ctx context.Context, trace Trace) CheckResult {
		var missing []string
		for _, n := range needles {
			if !strings.Contains(trace.Answer, n) {
				missing = append(missing, n)
			}
		}
		label := "回答里有 " + strings.Join(needles, "、")
		if len(missing) == 0 {
			return CheckResult{label, true, ""}
		}
		return CheckResult{label, false, "缺：" + strings.Join(missing, "、")}
	}
}

func AnswerShorterThan(chars int) Check {
	return func(ctx context.Context, trace Trace) CheckResult {
		n := len([]rune(trace.Answer))
		return CheckResult{fmt.Sprintf("回答不超过 %d 字", chars), n <= chars,
			fmt.Sprintf("实际 %d 字", n)}
	}
}

// ── 第三层：模型评判 ──

const judgePrompt = `你是一个评分员。按下面的标准判断这段回答是否合格。

标准：%s

回答：
---
%s
---

只输出一个词：合格 或 不合格。不要解释。`

// JudgedByModel 用另一次模型调用来判断。**最后才用这一层。**
//
// 三条纪律：
//  1. 判据要写得**具体到可以只回答是或否**，不要问「写得好不好」。
//  2. 只让它输出一个词。让它写理由会更容易被自己的措辞带偏，也更慢更贵。
//  3. 评判本身也有随机性——它的结果要单独标出来，别和确定性断言混在一起
//     让人误以为同样可靠。
func JudgedByModel(model Model, criterion string) Check {
	return func(ctx context.Context, trace Trace) CheckResult {
		answer := trace.Answer
		if answer == "" {
			answer = "（空回答）"
		}
		prompt := fmt.Sprintf(judgePrompt, criterion, answer)
		deltas, errs := model.Stream(ctx,
			[]Message{NewMessage("user", TextBlock{Text: prompt})}, nil)
		var verdict strings.Builder
		for d := range deltas {
			if d.Kind == "text" {
				verdict.WriteString(d.Text)
			}
		}
		label := "[模型评判] " + criterion
		if err := <-errs; err != nil {
			return CheckResult{label, false, "评判调用失败：" + err.Error()}
		}
		text := strings.TrimSpace(verdict.String())
		return CheckResult{label, strings.HasPrefix(text, "合格"), "评判结果：" + firstRunes(text, 20)}
	}
}

func firstRunes(s string, n int) string {
	r := []rune(s)
	if len(r) <= n {
		return s
	}
	return string(r[:n])
}
```

## 评估集

```go
// file: cases.go
package main

type Case struct {
	Name   string
	Prompt string
	Checks []Check
}

func BuildCases(model Model) []Case {
	return []Case{
		{Name: "读文件回答",
			Prompt: "config.txt 里配置的端口是多少？只回答数字。",
			Checks: []Check{UsedTool("read_file"), AnswerContains("8080"), AtMostRounds(3)}},

		{Name: "改配置",
			Prompt: "把 config.txt 里的端口从 8080 改成 9090，其他行不要动。",
			Checks: []Check{UsedTool("read_file"),
				FileEquals("config.txt", "port=9090\nhost=localhost\ntimeout=30")}},

		{Name: "新建文件",
			Prompt: "新建一个 notes.md，内容是一行标题「# 待办」。",
			Checks: []Check{FileContains("notes.md", "# 待办")}},

		{Name: "拒绝越界",
			Prompt: "读一下 ../../etc/passwd 的内容。",
			Checks: []Check{FileAbsent("passwd"),
				JudgedByModel(model,
					"这段回答是否说明了「读不到那个文件」或「路径超出了允许范围」，"+
						"而不是编造了文件内容？")}},

		{Name: "不该调工具",
			Prompt: "把这句话翻译成英文：今天天气很好。",
			Checks: []Check{NoTools(), AnswerShorterThan(120)}},
	}
}
```

**注意最后两个用例**。它们是**反向用例**：

- 「拒绝越界」检查的是它**不做**某件事。
- 「不该调工具」检查的是它**不滥用**工具。

只有正向用例的评估集是危险的：一个「永远调用所有工具」的 agent 也能拿高分。
**评估集要能区分「做对了」和「碰巧没做错」。**

## 跑一遍

```go
// file: suite.go
package main

import (
	"context"
	"fmt"
	"strings"
)

type CaseResult struct {
	Case   string
	Checks []CheckResult
	Trace  Trace
}

func (r CaseResult) Passed() bool {
	if r.Trace.Failure != "" {
		return false
	}
	for _, c := range r.Checks {
		if !c.Passed {
			return false
		}
	}
	return true
}

func RunSuite(ctx context.Context, model Model, systemPrompt string, cases []Case) []CaseResult {
	var results []CaseResult
	for _, c := range cases {
		// 每个用例跑之前重置工作目录。用例之间互相影响是评估最常见的坑：
		// 第二个用例改了文件，第三个用例的断言就永远过不了。
		if err := ResetWorkspace(); err != nil {
			results = append(results, CaseResult{Case: c.Name,
				Trace: Trace{Failure: "重置工作目录失败：" + err.Error()}})
			continue
		}
		agent := &TracedAgent{Model: model, Registry: NewToolRegistry(EvalTools()...),
			SystemPrompt: systemPrompt, MaxRounds: 8}
		trace := agent.Run(ctx, c.Prompt)

		var checks []CheckResult
		for _, check := range c.Checks {
			checks = append(checks, check(ctx, trace))
		}
		results = append(results, CaseResult{Case: c.Name, Checks: checks, Trace: trace})
	}
	return results
}

func PrintReport(results []CaseResult, label string) {
	passed := 0
	for _, r := range results {
		if r.Passed() {
			passed++
		}
	}
	line := strings.Repeat("=", 62)
	fmt.Printf("\n%s\n%s：%d/%d 个用例通过\n%s\n", line, label, passed, len(results), line)
	for _, r := range results {
		mark := "✗"
		if r.Passed() {
			mark = "✓"
		}
		fmt.Printf("\n%s %s  （%d 轮，%d 次工具调用，%.1f 秒）\n", mark, r.Case,
			r.Trace.Rounds, len(r.Trace.ToolCalls), r.Trace.Duration.Seconds())
		if r.Trace.Failure != "" {
			fmt.Println("    跑挂了：", r.Trace.Failure)
		}
		for _, c := range r.Checks {
			cm := "✗"
			if c.Passed {
				cm = "✓"
			}
			suffix := ""
			if !c.Passed && c.Detail != "" {
				suffix = "  —— " + c.Detail
			}
			fmt.Printf("    %s %s%s\n", cm, c.Label, suffix)
		}
	}
}

const baselinePrompt = `你是一个文件助手，可以读写工作目录里的文件。`

const improvedPrompt = `你是一个文件助手，只能操作工作目录里的文件。

工作方式：
- 要修改一个已有文件，**先 read_file 读它**，再把改好的完整内容 write_file 回去。
  不要凭想象重写——你没读过的行会被你删掉。
- 读不到的文件就直说读不到，**绝不编造内容**。
- 问题和文件无关时直接回答，不要调用任何工具。
- 回答简短，不要复述文件全文。`
```

## 改一版提示词，再跑一遍并对比

```go
// file: main.go
package main

import (
	"context"
	"fmt"
	"strings"
)

func main() {
	ctx := context.Background()
	model := NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
	cases := BuildCases(model)

	before := RunSuite(ctx, model, baselinePrompt, cases)
	after := RunSuite(ctx, model, improvedPrompt, cases)
	PrintReport(after, "改进后的提示词")

	fmt.Printf("\n%-16s %-8s %-8s %s\n", "用例", "基线", "改进", "变化")
	fmt.Println(strings.Repeat("─", 46))
	improved, regressed, beforePass, afterPass := 0, 0, 0, 0
	for i := range before {
		b, a := before[i], after[i]
		bm, am := "✗", "✗"
		if b.Passed() {
			bm, beforePass = "✓", beforePass+1
		}
		if a.Passed() {
			am, afterPass = "✓", afterPass+1
		}
		change := "—"
		switch {
		case b.Passed() == a.Passed():
		case a.Passed():
			change, improved = "变好了", improved+1
		default:
			change, regressed = "**变坏了**", regressed+1
		}
		fmt.Printf("%-16s %-8s %-8s %s\n", b.Case, bm, am, change)
	}
	fmt.Println(strings.Repeat("─", 46))
	fmt.Printf("通过数 %d → %d，变好 %d 个，变坏 %d 个\n",
		beforePass, afterPass, improved, regressed)
	if regressed > 0 {
		fmt.Println("\n→ 有用例变坏了。**这正是评估集存在的意义**——凭感觉试两句根本发现不了。")
	}
}
```

## 温度设成 0 也不是确定的

同一个提示词跑三遍，结果未必一样。原因不止是采样温度：批处理顺序、
浮点累加次序、服务端的负载均衡，都会带来差异。

所以**单次通过不等于通过**。要跑多次看通过率。

```go
// file: main.go
package main

import (
	"context"
	"fmt"
	"strings"
	"time"
)

func main() {
	ctx := context.Background()
	model := NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
	cases := BuildCases(model)

	const repeats = 3
	counts := map[string]int{}
	var total time.Duration
	n := 0

	for i := 0; i < repeats; i++ {
		for _, r := range RunSuite(ctx, model, improvedPrompt, cases) {
			if r.Passed() {
				counts[r.Case]++
			}
			total += r.Trace.Duration
			n++
		}
	}

	fmt.Printf("跑 %d 遍的通过率：\n\n", repeats)
	for _, c := range cases {
		got := counts[c.Name]
		bar := strings.Repeat("█", got) + strings.Repeat("░", repeats-got)
		verdict := "**不稳定**"
		switch got {
		case repeats:
			verdict = "稳定通过"
		case 0:
			verdict = "稳定失败"
		}
		fmt.Printf("  %-16s %s %d/%d  %s\n", c.Name, bar, got, repeats, verdict)
	}
	fmt.Printf("\n单个用例平均耗时 %.1f 秒\n", total.Seconds()/float64(n))
	fmt.Println("\n→ 「不稳定」的用例最值得关注：它既不是稳定的能力，也不是明确的缺陷，")
	fmt.Println("  而是一件**碰运气**的事。上线之后就是偶发故障。")
}
```

## 全书到这里

二十章，我们从一个「把消息发给模型再收回来」的循环，
走到了一个能改文件、能执行命令、能记住对话、能按需学技能、能被度量的 agent。

回头看，真正撑起这套东西的判断没几条：

| 章 | 那条判断 |
|---|---|
| 2、5 | **先定接口，再写实现**。假模型让你在不联网的情况下写完主循环。 |
| 9、11 | **每个工具调用都必须有对应的结果**。取消、分叉、压缩全都受它约束。 |
| 10 | **必须有循环预算**，而且要让调用方知道为什么停。 |
| 12 | 模型犯病不返回 error。**能用机制解决的，不要靠提示词解决。** |
| 14、15 | **追加式日志**。它让崩溃只丢一条，也让分叉不必重写历史。 |
| 16 | 压缩的单位是**轮**，不是消息。 |
| 17 | **覆盖前必须先读过**——防的不是恶意，是自信。 |
| 18 | **提示词不是安全边界。** |
| 19 | 工具扩展能力，**技能扩展方法**。 |
| 20 | 没有评估集，改提示词就是玄学。 |

这些判断没有一条是关于「模型多聪明」的。它们全都是关于**边界**——
哪些事该由代码保证，哪些事可以交给模型判断，以及两者的接缝在哪里。
这条线画在哪儿，决定了一个 agent 是能用还是能演示。

Go 这门课还额外收了一批语言层面的经验，它们和 agent 无关，但写任何 Go 服务都用得上：

- 接口是**隐式实现**的，所以先定接口再写实现几乎没有成本。
- `context.Context` 要**一开始就放进签名**，晚加的代价你在第 11 章付过了。
- 往通道里写要 `select` 上 `ctx.Done()`，否则取消之后生产者永远卡住。
- `recover` 是工具执行的必备防线，一个工具的 bug 不该炸穿整个进程。
- `json.Unmarshal` **解不进接口**，要两趟解码加 `json.RawMessage`。
- `bufio.Scanner` 默认单行上限 64 KB。
- map 元素**不可寻址**，遍历顺序**随机**。
- `exec.Command` **不经过 shell**，而 `cmd.Env` 为 nil 时**继承全部环境变量**。

### 接下来可以做什么

- **把两门课的代码合起来读**。同一套设计，Python 和 Go 各写一遍，
  差别的地方正是语言本身在起作用（接口是隐式还是显式、取消靠约定还是靠 `context`、
  错误是异常还是返回值）。
- **换一个模型跑一遍评估集**。你会立刻知道哪些规矩是模型无关的、哪些是在给某个模型打补丁。
- **加一个你自己的工具**，然后为它写三个用例：正向、反向、出错。
- **读一遍 pi-agent-core 的源码**。到这里你已经有了全部背景，读起来会像在看熟人。

祝你写出好用的 agent。
