# 编码工具

前面十六章的工具都是玩具：查时间、算算术。这一章做真正干活的四个工具——
**读文件、写文件、改文件、列目录**。它们是编码 agent 的全部基础。

这四个工具看着简单，但每一个都有一处不做就会出事的设计：

| 工具 | 那处设计 | 不做的后果 |
|---|---|---|
| 读文件 | 带行号返回，并且能只读一段 | 大文件一次读完撑爆上下文；没有行号，模型没法精确指认位置 |
| 写文件 | 覆盖前必须先读过 | 模型凭想象重写整个文件，把没看过的内容删掉 |
| 改文件 | 按**唯一的原文片段**替换，不按行号 | 行号在上一次编辑后就失效了，改错位置 |
| 列目录 | 有深度和数量上限 | 一个 node_modules 就把上下文吃光 |

```go
// file: core.go
package main

import (
	"bufio"
	"bytes"
	"context"
	"crypto/rand"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"net/http"
	"sort"
	"strings"
	"time"
)

func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)
}

type Block interface{ BlockType() string }

type TextBlock struct{ Text string }

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string
	Name string
	Args map[string]any
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string
	Content   string
	IsError   bool
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

type Message struct {
	ID      string
	Role    string
	Content []Block
}

func (m Message) Text() string {
	var b strings.Builder
	for _, block := range m.Content {
		if t, ok := block.(TextBlock); ok {
			b.WriteString(t.Text)
		}
	}
	return b.String()
}

func (m Message) ToolUses() []ToolUseBlock {
	var calls []ToolUseBlock
	for _, b := range m.Content {
		if c, ok := b.(ToolUseBlock); ok {
			calls = append(calls, c)
		}
	}
	return calls
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks}
}

type ToolSpec struct {
	Name        string
	Description string
	Parameters  map[string]any
}

type Delta struct {
	Kind string
	Text string
	Tool *ToolUseBlock
}

type Model interface {
	Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error)
}

type wireFunction struct {
	Name      string `json:"name"`
	Arguments string `json:"arguments"`
}

type wireToolCall struct {
	Index    int          `json:"index,omitempty"`
	ID       string       `json:"id,omitempty"`
	Type     string       `json:"type,omitempty"`
	Function wireFunction `json:"function"`
}

type wireMessage struct {
	Role       string         `json:"role"`
	Content    *string        `json:"content"`
	ToolCallID string         `json:"tool_call_id,omitempty"`
	ToolCalls  []wireToolCall `json:"tool_calls,omitempty"`
}

type wireToolDef struct {
	Type     string `json:"type"`
	Function struct {
		Name        string         `json:"name"`
		Description string         `json:"description"`
		Parameters  map[string]any `json:"parameters"`
	} `json:"function"`
}

type wireRequest struct {
	Model    string        `json:"model"`
	Messages []wireMessage `json:"messages"`
	Tools    []wireToolDef `json:"tools,omitempty"`
	Stream   bool          `json:"stream,omitempty"`
}

type wireChoice struct {
	Delta wireMessage `json:"delta"`
}

type wireResponse struct {
	Choices []wireChoice `json:"choices"`
}

func toWireMessages(messages []Message) []wireMessage {
	var out []wireMessage
	for _, m := range messages {
		if m.Role == "tool" {
			for _, b := range m.Content {
				if r, ok := b.(ToolResultBlock); ok {
					content := r.Content
					out = append(out, wireMessage{Role: "tool", Content: &content, ToolCallID: r.ToolUseID})
				}
			}
			continue
		}
		text, calls := m.Text(), m.ToolUses()
		wm := wireMessage{Role: m.Role}
		if text != "" {
			wm.Content = &text
		}
		for _, c := range calls {
			args, err := json.Marshal(c.Args)
			if err != nil {
				args = []byte("{}")
			}
			wm.ToolCalls = append(wm.ToolCalls, wireToolCall{ID: c.ID, Type: "function",
				Function: wireFunction{Name: c.Name, Arguments: string(args)}})
		}
		out = append(out, wm)
	}
	return out
}

func toWireTools(specs []ToolSpec) []wireToolDef {
	var out []wireToolDef
	for _, s := range specs {
		var d wireToolDef
		d.Type = "function"
		d.Function.Name = s.Name
		d.Function.Description = s.Description
		if s.Parameters == nil {
			d.Function.Parameters = map[string]any{"type": "object", "properties": map[string]any{}}
		} else {
			d.Function.Parameters = s.Parameters
		}
		out = append(out, d)
	}
	return out
}

func parseArguments(raw string) map[string]any {
	if raw == "" {
		return map[string]any{}
	}
	var args map[string]any
	if err := json.Unmarshal([]byte(raw), &args); err != nil {
		return map[string]any{"_invalid_json": raw}
	}
	return args
}

type toolCallSlot struct{ id, name, arguments string }

type toolCallAccumulator struct{ slots map[int]*toolCallSlot }

func newToolCallAccumulator() *toolCallAccumulator {
	return &toolCallAccumulator{slots: map[int]*toolCallSlot{}}
}

func (a *toolCallAccumulator) Feed(calls []wireToolCall) {
	for _, item := range calls {
		slot, ok := a.slots[item.Index]
		if !ok {
			slot = &toolCallSlot{}
			a.slots[item.Index] = slot
		}
		if item.ID != "" {
			slot.id = item.ID
		}
		if item.Function.Name != "" {
			slot.name = item.Function.Name
		}
		if item.Function.Arguments != "" {
			slot.arguments += item.Function.Arguments
		}
	}
}

func (a *toolCallAccumulator) Finish() []ToolUseBlock {
	indexes := make([]int, 0, len(a.slots))
	for i := range a.slots {
		indexes = append(indexes, i)
	}
	sort.Ints(indexes)
	var blocks []ToolUseBlock
	for _, i := range indexes {
		slot := a.slots[i]
		id := slot.id
		if id == "" {
			id = newID("call")
		}
		blocks = append(blocks, ToolUseBlock{ID: id, Name: slot.name, Args: parseArguments(slot.arguments)})
	}
	return blocks
}

func parseSSELine(line string) (wireResponse, bool) {
	line = strings.TrimSpace(line)
	if line == "" || strings.HasPrefix(line, ":") || !strings.HasPrefix(line, "data:") {
		return wireResponse{}, false
	}
	payload := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
	if payload == "[DONE]" {
		return wireResponse{}, false
	}
	var parsed wireResponse
	if err := json.Unmarshal([]byte(payload), &parsed); err != nil {
		return wireResponse{}, false
	}
	return parsed, true
}

type OpenAICompatModel struct {
	BaseURL string
	ModelID string
	Timeout time.Duration
}

func NewOpenAICompatModel(baseURL, modelID string) *OpenAICompatModel {
	return &OpenAICompatModel{BaseURL: baseURL, ModelID: modelID, Timeout: 180 * time.Second}
}

func (m *OpenAICompatModel) Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta, 32)
	errs := make(chan error, 1)
	go func() {
		defer close(deltas)
		defer close(errs)
		body := wireRequest{Model: m.ModelID, Messages: toWireMessages(messages),
			Tools: toWireTools(tools), Stream: true}
		raw, err := json.Marshal(body)
		if err != nil {
			errs <- fmt.Errorf("请求体序列化失败：%w", err)
			return
		}
		req, err := http.NewRequestWithContext(ctx, "POST", m.BaseURL+"/chat/completions", bytes.NewReader(raw))
		if err != nil {
			errs <- fmt.Errorf("构造请求失败：%w", err)
			return
		}
		req.Header.Set("Content-Type", "application/json")
		resp, err := (&http.Client{Timeout: m.Timeout}).Do(req)
		if err != nil {
			errs <- fmt.Errorf("请求失败：%w", err)
			return
		}
		defer resp.Body.Close()
		if resp.StatusCode != http.StatusOK {
			buf := make([]byte, 300)
			n, _ := resp.Body.Read(buf)
			errs <- fmt.Errorf("模型服务返回 %d：%s", resp.StatusCode, string(buf[:n]))
			return
		}
		acc := newToolCallAccumulator()
		scanner := bufio.NewScanner(resp.Body)
		scanner.Buffer(make([]byte, 0, 64*1024), 4*1024*1024)
		for scanner.Scan() {
			chunk, ok := parseSSELine(scanner.Text())
			if !ok || len(chunk.Choices) == 0 {
				continue
			}
			d := chunk.Choices[0].Delta
			if d.Content != nil && *d.Content != "" {
				deltas <- Delta{Kind: "text", Text: *d.Content}
			}
			if len(d.ToolCalls) > 0 {
				acc.Feed(d.ToolCalls)
			}
		}
		if err := scanner.Err(); err != nil {
			errs <- fmt.Errorf("读流中断：%w", err)
			return
		}
		for _, block := range acc.Finish() {
			b := block
			deltas <- Delta{Kind: "tool_use", Tool: &b}
		}
		errs <- nil
	}()
	return deltas, errs
}

type ToolResult struct {
	Content string
	IsError bool
}

type Tool interface {
	Spec() ToolSpec
	Run(ctx context.Context, args map[string]any) (string, error)
}

type FuncTool struct {
	spec ToolSpec
	fn   func(ctx context.Context, args map[string]any) (string, error)
}

func NewFuncTool(name, description string, parameters map[string]any,
	fn func(ctx context.Context, args map[string]any) (string, error)) *FuncTool {
	return &FuncTool{spec: ToolSpec{Name: name, Description: description, Parameters: parameters}, fn: fn}
}

func (t *FuncTool) Spec() ToolSpec { return t.spec }

func (t *FuncTool) Run(ctx context.Context, args map[string]any) (string, error) {
	return t.fn(ctx, args)
}

func argString(args map[string]any, key string) string {
	s, _ := args[key].(string)
	return s
}

func argInt(args map[string]any, key string, fallback int) int {
	if v, ok := args[key].(float64); ok {
		return int(v)
	}
	return fallback
}

type ToolRegistry struct {
	tools map[string]Tool
	order []string
}

func NewToolRegistry(tools ...Tool) *ToolRegistry {
	reg := &ToolRegistry{tools: map[string]Tool{}}
	for _, t := range tools {
		reg.tools[t.Spec().Name] = t
		reg.order = append(reg.order, t.Spec().Name)
	}
	return reg
}

func (reg *ToolRegistry) Names() []string { return reg.order }

func (reg *ToolRegistry) Specs() []ToolSpec {
	var out []ToolSpec
	for _, name := range reg.order {
		out = append(out, reg.tools[name].Spec())
	}
	return out
}

func (reg *ToolRegistry) Execute(ctx context.Context, name string, args map[string]any) (result ToolResult) {
	tool, ok := reg.tools[name]
	if !ok {
		return ToolResult{Content: "没有叫 " + name + " 的工具。可用的：" +
			strings.Join(reg.order, "、"), IsError: true}
	}
	defer func() {
		if r := recover(); r != nil {
			result = ToolResult{Content: fmt.Sprintf("工具崩溃：%v", r), IsError: true}
		}
	}()
	output, err := tool.Run(ctx, args)
	if err != nil {
		return ToolResult{Content: err.Error(), IsError: true}
	}
	return ToolResult{Content: output}
}

type Agent struct {
	Model        Model
	Registry     *ToolRegistry
	SystemPrompt string
	MaxRounds    int
	OnEvent      func(name string, args map[string]any, result ToolResult)
}

func (a *Agent) Run(ctx context.Context, userText string) (string, error) {
	history := []Message{
		NewMessage("system", TextBlock{Text: a.SystemPrompt}),
		NewMessage("user", TextBlock{Text: userText}),
	}
	for round := 0; round < a.MaxRounds; round++ {
		reply, err := a.oneRound(ctx, history)
		if err != nil {
			return "", err
		}
		history = append(history, reply)
		calls := reply.ToolUses()
		if len(calls) == 0 {
			return reply.Text(), nil
		}
		var results []Block
		for _, c := range calls {
			r := a.Registry.Execute(ctx, c.Name, c.Args)
			if a.OnEvent != nil {
				a.OnEvent(c.Name, c.Args, r)
			}
			results = append(results, ToolResultBlock{
				ToolUseID: c.ID, Content: r.Content, IsError: r.IsError})
		}
		history = append(history, NewMessage("tool", results...))
	}
	return "（轮数用完了）", nil
}

func (a *Agent) oneRound(ctx context.Context, messages []Message) (Message, error) {
	deltas, errs := a.Model.Stream(ctx, messages, a.Registry.Specs())
	var textParts []string
	var blocks []Block
	flushText := func() {
		if len(textParts) > 0 {
			joined := ""
			for _, p := range textParts {
				joined += p
			}
			blocks = append(blocks, TextBlock{Text: joined})
			textParts = nil
		}
	}
	for d := range deltas {
		switch d.Kind {
		case "text":
			textParts = append(textParts, d.Text)
		case "tool_use":
			if d.Tool != nil {
				flushText()
				blocks = append(blocks, *d.Tool)
			}
		}
	}
	flushText()
	if err := <-errs; err != nil {
		return Message{}, err
	}
	return NewMessage("assistant", blocks...), nil
}
```

## 先把路径管住

所有文件工具的第一件事，都是**把模型给的路径钉在工作目录里**。
模型会写出 `../../etc/passwd`，不是因为它有恶意，而是因为它在猜项目结构。

```go
// file: paths.go
package main

import (
	"errors"
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

// PathError 路径不合法。单独一个类型，因为它的处置和别的错误不同——
// 这是要拦住的越界，不是可以重试的失败。
type PathError struct {
	Raw    string
	Reason string
}

func (e *PathError) Error() string { return fmt.Sprintf("%s：%s", e.Reason, e.Raw) }

// evalExisting 展开符号链接。
//
// EvalSymlinks 要求路径**存在**，而我们经常要处理还没创建的文件。
// 办法是往上找到最近一个存在的祖先，展开它，再把剩下那截拼回去。
// 这一步不能省：工作目录里放一个指向 /etc 的符号链接，
// 光看字符串它还在工作目录里。
func evalExisting(path string) (string, error) {
	remainder := ""
	current := path
	for {
		resolved, err := filepath.EvalSymlinks(current)
		if err == nil {
			return filepath.Join(resolved, remainder), nil
		}
		if !errors.Is(err, os.ErrNotExist) {
			return "", err
		}
		parent := filepath.Dir(current)
		if parent == current {
			return "", fmt.Errorf("找不到 %s 的任何一层父目录", path)
		}
		remainder = filepath.Join(filepath.Base(current), remainder)
		current = parent
	}
}

// within p 是不是在 root 里面。
//
// 用 filepath.Rel 而不是 strings.HasPrefix：
// /tmp/workspace-evil 是以 /tmp/workspace 开头的，前缀判断会放它过去。
func within(root, p string) bool {
	rel, err := filepath.Rel(root, p)
	if err != nil {
		return false
	}
	return rel != ".." && !strings.HasPrefix(rel, ".."+string(filepath.Separator))
}

// ResolveInWorkspace 把模型给的路径解析成一个确定在工作目录里的绝对路径。
func ResolveInWorkspace(raw string) (string, error) {
	if strings.TrimSpace(raw) == "" {
		return "", &PathError{Raw: raw, Reason: "路径不能为空"}
	}
	// 绝对路径要单独挡：filepath.Join(root, "/etc/passwd") 得到的是
	// root/etc/passwd，看起来安全，但模型的意图明显是越界，直接拒绝更清楚。
	if filepath.IsAbs(raw) {
		return "", &PathError{Raw: raw, Reason: "只接受相对路径"}
	}

	rootResolved, err := filepath.EvalSymlinks(workspaceRoot)
	if err != nil {
		return "", err
	}
	resolved, err := evalExisting(filepath.Join(rootResolved, raw))
	if err != nil {
		return "", err
	}
	if !within(rootResolved, resolved) {
		return "", &PathError{Raw: raw, Reason: "路径越出了工作目录"}
	}
	return resolved, nil
}

// Display 把绝对路径变回给模型看的相对路径。
func Display(abs string) string {
	rootResolved, err := filepath.EvalSymlinks(workspaceRoot)
	if err != nil {
		return abs
	}
	rel, err := filepath.Rel(rootResolved, abs)
	if err != nil {
		return abs
	}
	return rel
}
```

```go
// file: main.go
package main

import "fmt"

func main() {
	if err := SetupWorkspace(); err != nil {
		fmt.Println("准备工作目录失败：", err)
		return
	}
	for _, raw := range []string{"app/greet.go", "./README.md", "../../etc/passwd",
		"/etc/passwd", "app/../README.md"} {
		resolved, err := ResolveInWorkspace(raw)
		if err != nil {
			fmt.Printf("  ✗ %-20s → %v\n", raw, err)
			continue
		}
		fmt.Printf("  ✓ %-20s → %s\n", raw, Display(resolved))
	}
}
```

## 读文件：带行号，能读一段

```go
// file: readfile.go
package main

import (
	"fmt"
	"os"
	"strings"
)

const (
	maxReadLines = 400 // 一次最多给模型这么多行
	maxLineChars = 300 // 单行超长就截断（压缩过的前端代码能有几万字符一行）
)

// ReadFile 读文件，返回带行号的内容。
//
// 行号不是为了好看：模型要用它来说「第 12 行有问题」，
// 也要用它来决定下一步读哪一段。没有行号，它只能整文件重读。
func ReadFile(path string, startLine, endLine int) (string, error) {
	target, err := ResolveInWorkspace(path)
	if err != nil {
		return "", err
	}
	info, err := os.Stat(target)
	if err != nil {
		// 报错要给出下一步能怎么办。「文件不存在」是死路，
		// 「文件不存在，同目录下有这些」是线索。
		hint := ""
		if entries, e := os.ReadDir(parentOf(target)); e == nil {
			var names []string
			for i, entry := range entries {
				if i >= 10 {
					break
				}
				names = append(names, entry.Name())
			}
			if len(names) > 0 {
				hint = "，同目录下有：" + strings.Join(names, "、")
			}
		}
		return "", fmt.Errorf("%s 不存在%s", path, hint)
	}
	if info.IsDir() {
		return "", fmt.Errorf("%s 是目录，不是文件。列目录请用 list_dir", path)
	}

	raw, err := os.ReadFile(target)
	if err != nil {
		return "", err
	}
	lines := strings.Split(strings.TrimSuffix(string(raw), "\n"), "\n")
	total := len(lines)

	start := startLine
	if start < 1 {
		start = 1
	}
	end := endLine
	if end <= 0 || end > total {
		end = total
	}
	if start > total {
		return fmt.Sprintf("（%s 只有 %d 行，第 %d 行之后没有内容）", path, total, start), nil
	}

	chunk := lines[start-1 : end]
	truncatedTail := false
	if len(chunk) > maxReadLines {
		chunk = chunk[:maxReadLines]
		end = start + maxReadLines - 1
		truncatedTail = true
	}

	width := len(fmt.Sprint(end))
	var body strings.Builder
	for i, line := range chunk {
		shown := line
		suffix := ""
		if r := []rune(line); len(r) > maxLineChars {
			shown = string(r[:maxLineChars])
			suffix = "…（本行已截断）"
		}
		body.WriteString(fmt.Sprintf("%*d│%s%s\n", width, start+i, shown, suffix))
	}

	out := fmt.Sprintf("%s（共 %d 行，显示 %d-%d）\n%s", path, total, start, end,
		strings.TrimSuffix(body.String(), "\n"))
	if truncatedTail || end < total {
		out += fmt.Sprintf("\n…还有 %d 行没显示，接着读请传 start_line=%d", total-end, end+1)
	}
	return out, nil
}

func parentOf(path string) string {
	if i := strings.LastIndex(path, "/"); i > 0 {
		return path[:i]
	}
	return "/"
}
```

## 写文件：覆盖前必须先读过

这是这一章最重要的一条规矩。

模型很爱干一件事：你让它「在文件里加个函数」，它直接 `write_file` 写一份
**它想象中的完整文件**——把它没读过的部分全删了。
防住这件事只有一个可靠办法：**记住哪些文件被读过，没读过就不许覆盖**。

```go
// file: writefile.go
package main

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"sync"
	"time"
)

// ReadTracker 记住哪些文件被读过，以及读的时候是什么样子。
//
// 这不是「安全机制」，是「防止模型自信地覆盖它没看过的内容」。
// 两者的区别在于：安全机制防的是恶意，这个防的是自信。
//
// 加锁是因为第 11 章的并发执行——两个工具可能同时在跑。
type ReadTracker struct {
	mu   sync.Mutex
	seen map[string]time.Time
}

func NewReadTracker() *ReadTracker { return &ReadTracker{seen: map[string]time.Time{}} }

func (t *ReadTracker) MarkRead(path string) {
	t.mu.Lock()
	defer t.mu.Unlock()
	if info, err := os.Stat(path); err == nil {
		t.seen[path] = info.ModTime()
	} else {
		t.seen[path] = time.Time{}
	}
}

func (t *ReadTracker) CheckCanOverwrite(path, display string) error {
	info, err := os.Stat(path)
	if err != nil {
		return nil // 新建文件不需要先读
	}
	t.mu.Lock()
	seenAt, known := t.seen[path]
	t.mu.Unlock()

	if !known {
		return fmt.Errorf("%s 已经存在，但你还没读过它。请先用 read_file 读一遍，"+
			"确认要保留哪些内容再写。", display)
	}
	if info.ModTime().After(seenAt) {
		// 别人改过了。直接覆盖会把那个人的改动无声地抹掉。
		return fmt.Errorf("%s 在你读过之后被改动过。请重新 read_file 再写。", display)
	}
	return nil
}

// WriteFile 整文件覆盖。新建文件用它；改已有文件优先用 EditFile。
func WriteFile(tracker *ReadTracker, path, content string) (string, error) {
	target, err := ResolveInWorkspace(path)
	if err != nil {
		return "", err
	}
	if err := tracker.CheckCanOverwrite(target, path); err != nil {
		return "", err
	}
	if err := os.MkdirAll(filepath.Dir(target), 0o755); err != nil {
		return "", err
	}

	oldLines := 0
	existed := false
	if raw, err := os.ReadFile(target); err == nil {
		existed = true
		oldLines = len(strings.Split(strings.TrimSuffix(string(raw), "\n"), "\n"))
	}
	if err := os.WriteFile(target, []byte(content), 0o644); err != nil {
		return "", err
	}
	tracker.MarkRead(target) // 刚写完就是最新的，不必再读一遍

	newLines := len(strings.Split(strings.TrimSuffix(content, "\n"), "\n"))
	if existed {
		return fmt.Sprintf("已覆盖 %s：%d 行 → %d 行", path, oldLines, newLines), nil
	}
	return fmt.Sprintf("已新建 %s：%d 行", path, newLines), nil
}
```

## 改文件：按原文片段，不按行号

```go
// file: editfile.go
package main

import (
	"fmt"
	"os"
	"strings"
)

// EditFile 把 oldText 换成 newText。
//
// 为什么按原文片段而不按行号：**行号会失效**。
// 模型读文件时看到第 12 行，改完第 5 行之后，原来的第 12 行已经变成第 13 行了。
// 多改几处，行号全乱。原文片段没有这个问题——它锚定的是内容本身。
//
// 为什么要求唯一：`return nil` 在一个文件里可能出现二十次。
// 替换错地方比不替换糟糕得多，所以数量对不上就拒绝，让模型带更多上下文重来。
func EditFile(tracker *ReadTracker, path, oldText, newText string, expectCount int) (string, error) {
	target, err := ResolveInWorkspace(path)
	if err != nil {
		return "", err
	}
	if oldText == "" {
		return "", fmt.Errorf("old_text 不能为空。要整文件覆盖请用 write_file")
	}
	raw, err := os.ReadFile(target)
	if err != nil {
		return "", fmt.Errorf("%s 读不了：%w", path, err)
	}
	content := string(raw)
	count := strings.Count(content, oldText)

	if count == 0 {
		return "", fmt.Errorf("在 %s 里找不到这段内容。请先 read_file 确认原文，"+
			"注意缩进和空格必须一模一样。", path)
	}
	if expectCount <= 0 {
		expectCount = 1
	}
	if count != expectCount {
		return "", fmt.Errorf("这段内容在 %s 里出现了 %d 次，不是期望的 %d 次。"+
			"请带上前后几行让它唯一，或者把 expect_count 改成 %d。", path, count, expectCount, count)
	}

	if err := os.WriteFile(target, []byte(strings.ReplaceAll(content, oldText, newText)), 0o644); err != nil {
		return "", err
	}
	tracker.MarkRead(target)

	// 返回改动的样子，让模型（和人）能确认改对了
	after := "（删除）"
	if newText != "" {
		after = firstLine(newText)
	}
	return fmt.Sprintf("已修改 %s（替换 %d 处）\n- %s\n+ %s",
		path, count, firstLine(oldText), after), nil
}

func firstLine(s string) string {
	if i := strings.IndexByte(s, '\n'); i >= 0 {
		s = s[:i]
	}
	r := []rune(s)
	if len(r) > 60 {
		return string(r[:60])
	}
	return s
}
```

```go
// file: main.go
package main

import "fmt"

func main() {
	SetupWorkspace()
	tracker := NewReadTracker()

	// 没读过就想覆盖
	if _, err := WriteFile(tracker, "app/greet.go", "package app\n"); err != nil {
		fmt.Println("✗ 拦住了：", err)
	}

	// 读过之后再改
	out, _ := ReadFile("app/greet.go", 1, 0)
	target, _ := ResolveInWorkspace("app/greet.go")
	tracker.MarkRead(target)
	_ = out

	msg, err := EditFile(tracker, "app/greet.go", `"Hello, " + name`, `"你好，" + name`, 1)
	if err != nil {
		fmt.Println("改失败：", err)
		return
	}
	fmt.Println(msg)

	// 不唯一的替换
	if _, err := EditFile(tracker, "app/greet.go", "\n", "", 1); err != nil {
		fmt.Println("\n✗ 拦住了：", err)
	}

	out, _ = ReadFile("app/greet.go", 1, 5)
	fmt.Println()
	fmt.Println(out)
}
```

## 列目录：有上限

```go
// file: listdir.go
package main

import (
	"fmt"
	"os"
	"sort"
	"strings"
)

var skipDirs = map[string]bool{".git": true, "node_modules": true, "__pycache__": true,
	".venv": true, "venv": true, "dist": true, "build": true, ".idea": true}

const maxEntries = 200

// ListDir 列出目录树。
//
// 两个上限都是必须的：
//   - 跳过 .git、node_modules 之类：一个 node_modules 有几万个文件，
//     列出来就把上下文吃光了，而且对模型毫无用处。
//   - 条目总数封顶：防住「上面那个清单漏了某个巨大目录」这种情况。
func ListDir(path string, maxDepth int) (string, error) {
	if path == "" || path == "." {
		path = "."
	}
	if maxDepth <= 0 {
		maxDepth = 2
	}
	target, err := ResolveInWorkspace(path)
	if err != nil {
		return "", err
	}
	info, err := os.Stat(target)
	if err != nil || !info.IsDir() {
		return "", fmt.Errorf("%s 不是目录", path)
	}

	var lines []string
	skipped := map[string]bool{}
	truncated := false

	var walk func(dir string, depth int, prefix string)
	walk = func(dir string, depth int, prefix string) {
		if depth > maxDepth || truncated {
			return
		}
		entries, err := os.ReadDir(dir)
		if err != nil {
			lines = append(lines, prefix+"（读不了这个目录）")
			return
		}
		// 目录在前，文件在后，各自按名字排
		sort.Slice(entries, func(i, j int) bool {
			if entries[i].IsDir() != entries[j].IsDir() {
				return entries[i].IsDir()
			}
			return entries[i].Name() < entries[j].Name()
		})
		for _, entry := range entries {
			if len(lines) >= maxEntries {
				truncated = true
				return
			}
			if entry.IsDir() {
				if skipDirs[entry.Name()] {
					skipped[entry.Name()] = true
					continue
				}
				lines = append(lines, prefix+entry.Name()+"/")
				walk(dir+"/"+entry.Name(), depth+1, prefix+"  ")
				continue
			}
			size := int64(0)
			if info, err := entry.Info(); err == nil {
				size = info.Size()
			}
			unit := fmt.Sprintf("%dB", size)
			if size >= 1024 {
				unit = fmt.Sprintf("%dKB", size/1024)
			}
			lines = append(lines, fmt.Sprintf("%s%s  (%s)", prefix, entry.Name(), unit))
		}
	}
	walk(target, 1, "")

	out := strings.Join(lines, "\n")
	if out == "" {
		out = "（空目录）"
	}
	if len(skipped) > 0 {
		var names []string
		for name := range skipped {
			names = append(names, name)
		}
		sort.Strings(names)
		out += "\n（跳过了 " + strings.Join(names, "、") + "）"
	}
	if truncated {
		out += fmt.Sprintf("\n（超过 %d 项，只显示了前面这些）", maxEntries)
	}
	return out, nil
}
```

## 装成工具

```go
// file: tools.go
package main

import "context"

// MakeCodingTools 把上面四个函数包成工具。
//
// 读文件的登记放在这一层而不是 ReadFile 里：登记的语义是「模型看过了」，
// 而只有经由工具调用的那次读，模型才真的看见了内容。
// 我们自己在讲义里调 ReadFile 看看内容，不该算模型看过。
func MakeCodingTools(tracker *ReadTracker) []Tool {
	return []Tool{
		NewFuncTool("read_file", "读一个文件，返回带行号的内容。大文件可以只读一段。",
			map[string]any{"type": "object", "properties": map[string]any{
				"path":       map[string]any{"type": "string", "description": "相对于工作目录的路径"},
				"start_line": map[string]any{"type": "integer", "description": "起始行号，从 1 开始"},
				"end_line":   map[string]any{"type": "integer", "description": "结束行号，0 表示到文件末尾"}},
				"required": []string{"path"}},
			func(ctx context.Context, args map[string]any) (string, error) {
				path := argString(args, "path")
				start, end := argInt(args, "start_line", 1), argInt(args, "end_line", 0)
				out, err := ReadFile(path, start, end)
				if err != nil {
					return "", err
				}
				if start <= 1 && end <= 0 {
					if target, e := ResolveInWorkspace(path); e == nil {
						tracker.MarkRead(target) // 只有读了全文才算「看过」
					}
				}
				return out, nil
			}),

		NewFuncTool("write_file", "整文件写入。新建文件用它；修改已有文件请优先用 edit_file。"+
			"覆盖已有文件之前必须先 read_file 读过。",
			map[string]any{"type": "object", "properties": map[string]any{
				"path":    map[string]any{"type": "string"},
				"content": map[string]any{"type": "string", "description": "文件的完整新内容"}},
				"required": []string{"path", "content"}},
			func(ctx context.Context, args map[string]any) (string, error) {
				return WriteFile(tracker, argString(args, "path"), argString(args, "content"))
			}),

		NewFuncTool("edit_file", "把文件里的一段原文替换成新内容。old_text 必须与原文完全一致"+
			"（含缩进），并且在文件里唯一。",
			map[string]any{"type": "object", "properties": map[string]any{
				"path":         map[string]any{"type": "string"},
				"old_text":     map[string]any{"type": "string", "description": "要被替换的原文，必须唯一"},
				"new_text":     map[string]any{"type": "string", "description": "替换成什么，空字符串表示删除"},
				"expect_count": map[string]any{"type": "integer", "description": "期望替换几处，默认 1"}},
				"required": []string{"path", "old_text", "new_text"}},
			func(ctx context.Context, args map[string]any) (string, error) {
				return EditFile(tracker, argString(args, "path"), argString(args, "old_text"),
					argString(args, "new_text"), argInt(args, "expect_count", 1))
			}),

		NewFuncTool("list_dir", "列出目录树。",
			map[string]any{"type": "object", "properties": map[string]any{
				"path":      map[string]any{"type": "string", "description": "相对路径，默认当前目录"},
				"max_depth": map[string]any{"type": "integer", "description": "最大深度，默认 2"}},
				"required": []string{}},
			func(ctx context.Context, args map[string]any) (string, error) {
				return ListDir(argString(args, "path"), argInt(args, "max_depth", 2))
			}),
	}
}

const codingSystemPrompt = `你是一个编码助手，在一个小 Go 项目里工作。

工作方式：
1. 先用 list_dir 看看有什么文件。
2. 用 read_file 读你要改的文件。
3. 用 edit_file 做小改动，用 write_file 新建文件。
4. 改完简短说明你做了什么。

规矩：修改已有文件之前必须先读过它。edit_file 的 old_text 必须和原文一模一样。`
```

## 让它真的改一次代码

```go
// file: main.go
package main

import (
	"context"
	"fmt"
	"strings"
)

func main() {
	if err := SetupWorkspace(); err != nil {
		fmt.Println("准备工作目录失败：", err)
		return
	}
	tracker := NewReadTracker()
	reg := NewToolRegistry(MakeCodingTools(tracker)...)

	agent := &Agent{
		Model:        NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B"),
		Registry:     reg,
		SystemPrompt: codingSystemPrompt,
		MaxRounds:    12,
		OnEvent: func(name string, args map[string]any, r ToolResult) {
			var parts []string
			for k, v := range args {
				text := fmt.Sprint(v)
				if len(text) > 40 {
					text = text[:40] + "…"
				}
				parts = append(parts, k+"="+text)
			}
			mark := "✓"
			if r.IsError {
				mark = "✗"
			}
			fmt.Printf("  %s %s(%s)\n", mark, name, strings.Join(parts, ", "))
			fmt.Printf("      → %s\n", firstLine(r.Content))
		},
	}

	fmt.Println("任务：给 Greet 函数加一个 greeting 参数")
	fmt.Println()
	answer, err := agent.Run(context.Background(),
		"给 app/greet.go 里的 Greet 函数加一个 greeting 参数（string 类型），"+
			"用它替换写死的 \"Hello, \"。Run 函数调用时传 \"Hello, \"。改完把文件读一遍确认。")
	if err != nil {
		fmt.Println("出错：", err)
		return
	}
	fmt.Printf("\n助手：%s\n", answer)

	fmt.Println("\n改完之后的文件：")
	out, _ := ReadFile("app/greet.go", 1, 0)
	fmt.Println(out)
}
```

## 验收：那四条规矩真的生效了吗

```go
// file: main.go
package main

import (
	"context"
	"fmt"
	"strings"
)

func main() {
	if err := SetupWorkspace(); err != nil {
		fmt.Println("准备工作目录失败：", err)
		return
	}
	ctx := context.Background()
	tracker := NewReadTracker()
	reg := NewToolRegistry(MakeCodingTools(tracker)...)

	type check struct {
		label string
		ok    bool
	}
	var checks []check

	// 1. 路径越界被拦住
	r := reg.Execute(ctx, "read_file", map[string]any{"path": "../../etc/passwd"})
	checks = append(checks, check{"路径越界被拦住",
		r.IsError && strings.Contains(r.Content, "越出")})

	// 2. 没读过就覆盖被拦住
	r = reg.Execute(ctx, "write_file", map[string]any{"path": "README.md", "content": "没了"})
	checks = append(checks, check{"没读过就覆盖被拦住",
		r.IsError && strings.Contains(r.Content, "还没读过")})

	// 3. 读过之后可以覆盖
	reg.Execute(ctx, "read_file", map[string]any{"path": "README.md"})
	r = reg.Execute(ctx, "write_file", map[string]any{"path": "README.md",
		"content": "# 示例项目\n\n改过了。\n"})
	checks = append(checks, check{"读过之后可以覆盖", !r.IsError})

	// 4. 不唯一的替换被拒绝
	r = reg.Execute(ctx, "edit_file", map[string]any{"path": "app/greet.go",
		"old_text": "\n", "new_text": ""})
	checks = append(checks, check{"不唯一的替换被拒绝",
		r.IsError && strings.Contains(r.Content, "出现了")})

	// 5. 新建文件不需要先读
	r = reg.Execute(ctx, "write_file", map[string]any{"path": "app/util.go",
		"content": "package app\n\nconst Pi = 3.14\n"})
	checks = append(checks, check{"新建文件不需要先读", !r.IsError})

	passed := 0
	for _, c := range checks {
		mark := "✗"
		if c.ok {
			mark = "✓"
			passed++
		}
		fmt.Printf("  %s %s\n", mark, c.label)
	}
	fmt.Printf("\n%d/%d 条规矩生效\n", passed, len(checks))
}
```

## 本章小结

- 每个文件工具的第一件事是**把路径钉在工作目录里**。三步都不能省：
  拒绝绝对路径、展开符号链接、用 `filepath.Rel` 判断在不在里面
  （`strings.HasPrefix` 会被 `/tmp/workspace-evil` 骗过）。
- `filepath.EvalSymlinks` **要求路径存在**。要处理还没创建的文件，
  就往上找到最近一个存在的祖先，展开它，再把剩下那截拼回去。
- 读文件要**带行号**，模型靠它精确指认位置；要**能读一段**；单行也要有长度上限。
- **覆盖已有文件之前必须先读过**。这不是安全机制，是防止模型自信地重写它没看过的内容。
  顺带还能挡住「你读完之后别人改了」的情况。
- 改文件**按唯一的原文片段，不按行号**。行号在上一次编辑之后就失效了。
  片段不唯一时**拒绝执行**。
- 列目录要跳过 `.git`、`node_modules` 之类，并且**条目总数封顶**。
- 报错要给出**下一步能怎么办**：「文件不存在」是死路，
  「文件不存在，同目录下有这些」是线索。
- 跨格子要用的常量和函数，**放在不会被 main.go 覆盖的文件里**。
