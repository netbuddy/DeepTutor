# 会话树

到现在为止，历史是一条直线：消息一条接一条往后排。但用户会做三件事，直线撑不住：

1. **「这个回答不好，重新生成一个」**——旧回答要留着，用户可能想比较。
2. **「我刚才那句话说错了，改一下」**——改完之后，后面的对话全部作废，但不该消失。
3. **「先按方案 A 往下聊，回头我想试试方案 B」**——两条路都要保留，随时切换。

这三件事的共同形状是**分叉**。历史不是一条线，是**一棵树**：
每条消息记住自己的父亲，会话再记住「当前走到哪个叶子」。

发给模型的历史，就是**从当前叶子回溯到根的那条路径**。

```go
// file: core.go
package main

import (
	"bufio"
	"bytes"
	"context"
	"crypto/rand"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"net/http"

func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)
}

type Block interface{ BlockType() string }

type TextBlock struct{ Text string }

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string
	Name string
	Args map[string]any
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string
	Content   string
	IsError   bool
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

type Message struct {
	ID        string
	Role      string
	Content   []Block
	CreatedAt time.Time
}

func (m Message) Text() string {
	out := ""
	for _, b := range m.Content {
		if t, ok := b.(TextBlock); ok {
			out += t.Text
		}
	}
	return out
}

func (m Message) ToolUses() []ToolUseBlock {
	var calls []ToolUseBlock
	for _, b := range m.Content {
		if c, ok := b.(ToolUseBlock); ok {
			calls = append(calls, c)
		}
	}
	return calls
}

func (m Message) ToolResults() []ToolResultBlock {
	var out []ToolResultBlock
	for _, b := range m.Content {
		if r, ok := b.(ToolResultBlock); ok {
			out = append(out, r)
		}
	}
	return out
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks, CreatedAt: time.Now()}
}

type ToolSpec struct {
	Name        string
	Description string
	Parameters  map[string]any
}

type Delta struct {
	Kind string
	Text string
	Tool *ToolUseBlock
}

type Model interface {
	Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error)
}

type wireFunction struct {
	Name      string `json:"name"`
	Arguments string `json:"arguments"`
}

type wireToolCall struct {
	Index    int          `json:"index,omitempty"`
	ID       string       `json:"id,omitempty"`
	Type     string       `json:"type,omitempty"`
	Function wireFunction `json:"function"`
}

type wireMessage struct {
	Role       string         `json:"role"`
	Content    *string        `json:"content"`
	ToolCallID string         `json:"tool_call_id,omitempty"`
	ToolCalls  []wireToolCall `json:"tool_calls,omitempty"`
}

type wireToolDef struct {
	Type     string `json:"type"`
	Function struct {
		Name        string         `json:"name"`
		Description string         `json:"description"`
		Parameters  map[string]any `json:"parameters"`
	} `json:"function"`
}

type wireRequest struct {
	Model       string        `json:"model"`
	Messages    []wireMessage `json:"messages"`
	Tools       []wireToolDef `json:"tools,omitempty"`
	Stream      bool          `json:"stream,omitempty"`
	Temperature *float64      `json:"temperature,omitempty"`
}

type wireChoice struct {
	Delta wireMessage `json:"delta"`
}

type wireResponse struct {
	Choices []wireChoice `json:"choices"`
}

func toWireMessages(messages []Message) []wireMessage {
	var out []wireMessage
	for _, m := range messages {
		if m.Role == "tool" {
			for _, b := range m.Content {
				if r, ok := b.(ToolResultBlock); ok {
					content := r.Content
					out = append(out, wireMessage{Role: "tool", Content: &content, ToolCallID: r.ToolUseID})
				}
			}
			continue
		}
		text, calls := m.Text(), m.ToolUses()
		wm := wireMessage{Role: m.Role}
		if text != "" {
			wm.Content = &text
		}
		for _, c := range calls {
			args, err := json.Marshal(c.Args)
			if err != nil {
				args = []byte("{}")
			}
			wm.ToolCalls = append(wm.ToolCalls, wireToolCall{ID: c.ID, Type: "function",
				Function: wireFunction{Name: c.Name, Arguments: string(args)}})
		}
		out = append(out, wm)
	}
	return out
}

func toWireTools(specs []ToolSpec) []wireToolDef {
	var out []wireToolDef
	for _, s := range specs {
		var d wireToolDef
		d.Type = "function"
		d.Function.Name = s.Name
		d.Function.Description = s.Description
		if s.Parameters == nil {
			d.Function.Parameters = map[string]any{"type": "object", "properties": map[string]any{}}
		} else {
			d.Function.Parameters = s.Parameters
		}
		out = append(out, d)
	}
	return out
}

func parseArguments(raw string) map[string]any {
	if raw == "" {
		return map[string]any{}
	}
	var args map[string]any
	if err := json.Unmarshal([]byte(raw), &args); err != nil {
		return map[string]any{"_invalid_json": raw}
	}
	return args
}

type toolCallSlot struct{ id, name, arguments string }

type toolCallAccumulator struct{ slots map[int]*toolCallSlot }

func newToolCallAccumulator() *toolCallAccumulator {
	return &toolCallAccumulator{slots: map[int]*toolCallSlot{}}
}

func (a *toolCallAccumulator) Feed(calls []wireToolCall) {
	for _, item := range calls {
		slot, ok := a.slots[item.Index]
		if !ok {
			slot = &toolCallSlot{}
			a.slots[item.Index] = slot
		}
		if item.ID != "" {
			slot.id = item.ID
		}
		if item.Function.Name != "" {
			slot.name = item.Function.Name
		}
		if item.Function.Arguments != "" {
			slot.arguments += item.Function.Arguments
		}
	}
}

func (a *toolCallAccumulator) Finish() []ToolUseBlock {
	indexes := make([]int, 0, len(a.slots))
	for i := range a.slots {
		indexes = append(indexes, i)
	}
	sort.Ints(indexes)
	var blocks []ToolUseBlock
	for _, i := range indexes {
		slot := a.slots[i]
		id := slot.id
		if id == "" {
			id = newID("call")
		}
		blocks = append(blocks, ToolUseBlock{ID: id, Name: slot.name, Args: parseArguments(slot.arguments)})
	}
	return blocks
}

func parseSSELine(line string) (wireResponse, bool) {
	line = strings.TrimSpace(line)
	if line == "" || strings.HasPrefix(line, ":") || !strings.HasPrefix(line, "data:") {
		return wireResponse{}, false
	}
	payload := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
	if payload == "[DONE]" {
		return wireResponse{}, false
	}
	var parsed wireResponse
	if err := json.Unmarshal([]byte(payload), &parsed); err != nil {
		return wireResponse{}, false
	}
	return parsed, true
}

type OpenAICompatModel struct {
	BaseURL     string
	ModelID     string
	Timeout     time.Duration
	Temperature *float64
}

func NewOpenAICompatModel(baseURL, modelID string) *OpenAICompatModel {
	return &OpenAICompatModel{BaseURL: baseURL, ModelID: modelID, Timeout: 180 * time.Second}
}

func (m *OpenAICompatModel) Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta, 32)
	errs := make(chan error, 1)
	go func() {
		defer close(deltas)
		defer close(errs)
		body := wireRequest{Model: m.ModelID, Messages: toWireMessages(messages),
			Tools: toWireTools(tools), Stream: true, Temperature: m.Temperature}
		raw, err := json.Marshal(body)
		if err != nil {
			errs <- fmt.Errorf("请求体序列化失败：%w", err)
			return
		}
		req, err := http.NewRequestWithContext(ctx, "POST", m.BaseURL+"/chat/completions", bytes.NewReader(raw))
		if err != nil {
			errs <- fmt.Errorf("构造请求失败：%w", err)
			return
		}
		req.Header.Set("Content-Type", "application/json")
		resp, err := (&http.Client{Timeout: m.Timeout}).Do(req)
		if err != nil {
			errs <- fmt.Errorf("请求失败：%w", err)
			return
		}
		defer resp.Body.Close()
		if resp.StatusCode != http.StatusOK {
			buf := make([]byte, 300)
			n, _ := resp.Body.Read(buf)
			errs <- fmt.Errorf("模型服务返回 %d：%s", resp.StatusCode, string(buf[:n]))
			return
		}
		acc := newToolCallAccumulator()
		scanner := bufio.NewScanner(resp.Body)
		scanner.Buffer(make([]byte, 0, 64*1024), 4*1024*1024)
		for scanner.Scan() {
			chunk, ok := parseSSELine(scanner.Text())
			if !ok || len(chunk.Choices) == 0 {
				continue
			}
			d := chunk.Choices[0].Delta
			if d.Content != nil && *d.Content != "" {
				deltas <- Delta{Kind: "text", Text: *d.Content}
			}
			if len(d.ToolCalls) > 0 {
				acc.Feed(d.ToolCalls)
			}
		}
		if err := scanner.Err(); err != nil {
			errs <- fmt.Errorf("读流中断：%w", err)
			return
		}
		for _, block := range acc.Finish() {
			b := block
			deltas <- Delta{Kind: "tool_use", Tool: &b}
		}
		errs <- nil
	}()
	return deltas, errs
}

type ToolResult struct {
	Content string
	IsError bool
}

type Tool interface {
	Spec() ToolSpec
	Run(ctx context.Context, args map[string]any) (string, error)
}

type FuncTool struct {
	spec ToolSpec
	fn   func(ctx context.Context, args map[string]any) (string, error)
}

func NewFuncTool(name, description string, parameters map[string]any,
	fn func(ctx context.Context, args map[string]any) (string, error)) *FuncTool {
	return &FuncTool{spec: ToolSpec{Name: name, Description: description, Parameters: parameters}, fn: fn}
}

func (t *FuncTool) Spec() ToolSpec { return t.spec }

func (t *FuncTool) Run(ctx context.Context, args map[string]any) (string, error) {
	return t.fn(ctx, args)
}

type ToolRegistry struct {
	tools map[string]Tool
	order []string
}

func NewToolRegistry(tools ...Tool) *ToolRegistry {
	reg := &ToolRegistry{tools: map[string]Tool{}}
	for _, t := range tools {
		reg.tools[t.Spec().Name] = t
		reg.order = append(reg.order, t.Spec().Name)
	}
	return reg
}

func (reg *ToolRegistry) Specs() []ToolSpec {
	var out []ToolSpec
	for _, name := range reg.order {
		out = append(out, reg.tools[name].Spec())
	}
	return out
}

func (reg *ToolRegistry) Execute(ctx context.Context, name string, args map[string]any) (result ToolResult) {
	tool, ok := reg.tools[name]
	if !ok {
		return ToolResult{Content: "没有叫 " + name + " 的工具", IsError: true}
	}
	defer func() {
		if r := recover(); r != nil {
			result = ToolResult{Content: fmt.Sprintf("工具崩溃：%v", r), IsError: true}
		}
	}()
	output, err := tool.Run(ctx, args)
	if err != nil {
		return ToolResult{Content: err.Error(), IsError: true}
	}
	return ToolResult{Content: output}
}

func NewTimeTool() Tool {
	return NewFuncTool("get_time", "查询某个城市的当前时间",
		map[string]any{"type": "object", "properties": map[string]any{
			"city": map[string]any{"type": "string"}}, "required": []string{"city"}},
		func(ctx context.Context, args map[string]any) (string, error) {
			city, _ := args["city"].(string)
			return fmt.Sprintf("%s 现在是 %s", city, time.Now().Format("15:04")), nil
		})
}
```

## 树

```go
// file: tree.go
package main

import "fmt"

// Node 树上的一个位置。消息本身不变，变的是它挂在哪里。
type Node struct {
	Message  Message
	ParentID string // 空字符串表示这是根
	Children []string
}

// ConversationTree 一棵会话树，外加一个「当前走到哪儿」的指针。
//
// 和直线历史比，多出来的只有两样东西：每个节点记住父亲，会话记住当前叶子。
// 别的复杂度都是从这两样推出来的。
type ConversationTree struct {
	ID     string
	Title  string
	Nodes  map[string]*Node
	Roots  []string
	LeafID string // 当前分支的末端
}

func NewConversationTree(title string) *ConversationTree {
	return &ConversationTree{ID: newID("ses"), Title: title, Nodes: map[string]*Node{}}
}

// Add 挂一条消息上去。parentID 传空串表示「挂在当前叶子后面」。
//
// 这里用 map[string]*Node 而不是 map[string]Node：
// 要往 Children 里追加，就必须能改到 map 里那个值本身。
// Go 的 map 元素**不可寻址**——m[k].Children = ... 是编译错误，
// 存指针是绕开它最直白的办法。
func (t *ConversationTree) Add(message Message, parentID string) (string, error) {
	if parentID == "" {
		parentID = t.LeafID
	}
	if parentID != "" {
		if _, ok := t.Nodes[parentID]; !ok {
			return "", fmt.Errorf("没有这个父节点：%s", parentID)
		}
	}
	t.Nodes[message.ID] = &Node{Message: message, ParentID: parentID}
	if parentID == "" {
		t.Roots = append(t.Roots, message.ID)
	} else {
		parent := t.Nodes[parentID]
		parent.Children = append(parent.Children, message.ID)
	}
	t.LeafID = message.ID
	return message.ID, nil
}

// PathTo 从根到这个节点的那条路径。这就是要发给模型的历史。
func (t *ConversationTree) PathTo(nodeID string) []Message {
	var chain []Message
	for cursor := nodeID; cursor != ""; {
		node, ok := t.Nodes[cursor]
		if !ok {
			break
		}
		chain = append(chain, node.Message)
		cursor = node.ParentID
	}
	// 回溯是从叶子往根走，发给模型要从根往叶子。就地反转。
	for i, j := 0, len(chain)-1; i < j; i, j = i+1, j-1 {
		chain[i], chain[j] = chain[j], chain[i]
	}
	return chain
}

func (t *ConversationTree) CurrentPath() []Message { return t.PathTo(t.LeafID) }

// Siblings 同一个父亲下的所有兄弟，包括自己。
// 「第 2 个回答，共 3 个」就是数这个。
func (t *ConversationTree) Siblings(nodeID string) []string {
	node, ok := t.Nodes[nodeID]
	if !ok {
		return nil
	}
	if node.ParentID == "" {
		return t.Roots
	}
	return t.Nodes[node.ParentID].Children
}

func (t *ConversationTree) SiblingIndex(nodeID string) int {
	for i, id := range t.Siblings(nodeID) {
		if id == nodeID {
			return i
		}
	}
	return -1
}

// DeepestLeafUnder 从某个节点往下一直走到底，每次选最新的那个孩子。
//
// 切换分支时要用：用户点的是分叉点，但他想去的是那条分支的末端。
func (t *ConversationTree) DeepestLeafUnder(nodeID string) string {
	cursor := nodeID
	for {
		node, ok := t.Nodes[cursor]
		if !ok || len(node.Children) == 0 {
			return cursor
		}
		cursor = node.Children[len(node.Children)-1]
	}
}

func (t *ConversationTree) SwitchTo(nodeID string) error {
	if _, ok := t.Nodes[nodeID]; !ok {
		return fmt.Errorf("没有这个节点：%s", nodeID)
	}
	t.LeafID = t.DeepestLeafUnder(nodeID)
	return nil
}
```

## 一条硬规矩：只能在轮的边界分叉

不能随便挑一个节点分叉。**助手发出的工具调用和它的工具结果必须待在一起**——
第 11 章讲过，少一条结果，这段历史就会被服务端拒绝。

所以合法的分叉点只有两种：

- **用户消息**：在它的父亲下面挂一条新的用户消息 = 「改写我的提问」。
- **助手的最终回答**（不带工具调用的那条）：在它的父亲下面重新生成 = 「换一个回答」。

```go
// file: boundary.go
package main

import "fmt"

// IsTurnBoundary 这个节点能不能当分叉点。
func IsTurnBoundary(t *ConversationTree, nodeID string) bool {
	node, ok := t.Nodes[nodeID]
	if !ok {
		return false
	}
	switch node.Message.Role {
	case "user":
		return true
	case "assistant":
		// 最终回答，这一轮到此结束；带工具调用的那条在轮的中间
		return len(node.Message.ToolUses()) == 0
	}
	return false
}

// CheckPairing 检查一条路径上的调用与结果是否成对。这是分叉之后必须过的验收。
func CheckPairing(messages []Message) []string {
	calls := map[string]bool{}
	results := map[string]bool{}
	for _, m := range messages {
		for _, c := range m.ToolUses() {
			calls[c.ID] = true
		}
		for _, r := range m.ToolResults() {
			results[r.ToolUseID] = true
		}
	}
	var problems []string
	for id := range calls {
		if !results[id] {
			problems = append(problems, fmt.Sprintf("调用 %s 没有结果", id))
		}
	}
	for id := range results {
		if !calls[id] {
			problems = append(problems, fmt.Sprintf("结果 %s 没有对应的调用", id))
		}
	}
	return problems
}
```

## 把 Agent 接到树上

```go
// file: agent.go
package main

import (
	"context"
	"fmt"
)

type AgentConfig struct {
	SystemPrompt string
	MaxRounds    int
}

// TreeAgent 在树上跑的 agent。
//
// 和上一章的区别只有一句话：历史不再是 session.Messages，
// 而是 tree.CurrentPath()。别的全都一样。
type TreeAgent struct {
	Model    Model
	Registry *ToolRegistry
	Config   AgentConfig
}

func NewTreeAgent(model Model, reg *ToolRegistry, config AgentConfig) *TreeAgent {
	if config.MaxRounds == 0 {
		config.MaxRounds = 6
	}
	if config.SystemPrompt == "" {
		config.SystemPrompt = "你是一个简洁的助手，回答控制在两句话以内。"
	}
	return &TreeAgent{Model: model, Registry: reg, Config: config}
}

func (a *TreeAgent) prepare(history []Message) []Message {
	out := make([]Message, 0, len(history)+1)
	out = append(out, NewMessage("system", TextBlock{Text: a.Config.SystemPrompt}))
	out = append(out, history...)
	return out
}

// generate 从当前叶子往下生成，把新消息挂上去，返回最终回答的节点标识。
func (a *TreeAgent) generate(ctx context.Context, tree *ConversationTree) (string, error) {
	for round := 0; round < a.Config.MaxRounds; round++ {
		reply, err := a.oneRound(ctx, a.prepare(tree.CurrentPath()))
		if err != nil {
			return "", err
		}
		nodeID, err := tree.Add(reply, "")
		if err != nil {
			return "", err
		}

		calls := reply.ToolUses()
		if len(calls) == 0 {
			return nodeID, nil
		}

		var results []Block
		for _, call := range calls {
			r := a.Registry.Execute(ctx, call.Name, call.Args)
			results = append(results, ToolResultBlock{
				ToolUseID: call.ID, Content: r.Content, IsError: r.IsError})
		}
		if _, err := tree.Add(NewMessage("tool", results...), ""); err != nil {
			return "", err
		}
	}
	return tree.LeafID, nil
}

func (a *TreeAgent) Send(ctx context.Context, tree *ConversationTree, userText string) (string, error) {
	if _, err := tree.Add(NewMessage("user", TextBlock{Text: userText}), ""); err != nil {
		return "", err
	}
	return a.generate(ctx, tree)
}

// Regenerate 换一个回答。做法是：回到它的父亲，从那里重新生成。
//
// 结果是一个**兄弟节点**——旧回答一个字没动，还挂在原处。
// 这就是树相对于「直接覆盖」的全部价值。
func (a *TreeAgent) Regenerate(ctx context.Context, tree *ConversationTree, nodeID string) (string, error) {
	node, ok := tree.Nodes[nodeID]
	if !ok {
		return "", fmt.Errorf("没有这个节点：%s", nodeID)
	}
	if node.Message.Role != "assistant" || len(node.Message.ToolUses()) > 0 {
		return "", fmt.Errorf("只能重新生成助手的最终回答")
	}
	tree.LeafID = node.ParentID
	return a.generate(ctx, tree)
}

// EditAndResend 改写一条用户消息。原消息和它下面那一整条分支都留着。
func (a *TreeAgent) EditAndResend(ctx context.Context, tree *ConversationTree,
	nodeID, newText string) (string, error) {
	node, ok := tree.Nodes[nodeID]
	if !ok {
		return "", fmt.Errorf("没有这个节点：%s", nodeID)
	}
	if node.Message.Role != "user" {
		return "", fmt.Errorf("只能改写用户消息")
	}
	tree.LeafID = node.ParentID
	if _, err := tree.Add(NewMessage("user", TextBlock{Text: newText}), ""); err != nil {
		return "", err
	}
	return a.generate(ctx, tree)
}

func (a *TreeAgent) oneRound(ctx context.Context, messages []Message) (Message, error) {
	deltas, errs := a.Model.Stream(ctx, messages, a.Registry.Specs())
	var textParts []string
	var blocks []Block
	flushText := func() {
		if len(textParts) > 0 {
			joined := ""
			for _, p := range textParts {
				joined += p
			}
			blocks = append(blocks, TextBlock{Text: joined})
			textParts = nil
		}
	}
	for d := range deltas {
		switch d.Kind {
		case "text":
			textParts = append(textParts, d.Text)
		case "tool_use":
			if d.Tool != nil {
				flushText()
				blocks = append(blocks, *d.Tool)
			}
		}
	}
	flushText()
	if err := <-errs; err != nil {
		return Message{}, err
	}
	return NewMessage("assistant", blocks...), nil
}
```

## 画出来

```go
// file: render.go
package main

import (
	"fmt"
	"strings"
)

var roleLabels = map[string]string{"user": "用户", "assistant": "助手", "tool": "工具"}

func summarize(m Message, limit int) string {
	text := strings.ReplaceAll(m.Text(), "\n", " ")
	if text == "" {
		var parts []string
		for _, c := range m.ToolUses() {
			parts = append(parts, "调用 "+c.Name)
		}
		for _, r := range m.ToolResults() {
			parts = append(parts, "结果 "+truncate(r.Content, 14))
		}
		text = strings.Join(parts, "、")
	}
	return truncate(text, limit)
}

func truncate(s string, n int) string {
	r := []rune(s)
	if len(r) <= n {
		return s
	}
	return string(r[:n]) + "…"
}

// Render 把树打出来。当前路径上的节点标一个箭头。
func Render(t *ConversationTree, highlight map[string]bool) {
	// 递归闭包必须先 var 声明再赋值。
	// 写成 walk := func(...){ ... walk(...) } 是编译错误：
	// := 右边求值时 walk 还没定义。这是 Go 里写递归闭包唯一的办法。
	var walk func(nodeID string, depth int)
	walk = func(nodeID string, depth int) {
		node := t.Nodes[nodeID]
		mark := " "
		if highlight[nodeID] {
			mark = "→"
		}
		tag := ""
		if brothers := t.Siblings(nodeID); len(brothers) > 1 {
			tag = fmt.Sprintf(" [%d/%d]", t.SiblingIndex(nodeID)+1, len(brothers))
		}
		fmt.Printf("%s %s%s%s：%s\n", mark, strings.Repeat("  ", depth),
			roleLabels[node.Message.Role], tag, summarize(node.Message, 38))
		for _, child := range node.Children {
			walk(child, depth+1)
		}
	}
	for _, root := range t.Roots {
		walk(root, 0)
	}
}

func pathSet(messages []Message) map[string]bool {
	set := map[string]bool{}
	for _, m := range messages {
		set[m.ID] = true
	}
	return set
}
```

```go
import (
	"context"

func main() {
	temp := 0.9 // 温度调高一点，「重新生成」才看得出差别
	model := NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
	model.Temperature = &temp

	tree := NewConversationTree("树的演示")

	if err != nil {
		fmt.Println("出错：", err)
		return
	}
	fmt.Println("第一次回答：")
	Render(tree, pathSet(tree.CurrentPath()))

	// 重新生成：旧回答还在
	second, err := agent.Regenerate(ctx, tree, first)
	if err != nil {
		fmt.Println("重新生成失败：", err)
		return
	}
	fmt.Println("\n重新生成之后：")
	Render(tree, pathSet(tree.CurrentPath()))

	fmt.Println("\n两个回答都在树上：")
	fmt.Println("  第一个：", truncate(tree.Nodes[first].Message.Text(), 50))
	fmt.Println("  第二个：", truncate(tree.Nodes[second].Message.Text(), 50))
	fmt.Printf("\n当前路径长度：%d 条消息（只包含第 %d 个回答）\n",
		len(tree.CurrentPath()), tree.SiblingIndex(second)+1)
}
```

**这是关键**：树上有两个回答，但**发给模型的路径里只有一个**。
用户能看到两个、能来回切，模型只看到当前这条路径——它不会被另一个版本干扰。

## 切回去，再改写提问

```go
// file: main.go
package main

import (
	"context"
	"fmt"
	"strings"
)

func main() {
	temp := 0.9
	model := NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
	model.Temperature = &temp
	ctx := context.Background()

	tree := NewConversationTree("切换与改写")
	agent := NewTreeAgent(model, NewToolRegistry(NewTimeTool()), AgentConfig{})

	first, _ := agent.Send(ctx, tree, "用一句话介绍一下什么是递归。")
	agent.Regenerate(ctx, tree, first)

	// 切回第一个回答
	tree.SwitchTo(first)
	fmt.Println("切回第一个回答后，当前路径的最后一条是：")
	path := tree.CurrentPath()
	fmt.Println(" ", truncate(path[len(path)-1].Text(), 50))

	// 在第一个回答后面接着聊
	agent.Send(ctx, tree, "那尾递归又是什么？")
	fmt.Println("\n现在的树：")
	Render(tree, pathSet(tree.CurrentPath()))

	// 改写那个提问
	var target string
	for id, node := range tree.Nodes {
		if node.Message.Role == "user" && strings.Contains(node.Message.Text(), "尾递归") {
			target = id
			break
		}
	}
	if target == "" {
		fmt.Println("没找到要改写的提问")
		return
	}
	if _, err := agent.EditAndResend(ctx, tree, target, "那迭代和递归比，各有什么优劣？"); err != nil {
		fmt.Println("改写失败：", err)
		return
	}
	fmt.Println("\n改写提问之后（原问题和它的回答都还在）：")
	Render(tree, pathSet(tree.CurrentPath()))
}
```

注意「那尾递归又是什么」挂在**第一个回答**下面。
如果用户切到第二个回答再接着问，就会长出另一条分支——两条对话各自独立往下走。

## 验收：路径上的调用与结果成对

分叉最容易出的错，就是把一次工具调用和它的结果拆到两条分支上。检查一遍：

```go
// file: main.go
package main

import (
	"context"
	"fmt"
)

func main() {
	ctx := context.Background()
	model := NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
	tree := NewConversationTree("带工具的分叉")

		AgentConfig{SystemPrompt: "需要查时间时调用 get_time 工具。"})

	answer, err := agent.Send(ctx, tree, "北京现在几点？")
	if err != nil {
		fmt.Println("出错：", err)
		return
	}
	again, err := agent.Regenerate(ctx, tree, answer)
	if err != nil {
		fmt.Println("重新生成失败：", err)
		return
	}

	fmt.Println("树的样子：")
	Render(tree, pathSet(tree.CurrentPath()))

	for _, c := range []struct {
		label string
		leaf  string
	}{{"第一条分支", answer}, {"第二条分支", again}} {
		path := tree.PathTo(c.leaf)
		calls := 0
		for _, m := range path {
			calls += len(m.ToolUses())
		}
		problems := CheckPairing(path)
		fmt.Printf("\n%s：%d 条消息，%d 次工具调用\n", c.label, len(path), calls)
		if len(problems) == 0 {
			fmt.Println("  成对检查：全部成对 ✓")
		} else {
			fmt.Println("  成对检查：", problems)
		}
	}

	fmt.Println("\n这棵树上哪些节点可以当分叉点：")
	for _, m := range tree.PathTo(again) {
		verdict := "不行（在轮的中间）"
		if IsTurnBoundary(tree, m.ID) {
			verdict = "可以"
		}
		fmt.Printf("  %-10s %-18s %s\n", m.Role, verdict, summarize(m, 20))
	}
}
```

两条分支各自完整。它们**共享**分叉点之前的那段历史（包括工具调用与结果），
分叉之后各走各的——所以两边都是成对的。

## 存到磁盘上要改什么

几乎不用改。上一章那个追加式日志，每条消息多存一个 `parent_id` 就够了：

```go
// file: main.go
package main

import (
	"context"
	"encoding/json"
	"fmt"
)

func main() {
	ctx := context.Background()
	model := NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
	tree := NewConversationTree("存储演示")
	agent := NewTreeAgent(model, NewToolRegistry(NewTimeTool()), AgentConfig{})

	first, err := agent.Send(ctx, tree, "一句话说明什么是幂等。")
	if err != nil {
		fmt.Println("出错：", err)
		return
	}
	agent.Regenerate(ctx, tree, first)

	type record struct {
		Kind     string `json:"kind"`
		ID       string `json:"id"`
		ParentID string `json:"parent_id"`
		Role     string `json:"role"`
	}

	fmt.Println("每条消息在文件里多存一个 parent_id：")
	for _, m := range tree.PathTo(tree.LeafID) {
		node := tree.Nodes[m.ID]
		raw, _ := json.Marshal(record{Kind: "message", ID: m.ID,
			ParentID: node.ParentID, Role: m.Role})
		fmt.Println("  ", string(raw))
	}
	fmt.Println("\n分支不需要额外的文件：树的形状全在 parent_id 里，读的时候重建就行。")
	fmt.Printf("另外要存的只有一个数：当前叶子 = %s\n", tree.LeafID)
}
```

**追加式日志天然适合树**：分叉不修改任何旧记录，只是新增一条父亲指向别处的记录。
如果当初选的是「每次覆写整个会话」，这里就要重写整个文件。

## 本章小结

- 历史不是一条线，是**一棵树**。多出来的只有两样东西：
  每个节点记住父亲，会话记住当前叶子。
- 发给模型的历史 = **从当前叶子回溯到根的路径**（记得反转）。
- 节点表用 `map[string]*Node` 而不是 `map[string]Node`：
  **Go 的 map 元素不可寻址**，存值的话没法往 `Children` 里追加。
- **递归闭包必须先 `var walk func(...)` 再赋值**，`:=` 那一行右边求值时 `walk` 还不存在。
- **只能在轮的边界分叉**：用户消息，或助手的最终回答。
  在带工具调用的消息中间分叉，会把调用和结果拆散，这段历史就废了。
- 「重新生成」= 回到父亲重新生成，结果是一个**兄弟节点**；旧回答一个字没动。
  **这是树相对于直接覆盖的全部价值。**
- 用户能看到全部分支，**模型只看到当前路径**。
- 切换分支时，用户点的是分叉点，但要去的是那条分支的**末端**。
- 存储几乎不用改：每条消息多存一个 `parent_id`，外加一个「当前叶子」。
  **追加式日志天然适合树**，因为分叉不修改任何旧记录。
