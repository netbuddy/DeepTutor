# 供应商适配

前五章一直在用假模型。这一章接上真的。

真模型的接口是别人定的，跟我们第 2 章定的内部协议长得不一样。所以需要一层**适配器**：
把内部协议翻译成对方要的格式，再把对方的回答翻译回内部协议。

这一章只做非流式（一次请求，等它把话说完），流式留到下一章。

## 为什么只对接一家

pi 的 `pi-ai` 包有两万多行，绝大部分在处理各家供应商的差异——Anthropic 的
`tool_use` 块、OpenAI 的 `tool_calls` 数组、Gemini 的 `functionCall`，还有各家不同的
流式分帧和错误码。这些差异对**理解 agent 原理毫无帮助**。

所以这门课只对接一种：**OpenAI 兼容接口**。它是事实上的通用格式，本地模型和大多数
云服务都提供它。

## 两边的格式差在哪

| 我们的内部协议 | OpenAI 兼容接口 |
|---|---|
| `Message{Role, Content: []Block}` | `{"role": …, "content": "字符串"}` |
| `ToolUseBlock{ID, Name, Args}` | 助手消息里的 `tool_calls: [{id, function:{name, arguments}}]` |
| `ToolResultBlock{ToolUseID, Content}` | 单独一条 `{"role":"tool", "tool_call_id": …, "content": …}` |

三个要留意的地方：

1. **参数是字符串不是对象。** `function.arguments` 是一段 JSON **文本**，要自己解析。
2. **工具结果是独立的消息。** 我们把一轮里的多个结果放在一条 `tool` 消息里，
   发过去时要**拆成多条**。
3. **一条助手消息可能既有文字又有工具调用**，翻译时两边都要带上。

## 本章起点

```go
// file: blocks.go
package main

import (
	"crypto/rand"
	"encoding/hex"
)

func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)
}

type Block interface{ BlockType() string }

type TextBlock struct{ Text string }

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string
	Name string
	Args map[string]any
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string
	Content   string
	IsError   bool
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

type Message struct {
	ID      string
	Role    string
	Content []Block
}

func (m Message) Text() string {
	out := ""
	for _, b := range m.Content {
		if t, ok := b.(TextBlock); ok {
			out += t.Text
		}
	}
	return out
}

func (m Message) ToolUses() []ToolUseBlock {
	var calls []ToolUseBlock
	for _, b := range m.Content {
		if c, ok := b.(ToolUseBlock); ok {
			calls = append(calls, c)
		}
	}
	return calls
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks}
}

func NewToolUse(name string, args map[string]any) ToolUseBlock {
	return ToolUseBlock{ID: newID("call"), Name: name, Args: args}
}

// ToolSpec 一个工具的声明。内部用扁平结构，发出去时再套接口要的壳。
type ToolSpec struct {
	Name        string
	Description string
	Parameters  map[string]any
}
```

## 接口那边的数据结构

Go 要先把对方的 JSON 结构声明出来。这些类型只在适配层用，不外泄。

```go
// file: wire.go
package main

// 下面这些类型对应 OpenAI 兼容接口的请求与响应。
// 字段上的 omitempty 很重要：content 为空时要么不传、要么传 null，
// 传空字符串有些服务端会当成"模型说了一句空话"。

type wireFunction struct {
	Name      string `json:"name"`
	Arguments string `json:"arguments"` // 注意是字符串，不是对象
}

type wireToolCall struct {
	Index    int          `json:"index,omitempty"`
	ID       string       `json:"id,omitempty"`
	Type     string       `json:"type,omitempty"`
	Function wireFunction `json:"function"`
}

type wireMessage struct {
	Role       string         `json:"role"`
	Content    *string        `json:"content"` // 指针：为 nil 时序列化成 null
	ToolCallID string         `json:"tool_call_id,omitempty"`
	ToolCalls  []wireToolCall `json:"tool_calls,omitempty"`
}

type wireToolDef struct {
	Type     string `json:"type"`
	Function struct {
		Name        string         `json:"name"`
		Description string         `json:"description"`
		Parameters  map[string]any `json:"parameters"`
	} `json:"function"`
}

type wireRequest struct {
	Model    string        `json:"model"`
	Messages []wireMessage `json:"messages"`
	Tools    []wireToolDef `json:"tools,omitempty"`
	Stream   bool          `json:"stream,omitempty"`
}

type wireChoice struct {
	Message      wireMessage `json:"message"`
	Delta        wireMessage `json:"delta"`
	FinishReason string      `json:"finish_reason"`
}

type wireResponse struct {
	Choices []wireChoice `json:"choices"`
}
```

## 出去的方向：内部协议 → 接口格式

```go
// file: encode.go
package main

import "encoding/json"

func toWireMessages(messages []Message) []wireMessage {
	var out []wireMessage
	for _, m := range messages {
		// 工具结果：我们把一轮里的多个结果放在一条 tool 消息里，
		// 接口要求一个结果一条消息，所以这里是一对多的展开。
		if m.Role == "tool" {
			for _, b := range m.Content {
				if r, ok := b.(ToolResultBlock); ok {
					content := r.Content
					out = append(out, wireMessage{
						Role:       "tool",
						Content:    &content,
						ToolCallID: r.ToolUseID,
					})
				}
			}
			continue
		}

		text := m.Text()
		calls := m.ToolUses()

		wm := wireMessage{Role: m.Role}
		if text != "" {
			wm.Content = &text
		}
		for _, c := range calls {
			args, err := json.Marshal(c.Args)
			if err != nil {
				args = []byte("{}")
			}
			wm.ToolCalls = append(wm.ToolCalls, wireToolCall{
				ID:   c.ID,
				Type: "function",
				Function: wireFunction{Name: c.Name, Arguments: string(args)},
			})
		}
		out = append(out, wm)
	}
	return out
}

func toWireTools(specs []ToolSpec) []wireToolDef {
	var out []wireToolDef
	for _, s := range specs {
		var d wireToolDef
		d.Type = "function"
		d.Function.Name = s.Name
		d.Function.Description = s.Description
		if s.Parameters == nil {
			d.Function.Parameters = map[string]any{"type": "object", "properties": map[string]any{}}
		} else {
			d.Function.Parameters = s.Parameters
		}
		out = append(out, d)
	}
	return out
}
```

package main

import (

func main() {
	call := NewToolUse("get_time", map[string]any{"city": "北京"})

跑一下能看到：**那条同时有文字和工具调用的助手消息**翻译成了一个带 `tool_calls`
数组的对象；工具结果被拆成了独立的一条 `role: "tool"` 消息。

## 回来的方向：接口格式 → 内部协议

```go
// file: decode.go
package main

import "encoding/json"

// parseArguments 把 arguments 那段 JSON 文本解析成参数字典。
//
// 模型偶尔会吐出不合法的 JSON——被截断、少个括号、或者干脆是自然语言。
// 这时不能让整轮崩掉：把原文留下，让上层决定怎么办。适配层的职责是如实转达，
// 不是替模型遮丑。
func parseArguments(raw string) map[string]any {
	if raw == "" {
		return map[string]any{}
	}
	var args map[string]any
	if err := json.Unmarshal([]byte(raw), &args); err != nil {
		return map[string]any{"_invalid_json": raw}
	}
	return args
}

func fromWireMessage(wm wireMessage) Message {
	var blocks []Block
	if wm.Content != nil && *wm.Content != "" {
		blocks = append(blocks, TextBlock{Text: *wm.Content})
	}
	for _, tc := range wm.ToolCalls {
		id := tc.ID
		if id == "" {
			id = newID("call")
		}
		blocks = append(blocks, ToolUseBlock{
			ID:   id,
			Name: tc.Function.Name,
			Args: parseArguments(tc.Function.Arguments),
		})
	}
	return NewMessage("assistant", blocks...)
}
```

## 真的调一次

```go
// file: client.go
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"time"
)

const (
	baseURL   = "http://192.168.213.116:8084/v1"
	modelName = "qwen3.6-27B"
)

// callModel 一次非流式请求。
func callModel(messages []Message, tools []ToolSpec) (Message, error) {
	body := wireRequest{
		Model:    modelName,
		Messages: toWireMessages(messages),
		Tools:    toWireTools(tools),
	}
	raw, err := json.Marshal(body)
	if err != nil {
		return Message{}, fmt.Errorf("请求体序列化失败：%w", err)
	}

	req, err := http.NewRequest("POST", baseURL+"/chat/completions", bytes.NewReader(raw))
	if err != nil {
		return Message{}, fmt.Errorf("构造请求失败：%w", err)
	}
	req.Header.Set("Content-Type", "application/json")

	client := &http.Client{Timeout: 180 * time.Second}
	resp, err := client.Do(req)
	if err != nil {
		return Message{}, fmt.Errorf("请求失败：%w", err)
	}
	defer resp.Body.Close()

	data, err := io.ReadAll(resp.Body)
	if err != nil {
		return Message{}, fmt.Errorf("读响应失败：%w", err)
	}
	if resp.StatusCode != http.StatusOK {
		// 把服务端的错误原文带上——「模型服务返回 400」没法排查，
		// 带上原文才知道是模型名写错了还是参数不对。
		return Message{}, fmt.Errorf("模型服务返回 %d：%s", resp.StatusCode, truncate(string(data), 300))
	}

	var parsed wireResponse
	if err := json.Unmarshal(data, &parsed); err != nil {
		return Message{}, fmt.Errorf("响应解析失败：%w", err)
	}
	if len(parsed.Choices) == 0 {
		return Message{}, fmt.Errorf("响应里没有 choices")
	}
	return fromWireMessage(parsed.Choices[0].Message), nil
}

func truncate(s string, n int) string {
	if len(s) <= n {
		return s
	}
	return s[:n] + "…"
}
```

## 让它调工具，再把结果回填

```go
// file: main.go
package main

import "fmt"

var timeTool = ToolSpec{
	Name:        "get_time",
	Description: "查询某个城市的当前时间",
	Parameters: map[string]any{
		"type": "object",
		"properties": map[string]any{
			"city": map[string]any{"type": "string", "description": "城市名"},
		},
		"required": []string{"city"},
	},
}

func main() {
	history := []Message{
		NewMessage("system", TextBlock{Text: "需要查时间时就调用工具，不要自己编。"}),
		NewMessage("user", TextBlock{Text: "北京现在几点？"}),
	}

	reply, err := callModel(history, []ToolSpec{timeTool})
	if err != nil {
		fmt.Println("出错：", err)
		return
	}
	history = append(history, reply)

	if t := reply.Text(); t != "" {
		fmt.Println("模型：", t)
	}
	calls := reply.ToolUses()
	if len(calls) == 0 {
		fmt.Println("这次模型没调工具，直接答了：", reply.Text())
		return
	}

	var results []Block
	for _, c := range calls {
		fmt.Printf("模型要求调用：%s(%v)\n", c.Name, c.Args)
		results = append(results, ToolResultBlock{ToolUseID: c.ID, Content: "14:32"})
	}
	history = append(history, NewMessage("tool", results...))

	final, err := callModel(history, []ToolSpec{timeTool})
	if err != nil {
		fmt.Println("第二轮出错：", err)
		return
	}
	fmt.Println("最终回答：", final.Text())
}
```

一轮完整的往返跑通了——**这就是第 1 章那张图的真实版本**。

## 本章小结

- 适配层的职责是**双向翻译**：内部协议 ↔ 接口格式。
- Go 里把接口的数据结构单独声明成 `wire*` 类型，**只在适配层用，不外泄**。
- `Content` 用**指针**：为空时序列化成 `null`，而不是空字符串。
- 三个容易踩的地方：**参数是 JSON 文本不是对象**、**工具结果要拆成多条消息**、
  **一条助手消息可能既有文字又有工具调用**。
- 参数不是合法 JSON 时**不要报错，把原文留着**。
- HTTP 错误要**带上服务端的响应原文**，否则没法排查。
