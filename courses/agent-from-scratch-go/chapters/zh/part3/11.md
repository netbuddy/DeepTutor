# 并发与取消

上一章的主循环有两个明显的短板：

1. 同一轮里的多个工具调用是**排队执行**的。三个各要两秒的工具，用户等六秒。
2. **停不下来**。用户点了「停止」，循环还在跑，工具还在写文件。

这一章解决这两件事。第二件比第一件难得多，难点不在「怎么停」，而在**停完之后对话历史还能不能用**。

## 先说一处接口变更

Go 里表达取消只有一种标准做法：`context.Context`。它的惯例是**作为第一个参数**一路传下去。

所以从这一章起，两个接口都改了签名：

```go
Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error)
Run(ctx context.Context, args map[string]any) (string, error)
```

改接口是有代价的：所有实现都得跟着改。这也是为什么 Go 社区的建议是
**接口一开始就带上 ctx**，哪怕暂时用不到——早改比晚改便宜得多。
这门课让你亲历一次晚改的代价，比直接给你一个带 ctx 的版本更有用。

```go
import (

func newID(prefix string) string {
	buf := make([]byte, 6)
	rand.Read(buf)
	return prefix + "_" + hex.EncodeToString(buf)
}

type Block interface{ BlockType() string }

type TextBlock struct{ Text string }

func (TextBlock) BlockType() string { return "text" }

type ToolUseBlock struct {
	ID   string
	Name string
	Args map[string]any
}

func (ToolUseBlock) BlockType() string { return "tool_use" }

type ToolResultBlock struct {
	ToolUseID string
	Content   string
	IsError   bool
}

func (ToolResultBlock) BlockType() string { return "tool_result" }

type Message struct {
	ID      string
	Role    string
	Content []Block
}

func (m Message) Text() string {
	out := ""
	for _, b := range m.Content {
		if t, ok := b.(TextBlock); ok {
			out += t.Text
		}
	}
	return out
}

func (m Message) ToolUses() []ToolUseBlock {
	var calls []ToolUseBlock
	for _, b := range m.Content {
		if c, ok := b.(ToolUseBlock); ok {
			calls = append(calls, c)
		}
	}
	return calls
}

func (m Message) ToolResults() []ToolResultBlock {
	var out []ToolResultBlock
	for _, b := range m.Content {
		if r, ok := b.(ToolResultBlock); ok {
			out = append(out, r)
		}
	}
	return out
}

func NewMessage(role string, blocks ...Block) Message {
	return Message{ID: newID("msg"), Role: role, Content: blocks}
}

func NewToolUse(name string, args map[string]any) ToolUseBlock {
	return ToolUseBlock{ID: newID("call"), Name: name, Args: args}
}

type ToolSpec struct {
	Name        string
	Description string
	Parameters  map[string]any
}

type Delta struct {
	Kind string
	Text string
	Tool *ToolUseBlock
}

// Model 签名里多了 ctx。
type Model interface {
	Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error)
}

type wireFunction struct {
	Name      string `json:"name"`
	Arguments string `json:"arguments"`
}

type wireToolCall struct {
	Index    int          `json:"index,omitempty"`
	ID       string       `json:"id,omitempty"`
	Type     string       `json:"type,omitempty"`
	Function wireFunction `json:"function"`

type wireMessage struct {
	Role       string         `json:"role"`

	ToolCallID string         `json:"tool_call_id,omitempty"`

	Type     string `json:"type"`
	Function struct {

		Description string         `json:"description"`

	Messages []wireMessage `json:"messages"`

func toWireMessages(messages []Message) []wireMessage {
	var out []wireMessage
	for _, m := range messages {
		if m.Role == "tool" {
			for _, b := range m.Content {
				if r, ok := b.(ToolResultBlock); ok {
					content := r.Content
					out = append(out, wireMessage{Role: "tool", Content: &content, ToolCallID: r.ToolUseID})
				}
			}
			continue
		}
		text, calls := m.Text(), m.ToolUses()
		wm := wireMessage{Role: m.Role}
		if text != "" {
			wm.Content = &text
		}
		for _, c := range calls {
			args, err := json.Marshal(c.Args)
			if err != nil {
				args = []byte("{}")
			}
			wm.ToolCalls = append(wm.ToolCalls, wireToolCall{ID: c.ID, Type: "function",
				Function: wireFunction{Name: c.Name, Arguments: string(args)}})
		}
		out = append(out, wm)
	}
	return out
}

func toWireTools(specs []ToolSpec) []wireToolDef {
	var out []wireToolDef
	for _, s := range specs {
		var d wireToolDef
		d.Type = "function"
		d.Function.Name = s.Name
		d.Function.Description = s.Description
		if s.Parameters == nil {
			d.Function.Parameters = map[string]any{"type": "object", "properties": map[string]any{}}
		} else {
			d.Function.Parameters = s.Parameters
		}
		out = append(out, d)
	}
	return out
}

func parseArguments(raw string) map[string]any {
	if raw == "" {
		return map[string]any{}
	}
	var args map[string]any
	if err := json.Unmarshal([]byte(raw), &args); err != nil {
		return map[string]any{"_invalid_json": raw}
	}
	return args
}

type toolCallSlot struct{ id, name, arguments string }

type toolCallAccumulator struct{ slots map[int]*toolCallSlot }

func newToolCallAccumulator() *toolCallAccumulator {
	return &toolCallAccumulator{slots: map[int]*toolCallSlot{}}
}

func (a *toolCallAccumulator) Feed(calls []wireToolCall) {
	for _, item := range calls {
		slot, ok := a.slots[item.Index]
		if !ok {
			slot = &toolCallSlot{}
			a.slots[item.Index] = slot
		}
		if item.ID != "" {
			slot.id = item.ID
		}
		if item.Function.Name != "" {
			slot.name = item.Function.Name
		}
		if item.Function.Arguments != "" {
			slot.arguments += item.Function.Arguments
		}
	}
}

func (a *toolCallAccumulator) Finish() []ToolUseBlock {
	indexes := make([]int, 0, len(a.slots))
	for i := range a.slots {
		indexes = append(indexes, i)
	}
	sort.Ints(indexes)
	var blocks []ToolUseBlock
	for _, i := range indexes {
		slot := a.slots[i]
		id := slot.id
		if id == "" {
			id = newID("call")
		}
		blocks = append(blocks, ToolUseBlock{ID: id, Name: slot.name, Args: parseArguments(slot.arguments)})
	}
	return blocks
}

func parseSSELine(line string) (wireResponse, bool) {
	line = strings.TrimSpace(line)
	if line == "" || strings.HasPrefix(line, ":") || !strings.HasPrefix(line, "data:") {
		return wireResponse{}, false
	}
	payload := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
	if payload == "[DONE]" {
		return wireResponse{}, false
	}
	var parsed wireResponse
	if err := json.Unmarshal([]byte(payload), &parsed); err != nil {
		return wireResponse{}, false
	}
	return parsed, true
}

	BaseURL string
	ModelID string
	APIKey  string
	Timeout time.Duration
}

func NewOpenAICompatModel(baseURL, modelID string) *OpenAICompatModel {
	return &OpenAICompatModel{BaseURL: baseURL, ModelID: modelID, Timeout: 180 * time.Second}
}

func (m *OpenAICompatModel) Stream(ctx context.Context, messages []Message, tools []ToolSpec) (<-chan Delta, <-chan error) {
	deltas := make(chan Delta, 32)
	errs := make(chan error, 1)

	go func() {
		defer close(deltas)
		defer close(errs)

		// send 把「往通道里写」和「ctx 被取消」放在一个 select 里。
		//
		// 这一步不能省：取消之后消费者不再读 deltas，如果还用裸的 deltas <- d，
		// 这个 goroutine 就会永远卡在那一行——goroutine 泄漏，进程越跑越胖。
		send := func(d Delta) bool {
			select {
			case deltas <- d:
				return true
			case <-ctx.Done():
				return false
			}
		}

		body := wireRequest{Model: m.ModelID, Messages: toWireMessages(messages),
			Tools: toWireTools(tools), Stream: true}
		raw, err := json.Marshal(body)
		if err != nil {
			errs <- fmt.Errorf("请求体序列化失败：%w", err)
			return
		}

		// NewRequestWithContext 让取消能真正传到网络层：
		// ctx 一取消，这条 HTTP 连接立刻被关掉，而不是等它自己超时。
		req, err := http.NewRequestWithContext(ctx, "POST", m.BaseURL+"/chat/completions", bytes.NewReader(raw))
		if err != nil {
			errs <- fmt.Errorf("构造请求失败：%w", err)
			return
		}
		req.Header.Set("Content-Type", "application/json")
		if m.APIKey != "" {
			req.Header.Set("Authorization", "Bearer "+m.APIKey)
		}

		resp, err := (&http.Client{Timeout: m.Timeout}).Do(req)
		if err != nil {
			errs <- fmt.Errorf("请求失败：%w", err)
			return
		}
		defer resp.Body.Close()

		if resp.StatusCode != http.StatusOK {
			buf := make([]byte, 300)
			n, _ := resp.Body.Read(buf)
			errs <- fmt.Errorf("模型服务返回 %d：%s", resp.StatusCode, string(buf[:n]))
			return
		}

		acc := newToolCallAccumulator()
		scanner := bufio.NewScanner(resp.Body)
		scanner.Buffer(make([]byte, 0, 64*1024), 4*1024*1024)

		for scanner.Scan() {
			chunk, ok := parseSSELine(scanner.Text())
			if !ok || len(chunk.Choices) == 0 {
				continue
			}
			d := chunk.Choices[0].Delta
			if d.Content != nil && *d.Content != "" {
				if !send(Delta{Kind: "text", Text: *d.Content}) {
					errs <- ctx.Err()
					return
				}
			}
			if len(d.ToolCalls) > 0 {
				acc.Feed(d.ToolCalls)
			}
		}
		if err := scanner.Err(); err != nil {
			// ctx 取消时连接被关掉，scanner 也会报错。这时候要报 ctx.Err()，
			// 别把「用户取消了」说成「网络出错了」。
			if ctx.Err() != nil {
				errs <- ctx.Err()
			} else {
				errs <- fmt.Errorf("读流中断：%w", err)
			}
			return
		}
		for _, block := range acc.Finish() {
			b := block
			if !send(Delta{Kind: "tool_use", Tool: &b}) {
				errs <- ctx.Err()
				return
			}
		}
		errs <- nil
	}()

	return deltas, errs
}
```

```go
type ToolResult struct {
	Content       string
	IsError       bool
	IsInvalidArgs bool
}

// Tool 签名里也多了 ctx。工具想响应取消，就去看这个 ctx。
type Tool interface {
	Spec() ToolSpec
	Run(ctx context.Context, args map[string]any) (string, error)
}

type FuncTool struct {
	spec ToolSpec
	fn   func(ctx context.Context, args map[string]any) (string, error)
}

func NewFuncTool(name, description string, parameters map[string]any,
	fn func(ctx context.Context, args map[string]any) (string, error)) *FuncTool {
	return &FuncTool{spec: ToolSpec{Name: name, Description: description, Parameters: parameters}, fn: fn}
}

func (t *FuncTool) Spec() ToolSpec { return t.spec }

func (t *FuncTool) Run(ctx context.Context, args map[string]any) (string, error) {
	return t.fn(ctx, args)
}

func argString(args map[string]any, key string) string {
	s, _ := args[key].(string)
	return s
}

func argFloat(args map[string]any, key string, fallback float64) float64 {
	if v, ok := args[key].(float64); ok {
		return v
	}
	return fallback
}

func jsonTypeOf(value any) string {
	switch v := value.(type) {
	case string:
		return "string"
	case bool:
		return "boolean"
	case float64:
		if v == math.Trunc(v) {
			return "integer"
		}
		return "number"
	case int:
		return "integer"
	case []any:
		return "array"
	case map[string]any:
		return "object"
	case nil:
		return "null"
	}
	return "unknown"
}

func requiredNames(schema map[string]any) []string {
	switch v := schema["required"].(type) {
	case []string:
		return v
	case []any:
		var out []string
		for _, item := range v {
			if s, ok := item.(string); ok {
				out = append(out, s)
			}
		}
		return out
	}
	return nil
}

func sortedKeys(m map[string]any) []string {
	keys := make([]string, 0, len(m))
	for k := range m {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	return keys
}

func ValidateArgs(args map[string]any, schema map[string]any) string {
	if raw, bad := args["_invalid_json"]; bad {
		text := fmt.Sprint(raw)
		if len(text) > 120 {
			text = text[:120]
		}
		return "参数不是合法的 JSON：" + text
	}
	props, _ := schema["properties"].(map[string]any)
	if props == nil {
		props = map[string]any{}
	}
	for _, name := range requiredNames(schema) {
		if _, ok := args[name]; !ok {
			return "缺少必填参数 " + name
		}
	}
	for _, key := range sortedKeys(args) {
		spec, known := props[key].(map[string]any)
		if !known {
			return fmt.Sprintf("不认识的参数 %s；这个工具接受 %s", key, strings.Join(sortedKeys(props), "、"))
		}
		value := args[key]
		if want, _ := spec["type"].(string); want != "" {
			got := jsonTypeOf(value)
			if !(got == want || (want == "number" && got == "integer")) {
				return fmt.Sprintf("参数 %s 应该是 %s，收到的是 %s", key, want, got)
			}
		}
		if enum, ok := spec["enum"].([]any); ok && len(enum) > 0 {
			hit := false
			var allowed []string
			for _, candidate := range enum {
				allowed = append(allowed, fmt.Sprint(candidate))
				if fmt.Sprint(candidate) == fmt.Sprint(value) {
					hit = true
				}
			}
			if !hit {
				return fmt.Sprintf("参数 %s 只能是 %s 之一", key, strings.Join(allowed, "、"))
			}
		}
	}
	return ""
}

type ToolRegistry struct {
	tools map[string]Tool
	order []string
}

func NewToolRegistry(tools ...Tool) *ToolRegistry {
	reg := &ToolRegistry{tools: map[string]Tool{}}
	for _, t := range tools {
		reg.Add(t)
	}
	return reg
}

func (reg *ToolRegistry) Add(tool Tool) {
	name := tool.Spec().Name
	if _, exists := reg.tools[name]; !exists {
		reg.order = append(reg.order, name)
	}
	reg.tools[name] = tool
}

func (reg *ToolRegistry) Get(name string) Tool { return reg.tools[name] }

func (reg *ToolRegistry) Specs() []ToolSpec {
	var out []ToolSpec
	for _, name := range reg.order {
		out = append(out, reg.tools[name].Spec())
	}
	return out
}

func (reg *ToolRegistry) Execute(ctx context.Context, name string, args map[string]any) (result ToolResult) {
	tool, ok := reg.tools[name]
	if !ok {
		return ToolResult{Content: fmt.Sprintf("没有叫 %s 的工具。可用的：%s",
			name, strings.Join(reg.order, "、")), IsError: true}
	}
	if problem := ValidateArgs(args, tool.Spec().Parameters); problem != "" {
		return ToolResult{Content: "参数不合格：" + problem, IsError: true, IsInvalidArgs: true}
	}
	defer func() {
		if r := recover(); r != nil {
			result = ToolResult{Content: fmt.Sprintf("工具 %s 执行时崩溃了：%v", name, r), IsError: true}
		}
	}()
	output, err := tool.Run(ctx, args)
	if err != nil {
		// 取消导致的失败要说成「已取消」，别说成工具坏了。
		if ctx.Err() != nil {
			return ToolResult{Content: "用户取消了这次执行", IsError: true}
		}
		return ToolResult{Content: err.Error(), IsError: true}
	}
	return ToolResult{Content: output}
}
```

```go
// file: events.go
package main

import (
	"fmt"
	"sync"
)

type Event struct {
	Kind string
	Data map[string]any
}

type EventStream struct {
	mu   sync.RWMutex
	subs map[int]func(Event)
	next int
}

func NewEventStream() *EventStream { return &EventStream{subs: map[int]func(Event){}} }

func (s *EventStream) Subscribe(fn func(Event)) func() {
	s.mu.Lock()
	defer s.mu.Unlock()
	id := s.next
	s.next++
	s.subs[id] = fn
	return func() {
		s.mu.Lock()
		defer s.mu.Unlock()
		delete(s.subs, id)
	}
}

func (s *EventStream) Emit(kind string, data map[string]any) {
	s.mu.RLock()
	handlers := make([]func(Event), 0, len(s.subs))
	for _, fn := range s.subs {
		handlers = append(handlers, fn)
	}
	s.mu.RUnlock()
	event := Event{Kind: kind, Data: data}
	for _, fn := range handlers {
		func() {
			defer func() {
				if r := recover(); r != nil {
					fmt.Printf("  [订阅者出错，已忽略] %v\n", r)
				}
			}()
			fn(event)
		}()
	}
}
```

## 不是所有工具都能并发

并发执行工具之前，得先承认一件事：**有些工具不能并发跑**。

两个「写同一个文件」的调用同时执行，结果是坏文件。两个「查资料」的调用同时执行，没问题。
区别在于有没有副作用——这件事只有工具作者知道，所以让工具自己声明。

Python 那门课里这是数据类上的一个字段。Go 这里用**可选接口**：

```go
// file: concurrency.go
package main

// ConcurrencyAware 工具**可以**实现这个接口来声明自己不能并发跑。
//
// 这是 Go 里加可选能力的惯用做法：不改动主接口 Tool，
// 想声明的类型多写一个方法，不想声明的什么都不用做。
// 标准库里到处是这个套路（比如 io.Writer 之外的 io.ReaderFrom）。
type ConcurrencyAware interface {
	ConcurrencySafe() bool
}

// isConcurrencySafe 探测一个工具的声明。没实现那个接口就当成安全的——
// 只读工具是多数，让多数派免于写样板代码。
func isConcurrencySafe(tool Tool) bool {
	if aware, ok := tool.(ConcurrencyAware); ok {
		return aware.ConcurrencySafe()
	}
	return true
}
```

## 并发执行

```go
// file: runner.go
package main

import (
	"context"
	"sync"
)

type ToolRunner struct {
	Registry   *ToolRegistry
	MaxWorkers int
}

func NewToolRunner(reg *ToolRegistry) *ToolRunner {
	return &ToolRunner{Registry: reg, MaxWorkers: 4}
}

func (r *ToolRunner) executeOne(ctx context.Context, call ToolUseBlock) ToolResultBlock {
	res := r.Registry.Execute(ctx, call.Name, call.Args)
	return ToolResultBlock{ToolUseID: call.ID, Content: res.Content, IsError: res.IsError}
}

// RunBatch 执行一批工具调用。并发安全的并发跑，不安全的串行跑。
func (r *ToolRunner) RunBatch(ctx context.Context, calls []ToolUseBlock) []Block {
	results := make([]ToolResultBlock, len(calls))
	filled := make([]bool, len(calls))

	// 用一个有缓冲的通道当信号量，限制同时在跑的 goroutine 数。
	// Go 里没有现成的线程池——信号量就是标准做法。
	sem := make(chan struct{}, r.MaxWorkers)
	var wg sync.WaitGroup

	for i, call := range calls {
		tool := r.Registry.Get(call.Name)
		if tool != nil && !isConcurrencySafe(tool) {
			continue // 留给下面串行跑
		}
		wg.Add(1)
		go func(i int, call ToolUseBlock) {
			defer wg.Done()
			sem <- struct{}{}
			defer func() { <-sem }()
			// 每个 goroutine 只写自己那一格，所以这里**不需要加锁**。
			// 切片本身不被改动（长度、底层数组都固定），只有各自的元素被赋值。
			results[i] = r.executeOne(ctx, call)
			filled[i] = true
		}(i, call)
	}
	wg.Wait()

	for i, call := range calls {
		if filled[i] {
			continue
		}
		tool := r.Registry.Get(call.Name)
		if tool != nil && isConcurrencySafe(tool) {
			continue
		}
		if ctx.Err() != nil {
			break // 取消之后不再启动新的
		}
		results[i] = r.executeOne(ctx, call)
		filled[i] = true
	}

	// 关键一步：按**原始顺序**输出，并且给没跑到的调用补一条结果。
	// 下一节解释为什么这一步不能省。
	out := make([]Block, len(calls))
	for i, call := range calls {
		if filled[i] {
			out[i] = results[i]
		} else {
			out[i] = ToolResultBlock{ToolUseID: call.ID, Content: "用户取消了这次执行", IsError: true}
		}
	}
	return out
}
```

## 为什么必须给每个调用补一条结果

这是这一章最要紧的一句话：

> **模型发出的每一个 tool_use，历史里都必须有一条对应的 tool_result。**
> 少一条，下一次请求就会被服务端拒绝。

取消最容易违反这条：用户在三个工具跑到一半时点了停止，两个跑完了，一个没跑。
如果只把跑完的两条写进历史，这段对话就**永久坏掉了**——以后每次带上它都会报错。

```go
// file: main.go
package main

import "fmt"

func main() {
	calls := []ToolUseBlock{
		NewToolUse("get_time", map[string]any{"city": "北京"}),
		NewToolUse("slow_lookup", map[string]any{"topic": "量子计算", "seconds": 5.0}),
	}
	assistant := NewMessage("assistant", TextBlock{Text: "我查一下。"}, calls[0], calls[1])

	broken := NewMessage("tool", ToolResultBlock{ToolUseID: calls[0].ID, Content: "14:32"})
	fixed := NewMessage("tool",
		ToolResultBlock{ToolUseID: calls[0].ID, Content: "14:32"},
		ToolResultBlock{ToolUseID: calls[1].ID, Content: "用户取消了这次执行", IsError: true})

	for _, c := range []struct {
		label string
		msg   Message
	}{{"缺一条结果", broken}, {"补齐了结果", fixed}} {
		wire := toWireMessages([]Message{assistant, c.msg})
		nCalls := len(wire[0].ToolCalls)
		nResults := 0
		for _, m := range wire[1:] {
			if m.Role == "tool" {
				nResults++
			}
		}
		verdict := "合法"
		if nCalls != nResults {
			verdict = "服务端会拒绝"
		}
		fmt.Printf("%s：%d 个调用 / %d 条结果 → %s\n", c.label, nCalls, nResults, verdict)
	}
}
```

## 带取消的主循环

```go
// file: loop.go
package main

import "context"

type AgentLoop struct {
	Model     Model
	Registry  *ToolRegistry
	Stream    *EventStream
	Runner    *ToolRunner
	MaxRounds int
}

func NewAgentLoop(model Model, reg *ToolRegistry, stream *EventStream) *AgentLoop {
	if stream == nil {
		stream = NewEventStream()
	}
	return &AgentLoop{Model: model, Registry: reg, Stream: stream,
		Runner: NewToolRunner(reg), MaxRounds: 8}
}

func (l *AgentLoop) Run(ctx context.Context, history []Message) TurnResult {
	turnID := newID("turn")
	var newMessages []Message
	l.Stream.Emit("turn_start", map[string]any{"turn_id": turnID})

	for round := 1; round <= l.MaxRounds; round++ {
		if ctx.Err() != nil {
			return l.cancelled(turnID, round-1, newMessages)
		}

		reply, interrupted, err := l.oneRound(ctx, append(append([]Message{}, history...), newMessages...))
		if err != nil && ctx.Err() == nil {
			l.Stream.Emit("error", map[string]any{"turn_id": turnID, "message": err.Error()})
			return TurnResult{StopReason: StopError, RoundsUsed: round, Messages: newMessages, Err: err}
		}

		newMessages = append(newMessages, reply)

		// 即使是被打断的那一轮，也要把已经解析出来的调用补上结果——
		// 否则历史就缺了对应关系。
		if calls := reply.ToolUses(); len(calls) > 0 {
			results := l.Runner.RunBatch(ctx, calls)
			newMessages = append(newMessages, NewMessage("tool", results...))
		} else if !interrupted && ctx.Err() == nil {
			l.Stream.Emit("turn_done", map[string]any{"turn_id": turnID, "reason": string(StopCompleted)})
			return TurnResult{Text: reply.Text(), StopReason: StopCompleted,
				RoundsUsed: round, Messages: newMessages}
		}

		if interrupted || ctx.Err() != nil {
			return l.cancelled(turnID, round, newMessages)
		}
	}

	l.Stream.Emit("turn_done", map[string]any{"turn_id": turnID, "reason": string(StopBudget)})
	text := ""
	if n := len(newMessages); n >= 2 {
		text = newMessages[n-2].Text()
	}
	return TurnResult{Text: text, StopReason: StopBudget, RoundsUsed: l.MaxRounds, Messages: newMessages}
}

func (l *AgentLoop) cancelled(turnID string, rounds int, messages []Message) TurnResult {
	l.Stream.Emit("turn_done", map[string]any{"turn_id": turnID, "reason": string(StopCancelled)})
	return TurnResult{StopReason: StopCancelled, RoundsUsed: rounds, Messages: messages}
}

// oneRound 返回（消息, 是否被中途打断, 错误）。
//
// 被打断时**已经收到的内容照样组装成消息**——那些字是模型真的说了的，
// 丢掉它们会让历史与事实不符。
func (l *AgentLoop) oneRound(ctx context.Context, messages []Message) (Message, bool, error) {
	deltas, errs := l.Model.Stream(ctx, messages, l.Registry.Specs())

	var textParts []string
	var blocks []Block
	flushText := func() {
		if len(textParts) > 0 {
			joined := ""
			for _, p := range textParts {
				joined += p
			}
			blocks = append(blocks, TextBlock{Text: joined})
			textParts = nil
		}
	}

	interrupted := false
	for d := range deltas {
		if ctx.Err() != nil {
			interrupted = true
			// 注意：这里 break 之后**必须继续把 deltas 读空**，
			// 否则生产者会卡在写通道那一行。下面那个 for range 就是干这个的。
			break
		}
		switch d.Kind {
		case "text":
			textParts = append(textParts, d.Text)
			l.Stream.Emit("text_delta", map[string]any{"delta": d.Text})
		case "tool_use":
			if d.Tool != nil {
				flushText()
				blocks = append(blocks, *d.Tool)
			}
		}
	}
	if interrupted {
		for range deltas { // 排空，让生产者能走完 defer close
		}
	}
	flushText()

	err := <-errs
	message := NewMessage("assistant", blocks...)
	l.Stream.Emit("message_done", map[string]any{"message_id": message.ID, "text": message.Text()})
	return message, interrupted, err
}
```

## 并发到底快了多少

```go
// file: main.go
package main

import (
	"context"
	"fmt"
	"time"
)

func main() {
	reg := NewToolRegistry(GetTimeTool{}, SlowLookupTool{}, AppendNoteTool{})
	runner := NewToolRunner(reg)

	var calls []ToolUseBlock
	for _, topic := range []string{"量子计算", "光刻机", "核聚变"} {
		calls = append(calls, NewToolUse("slow_lookup", map[string]any{"topic": topic, "seconds": 2.0}))
	}

	start := time.Now()
	results := runner.RunBatch(context.Background(), calls)
	fmt.Printf("三个各 2 秒的工具，并发用时 %.1f 秒\n", time.Since(start).Seconds())
	for _, b := range results {
		r := b.(ToolResultBlock)
		fmt.Printf("  %s\n", r.Content)
	}

	start = time.Now()
	for _, c := range calls {
		runner.executeOne(context.Background(), c)
	}
	fmt.Printf("同样三个，串行用时 %.1f 秒\n", time.Since(start).Seconds())
}
```

Go 这里比 Python 省心：goroutine 是真并行，不受全局解释器锁限制，
纯计算的工具也照样能重叠。代价是**你要自己想清楚共享数据的问题**——
上面那个「每个 goroutine 只写自己那一格」就是在回避共享。

## 中途按停止

```go
// file: main.go
package main

import (
	"context"
	"fmt"
	"time"
)

func main() {
	reg := NewToolRegistry(GetTimeTool{}, SlowLookupTool{})
	model := NewOpenAICompatModel("http://192.168.213.116:8084/v1", "qwen3.6-27B")
	loop := NewAgentLoop(model, reg, nil)

	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()

	history := []Message{
		NewMessage("system", TextBlock{Text: "你可以使用工具。需要资料时调用 slow_lookup。"}),
		NewMessage("user", TextBlock{Text: "帮我查一下量子计算的资料，seconds 传 8。"}),
	}

	done := make(chan TurnResult, 1)
	start := time.Now()
	go func() { done <- loop.Run(ctx, history) }()

	time.Sleep(3 * time.Second)
	fmt.Println("→ 用户点了停止")
	cancel()

	result := <-done
	fmt.Printf("从开始到停下用了 %.1f 秒（工具本来要 8 秒）\n", time.Since(start).Seconds())
	fmt.Printf("停止原因：%s\n", result.StopReason)

	// 停完之后，历史还能用吗？这才是验收标准。
	var callIDs, resultIDs []string
	for _, m := range result.Messages {
		for _, c := range m.ToolUses() {
			callIDs = append(callIDs, c.ID)
		}
		for _, r := range m.ToolResults() {
			resultIDs = append(resultIDs, r.ToolUseID)
			fmt.Printf("  结果：%s\n", r.Content)
		}
	}
	var missing []string
	for _, id := range callIDs {
		found := false
		for _, rid := range resultIDs {
			if rid == id {
				found = true
			}
		}
		if !found {
			missing = append(missing, id)
		}
	}
	fmt.Printf("工具调用 %d 个，结果 %d 条，缺结果的：%v\n", len(callIDs), len(resultIDs), missing)
	if len(missing) == 0 {
		fmt.Println("→ 这段历史可以继续用")
	} else {
		fmt.Println("→ 这段历史已经坏了")
	}
}
```

取消之后历史自洽，用户就能接着说「算了，换个话题」，而不是被迫新开一个会话。

## 本章小结

- Go 里表达取消只有一种标准做法：`context.Context`，作为第一个参数一路传下去。
  **接口一开始就带上 ctx**，晚改的代价你这一章已经付过了。
- 往通道里写的时候要 `select` 上 `ctx.Done()`，否则取消之后生产者会永远卡住——**goroutine 泄漏**。
- `http.NewRequestWithContext` 让取消传到网络层，连接立刻断掉而不是等超时。
- 可被打断的等待 = `select { case <-timer.C: ... case <-ctx.Done(): ... }`。定时器记得 `Stop()`。
- 工具用**可选接口**声明自己不能并发跑，不想声明的类型什么都不用写。
- 每个 goroutine 只写结果切片里自己那一格，就**不需要加锁**。
- **每一个 tool_use 都必须有一条对应的 tool_result**，包括被取消的那些。
  少一条，这段对话就永久坏掉。
- 中途 break 出 `for range 通道` 之后要把通道**排空**，否则生产者卡死。
- 取消导致的失败要报「已取消」，别报成「网络出错」。
